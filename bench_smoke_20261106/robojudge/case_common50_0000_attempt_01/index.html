+def ease(t):
+    t = max(0.0, min(1.0, t))
+    return t * t * (3.0 - 2.0 * t)

+def mix(a, b, t):
+    t = ease(t)

+def poly_mask(size, polygons):
+    im = Image.new("L", size, 0)

+def remove_fruit(plate, kind):

+def P(point):

+def capsule(draw, a, b, width, fill, outline=None, outline_width=0):

+def arm_geometry(grip):
