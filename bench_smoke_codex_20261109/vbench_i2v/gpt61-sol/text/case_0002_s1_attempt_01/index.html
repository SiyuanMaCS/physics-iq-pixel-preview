+"""Deterministic, image-based macro liquid animation, with a rightward dolly/pan.
+
+The photograph is the sole texture.  A smooth Lagrangian flow and a dispersive
+capillary-wave refraction field animate it without successively resampling frames.
+The small area uncovered by the camera uses a smoothly bent reflected extension
+of the same liquid, not a clamped or stretched border.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from PIL import Image
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def reflect_index(i, size):
+    i = np.mod(i, 2 * (size - 1))
+    return np.where(i < size, i, 2 * (size - 1) - i).astype(np.int32)
+
+
+def cubic_weights(f):
+    # Interpolating cubic, so the zero-time sample retains the actual pixels.
+    f2, f3 = f * f, f * f * f
+    return (-0.5*f + f2 - 0.5*f3,
+            1.0 - 2.5*f2 + 1.5*f3,
+            0.5*f + 2.0*f2 - 1.5*f3,
+            -0.5*f2 + 0.5*f3)
+
+
+def sample_rgb(image, x, y):
+    h, w = image.shape[:2]
+    ix, iy = np.floor(x).astype(np.int32), np.floor(y).astype(np.int32)
+    wx, wy = cubic_weights(x-ix), cubic_weights(y-iy)
+    result = np.zeros(x.shape + (3,), np.float32)
+    for j in range(4):
+        yy = reflect_index(iy+j-1, h)
+        line = np.zeros_like(result)
+        for i in range(4):
+            xx = reflect_index(ix+i-1, w)
+            line += image[yy, xx] * wx[i][..., None]
+        result += line * wy[j][..., None]
+    return result
+
+
+def extended_coordinates(x, y):
+    """Irregular continuation of the liquid beyond the photographed area.
+
+    The bending starts with a zero derivative at the edge, so a feature does
+    not pop or tear when it first enters the camera's field of view.
+    """
+    d = np.maximum(x-255.0, 0.0)
+    ease = 1.0 - np.exp(-(d/22.0)**2)
+    bend = ease * (5.0*np.sin(y/25.0 + d/34.0)
+                   + 2.8*np.sin(y/11.0 - d/29.0))
+    sy = y + bend
+    sx = np.where(x <= 255, x,
+                  255 - d*(0.91 + 0.085*np.sin(y/38.0))
+                  + ease*2.1*np.sin(y/19.0 + d/23.0))
+    return sx, sy
+
+
+def liquid_coordinates(x, y, t):
+    # These large, very slow eddies transport existing colour and reflections.
+    # Analytic rotations avoid the diffusion of repeated image advection.
+    sx, sy = x.copy(), y.copy()
+    for cx, cy, radius, rate in [
+        (72., 58., 46., .060),
+        (187., 116., 42., -.074),
+        (284., 44., 48., .051),
+    ]:
+        rx, ry = sx-cx, sy-cy
+        angle = -rate*t*np.exp(-(rx*rx+ry*ry)/(2*radius*radius))
+        co, si = np.cos(angle), np.sin(angle)
+        sx, sy = cx+co*rx-si*ry, cy+si*rx+co*ry
+
+    sx -= t * (.37*np.sin(y/29.0 + x/88.0) + .22*np.cos(y/19.0))
+    sy -= t * (.28*np.sin(x/48.0 - y/71.0))
+
+    # Long gravity ripples overlaid with shorter, faster capillary ripples.
+    # Difference from t=0 supplies exact initial-image registration.
+    waves = [
+        (105.,  .25, .968, 2.65, 1.10, .4),
+        (67.,  -.46, .888, 1.95, 1.47, 2.3),
+        (43.,   .57, .822, 1.25, 1.90, 4.5),
+        (28.,  -.30, .954,  .75, 2.35, 1.4),
+        (18.,   .45, .893,  .38, 2.92, 3.3),
+        (11.5, -.22, .975,  .20, 3.61, 5.1),
+    ]
+    lighting = np.zeros_like(x)
+    for length, nx, ny, amplitude, speed, phase in waves:
+        p = 2*np.pi/length*(nx*x + ny*y) + phase
+        q = p-speed*t
+        delta = amplitude*(np.sin(q)-np.sin(p))
+        sx += nx*delta
+        sy += ny*delta
+        lighting += (0.45 if length < 30 else 0.75)*(np.cos(q)-np.cos(p))
+    return sx, sy, lighting
+
+
+def generate():
+    spec = json.loads((ROOT / 'task_spec.json').read_text())
+    first = np.asarray(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    h, w = first.shape[:2]
+    assert (h, w) == (spec['height'], spec['width'])
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    texture = first.astype(np.float32)
+    frames = np.empty((spec['frames'], h, w, 3), dtype=np.uint8)
+    frames[0] = first
+    for i in range(1, spec['frames']):
+        t = i / spec['fps']
+        # Panning right means fixed liquid features move to the left.
+        camera = 12.0*(t-.25*(1.0-np.exp(-t/.25)))
+        world_x = xx+camera
+        sx, sy, light = liquid_coordinates(world_x, yy, t)
+        tx, ty = extended_coordinates(sx, sy)
+        rgb = sample_rgb(texture, tx, ty)
+
+        # Subtle moving surface glints; brightest reflections respond most.
+        # Pigment remains blue/orange rather than a new overlay of white noise.
+        lum = rgb @ np.array([.2126, .7152, .0722], np.float32)
+        sheen = np.clip((lum-45)/170, 0., 1.)
+        gain = .013*light*(.35+.65*sheen)
+        rgb *= (1.0+gain[..., None])
+        rgb += (light*.35*sheen)[..., None]*np.array([.88, .98, 1.0])
+        frames[i] = np.clip(np.rint(rgb), 0, 255).astype(np.uint8)
+
+    # Save numerical inspection measurements alongside the canonical artifact.
+    report = {
+        'shape': list(frames.shape),
+        'dtype': str(frames.dtype),
+        'first_frame_identical': bool(np.array_equal(frames[0], first)),
+        'mean_adjacent_rgb_change': np.abs(np.diff(frames.astype(np.float32), axis=0)).mean(axis=(1, 2, 3)).tolist(),
+        'camera_travel_pixels': float(12.0*(5.0-.25*(1.0-np.exp(-5.0/.25)))),
+    }
+    (ROOT/'.tmp'/'validation.json').write_text(json.dumps(report, indent=2))
+    artifact_io.save(frames, ROOT)
+    print('Saved', frames.shape, frames.dtype, '| initial frame exact:', report['first_frame_identical'])
+    print('Mean per-frame RGB change:', np.mean(report['mean_adjacent_rgb_change']))
+
+
+if __name__ == '__main__':
+    generate()
