+"""Animate the supplied macro photograph with smooth, continuous liquid flow.
+
+The pigment/reflection photograph is sampled only once per frame, at coordinates
+obtained by integrating a smooth vortical velocity field backwards in time.
+Long gravity waves and smaller capillary waves perturb the liquid surface.
+No trained model or other source imagery is used.
+"""
+from pathlib import Path
+import json
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def edge_window(p, length, scale):
+    """Allow tangential flow at a boundary, but no missing image at that edge."""
+    d = np.maximum(0.0, np.minimum(p, length - 1.0 - p))
+    return -np.expm1(-d / scale)
+
+
+def velocity(x, y, t, width, height):
+    # Smooth eddies, with mildly changing strengths, stretch the two pigments
+    # into curved ribbons rather than moving the entire photograph as a panel.
+    eddies = (
+        (143.0, 78.0, 96.0, 0.125, 0.11, 0.2),
+        (68.0, 43.0, 48.0, -0.235, 0.15, 1.5),
+        (193.0, 30.0, 37.0, -0.300, 0.10, 2.6),
+        (119.0, 106.0, 35.0, 0.230, 0.13, 0.8),
+        (224.0, 119.0, 42.0, -0.170, 0.10, 3.3),
+        (29.0, 104.0, 30.0, 0.140, 0.10, 4.1),
+    )
+    u = np.full_like(x, 0.95)
+    v = np.full_like(y, 0.40)
+    for cx, cy, radius, omega, modulation, phase in eddies:
+        dx, dy = x - cx, y - cy
+        strength = omega * (1.0 + modulation * np.sin(0.65 * t + phase))
+        g = strength * np.exp(-(dx * dx + dy * dy) / (2.0 * radius * radius))
+        u -= dy * g
+        v += dx * g
+    # A few broad divergence-free wave currents gently evolve the eddies.
+    for kx, ky, amplitude, frequency, phase in (
+        (0.041, 0.036, 25.0, 0.70, 0.5),
+        (-0.027, 0.061, 16.0, 0.49, 2.1),
+        (0.071, 0.018, 10.0, 0.87, 4.0),
+    ):
+        c = amplitude * np.cos(kx * x + ky * y - frequency * t + phase)
+        u += ky * c
+        v -= kx * c
+    u *= edge_window(x, width, 11.0)
+    v *= edge_window(y, height, 10.0)
+    return u, v
+
+
+def backtrace(x, y, time, width, height):
+    # Midpoint integration, using a fixed upper bound on the time step.  Unlike
+    # recurrent image warping, this retains the original photographic texture.
+    steps = max(1, int(np.ceil(time / 0.085)))
+    dt = time / steps
+    px, py = x.copy(), y.copy()
+    for step in range(steps):
+        now = time - step * dt
+        ux, uy = velocity(px, py, now, width, height)
+        mx, my = px - 0.5 * dt * ux, py - 0.5 * dt * uy
+        ux, uy = velocity(mx, my, now - 0.5 * dt, width, height)
+        px -= dt * ux
+        py -= dt * uy
+    return px, py
+
+
+def surface(x, y, time, width, height):
+    # Traveling waves of multiple orientations. Subtracting their initial
+    # displacement makes the first image an exact initial condition.
+    dx, dy, light = np.zeros_like(x), np.zeros_like(y), np.zeros_like(x)
+    for kx, ky, amplitude, speed, phase in (
+        (0.046, 0.071, 1.65, 1.19, 0.3),
+        (-0.089, 0.038, 1.05, 1.42, 1.8),
+        (0.116, 0.094, 0.58, 1.80, 3.1),
+        (0.205, -0.131, 0.23, 2.26, 4.8),
+    ):
+        theta = kx * x + ky * y + phase
+        norm = np.hypot(kx, ky)
+        delta = np.sin(theta - speed * time) - np.sin(theta)
+        dx += amplitude * kx / norm * delta
+        dy += amplitude * ky / norm * delta
+        light += amplitude * norm * (
+            np.cos(theta - speed * time) - np.cos(theta)
+        )
+    dx *= edge_window(x, width, 13.0)
+    dy *= edge_window(y, height, 11.0)
+    return dx, dy, light
+
+
+def generate():
+    spec = json.loads((ROOT / "task_spec.json").read_text())
+    source = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    height, width = source.shape[:2]
+    assert (height, width) == (spec["height"], spec["width"])
+    y, x = np.mgrid[:height, :width].astype(np.float32)
+    n, fps = spec["frames"], spec["fps"]
+    frames = np.empty((n, height, width, 3), dtype=np.uint8)
+    frames[0] = source
+    source_float = source.astype(np.float32)
+    statistics = []
+    for index in range(1, n):
+        time = index / fps
+        dx, dy, light = surface(x, y, time, width, height)
+        px, py = backtrace(x + dx, y + dy, time, width, height)
+        frame = cv2.remap(
+            source_float, px, py, cv2.INTER_CUBIC,
+            borderMode=cv2.BORDER_REFLECT_101,
+        )
+        # Weak changing specular illumination: reflections glint on the rolling
+        # surface without random flicker or an overall change in exposure.
+        brightness = frame.mean(axis=2)
+        reflective = np.clip((brightness - 40.0) / 155.0, 0.0, 1.0)
+        glint = 8.5 * light * (0.25 + 0.75 * reflective)
+        frame += glint[..., None] * np.array([0.91, 0.97, 1.0], np.float32)
+        frames[index] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+        difference = np.abs(frames[index].astype(float) - frames[index - 1])
+        statistics.append({
+            "frame": index,
+            "mean_frame_change": float(difference.mean()),
+            "mean_displacement": float(np.hypot(px - x, py - y).mean()),
+            "max_displacement": float(np.hypot(px - x, py - y).max()),
+        })
+    artifact_io.save(frames, ROOT)
+    (ROOT / "generation_stats.json").write_text(json.dumps(statistics, indent=2))
+    assert np.array_equal(frames[0], source)
+    return frames
+
+
+if __name__ == "__main__":
+    generate()
