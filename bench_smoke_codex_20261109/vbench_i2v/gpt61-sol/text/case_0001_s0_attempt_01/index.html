+"""Image-derived macro liquid motion and a continuous leftward camera pan.
+
+The initial photograph is the only source of appearance.  An incompressible
+analytic current advects its colored liquid/reflective detail, and a small
+surface-normal perturbation supplies the finer moving ripples.  No models or
+external assets are used.
+"""
+
+from pathlib import Path
+import json
+import numpy as np
+from PIL import Image, ImageFilter
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def cubic_weight(z):
+    """Interpolating cubic reconstruction, rather than repeated image warps."""
+    z = np.abs(z)
+    return np.where(
+        z <= 1, 1.5 * z**3 - 2.5 * z**2 + 1,
+        np.where(z < 2, -.5 * z**3 + 2.5 * z**2 - 4 * z + 2, 0),
+    )
+
+
+def sample_rgb(image, x, y):
+    h, w = image.shape[:2]
+    x = np.clip(x, 0, w - 1)
+    y = np.clip(y, 0, h - 1)
+    ix, iy = np.floor(x).astype(np.int32), np.floor(y).astype(np.int32)
+    fx, fy = x - ix, y - iy
+    out = np.zeros((*x.shape, 3), dtype=np.float32)
+    for j in (-1, 0, 1, 2):
+        wy = cubic_weight(fy - j)
+        row = np.clip(iy + j, 0, h - 1)
+        for i in (-1, 0, 1, 2):
+            wx = cubic_weight(fx - i)
+            col = np.clip(ix + i, 0, w - 1)
+            out += image[row, col] * (wx * wy)[..., None]
+    return out
+
+
+def velocity(x, y, height):
+    """The curl of a stream function: broad, nonuniform liquid currents."""
+    a = np.pi / (height - 1)
+    bound = np.sin(a * y)**2
+    dbound = a * np.sin(2 * a * y)
+    u = 65 * np.sin(x / 78 + .55) * dbound
+    v = -65 / 78 * np.cos(x / 78 + .55) * bound
+    # x, y, length scale, signed circulation.  The orange pools and the
+    # connecting blue channel each have their own gentle rotational current.
+    for cx, cy, r, strength in (
+        (86, 58, 46, 5.4),
+        (203, 22, 34, -4.0),
+        (202, 125, 53, 5.8),
+        (32, 92, 30, -3.2),
+        (-35, 55, 52, -3.8),
+    ):
+        dx, dy = x - cx, y - cy
+        e = np.exp(-(dx * dx + dy * dy) / (2 * r * r))
+        u += strength * e * (-dy / r * bound + r * dbound)
+        v += strength * e * dx / r * bound
+    return u, v
+
+
+def edge_tangents(first):
+    """Continue the inclination of fluid contours entering the left edge."""
+    blurred = np.asarray(
+        Image.fromarray(first).filter(ImageFilter.GaussianBlur(2.2)),
+        dtype=np.float32,
+    )
+    gx = (blurred[:, 4] - blurred[:, 0]) / 4
+    gy = np.gradient(blurred[:, :5].mean(axis=1), axis=0)
+    slope = -2 * np.sum(gx * gy, axis=1) / (np.sum(gy * gy, axis=1) + 35)
+    slope = np.clip(slope, -1.05, 1.05)
+    kernel = np.exp(-np.arange(-8, 9, dtype=float)**2 / 20)
+    kernel /= kernel.sum()
+    return np.convolve(np.pad(slope, (8, 8), mode='edge'), kernel, mode='valid')
+
+
+def main():
+    spec = json.loads((ROOT / 'task_spec.json').read_text())
+    first = np.array(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    h, w = first.shape[:2]
+    assert (w, h) == (spec['width'], spec['height'])
+    image = first.astype(np.float32)
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    slopes = edge_tangents(first)
+    frames = np.empty((spec['frames'], h, w, 3), dtype=np.uint8)
+    frames[0] = first
+
+    for n in range(1, spec['frames']):
+        t = n / spec['fps']
+        # A leftward camera movement makes the material travel to screen right.
+        pan = 9.25 * t + .8 * np.sin(.65 * t)
+        x, y = xx - pan, yy.copy()
+
+        # Trace material back to the original photograph, using midpoint
+        # integration.  Each frame samples that original just once, avoiding
+        # accumulated resampling blur.
+        steps = max(1, int(np.ceil(t / .17)))
+        dt = t / steps
+        for _ in range(steps):
+            u, v = velocity(x, y, h)
+            um, vm = velocity(x - .5 * dt * u, y - .5 * dt * v, h)
+            x -= dt * um
+            y -= dt * vm
+
+        # Fine capillary ripples: low amplitude compared with the large pools.
+        # Differences of phases guarantee an identity deformation at t=0.
+        envelope = np.sin(np.pi * np.clip(y, 0, h - 1) / (h - 1))
+        phase1 = x / 15.5 + y / 23
+        phase2 = x / 31 - y / 13
+        rx = .42 * (np.sin(phase1 + .86*t) - np.sin(phase1))
+        rx += .19 * (np.sin(phase2 - .67*t) - np.sin(phase2))
+        ry = .36 * (np.cos(phase2 + .72*t) - np.cos(phase2))
+        x += rx * (.35 + .65 * envelope)
+        y += ry * envelope
+
+        # Newly revealed material is extended from the photograph, reflecting
+        # through a smoothly sheared coordinate field rather than tiling it.
+        # The contour-dependent shear stops the reveal looking like a simple
+        # symmetrical repetition along the original left border.
+        left = np.maximum(-x, 0)
+        sy = np.clip(y, 0, h - 1)
+        k = np.interp(sy, np.arange(h), slopes).astype(np.float32)
+        reach = 24 * (1 - np.exp(-left / 24))
+        sy += reach * k * np.sin(np.pi * sy / (h - 1))
+        sy += 2.0 * (1 - np.exp(-left / 22)) * np.sin(left / 19 + y / 33) * envelope
+        sx = np.where(x < 0, left * (1 + .10 * np.sin(y / 29 + left / 35)), x)
+        result = sample_rgb(image, sx, sy)
+
+        # Very slight evolving illumination on the brighter liquid ridges.
+        # Colored dark areas keep their original saturation and density.
+        highlight = np.clip((result.max(axis=-1) - 115) / 105, 0, 1)
+        light = .012 * highlight * envelope * (
+            np.sin(xx / 46 + yy / 38 + .75*t)
+            - np.sin(xx / 46 + yy / 38)
+        )
+        result *= (1 + light[..., None])
+        frames[n] = np.clip(np.rint(result), 0, 255).astype(np.uint8)
+
+    assert np.array_equal(frames[0], first)
+    artifact_io.save(frames, ROOT)
+    print('Saved', frames.shape, 'RGB frames.')
+    d = np.abs(np.diff(frames.astype(np.float32), axis=0))
+    print('Mean adjacent-frame RGB difference:', round(float(d.mean()), 3))
+    print('First/final frame RGB difference:',
+          round(float(np.abs(frames[-1].astype(float)-first).mean()), 3))
+
+
+if __name__ == '__main__':
+    main()
