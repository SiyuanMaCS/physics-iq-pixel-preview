+"""Image-native advection of a close-up, viscous marbled liquid."""
+from pathlib import Path
+import json
+
+import cv2
+import numpy as np
+from PIL import Image
+
+from artifact_io import save
+
+
+def velocity(x, y, width, height):
+    """Overlapping broad eddies; small eddies carry the delicate paint veins."""
+    u = np.zeros_like(x)
+    v = np.zeros_like(y)
+    # Streamfunctions of elliptical vortices. The two warm pools circulate
+    # differently, while the central blue ribbon gently stretches between them.
+    vortices = (
+        (197., 12., 36., 39., 390.),
+        (223., 135., 55., 31., -440.),
+        (118., 51., 62., 25., 150.),
+        (53., 91., 38., 32., -105.),
+        (159., 77., 37., 22., -100.),
+        (20., 20., 29., 23., 65.),
+        (89., 16., 28., 18., -54.),
+    )
+    for cx, cy, rx, ry, strength in vortices:
+        dx, dy = x - cx, y - cy
+        e = np.exp(-0.5 * ((dx / rx) ** 2 + (dy / ry) ** 2))
+        u -= strength * dy / (ry * ry) * e
+        v += strength * dx / (rx * rx) * e
+
+    # Very broad, gently sinuous currents, rather than an animated wave overlay.
+    kx, ky = 2 * np.pi / 183., 2 * np.pi / 173.
+    phase_x, phase_y = kx * x + 0.5, ky * y - 0.8
+    u += 31. * ky * np.sin(phase_x) * np.cos(phase_y)
+    v -= 31. * kx * np.cos(phase_x) * np.sin(phase_y)
+    u += 0.65
+    v -= 0.12
+
+    # No artificial pixels enter the close-up. Only normal flow is smoothly
+    # brought to rest at the outermost edge; tangential circulation is retained.
+    edge_x = np.maximum(0., np.minimum(x, width - 1 - x))
+    edge_y = np.maximum(0., np.minimum(y, height - 1 - y))
+    u *= 1. - np.exp(-(edge_x / 8.) ** 2)
+    v *= 1. - np.exp(-(edge_y / 8.) ** 2)
+    return u, v
+
+
+def backward_step(x, y, dt, w, h):
+    # For a stationary velocity field this is the exact inverse flow's ODE.
+    # Keeping material coordinates, instead of repeatedly warping RGB frames,
+    # avoids cumulative diffusion of the very fine marbled lines.
+    a, b = velocity(x, y, w, h)
+    c, d = velocity(x - 0.5 * dt * a, y - 0.5 * dt * b, w, h)
+    e, f = velocity(x - 0.5 * dt * c, y - 0.5 * dt * d, w, h)
+    g, j = velocity(x - dt * e, y - dt * f, w, h)
+    x -= (dt / 6.) * (a + 2 * c + 2 * e + g)
+    y -= (dt / 6.) * (b + 2 * d + 2 * f + j)
+    return x, y
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    spec = json.loads((case_dir / "task_spec.json").read_text())
+    source = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"))
+    n, w, h, fps = spec["frames"], spec["width"], spec["height"], spec["fps"]
+    assert source.shape == (h, w, 3)
+    scale = 2
+    # A high-resolution material map and antialiased rendering make the little
+    # bubbles and copper-colored hairline bands move continuously.
+    texture = cv2.resize(source, (w * scale, h * scale), interpolation=cv2.INTER_CUBIC)
+    gy, gx = np.mgrid[:h * scale, :w * scale].astype(np.float32)
+    x = (gx + 0.5) / scale - 0.5
+    y = (gy + 0.5) / scale - 0.5
+    x = np.clip(x, 0., w - 1.)
+    y = np.clip(y, 0., h - 1.)
+    frames = np.empty((n, h, w, 3), dtype=np.uint8)
+    frames[0] = source
+    for i in range(1, n):
+        for _ in range(2):
+            x, y = backward_step(x, y, 1. / (fps * 2), w, h)
+        moving = cv2.remap(
+            texture, (x + .5) * scale - .5, (y + .5) * scale - .5,
+            interpolation=cv2.INTER_CUBIC, borderMode=cv2.BORDER_REPLICATE,
+        )
+        frames[i] = cv2.resize(moving, (w, h), interpolation=cv2.INTER_AREA)
+    assert np.array_equal(frames[0], source)
+    save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
