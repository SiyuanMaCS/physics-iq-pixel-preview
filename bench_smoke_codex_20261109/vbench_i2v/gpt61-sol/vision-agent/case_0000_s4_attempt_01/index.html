+"""Animate the source marbling by continuous, smooth liquid advection."""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+W, H = 256, 144
+
+
+def velocity(x, y, t):
+    # Elliptical eddies preserve the stretched, poured-paint character of
+    # the source. All detail is transported, rather than regenerated.
+    u = np.zeros_like(x)
+    v = np.zeros_like(y)
+    eddies = (
+        (193., 15., 35., 42., .30, .3),
+        (208., 133., 62., 34., -.235, 1.8),
+        (119., 72., 68., 28., .155, 3.0),
+        (36., 23., 39., 34., -.19, 4.2),
+        (48., 119., 58., 38., .075, 5.0),
+    )
+    for cx, cy, rx, ry, spin, phase in eddies:
+        cx += 2.4 * np.sin(.51 * t + phase) - 2.4 * np.sin(phase)
+        cy += 1.7 * np.sin(.43 * t + phase) - 1.7 * np.sin(phase)
+        dx, dy = x - cx, y - cy
+        strength = spin * (1.0 + .12 * np.sin(.8 * t + phase))
+        envelope = np.exp(-.5 * ((dx / rx) ** 2 + (dy / ry) ** 2))
+        u -= strength * (rx / ry) * dy * envelope
+        v += strength * (ry / rx) * dx * envelope
+    # A low-frequency through-flow softly flexes the intervening filaments.
+    u += 1.0 * np.sin(y / 29. + .35 * t) + .5
+    v += .7 * np.sin(x / 43. - .29 * t)
+    # No inflow from outside the photographed liquid: normal velocity
+    # vanishes at the image edge, while tangential drift remains free.
+    u *= 1.0 - np.exp(-np.maximum(x, 0) / 11.) - np.exp(-np.maximum(W - 1 - x, 0) / 11.)
+    v *= 1.0 - np.exp(-np.maximum(y, 0) / 10.) - np.exp(-np.maximum(H - 1 - y, 0) / 10.)
+    return u, v
+
+
+def cubic_weights(f):
+    return (
+        -.5 * f + f * f - .5 * f * f * f,
+        1. - 2.5 * f * f + 1.5 * f * f * f,
+        .5 * f + 2. * f * f - 1.5 * f * f * f,
+        -.5 * f * f + .5 * f * f * f,
+    )
+
+
+def sample(image, x, y):
+    x = np.clip(x, 0, W - 1)
+    y = np.clip(y, 0, H - 1)
+    ix, iy = np.floor(x).astype(np.int32), np.floor(y).astype(np.int32)
+    wx, wy = cubic_weights(x - ix), cubic_weights(y - iy)
+    out = np.zeros((*x.shape, 3), np.float32)
+    for j in range(4):
+        row = np.zeros_like(out)
+        yy = np.clip(iy + j - 1, 0, H - 1)
+        for i in range(4):
+            xx = np.clip(ix + i - 1, 0, W - 1)
+            row += image[yy, xx] * wx[i][..., None]
+        out += row * wy[j][..., None]
+    return np.clip(out, 0, 255)
+
+
+def generate():
+    first = np.asarray(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    image = first.astype(np.float32)
+    # Two samples per axis reduce aliasing in the finest pigment threads.
+    yy, xx = np.meshgrid((np.arange(H * 2, dtype=np.float32) + .5) / 2 - .5,
+                         (np.arange(W * 2, dtype=np.float32) + .5) / 2 - .5,
+                         indexing='ij')
+    frames = [first.copy()]
+    for index in range(1, 81):
+        t = index / 16.
+        # A very subtle push-in keeps the composition a close-up.
+        zoom = 1.0 + .024 * (t / 5.)
+        x = (xx - 127.5) / zoom + 127.5
+        y = (yy - 71.5) / zoom + 71.5
+        steps = max(1, int(np.ceil(t / .14)))
+        dt = t / steps
+        # Backward midpoint characteristic integration gives each frame
+        # a fresh sample of the original, preventing accumulated blur.
+        for k in range(steps):
+            now = t - k * dt
+            u, v = velocity(x, y, now)
+            um, vm = velocity(x - .5 * dt * u, y - .5 * dt * v, now - .5 * dt)
+            x -= dt * um
+            y -= dt * vm
+        rgb = sample(image, x, y)
+        rgb = np.rint(rgb).astype(np.uint8)
+        frame = np.asarray(Image.fromarray(rgb).resize((W, H), Image.Resampling.LANCZOS))
+        frames.append(frame)
+        if index % 16 == 0:
+            print(f'Generated {index}/80', flush=True)
+    frames = np.stack(frames)
+    assert np.array_equal(frames[0], first)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == '__main__':
+    generate()
