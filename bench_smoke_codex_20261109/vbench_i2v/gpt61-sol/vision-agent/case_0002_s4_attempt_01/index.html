+"""Reproduce a slowly moving macro view of marbled blue/orange liquid.
+
+The photograph supplies all colour and fine texture. A smooth, divergence-free
+flow carries that texture, while the camera moves to the right over the surface.
+"""
+from pathlib import Path
+import json
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def make_surface(source):
+    # Continue the photographed fluid beyond the right edge. The continuation
+    # curls back into the existing paint, with a gently bent reflection rather
+    # than a tiled texture or a stationary stretched border.
+    margin = 64
+    height, width = source.shape[:2]
+    yy, xx = np.mgrid[-margin:height + margin,
+                      -margin:width + 112].astype(np.float32)
+    beyond = np.maximum(xx - (width - 1), 0)
+    yy += (0.30 * beyond + 0.008 * beyond ** 2 *
+           np.sin((yy - 45) / 41))
+    # An additional gentle turn prevents the newly revealed gold stream from
+    # appearing as the bilateral twin of the one in the reference photograph.
+    xx += 0.14 * beyond * np.sin((yy + 12) / 38)
+    atlas = cv2.remap(source, xx, yy, cv2.INTER_CUBIC,
+                      borderMode=cv2.BORDER_REFLECT_101)
+    # Sampling from a high-resolution atlas gives a stable subpixel pan.
+    atlas = cv2.resize(atlas, None, fx=3, fy=3,
+                       interpolation=cv2.INTER_CUBIC)
+    return atlas, margin
+
+
+def velocity(x, y):
+    # Each Gaussian eddy is the perpendicular gradient of a stream function;
+    # consequently it deforms the ribbons without swelling their area.
+    vx = np.zeros_like(x)
+    vy = np.zeros_like(y)
+    eddies = [
+        (194, 20, 31, 27, 0.073),
+        (224, 129, 54, 30, -0.060),
+        (127, 57, 67, 31, 0.039),
+        (48, 31, 37, 31, -0.024),
+        (302, 74, 47, 53, 0.044),
+    ]
+    for cx, cy, sx, sy, strength in eddies:
+        dx, dy = x - cx, y - cy
+        falloff = np.exp(-0.5 * ((dx / sx) ** 2 + (dy / sy) ** 2))
+        vx -= strength * dy * (sx / sy) * falloff
+        vy += strength * dx * (sy / sx) * falloff
+    vx += 0.32 * np.sin(y / 26 + x / 97)
+    vy -= (0.32 * 26 / 97) * np.sin(y / 26 + x / 97)
+    return vx, vy
+
+
+def generate():
+    spec = json.loads((ROOT / 'task_spec.json').read_text())
+    original = np.asarray(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    h, w = original.shape[:2]
+    atlas, margin = make_surface(original)
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    frames = np.empty((spec['frames'], h, w, 3), dtype=np.uint8)
+    frames[0] = original
+    duration = (spec['frames'] - 1) / spec['fps']
+    for index in range(1, spec['frames']):
+        t = index / spec['fps']
+        # Constant, level camera motion; a tiny soft start avoids a jerk.
+        pan = 8.8 * (t - 0.15 * (1 - np.exp(-t / 0.15)))
+        x = xx + pan
+        y = yy.copy()
+        # Midpoint integration of inverse material trajectories.
+        steps = 18
+        dt = t / steps
+        for _ in range(steps):
+            ux, uy = velocity(x, y)
+            ux, uy = velocity(x - ux * dt * 0.5,
+                              y - uy * dt * 0.5)
+            x = x - dt * ux
+            y = y - dt * uy
+        # A very small, slowly changing ripple acts at a much finer scale than
+        # the large eddies; no crossfade is used, so paint stays fully opaque.
+        amplitude = 0.34 * np.sin(0.5 * np.pi * t / duration)
+        x += amplitude * np.sin(y / 10.5 + x / 28 - t * 0.21)
+        y += amplitude * 0.35 * np.sin(x / 15 + y / 23 + t * 0.17)
+        mx = ((x + margin + 0.5) * 3 - 0.5).astype(np.float32)
+        my = ((y + margin + 0.5) * 3 - 0.5).astype(np.float32)
+        frames[index] = cv2.remap(atlas, mx, my, cv2.INTER_CUBIC,
+                                 borderMode=cv2.BORDER_REFLECT_101)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == '__main__':
+    generate()
