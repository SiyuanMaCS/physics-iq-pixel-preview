+"""Advect the supplied marbled liquid while a macro camera pans to the left."""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def reflect(q, size):
+    p = np.mod(q, 2.0 * (size - 1))
+    return np.where(p <= size - 1, p, 2.0 * (size - 1) - p)
+
+
+def bilinear(image, x, y):
+    x = np.clip(x, 0, image.shape[1] - 1.001)
+    y = np.clip(y, 0, image.shape[0] - 1.001)
+    ix, iy = x.astype(np.int32), y.astype(np.int32)
+    fx, fy = (x - ix)[..., None], (y - iy)[..., None]
+    a = image[iy, ix] * (1 - fx) + image[iy, ix + 1] * fx
+    b = image[iy + 1, ix] * (1 - fx) + image[iy + 1, ix + 1] * fx
+    return a * (1 - fy) + b * fy
+
+
+def make_texture(source):
+    # Continue the marbling beyond the frame. A sheared reflected continuation
+    # retains the intricate pigment ribbons without a repeated/tiled boundary.
+    scale = 3
+    xmin, ymin = -112, -66
+    xmax, ymax = 325, 214
+    yy, xx = np.meshgrid(
+        np.arange(ymin * scale, ymax * scale + 1, dtype=np.float32) / scale,
+        np.arange(xmin * scale, xmax * scale + 1, dtype=np.float32) / scale,
+        indexing="ij",
+    )
+    left = np.maximum(-xx, 0)
+    sx = np.where(xx < 0, left * (0.89 + 0.075 * np.sin(yy / 32)), xx)
+    sy = yy + left * (0.19 + 0.09 * np.sin(yy / 47 + 0.3))
+    sx, sy = reflect(sx, 256), reflect(sy, 144)
+    enlarged = np.asarray(
+        Image.fromarray(source).resize((256 * scale, 144 * scale), Image.Resampling.LANCZOS),
+        dtype=np.float32,
+    )
+    texture = bilinear(enlarged, (sx + 0.5) * scale - 0.5, (sy + 0.5) * scale - 0.5)
+    return texture, scale, xmin, ymin
+
+
+def velocity(x, y):
+    # A steady, smooth flow. Elliptical eddies turn the orange ribbons slowly;
+    # the broad teal current carries fine tendrils along their existing curves.
+    vx = np.full_like(x, 0.50)
+    vy = np.full_like(y, -0.12)
+    for cx, cy, rx, ry, omega in (
+        (199.0, 8.0, 43.0, 53.0, 0.053),
+        (216.0, 130.0, 65.0, 29.0, -0.050),
+        (112.0, 39.0, 82.0, 39.0, -0.025),
+        (-9.0, 80.0, 35.0, 47.0, 0.034),
+    ):
+        dx, dy = x - cx, y - cy
+        envelope = np.exp(-0.5 * ((dx / rx) ** 2 + (dy / ry) ** 2))
+        vx += -omega * dy * (rx / ry) * envelope
+        vy += omega * dx * (ry / rx) * envelope
+    # Very low-amplitude incompressible perturbation, not an oscillating screen warp.
+    vx += 0.22 * np.sin(x / 31 + y / 24) + 0.16 * np.cos(x / 49 - y / 18)
+    vy += -0.22 * (24 / 31) * np.sin(x / 31 + y / 24)
+    vy += 0.16 * (18 / 49) * np.cos(x / 49 - y / 18)
+    return vx, vy
+
+
+def main():
+    source = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    texture, scale, xmin, ymin = make_texture(source)
+    yy, xx = np.mgrid[:144, :256].astype(np.float32)
+    frames = np.empty((81, 144, 256, 3), dtype=np.uint8)
+    frames[0] = source
+    for i in range(1, 81):
+        t = i / 16.0
+        # Leftward travel reveals the neighboring pigments on the left, while
+        # all stationary scene features move consistently to the right.
+        pan = 10.0 * (t - 0.20 * (1.0 - np.exp(-t / 0.20)))
+        zoom = 1.0 + 0.008 * t / 5.0
+        x = (xx - 127.5) / zoom + 127.5 - pan
+        y = (yy - 71.5) / zoom + 71.5
+        # Backtrace a stationary liquid flow using midpoint integration. Every
+        # frame is sampled from the original texture, avoiding accumulated blur.
+        steps = max(1, int(np.ceil(t / 0.10)))
+        dt = t / steps
+        for _ in range(steps):
+            u, v = velocity(x, y)
+            um, vm = velocity(x - 0.5 * dt * u, y - 0.5 * dt * v)
+            x -= dt * um
+            y -= dt * vm
+        rgb = bilinear(texture, (x - xmin) * scale, (y - ymin) * scale)
+        frames[i] = np.clip(np.rint(rgb), 0, 255).astype(np.uint8)
+    artifact_io.save(frames, ROOT)
+    assert np.array_equal(frames[0], source)
+
+
+if __name__ == "__main__":
+    main()
