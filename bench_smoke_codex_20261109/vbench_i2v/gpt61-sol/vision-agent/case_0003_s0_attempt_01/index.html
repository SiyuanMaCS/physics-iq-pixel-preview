+"""Animate a macro fluid photograph with smooth pigment advection and camera tilt.
+
+All visual detail is sampled from the supplied photograph.  A divergence-free
+velocity field moves the pigments, while a pinhole projection moves the camera.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from PIL import Image
+import cv2
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def velocity(x, y, time):
+    # Broad, slow eddies preserve the fine cellular structure in the paint.
+    vx = np.zeros_like(x)
+    vy = np.zeros_like(y)
+    vortices = (
+        (201., 3., 28., 0.105),
+        (145., 62., 40., -0.082),
+        (223., 145., 46., 0.092),
+        (11., 74., 25., -0.064),
+        (66., 17., 34., 0.035),
+    )
+    for cx, cy, radius, angular_speed in vortices:
+        dx, dy = x-cx, y-cy
+        strength = angular_speed * np.exp(-(dx*dx + dy*dy)/(2*radius*radius))
+        strength *= 1.0 + 0.10*np.sin(time*0.48 + cx*0.021)
+        vx -= dy*strength
+        vy += dx*strength
+    # Streamfunction-derived currents are incompressible, rather than expanding
+    # or contracting the original photograph as a rubber-sheet warp would.
+    for amp, kx, ky, phase, speed in (
+        (23., 0.025, 0.032, 0.5, 0.16),
+        (7., -0.047, 0.058, 1.8, -0.21),
+        (2., 0.105, 0.071, 3.1, 0.25),
+    ):
+        wave = amp*np.cos(kx*x + ky*y + phase + speed*time)
+        vx += ky*wave
+        vy -= kx*wave
+    return vx, vy
+
+
+def sample_photo(photo, x, y):
+    # The upward tilt reveals a narrow band beyond the photographed top edge.
+    # Continue its pigment ribbons with a curved reflective texture extension;
+    # the extension is continuous at the image boundary, without a flat border.
+    outside = np.maximum(-y, 0.)
+    bend = outside*(0.36 + 0.25*np.sin(x/31.0))
+    sx = x + bend
+    sy = np.where(y < 0, outside*(0.75 + 0.12*np.sin(x/42.0)), y)
+    return cv2.remap(photo, sx.astype(np.float32), sy.astype(np.float32),
+                     cv2.INTER_CUBIC, borderMode=cv2.BORDER_REFLECT_101)
+
+
+def generate():
+    spec = json.loads((ROOT/'task_spec.json').read_text())
+    original = np.asarray(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    count, height, width = spec['frames'], spec['height'], spec['width']
+    yy, xx = np.mgrid[:height, :width].astype(np.float32)
+    frames = np.empty((count, height, width, 3), dtype=np.uint8)
+    frames[0] = original
+    cx, cy, focal = (width-1)/2, (height-1)/2, 240.
+    duration = (count-1)/spec['fps']
+    for index in range(1, count):
+        t = index/spec['fps']
+        progress = t/duration
+        # Soft take-up followed by a steady upward tilt, never a camera cut.
+        travel = progress - 0.09*(1-np.exp(-progress/0.09))
+        angle = 0.111*travel
+        scale = 1. + 0.026*travel
+        u = (xx-cx)/(focal*scale)
+        v = (yy-cy)/(focal*scale)
+        denominator = np.cos(angle) + v*np.sin(angle)
+        x = cx + focal*u/denominator + 0.9*travel
+        y = cy + focal*(v*np.cos(angle)-np.sin(angle))/denominator
+        # Backtrace each viewing ray through the liquid using midpoint steps.
+        steps = max(1, int(np.ceil(t/0.12)))
+        dt = t/steps
+        for step in range(steps):
+            time = t-step*dt
+            vx, vy = velocity(x, y, time)
+            mx, my = x-0.5*dt*vx, y-0.5*dt*vy
+            vx, vy = velocity(mx, my, time-0.5*dt)
+            x -= dt*vx
+            y -= dt*vy
+        frames[index] = sample_photo(original, x, y)
+    artifact_io.save(frames, ROOT)
+    return frames
+
+
+if __name__ == '__main__':
+    generate()
