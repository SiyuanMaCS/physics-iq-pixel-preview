+"""Advect a photographed marbled liquid under a leftward tracking camera.
+
+All texture and color come from the supplied first frame.  A smooth
+divergence-free velocity field bends and turns the fine pigment ribbons.
+Backward particle tracing avoids repeatedly resampling the color image.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+from scipy.ndimage import map_coordinates
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def velocity(x, y, t):
+    # Broad eddies, with their angular rates smoothly tapered by radius.
+    u = np.zeros_like(x)
+    v = np.zeros_like(y)
+    for cx, cy, radius, rate in [
+        (194.0, 12.0, 43.0, -0.085),
+        (211.0, 133.0, 49.0, 0.085),
+        (116.0, 64.0, 63.0, 0.048),
+        (27.0, 21.0, 43.0, -0.038),
+        (-36.0, 80.0, 61.0, 0.045),
+    ]:
+        dx, dy = x - cx, y - cy
+        strength = rate * np.exp(-(dx * dx + dy * dy) / (2 * radius * radius))
+        u -= dy * strength
+        v += dx * strength
+    # Analytic streamfunctions add modest, coherent ribbon stretching.
+    phase = 0.16 * t
+    for kx, ky, a, p in [
+        (0.035, 0.044, 21.0, 0.7),
+        (0.076, 0.062, 6.0, 2.4),
+        (0.128, 0.111, 1.6, 4.2),
+    ]:
+        ax = kx * x + p + phase
+        ay = ky * y - 0.6 * p - phase * 0.7
+        u += a * ky * np.sin(ax) * np.cos(ay)
+        v -= a * kx * np.cos(ax) * np.sin(ay)
+    return u, v
+
+
+def reflect(a, limit):
+    r = np.mod(a, 2.0 * limit)
+    return np.where(r <= limit, r, 2.0 * limit - r)
+
+
+def render():
+    source = np.asarray(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    h, w = source.shape[:2]
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    frames = [source.copy()]
+    sourcef = source.astype(np.float32)
+    for frame in range(1, 81):
+        t = frame / 16.0
+        # Left pan: stationary landmarks traverse the image to the right.
+        camera = 10.3 * t
+        x = xx - camera
+        y = yy.copy()
+        steps = max(2, int(np.ceil(t / 0.15)))
+        dt = t / steps
+        # Midpoint backward characteristics, using world-space velocities.
+        for step in range(steps):
+            s = t - step * dt
+            u, v = velocity(x, y, s)
+            um, vm = velocity(x - 0.5 * dt * u, y - 0.5 * dt * v, s - dt / 2)
+            x -= dt * um
+            y -= dt * vm
+        # The new margin extends the existing liquid, not a blank background.
+        # A gradually curved reflection preserves its pigment palette while
+        # avoiding a conspicuous straight mirrored motif at the moving edge.
+        outside = np.maximum(-x, 0.0)
+        y += 0.0045 * outside ** 2 + 4.0 * (1 - np.exp(-outside / 28.0)) * np.sin(y / 26.0)
+        sx = reflect(x, w - 1)
+        sy = reflect(y, h - 1)
+        coordinates = np.array([sy, sx])
+        out = np.stack([
+            map_coordinates(sourcef[:, :, c], coordinates, order=3, mode='mirror')
+            for c in range(3)
+        ], axis=-1)
+        frames.append(np.clip(np.rint(out), 0, 255).astype(np.uint8))
+    result = np.stack(frames)
+    assert np.array_equal(result[0], source)
+    artifact_io.save(result, ROOT)
+
+
+if __name__ == '__main__':
+    render()
