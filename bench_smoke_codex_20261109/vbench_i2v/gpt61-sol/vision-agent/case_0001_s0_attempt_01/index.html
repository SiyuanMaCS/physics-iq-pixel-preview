+"""Image-only liquid advection and a continuous leftward camera move."""
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+from scipy.ndimage import map_coordinates, spline_filter
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def velocity(x, y):
+    # Broad, very slow vortices carry the pigment, without boiling or noise.
+    vx = np.zeros_like(x)
+    vy = np.zeros_like(y)
+    for cx, cy, radius, omega in (
+        (193.0, 14.0, 43.0, -0.034),
+        (215.0, 133.0, 57.0, 0.025),
+        (99.0, 65.0, 66.0, -0.013),
+        (8.0, 84.0, 38.0, 0.020),
+        (-45.0, 28.0, 45.0, -0.017),
+    ):
+        dx, dy = x - cx, y - cy
+        spin = omega * np.exp(-(dx * dx + dy * dy) / (2 * radius * radius))
+        vx -= spin * dy
+        vy += spin * dx
+    vx += 0.20 * np.sin(y / 35.0 + x / 81.0)
+    vy += 0.13 * np.sin(x / 51.0 - y / 67.0)
+    # Keep the extremely shallow surface in frame at the upper/lower edges.
+    edge = np.clip(y * (143.0 - y) / 1250.0, 0, 1)
+    vy *= edge
+    return vx, vy
+
+
+def main():
+    image = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    height, width = image.shape[:2]
+    yy, xx = np.mgrid[:height, :width].astype(np.float64)
+    coefficients = [spline_filter(image[:, :, c].astype(np.float64), order=3,
+                                  mode="reflect") for c in range(3)]
+    frames = np.empty((81, height, width, 3), dtype=np.uint8)
+    frames[0] = image
+    for frame in range(1, 81):
+        t = frame / 16.0
+        u = frame / 80.0
+        # The view moves left, so every large pigment feature travels right.
+        # A brief, gentle acceleration avoids a hard camera start.
+        pan = 49.0 * (u - 0.035 * (1.0 - np.exp(-u / 0.035))) / 0.965
+        x, y = xx - pan, yy.copy()
+        dt = t / 16.0
+        # Backtrace a smooth flow; RK2 avoids numerical stretching of filaments.
+        for _ in range(16):
+            vx, vy = velocity(x, y)
+            mx, my = x - 0.5 * dt * vx, y - 0.5 * dt * vy
+            vx, vy = velocity(mx, my)
+            x, y = x - dt * vx, y - dt * vy
+
+        # Continue the marbled liquid past the left edge. Reflection is gently
+        # curved and sheared away from the join rather than a repeated tile.
+        outside = np.maximum(-x, 0.0)
+        sx = np.where(x < 0, outside * (1.0 + 0.14 * np.sin(y / 29.0))
+                      + 0.003 * outside**2, x)
+        sy = y + (1.0 - np.exp(-outside / 18.0)) * (
+            0.22 * outside + 5.5 * np.sin(y / 26.0))
+        for channel in range(3):
+            value = map_coordinates(coefficients[channel], [sy, sx],
+                                    order=3, mode="reflect", prefilter=False)
+            frames[frame, :, :, channel] = np.clip(np.rint(value), 0, 255).astype(np.uint8)
+
+    artifact_io.save(frames, ROOT)
+    assert np.array_equal(frames[0], image)
+
+
+if __name__ == "__main__":
+    main()
