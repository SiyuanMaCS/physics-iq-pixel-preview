+"""Animate the source's pigment using smooth, incompressible liquid transport.
+
+No synthesized textures: the paint is traced backwards through an analytic
+time-dependent flow and sampled from the original photograph on every frame.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from PIL import Image
+from scipy.ndimage import map_coordinates, spline_filter
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def odd_gaussian(q, center, radius, extent):
+    """Image vortices keep flow tangent to the photograph's outer edges."""
+    value = np.zeros_like(q)
+    derivative = np.zeros_like(q)
+    for c, sign in ((center, 1), (-center, -1),
+                    (2 * extent - center, -1), (2 * extent + center, 1),
+                    (center - 2 * extent, 1), (-center - 2 * extent, -1)):
+        d = q - c
+        g = np.exp(-0.5 * (d / radius) ** 2)
+        value += sign * g
+        derivative -= sign * d / radius ** 2 * g
+    return value, derivative
+
+
+def velocity(x, y, t, w, h):
+    # Streamfunction derivatives give divergence-free pigment transport.
+    px, py = np.pi * x / w, np.pi * y / h
+    a = 168.0 * (1.0 + 0.13 * np.sin(0.65 * t))
+    vx = a * np.pi / h * np.sin(px) * np.cos(py)
+    vy = -a * np.pi / w * np.cos(px) * np.sin(py)
+
+    # Separate, broad pigment rolls are gradually folded into the cyan current.
+    vortices = (
+        (199.0, 18.0, 31.0, 33.0, 245.0, 0.0),
+        (216.0, 119.0, 44.0, 32.0, -295.0, 1.6),
+        (121.0, 57.0, 54.0, 27.0, 170.0, 3.1),
+        (24.0, 89.0, 32.0, 36.0, -113.0, 4.5),
+        (79.0, 125.0, 31.0, 31.0, 83.0, 2.0),
+    )
+    for cx, cy, sx, sy, strength, phase in vortices:
+        cx += 2.6 * (np.sin(0.39 * t + phase) - np.sin(phase))
+        cy += 1.7 * (np.sin(0.47 * t + phase) - np.sin(phase))
+        gx, dx = odd_gaussian(x, cx, sx, w)
+        gy, dy = odd_gaussian(y, cy, sy, h)
+        strength *= 1 + 0.12 * np.sin(0.7 * t + phase)
+        vx += strength * gx * dy
+        vy -= strength * dx * gy
+
+    # Very low-amplitude evolving rolls break up perfectly rigid circulation.
+    for kx, ky, amp, phase in ((3, 2, 10.0, 0.3), (2, 3, 7.0, 2.1)):
+        strength = amp * np.sin(0.55 * t + phase)
+        vx += strength * ky * np.pi / h * np.sin(kx * px) * np.cos(ky * py)
+        vy -= strength * kx * np.pi / w * np.cos(kx * px) * np.sin(ky * py)
+    return vx, vy
+
+
+def generate():
+    spec = json.loads((ROOT / 'task_spec.json').read_text())
+    first = np.array(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    h, w = first.shape[:2]
+    n, fps = spec['frames'], spec['fps']
+    frames = np.empty((n, h, w, 3), np.uint8)
+    frames[0] = first
+
+    # 2x render grid for stable subpixel detail in the very fine marbling.
+    scale = 2
+    yy, xx = np.mgrid[:h * scale, :w * scale].astype(np.float32)
+    xx = (xx + 0.5) / scale - 0.5
+    yy = (yy + 0.5) / scale - 0.5
+    xx = np.clip(xx, 0, w - 1)
+    yy = np.clip(yy, 0, h - 1)
+    coefficients = [spline_filter(first[:, :, c].astype(np.float32), order=3)
+                    for c in range(3)]
+
+    for index in range(1, n):
+        t = index / fps
+        x, y = xx.copy(), yy.copy()
+        steps = max(1, int(np.ceil(t / 0.16)))
+        dt = -t / steps
+        # Explicit midpoint characteristics, run backwards in physical time.
+        for step in range(steps):
+            now = t + dt * step
+            vx, vy = velocity(x, y, now, w - 1, h - 1)
+            mx, my = x + dt * 0.5 * vx, y + dt * 0.5 * vy
+            vx, vy = velocity(mx, my, now + dt * 0.5, w - 1, h - 1)
+            x += dt * vx
+            y += dt * vy
+        coords = np.array([np.clip(y, 0, h - 1), np.clip(x, 0, w - 1)])
+        image = np.stack([map_coordinates(c, coords, order=3,
+                                         prefilter=False, mode='nearest')
+                          for c in coefficients], axis=-1)
+        image = Image.fromarray(np.clip(np.rint(image), 0, 255).astype(np.uint8))
+        frames[index] = np.asarray(image.resize((w, h), Image.Resampling.LANCZOS))
+        if index % 16 == 0:
+            print(f'Generated {index}/{n - 1}', flush=True)
+
+    assert np.array_equal(frames[0], first)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == '__main__':
+    generate()
