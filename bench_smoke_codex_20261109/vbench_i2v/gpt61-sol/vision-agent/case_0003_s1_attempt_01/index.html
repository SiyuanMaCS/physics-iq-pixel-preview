+"""Animate the supplied macro photograph with a smooth liquid flow and upward tilt."""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+
+from artifact_io import save
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def velocity(x, y, t):
+    # Solenoidal eddies: the colors travel together rather than dissolving.
+    u = np.zeros_like(x)
+    v = np.zeros_like(y)
+    vortices = [
+        (195., 4., 43., 0.064),
+        (214., 135., 57., -0.050),
+        (135., 5., 25., -0.043),
+        (115., 68., 69., 0.029),
+        (12., 76., 28., -0.034),
+    ]
+    for cx, cy, radius, omega in vortices:
+        dx, dy = x - cx, y - cy
+        strength = omega * np.exp(-(dx * dx + dy * dy) / (2 * radius * radius))
+        strength *= 1.0 + 0.09 * np.sin(0.7 * t + cx * 0.02)
+        u -= dy * strength
+        v += dx * strength
+    # A broad, slowly changing stream adds gentle stretching of the fine veins.
+    a = x / 62. + 0.17 * t
+    b = y / 49. - 0.12 * t
+    u += 0.85 * np.sin(a) * np.cos(b)
+    v -= 0.85 * 49. / 62. * np.cos(a) * np.sin(b)
+    return u, v
+
+
+def cubic_weights(f):
+    f2, f3 = f * f, f * f * f
+    return (-0.5*f + f2 - 0.5*f3,
+            1 - 2.5*f2 + 1.5*f3,
+            0.5*f + 2*f2 - 1.5*f3,
+            -0.5*f2 + 0.5*f3)
+
+
+def texture(image, x, y):
+    # Continue the cropped liquid at the margins. A curved reflection lets
+    # the camera reveal a little more liquid, without a frozen edge strip.
+    above = np.maximum(-y, 0)
+    x = x + 0.006 * above**2
+    y = np.abs(y)
+    y = np.where(y > 143, 286-y, y)
+    x = np.where(x < 0, -x, x)
+    x = np.where(x > 255, 510-x, x)
+    ix, iy = np.floor(x).astype(int), np.floor(y).astype(int)
+    wx, wy = cubic_weights(x-ix), cubic_weights(y-iy)
+    rgb = np.zeros((*x.shape, 3), dtype=np.float32)
+    for j in range(4):
+        yy = np.clip(iy+j-1, 0, 143)
+        for i in range(4):
+            xx = np.clip(ix+i-1, 0, 255)
+            rgb += image[yy, xx] * (wx[i]*wy[j])[..., None]
+    return np.uint8(np.clip(np.rint(rgb), 0, 255))
+
+
+def main():
+    initial = np.asarray(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    image = initial.astype(np.float32)
+    yy, xx = np.mgrid[:144, :256].astype(np.float32)
+    frames = [initial.copy()]
+    for index in range(1, 81):
+        t = index / 16.
+        q = index / 80.
+        # Tilt upward: the image travels downward, with a very small perspective
+        # magnification appropriate to a close-up shot.
+        shift = 29.0 * (0.12*q*q + 0.88*q)
+        zoom = 1 + 0.025*q
+        x = (xx-127.5) / zoom + 127.5 + 1.7*q
+        y = (yy-71.5) / zoom + 71.5 - shift
+        # Backtrace through the evolving flow, always sampling the original
+        # photograph so fine pigment filaments do not blur frame after frame.
+        steps = max(2, int(np.ceil(t / 0.10)))
+        dt = t / steps
+        for step in range(steps):
+            clock = t-step*dt
+            u, v = velocity(x, y, clock)
+            um, vm = velocity(x-0.5*dt*u, y-0.5*dt*v, clock-0.5*dt)
+            x -= dt*um
+            y -= dt*vm
+        frames.append(texture(image, x, y))
+    result = np.stack(frames)
+    assert np.array_equal(result[0], initial)
+    save(result, ROOT)
+
+
+if __name__ == '__main__':
+    main()
