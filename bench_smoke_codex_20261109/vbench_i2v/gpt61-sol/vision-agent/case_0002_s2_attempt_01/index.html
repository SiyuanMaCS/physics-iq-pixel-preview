+"""A slow macro camera move over a gently advecting marbled paint surface.
+
+The only texture is the supplied photograph.  A curved reflected continuation
+of its right edge provides the small area revealed by the horizontal pan.
+"""
+from pathlib import Path
+import json
+
+import numpy as np
+from PIL import Image
+from scipy.ndimage import map_coordinates, spline_filter
+
+import artifact_io
+
+
+def continuation(image, extra=112, margin=20):
+    h, w = image.shape[:2]
+    yy, xx = np.mgrid[-margin:h+margin, -margin:w+extra].astype(float)
+    distance = np.maximum(xx - (w - 1), 0.0)
+
+    # Continue the existing colour bands around broad, asymmetric curves,
+    # rather than tile the photograph or expose a frozen edge strip.
+    bend = 0.16 * distance * np.sin(yy / 39.0 + 0.7)
+    bend += 0.0024 * distance**2 * np.sin(yy / 55.0 - 0.5)
+    sy = yy + bend
+    sx = xx.copy()
+    sx -= 2 * distance
+    sx += distance * (0.11 * np.sin(yy / 37.0) + 0.08)
+    sx += 0.001 * distance**2
+    scene = np.stack([
+        map_coordinates(image[:, :, c].astype(float), [sy, sx],
+                        order=3, mode='reflect')
+        for c in range(3)
+    ], axis=-1)
+    # Keep all supplied pixels exactly intact in the source plane.
+    scene[margin:margin+h, margin:margin+w] = image
+    return np.clip(scene, 0, 255), margin
+
+
+def flow(x, y, elapsed):
+    """Small, steady rotational displacements about the existing paint curls."""
+    dx = np.zeros_like(x)
+    dy = np.zeros_like(y)
+    vortices = [
+        (202, 10, 43, 33, 0.022),
+        (206, 129, 63, 32, -0.027),
+        (122, 66, 78, 26, 0.012),
+        (252, 83, 27, 20, -0.023),
+        (31, 23, 46, 36, -0.009),
+    ]
+    for cx, cy, rx, ry, rate in vortices:
+        px, py = x-cx, y-cy
+        falloff = np.exp(-0.5 * ((px/rx)**2 + (py/ry)**2))
+        angle = rate * elapsed * falloff
+        ca, sa = np.cos(angle), np.sin(angle)
+        dx += px*(ca-1) - py*sa
+        dy += px*sa + py*(ca-1)
+
+    # Broad shear gives the thin veins a slow viscous drift, without adding
+    # invented grain, changing their colours, or pulsing their brightness.
+    dx += 0.21 * elapsed * np.sin(y/23.0 + x/79.0)
+    dy += 0.11 * elapsed * np.sin(x/47.0 - y/58.0)
+    return x+dx, y+dy
+
+
+def main():
+    root = Path(__file__).resolve().parent
+    spec = json.loads((root/'task_spec.json').read_text())
+    image = np.asarray(Image.open(root/'first_frame.png').convert('RGB'))
+    n, h, w = spec['frames'], spec['height'], spec['width']
+    fps = spec['fps']
+    scene, margin = continuation(image)
+    coefficients = [spline_filter(scene[:, :, c], order=3, mode='mirror')
+                    for c in range(3)]
+    yy, xx = np.mgrid[:h, :w].astype(float)
+    frames = np.empty((n, h, w, 3), np.uint8)
+    frames[0] = image
+    for i in range(1, n):
+        t = i / fps
+        # A brief soft start, followed by a constant-rate rightward camera pan.
+        travel = 12.3 * (t - 0.20*(1-np.exp(-t/0.20)))
+        sx, sy = flow(xx + travel, yy, t)
+        sampled = np.stack([
+            map_coordinates(c, [sy+margin, sx+margin], order=3,
+                            mode='mirror', prefilter=False)
+            for c in coefficients
+        ], axis=-1)
+        frames[i] = np.clip(np.rint(sampled), 0, 255).astype(np.uint8)
+    artifact_io.save(frames, root)
+
+
+if __name__ == '__main__':
+    main()
