+"""Animate the supplied pigment photograph with continuous liquid advection.
+
+The texture is sampled from the original only (never from a previous RGB
+frame). An analytic, incompressible streamfunction supplies a smooth flow;
+backward characteristic integration preserves its delicate marbling.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from PIL import Image
+from scipy.ndimage import map_coordinates
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+def velocity(x, y):
+    # The streamfunction goes to zero at the image boundaries: no invented
+    # border imagery, and no hard clipped samples as the fluid circulates.
+    W, H = 256., 144.
+    X, Y = x + .5, y + .5
+    lx, ly = 19., 16.
+    ex0, ex1 = np.exp(-X/lx), np.exp(-(W-X)/lx)
+    ey0, ey1 = np.exp(-Y/ly), np.exp(-(H-Y)/ly)
+    wx, wy = (1-ex0)*(1-ex1), (1-ey0)*(1-ey1)
+    dxw = (ex0-ex1)/lx
+    dyw = (ey0-ey1)/ly
+    psi = np.zeros_like(x)
+    px = np.zeros_like(x)
+    py = np.zeros_like(x)
+    # Broad blue currents and smaller pigment eddies. The orange islands
+    # turn in opposite directions, slowly pulling the bands into filaments.
+    eddies = [
+        (199., 16., 29., 32., 158.),
+        (207., 125., 44., 29., -180.),
+        (133., 68., 66., 40., 210.),
+        (49., 29., 32., 26., -57.),
+        (123., 11., 25., 22., 43.),
+        (68., 90., 29., 25., -55.),
+        (239., 79., 19., 22., 25.),
+        (12., 67., 20., 27., 26.),
+        (152., 102., 21., 18., 21.),
+    ]
+    for cx,cy,sx,sy,a in eddies:
+        dx,dy=x-cx,y-cy
+        p=a*np.exp(-.5*((dx/sx)**2+(dy/sy)**2))
+        psi += p
+        px -= p*dx/(sx*sx)
+        py -= p*dy/(sy*sy)
+    # Very gentle distributed strain prevents the interstitial liquid from
+    # behaving like a single rigid vortex. All terms share a streamfunction.
+    for a,kx,ky,p in [(6.0,.055,.043,.4),(3.5,.099,-.076,1.8),
+                      (2.3,.141,.110,2.5),(1.0,.211,-.151,.9)]:
+        phase=kx*x+ky*y+p
+        psi += a*np.sin(phase)
+        c=a*np.cos(phase)
+        px += c*kx
+        py += c*ky
+    u = wx*(wy*py + dyw*psi)
+    v = -wy*(wx*px + dxw*psi)
+    return u, v
+
+
+def generate():
+    spec=json.loads((ROOT/'task_spec.json').read_text())
+    im=np.asarray(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    n,h,w=spec['frames'],spec['height'],spec['width']
+    yy,xx=np.mgrid[0:h,0:w].astype(np.float32)
+    bx,by=xx.copy(),yy.copy()
+    frames=np.empty((n,h,w,3),np.uint8)
+    frames[0]=im
+    original=im.astype(np.float32)
+    # Each characteristic is advanced backwards through a stationary
+    # divergence-free velocity. Four RK4 steps per frame give smooth subpixel
+    # motion, without cumulative resampling blur or camera movement.
+    clock=0.
+    for f in range(1,n):
+        t=f/spec['fps']
+        # A slight deceleration is characteristic of viscous pouring liquid.
+        next_clock=0.93*t+0.15*(1-np.exp(-t/.9))
+        dt=-(next_clock-clock)/4
+        for _ in range(4):
+            k1x,k1y=velocity(bx,by)
+            k2x,k2y=velocity(bx+.5*dt*k1x,by+.5*dt*k1y)
+            k3x,k3y=velocity(bx+.5*dt*k2x,by+.5*dt*k2y)
+            k4x,k4y=velocity(bx+dt*k3x,by+dt*k3y)
+            bx+=dt*(k1x+2*k2x+2*k3x+k4x)/6
+            by+=dt*(k1y+2*k2y+2*k3y+k4y)/6
+        clock=next_clock
+        coords=np.array([by,bx])
+        for c in range(3):
+            values=map_coordinates(original[:,:,c],coords,order=3,mode='nearest',prefilter=True)
+            frames[f,:,:,c]=np.clip(np.rint(values),0,255).astype(np.uint8)
+    artifact_io.save(frames,ROOT)
+    assert np.array_equal(frames[0],im)
+    print('Saved',frames.shape,frames.dtype)
+
+if __name__=='__main__':
+    generate()
