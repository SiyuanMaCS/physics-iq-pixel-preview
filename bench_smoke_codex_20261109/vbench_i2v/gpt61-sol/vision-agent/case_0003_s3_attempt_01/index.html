+"""Generate an upward-tilting macro view of slowly advecting marbled liquid.
+
+The only texture is the supplied first frame.  Smooth analytic fluid vortices
+are integrated backwards, and a pinhole camera supplies the upward tilt.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+from artifact_io import save
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def velocity(x, y, t):
+    # Broad circulation carries the thin paint filaments along with the pools.
+    vx = np.zeros_like(x)
+    vy = np.zeros_like(y)
+    vortices = [
+        (198.0, 9.0, 46.0, 0.074),
+        (218.0, 139.0, 57.0, -0.063),
+        (49.0, 32.0, 48.0, -0.031),
+        (136.0, 73.0, 61.0, 0.028),
+        (17.0, 98.0, 34.0, 0.034),
+    ]
+    for cx, cy, radius, angular_speed in vortices:
+        dx, dy = x - cx, y - cy
+        envelope = np.exp(-(dx * dx + dy * dy) / (2.0 * radius * radius))
+        omega = angular_speed * (0.93 + 0.07 * np.cos(0.4 * t))
+        vx -= omega * dy * envelope
+        vy += omega * dx * envelope
+
+    # Very low-frequency incompressible waviness in the long blue ribbons.
+    phase = x / 67.0 + y / 54.0 + 0.16 * t
+    vx += 0.40 * np.sin(phase) + 0.18 * np.sin(y / 25.0 - 0.2 * t)
+    vy -= (54.0 / 67.0) * 0.40 * np.sin(phase)
+    # Don't pull conspicuous repeated features in from the left/right edges.
+    distance = np.maximum(0.0, np.minimum(x, 255.0 - x))
+    vx *= 1.0 - np.exp(-(distance / 14.0) ** 2)
+    return vx, vy
+
+
+def reflect(z, n):
+    z = np.mod(z, 2 * (n - 1))
+    return np.where(z <= n - 1, z, 2 * (n - 1) - z)
+
+
+def cubic_weights(d):
+    d = np.abs(d)
+    return np.where(
+        d <= 1,
+        1.5 * d**3 - 2.5 * d**2 + 1,
+        np.where(d < 2, -0.5 * d**3 + 2.5 * d**2 - 4 * d + 2, 0),
+    )
+
+
+def texture_sample(texture, x, y):
+    # The little unseen overscan at the top follows the existing pigment bands.
+    # A progressively sheared reflected continuation avoids a mechanical tile.
+    above = np.maximum(-y, 0.0)
+    x = x + 0.047 * above**1.45 * np.sin(x / 48.0 + 0.6)
+    x, y = reflect(x, texture.shape[1]), reflect(y, texture.shape[0])
+    ix, iy = np.floor(x).astype(np.int32), np.floor(y).astype(np.int32)
+    result = np.zeros((*x.shape, 3), dtype=np.float64)
+    for j in (-1, 0, 1, 2):
+        yy = reflect(iy + j, texture.shape[0]).astype(np.int32)
+        wy = cubic_weights(y - (iy + j))
+        for i in (-1, 0, 1, 2):
+            xx = reflect(ix + i, texture.shape[1]).astype(np.int32)
+            weight = wy * cubic_weights(x - (ix + i))
+            result += texture[yy, xx] * weight[..., None]
+    return np.clip(np.rint(result), 0, 255).astype(np.uint8)
+
+
+def generate():
+    source = np.asarray(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    h, w = source.shape[:2]
+    yy, xx = np.mgrid[:h, :w].astype(np.float64)
+    frames = np.empty((81, h, w, 3), dtype=np.uint8)
+    frames[0] = source
+    focal_length = 370.0
+
+    for k in range(1, 81):
+        t = k / 16.0
+        u = k / 80.0
+        # Smooth, sustained upward camera tilt (about three degrees).
+        advance = u - 0.10 * np.sin(2 * np.pi * u) / (2 * np.pi)
+        angle = 0.056 * advance
+        zoom = 1.0 + 0.025 * advance
+        px = (xx - (w - 1) / 2) / zoom
+        py = (yy - (h - 1) / 2) / zoom
+        c, s = np.cos(angle), np.sin(angle)
+        denominator = c + s * py / focal_length
+        x = px / denominator + (w - 1) / 2
+        y = (c * py - s * focal_length) / denominator + (h - 1) / 2
+
+        # Integrate the inverse material trajectories using midpoint RK2.
+        steps = max(1, int(np.ceil(t / 0.22)))
+        dt = t / steps
+        for n in range(steps):
+            clock = t - n * dt
+            vx, vy = velocity(x, y, clock)
+            mx, my = x - 0.5 * dt * vx, y - 0.5 * dt * vy
+            vx, vy = velocity(mx, my, clock - 0.5 * dt)
+            x -= dt * vx
+            y -= dt * vy
+        frames[k] = texture_sample(source, x, y)
+    return frames
+
+
+if __name__ == '__main__':
+    video = generate()
+    assert np.array_equal(video[0], np.asarray(Image.open(ROOT / 'first_frame.png')))
+    save(video, ROOT)
