+"""A continuous macro-camera move across an advected, marbled liquid surface.
+
+The sole texture is the provided RGB frame.  Smooth analytic incompressible
+currents carry its fine pigment striations; the camera travels independently.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+from artifact_io import save
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def velocity(x, y, time):
+    # Large, gentle eddies, rather than independently moving image patches.
+    u = np.zeros_like(x)
+    v = np.zeros_like(y)
+    for cx, cy, radius, omega in (
+        (194.0, 13.0, 48.0, 0.034),
+        (126.0, 61.0, 76.0, -0.026),
+        (209.0, 136.0, 59.0, 0.033),
+        (-17.0, 68.0, 53.0, -0.026),
+    ):
+        dx, dy = x - cx, y - cy
+        strength = omega * np.exp(-(dx * dx + dy * dy) / (2 * radius * radius))
+        u -= dy * strength
+        v += dx * strength
+    # Streamfunction-derived broad undulation. It keeps neighboring bands
+    # together and avoids elastic-looking independent per-pixel oscillations.
+    kx, ky = 2 * np.pi / 220, 2 * np.pi / 172
+    px, py = kx * x + 0.13 * time, ky * y - 0.10 * time
+    u += 15 * ky * np.sin(px) * np.cos(py)
+    v -= 15 * kx * np.cos(px) * np.sin(py)
+    return u, v
+
+
+def backtrace(x, y, time):
+    # Midpoint integration of trajectories back to the original pigment.
+    steps = max(1, int(np.ceil(time / 0.22)))
+    dt = time / steps
+    for step in range(steps):
+        now = time - step * dt
+        u, v = velocity(x, y, now)
+        um, vm = velocity(x - 0.5 * dt * u, y - 0.5 * dt * v, now - dt * 0.5)
+        x = x - dt * um
+        y = y - dt * vm
+    return x, y
+
+
+def reflect_index(i, size):
+    a = np.mod(i, 2 * (size - 1))
+    return np.where(a < size, a, 2 * (size - 1) - a)
+
+
+def cubic_weights(t):
+    # Catmull--Rom reconstruction retains delicate marbling during subpixel pans.
+    return (
+        -0.5 * t + t * t - 0.5 * t ** 3,
+        1 - 2.5 * t * t + 1.5 * t ** 3,
+        0.5 * t + 2 * t * t - 1.5 * t ** 3,
+        -0.5 * t * t + 0.5 * t ** 3,
+    )
+
+
+def sample(image, x, y):
+    # Continue pigment beyond the photographed edge. A slowly increasing
+    # tangential offset makes the continuation less mechanically symmetrical.
+    outside = np.maximum(-x, 0)
+    y = y + 0.004 * outside ** 2 + 1.8 * np.sin(outside / 23) ** 2
+    ix, iy = np.floor(x).astype(np.int32), np.floor(y).astype(np.int32)
+    wx, wy = cubic_weights(x - ix), cubic_weights(y - iy)
+    result = np.zeros(x.shape + (3,), dtype=np.float32)
+    for j in range(4):
+        jj = reflect_index(iy + j - 1, image.shape[0])
+        for i in range(4):
+            ii = reflect_index(ix + i - 1, image.shape[1])
+            result += image[jj, ii] * (wy[j] * wx[i])[..., None]
+    return np.clip(np.rint(result), 0, 255).astype(np.uint8)
+
+
+def main():
+    image = np.asarray(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    yy, xx = np.mgrid[:144, :256].astype(np.float32)
+    frames = np.empty((81, 144, 256, 3), dtype=np.uint8)
+    frames[0] = image
+    for frame in range(1, 81):
+        t = frame / 16.0
+        # Camera pans LEFT: pigment landmarks therefore travel RIGHT.
+        camera_x = 9.6 * t
+        camera_y = 0.28 * t + 0.15 * np.sin(0.65 * t)
+        x, y = backtrace(xx - camera_x, yy + camera_y, t)
+        frames[frame] = sample(image, x, y)
+    save(frames, ROOT)
+    assert np.array_equal(frames[0], image)
+
+
+if __name__ == '__main__':
+    main()
