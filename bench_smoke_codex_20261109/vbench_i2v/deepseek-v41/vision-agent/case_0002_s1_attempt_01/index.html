+"""Generate a rightward camera pan across the supplied marbled liquid frame."""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+WIDTH = 256
+HEIGHT = 144
+FRAMES = 81
+FPS = 16.0
+PAN_PIXELS = 50.0
+
+
+def _smoothstep(edge0: float, edge1: float, value: np.ndarray) -> np.ndarray:
+    t = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)
+
+
+def _extend_texture(source: np.ndarray, width: int) -> np.ndarray:
+    """Extend the source to the right while preserving its rightmost edge.
+
+    The continuation is a gently advected reflection. The deformation fades in
+    from zero at the seam, so the first frame needs no blending or correction.
+    """
+    yy, xx = np.mgrid[0:HEIGHT, 0:width].astype(np.float32)
+    distance = np.maximum(xx - (WIDTH - 1), 0.0)
+    ramp = _smoothstep(0.0, 38.0, distance)
+
+    # Two scales of undulation break up the visual symmetry of a plain mirror.
+    wave_x = (
+        4.8 * np.sin(yy * 0.115 + distance * 0.031)
+        + 2.2 * np.sin(yy * 0.043 - distance * 0.073 + 1.4)
+    ) * ramp
+    wave_y = (
+        3.2 * np.sin(xx * 0.041 + 0.7)
+        + 1.3 * np.sin(xx * 0.101 - yy * 0.027)
+    ) * ramp
+
+    # Reflect around the seam. The x displacement is directed back into the
+    # source, and the y displacement makes the continuation feel fluid rather
+    # than literally mirrored.
+    sample_x = (2.0 * (WIDTH - 1) - xx + wave_x).astype(np.float32)
+    sample_y = np.clip(yy + wave_y, 0.0, HEIGHT - 1.0).astype(np.float32)
+    panorama = cv2.remap(
+        source,
+        sample_x,
+        sample_y,
+        interpolation=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    panorama[:, :WIDTH] = source
+    return panorama
+
+
+def _sample_field(field: np.ndarray, shift: float, phase: float) -> np.ndarray:
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+
+    # The warp starts exactly at identity and grows smoothly over the first
+    # moments. A curl-like pair of coordinate waves keeps the marbling alive
+    # without moving any visible structure independently of the camera.
+    ramp = 0.75 * _smoothstep(0.0, 0.16, phase)
+    warp_x = ramp * (
+        1.30 * np.sin(yy * 0.089 + phase * 4.9)
+        + 0.65 * np.sin(yy * 0.193 - phase * 7.3 + 1.1)
+        + 0.35 * np.sin((xx + yy) * 0.047 + phase * 3.1)
+    )
+    warp_y = ramp * (
+        1.00 * np.sin(xx * 0.052 - phase * 5.7 + 0.4)
+        + 0.50 * np.sin(xx * 0.127 + yy * 0.021 + phase * 3.9)
+    )
+
+    sample_x = xx + shift + warp_x
+    sample_y = np.clip(yy + warp_y, 0.0, HEIGHT - 1.0)
+    frame = cv2.remap(
+        field,
+        sample_x.astype(np.float32),
+        sample_y.astype(np.float32),
+        interpolation=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+    # A tiny travelling sheen suggests a wet, reflective surface.
+    sheen = (
+        0.010 * np.sin(xx * 0.034 + yy * 0.021 + phase * 7.2)
+        + 0.006 * np.sin(xx * 0.071 - yy * 0.043 - phase * 4.6)
+    ) * _smoothstep(0.0, 0.2, phase)
+    frame = frame.astype(np.float32) + sheen[..., None] * 255.0
+    return np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    if source.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"Unexpected first frame shape: {source.shape}")
+
+    panorama = _extend_texture(source, WIDTH + int(PAN_PIXELS) + 16)
+    frames = np.empty((FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    for index in range(FRAMES):
+        phase = index / (FRAMES - 1)
+        shift = PAN_PIXELS * phase + 0.65 * np.sin(np.pi * phase)
+        frames[index] = _sample_field(panorama, shift, phase)
+
+    # The contract requires the source frame to remain untouched.
+    frames[0] = source
+    return frames
+
+
+def main() -> None:
+    frames = generate()
+    assert frames.shape == (FRAMES, HEIGHT, WIDTH, 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[0], np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB")))
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()

+def _bilinear_sample(image: np.ndarray, sample_x: np.ndarray, sample_y: np.ndarray) -> np.ndarray:
+    """Sample an RGB image at floating-point coordinates with bilinear weights."""
+    height, width = image.shape[:2]
+    x = np.clip(sample_x, 0.0, width - 1.0)
+    y = np.clip(sample_y, 0.0, height - 1.0)
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = np.minimum(x0 + 1, width - 1)
+    y1 = np.minimum(y0 + 1, height - 1)
+    fx = (x - x0)[..., None]
+    fy = (y - y0)[..., None]
+    pixels = image.astype(np.float32)

+    bottom = pixels[y1, x0] * (1.0 - fx) + pixels[y1, x1] * fx
+    return top * (1.0 - fy) + bottom * fy
