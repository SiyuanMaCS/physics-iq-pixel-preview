+"""Generate a short liquid-macro continuation from first_frame.png."""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N_FRAMES = 81
+FPS = 16.0
+WIDTH = 256
+HEIGHT = 144
+
+
+def _velocity(x, y, t):
+    """Smooth, mostly incompressible liquid velocity in normalized units."""
+    # Broad rotating regions make the marbled bands curl without breaking apart.
+    vortices = (
+        (0.58, 0.51, 0.25, 0.18, 0.020),
+        (0.24, 0.27, 0.22, 0.14, -0.013),
+        (0.80, 0.80, 0.18, 0.14, -0.010),
+        (0.15, 0.77, 0.16, 0.10, 0.007),
+    )
+    u = np.zeros_like(x)
+    v = np.zeros_like(y)
+    for cx, cy, sx, sy, strength in vortices:
+        dx = x - cx
+        dy = y - cy
+        falloff = np.exp(-0.5 * ((dx / sx) ** 2 + (dy / sy) ** 2))
+        pulse = 0.82 + 0.18 * np.cos(2.0 * np.pi * t / 5.0625 + cx * 4.0)
+        u += strength * pulse * (-dy / sy) * falloff
+        v += strength * pulse * (dx / sx) * falloff
+
+    # Long wavelength shear carries the fine ribbons around the larger eddies.
+    phase = 2.0 * np.pi * t / 7.5
+    wave = np.sin(2.0 * np.pi * (0.90 * y + 0.16 * x) + phase)
+    u += 0.0055 * wave * np.sin(np.pi * x) ** 2
+    v += 0.0035 * np.sin(
+        2.0 * np.pi * (0.72 * x - 0.12 * y) - 0.75 * phase
+    ) * np.sin(np.pi * y) ** 2
+
+    # Motion is strongest in the open center and fades smoothly at the frame.
+    envelope = np.sin(np.pi * x) ** 1.7 * np.sin(np.pi * y) ** 1.7
+    return u * envelope, v * envelope
+
+
+def _flow_map(t):
+    """Backward flow map from each output pixel to its source pixel."""
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float64)
+    x = xx / (WIDTH - 1)
+    y = yy / (HEIGHT - 1)
+
+    # Integrate the velocity backwards from t to 0. Sub-stepping keeps the
+    # curved flow map accurate while remaining inexpensive at this resolution.
+    sx = xx.copy()
+    sy = yy.copy()
+    steps = 8
+    dt = t / steps
+    for step in range(steps):
+        tau = t - (step + 0.5) * dt
+        px = sx / (WIDTH - 1)
+        py = sy / (HEIGHT - 1)
+        u, v = _velocity(px, py, tau)
+        sx -= u * (WIDTH - 1) * dt
+        sy -= v * (HEIGHT - 1) * dt
+
+    # A very small secondary displacement increases the sense of viscous drag.
+    px = sx / (WIDTH - 1)
+    py = sy / (HEIGHT - 1)
+    ribbon = np.sin(2.0 * np.pi * (1.7 * py + 0.35 * np.sin(2.0 * np.pi * px)))
+    drag = 0.55 * np.sin(np.pi * px) * np.sin(np.pi * py) * ribbon
+    sx += drag * np.sin(2.0 * np.pi * t / 5.0625)
+    sy -= 0.35 * drag * np.cos(2.0 * np.pi * t / 5.0625)
+    return sx, sy
+
+
+def _cubic_weights(fraction):
+    """Catmull-Rom cubic weights for four neighboring samples."""
+    f2 = fraction * fraction
+    f3 = f2 * fraction
+    return (
+        -0.5 * f3 + f2 - 0.5 * fraction,
+        1.5 * f3 - 2.5 * f2 + 1.0,
+        -1.5 * f3 + 2.0 * f2 + 0.5 * fraction,
+        0.5 * f3 - 0.5 * f2,
+    )
+
+
+def _sample_cubic(image, sx, sy):
+    """Warp an RGB image with separable Catmull-Rom interpolation."""
+    x0 = np.floor(sx).astype(np.int32)
+    y0 = np.floor(sy).astype(np.int32)
+    wx = _cubic_weights(sx - x0)
+    wy = _cubic_weights(sy - y0)
+    result = np.zeros((HEIGHT, WIDTH, 3), dtype=np.float64)
+
+    for j in range(4):
+        yy = np.clip(y0 + j - 1, 0, HEIGHT - 1)
+        row = np.zeros((HEIGHT, WIDTH, 3), dtype=np.float64)
+        for i in range(4):
+            xx = np.clip(x0 + i - 1, 0, WIDTH - 1)
+            row += image[yy, xx] * wx[i][..., None]
+        result += row * wy[j][..., None]
+    return result
+
+
+def generate():
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    if source.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"unexpected first-frame shape: {source.shape}")
+
+    image = source.astype(np.float64) / 255.0
+    frames = np.empty((N_FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = source
+
+    for index in range(1, N_FRAMES):
+        t = index / (N_FRAMES - 1) * 5.0625
+        sx, sy = _flow_map(t)
+        warped = _sample_cubic(image, sx, sy)
+
+        # Subtle, slowly travelling light reveals the glossy liquid surface.
+        yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+        nx = xx / (WIDTH - 1)
+        ny = yy / (HEIGHT - 1)
+        glint = (
+            np.sin(2.0 * np.pi * (0.75 * nx + 0.22 * ny) - 0.9 * t)
+            * np.sin(np.pi * nx)
+            * np.sin(np.pi * ny)
+        )
+        light = 1.0 + 0.018 * glint
+        warped *= light[..., None]
+
+        # Preserve the saturated orange/blue contrast after repeated sampling.
+        mean = warped.mean(axis=2, keepdims=True)
+        warped = mean + (warped - mean) * 1.012
+        frames[index] = np.clip(warped * 255.0 + 0.5, 0, 255).astype(np.uint8)
+
+    artifact_io.save(frames, CASE_DIR)
+    return frames
+
+
+if __name__ == "__main__":
+    generate()
