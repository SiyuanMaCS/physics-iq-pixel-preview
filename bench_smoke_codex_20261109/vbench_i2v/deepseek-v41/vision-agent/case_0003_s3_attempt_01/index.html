+"""Generate a short upward-tilting close-up of flowing marbled liquid."""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageFilter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+WIDTH = 256
+HEIGHT = 144
+FRAMES = 81
+FPS = 16.0
+
+
+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def bilinear_sample(image, x, y):
+    """Sample an HxWx3 float image at floating-point pixel coordinates."""
+    h, w = image.shape[:2]
+    x = np.clip(x, 0.0, w - 1.001)
+    y = np.clip(y, 0.0, h - 1.001)
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = x0 + 1
+    y1 = y0 + 1
+    fx = (x - x0)[..., None]
+    fy = (y - y0)[..., None]
+
+    top = image[y0, x0] * (1.0 - fx) + image[y0, x1] * fx
+    bottom = image[y1, x0] * (1.0 - fx) + image[y1, x1] * fx
+    return top * (1.0 - fy) + bottom * fy
+
+
+def prepare_canvas(source):
+    """Extend the abstract fluid texture so camera movement has no hard edges."""
+    image = Image.fromarray(np.asarray(source, dtype=np.uint8))
+    margin = 24
+    canvas = Image.new("RGB", (WIDTH + margin * 2, HEIGHT + margin * 2))
+    canvas.paste(image, (margin, margin))
+
+    # Mirrored edge strips preserve the painted bands at the moving frame edges.
+    left = image.crop((0, 0, margin, HEIGHT)).transpose(Image.Transpose.FLIP_LEFT_RIGHT)
+    right = image.crop((WIDTH - margin, 0, WIDTH, HEIGHT)).transpose(Image.Transpose.FLIP_LEFT_RIGHT)
+    canvas.paste(left, (0, margin))
+    canvas.paste(right, (WIDTH + margin, margin))
+
+    top = canvas.crop((0, margin, WIDTH + margin * 2, margin + margin)).transpose(
+        Image.Transpose.FLIP_TOP_BOTTOM
+    )
+    bottom = canvas.crop(
+        (0, HEIGHT, WIDTH + margin * 2, HEIGHT + margin)
+    ).transpose(Image.Transpose.FLIP_TOP_BOTTOM)
+    canvas.paste(top, (0, 0))
+    canvas.paste(bottom, (0, HEIGHT + margin))
+
+    # A soft treatment of the synthetic margin keeps it continuous with the shot.
+    blurred = canvas.filter(ImageFilter.GaussianBlur(2.2))
+    canvas = Image.blend(canvas, blurred, 0.20)
+    return np.asarray(canvas, dtype=np.float32)
+
+
+def fluid_offsets(x, y, phase):
+    """Smooth, non-rigid motion concentrated around a few paint swirls."""
+    dx = np.zeros_like(x, dtype=np.float32)
+    dy = np.zeros_like(y, dtype=np.float32)
+
+    swirls = (
+        # cx, cy, radius, strength, rate, phase
+        (55.0, 28.0, 27.0, 2.5, 1.00, 0.20),
+        (219.0, 25.0, 33.0, 3.1, 0.82, 1.40),
+        (116.0, 88.0, 42.0, 2.0, 0.67, 2.10),
+        (205.0, 120.0, 34.0, 3.4, 0.91, 0.75),
+        (18.0, 99.0, 30.0, 2.4, 0.76, 2.60),
+    )
+    for cx, cy, radius, strength, rate, offset in swirls:
+        ux = x - cx
+        uy = y - cy
+        radius_sq = ux * ux + uy * uy
+        weight = np.exp(-radius_sq / (2.0 * radius * radius))
+        angle = strength * weight * np.sin(phase * rate + offset)
+        dx += -uy * angle / radius
+        dy += ux * angle / radius
+
+    # Low-amplitude travelling ripples make the upper fluid band breathe.
+    wave = np.sin(phase * 0.58 + x * 0.050 + y * 0.026)
+    dx += 0.72 * wave
+    dy += 0.34 * np.cos(phase * 0.49 + x * 0.028 - y * 0.043)
+    return dx, dy
+
+
+def render_frame(source, canvas, frame_index):
+    if frame_index == 0:
+        return source.copy()
+
+    t = frame_index / (FRAMES - 1)
+    eased = smoothstep(t)
+    phase = t * np.pi * 2.0
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+    cx = (WIDTH - 1) * 0.5
+    cy = (HEIGHT - 1) * 0.5
+
+    # Looking upward moves the paint downward through the frame. The tiny zoom
+    # and vertical perspective keep the motion read as a camera tilt.
+    shift_y = 13.5 * eased
+    zoom = 1.0 + 0.016 * eased
+    px = (xx - cx) / zoom + cx + 1.4 * eased
+    py = (yy - shift_y - cy) / zoom + cy
+    perspective = (py - cy) * (0.009 * eased)
+    px += perspective
+
+    fluid_x, fluid_y = fluid_offsets(px, py, phase)
+    sample_x = px + fluid_x + 24.0
+    sample_y = py + fluid_y + 24.0
+    frame = bilinear_sample(canvas, sample_x, sample_y)
+
+    # A slow sheen crossing the glossy paint adds coherent macro lighting.
+    sheen_center = -45.0 + 335.0 * t
+    sheen = np.exp(-((xx - sheen_center + 0.32 * (yy - cy)) / 58.0) ** 2)
+    frame *= 1.0 + 0.030 * sheen[..., None]
+    frame += np.array([2.0, 1.0, -1.0], dtype=np.float32) * sheen[..., None]
+
+    # Gentle filmic contrast and slight blue-orange color separation.
+    frame = (frame - 127.5) * 1.006 + 127.5
+    frame[..., 0] += 0.6 * eased
+    frame[..., 2] -= 0.35 * eased
+    return np.clip(frame, 0.0, 255.0).astype(np.uint8)
+
+
+def main():
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    canvas = prepare_canvas(source)
+    frames = np.stack(
+        [render_frame(source, canvas, index) for index in range(FRAMES)],
+        axis=0,
+    )
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
