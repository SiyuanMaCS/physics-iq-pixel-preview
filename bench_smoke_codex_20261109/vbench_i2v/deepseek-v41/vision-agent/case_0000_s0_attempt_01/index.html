+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter, map_coordinates
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+WIDTH, HEIGHT, FRAME_COUNT, FPS = 256, 144, 81, 16.0
+
+
+def smoothstep(edge0, edge1, value):
+    value = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def make_flow(x, y, t):
+    """Return a smooth, mostly incompressible image displacement in pixels."""
+    # Broad pigment circulation, strongest around the center of the frame.
+    dx = 2.4 * t * np.sin(2.0 * np.pi * (0.72 * y / HEIGHT + 0.06))
+    dy = 1.7 * t * np.sin(2.0 * np.pi * (0.53 * x / WIDTH - 0.11 * y / HEIGHT + 0.22))
+
+    # A few slow, large-scale waves keep the field from reading as one rigid swirl.
+    dx += 1.35 * t * np.sin(
+        2.0 * np.pi * (0.82 * y / HEIGHT + 0.55 * x / WIDTH) + 0.6 * t + 0.8
+    )
+    dy += 1.05 * t * np.sin(
+        2.0 * np.pi * (0.61 * x / WIDTH - 0.35 * y / HEIGHT) - 0.8 * t + 2.1
+    )
+
+    # Localized vortices follow the prominent curls in the source image.
+    vortices = (
+        (52.0, 32.0, 30.0, 0.36, 0.25, 0.35),
+        (205.0, 24.0, 28.0, -0.40, -0.20, 0.70),
+        (25.0, 79.0, 31.0, -0.26, 0.25, 0.10),
+        (121.0, 96.0, 40.0, 0.20, 0.40, 0.55),
+        (214.0, 122.0, 39.0, -0.38, -0.30, 0.20),
+        (85.0, 35.0, 24.0, 0.22, 0.15, 0.85),
+    )
+    for cx, cy, sigma, strength, drift_x, drift_y in vortices:
+        cx += 2.2 * t * np.sin(0.65 * np.pi * t + drift_x * 5.0)
+        cy += 1.6 * t * np.cos(0.55 * np.pi * t + drift_y * 4.0)
+        rx = x - cx
+        ry = y - cy
+        falloff = np.exp(-(rx * rx + ry * ry) / (2.0 * sigma * sigma))
+        angle = strength * t
+        dx += -ry * angle * falloff
+        dy += rx * angle * falloff
+
+    # Very low-amplitude ripple adds the fine, viscous motion of a liquid surface.
+    dx += 0.34 * t * np.sin(2.0 * np.pi * (x / 31.0 + 0.7 * t) + 0.23 * y)
+    dy += 0.22 * t * np.sin(2.0 * np.pi * (y / 27.0 - 0.55 * t) + 0.17 * x)
+    return dx, dy
+
+
+def relight(image, x, y, t):
+    """Add a restrained moving sheen while keeping the original pigment colors."""
+    luminance = (
+        0.2126 * image[..., 0] + 0.7152 * image[..., 1] + 0.0722 * image[..., 2]
+    )
+    gy, gx = np.gradient(gaussian_filter(luminance, 0.75))
+    light_x = 0.52 * np.cos(2.0 * np.pi * (0.08 * t + 0.15))
+    light_y = 0.52 * np.sin(2.0 * np.pi * (0.08 * t + 0.15))
+    relief = 0.78 * gx * light_x + 0.78 * gy * light_y
+
+    highlight = np.exp(
+        -(
+            (x - (69.0 + 6.0 * t)) ** 2 / (2.0 * 43.0**2)
+            + (y - (52.0 - 2.0 * t)) ** 2 / (2.0 * 31.0**2)
+        )
+    )
+    sheen = (0.0035 + 0.0070 * t) * highlight
+    return image + t * (0.30 * relief + sheen)[..., None]
+
+
+def generate_frames(first_frame):
+    source = first_frame.astype(np.float32) / 255.0
+    # Reflect-pad so the advection can pull believable texture in at the frame edges.
+    padded = np.pad(source, ((24, 24), (24, 24), (0, 0)), mode="reflect")
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first_frame
+
+    for frame_index in range(1, FRAME_COUNT):
+        t = frame_index / (FRAME_COUNT - 1)
+        # Ease in during the first few frames, then keep a nearly constant flow speed.
+        motion_t = 0.28 * t + 0.72 * smoothstep(0.0, 0.10, t)
+        dx, dy = make_flow(xx, yy, motion_t)
+        sample_x = xx + dx + 24.0
+        sample_y = yy + dy + 24.0
+
+        warped = np.empty_like(source)
+        for channel in range(3):
+            warped[..., channel] = map_coordinates(
+                padded[..., channel],
+                (sample_y, sample_x),
+                order=3,
+                mode="reflect",
+                prefilter=True,
+            )
+        warped = np.clip(relight(warped, xx, yy, motion_t), 0.0, 1.0)
+        frames[frame_index] = np.rint(warped * 255.0).astype(np.uint8)
+
+    return frames
+
+
+def main():
+    first_frame = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    assert first_frame.shape == (HEIGHT, WIDTH, 3)
+    frames = generate_frames(first_frame)
+
+    artifact_io.save(frames, ROOT)
+
+    # A local MP4 is useful for quick visual review in addition to the required player.
+    writer = cv2.VideoWriter(
+        str(ROOT / "preview.mp4"),
+        cv2.VideoWriter_fourcc(*"mp4v"),
+        FPS,
+        (WIDTH, HEIGHT),
+    )
+    for frame in frames:
+        writer.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
+    writer.release()
+
+    print(f"Generated {len(frames)} frames at {WIDTH}x{HEIGHT}, {FPS:g} fps")
+    print(f"First frame exact: {np.array_equal(frames[0], first_frame)}")
+
+
+if __name__ == "__main__":
+    main()
