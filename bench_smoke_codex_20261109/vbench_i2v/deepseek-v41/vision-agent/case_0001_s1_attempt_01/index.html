+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+TAU = 2.0 * np.pi
+
+
+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def make_warm_mask(image):
+    image = image.astype(np.float32) / 255.0
+    red, green, blue = image[..., 0], image[..., 1], image[..., 2]
+    warmth = np.clip((red - 0.62 * blue) * 1.8 + (red - green) * 0.7, 0.0, 1.0)
+    return cv2.GaussianBlur(warmth, (0, 0), 5.0)
+
+
+def reflected_x(source_x, y_norm, time_value):
+    """Break up the symmetry of the reflected strip revealed by the pan."""
+    bend = (
+        4.5 * np.sin(TAU * (1.15 * y_norm + 0.08 * time_value) + 0.6)
+        + 2.2 * np.sin(TAU * (2.7 * y_norm - 0.12 * time_value) + 1.8)
+    )
+    reflected = np.abs(source_x)
+    blend = smoothstep(np.minimum(reflected / 24.0, 1.0))
+    return reflected + bend * blend
+
+
+def render_frame(image, warm_mask, time_value):
+    height, width = image.shape[:2]
+    grid_y, grid_x = np.mgrid[0:height, 0:width].astype(np.float32)
+    x_norm = grid_x / (width - 1.0)
+    y_norm = grid_y / (height - 1.0)
+
+    # A slow, eased camera pan to the left; scene content therefore moves right.
+    pan = 28.0 * smoothstep(time_value)
+    rise = smoothstep(np.minimum(time_value / 0.18, 1.0))
+
+    # Broad marbling waves, followed by localized liquid eddies.
+    wave_x = (
+        3.4 * np.sin(TAU * (1.05 * y_norm + 0.10 * x_norm + 0.10 * time_value) + 0.25)
+        + 1.9 * np.sin(TAU * (2.25 * y_norm - 0.16 * x_norm + 0.18 * time_value) + 1.35)
+        + 0.9 * np.sin(TAU * (4.20 * y_norm + 0.24 * x_norm - 0.14 * time_value) + 2.20)
+    )
+    wave_y = (
+        2.0 * np.sin(TAU * (0.78 * x_norm + 0.28 * y_norm + 0.08 * time_value) + 1.05)
+        + 1.2 * np.sin(TAU * (1.85 * x_norm - 0.22 * y_norm - 0.16 * time_value) + 2.55)
+    )
+
+    displacement_x = rise * wave_x
+    displacement_y = rise * wave_y
+
+    vortices = (
+        (0.19, 0.22, 0.17, 3.7, 0.20, 0.20),
+        (0.47, 0.13, 0.14, 2.6, 1.90, 0.18),
+        (0.66, 0.61, 0.16, 2.9, 3.20, 0.22),
+        (0.88, 0.80, 0.18, 3.3, 4.60, 0.17),
+    )
+    for center_x, center_y, sigma, amplitude, phase, rate in vortices:
+        offset_x = x_norm - center_x
+        offset_y = y_norm - center_y
+        gaussian = np.exp(-(offset_x * offset_x + offset_y * offset_y) / (2.0 * sigma * sigma))
+        strength = rise * amplitude * np.sin(TAU * rate * time_value + phase)
+        displacement_x += strength * (-offset_y / sigma) * gaussian
+        displacement_y += strength * (offset_x / sigma) * gaussian
+
+    # The warmer lower liquid responds a little more than the cool marbling.
+    warm_sampled = cv2.remap(
+        np.ascontiguousarray(warm_mask, dtype=np.float32),
+        np.ascontiguousarray(grid_x - pan, dtype=np.float32),
+        np.ascontiguousarray(grid_y, dtype=np.float32),
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    differential = rise * warm_sampled
+    displacement_x += differential * (
+        1.7 * np.sin(TAU * (0.28 * time_value) + 0.4)
+        + 0.7 * np.sin(TAU * (1.45 * y_norm + 0.32 * time_value) + 2.1)
+    )
+    displacement_y += differential * 1.2 * np.sin(
+        TAU * (0.72 * x_norm - 0.24 * time_value) + 1.1
+    )
+
+    source_x = grid_x - pan + displacement_x
+    source_y = grid_y + displacement_y
+
+    negative = source_x < 0.0
+    if np.any(negative):
+        source_x = np.where(
+            negative,
+            reflected_x(source_x, y_norm, time_value),
+            source_x,
+        )
+
+    rendered = cv2.remap(
+        image,
+        np.ascontiguousarray(source_x, dtype=np.float32),
+        np.ascontiguousarray(source_y, dtype=np.float32),
+        interpolation=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+    # Keep the fine marbling crisp after subpixel resampling.
+    soft = cv2.GaussianBlur(rendered, (0, 0), 0.65)
+    rendered = cv2.addWeighted(rendered, 1.10, soft, -0.10, 0.0)
+    return np.clip(rendered, 0.0, 255.0).astype(np.uint8)
+
+
+def generate():
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    height, width = first.shape[:2]
+    warm_mask = make_warm_mask(first)
+
+    frame_count = 81
+    frames = np.empty((frame_count, height, width, 3), dtype=np.uint8)
+    frames[0] = first
+    for index in range(1, frame_count):
+        time_value = index / (frame_count - 1.0)
+        frames[index] = render_frame(first, warm_mask, time_value)
+
+    artifact_io.save(frames, ROOT)
+    return frames
+
+
+if __name__ == "__main__":
+    generated = generate()
+    print(f"saved {generated.shape} RGB frames to {ROOT}")
