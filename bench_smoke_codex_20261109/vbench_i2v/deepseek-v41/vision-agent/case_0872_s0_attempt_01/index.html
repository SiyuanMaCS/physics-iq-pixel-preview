+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw
+from scipy.ndimage import gaussian_filter, map_coordinates
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+WIDTH = 256
+HEIGHT = 144
+FRAME_COUNT = 81
+FPS = 16
+
+
+def polygon_mask(points, blur=2.0):
+    mask = Image.new("L", (WIDTH, HEIGHT), 0)
+    ImageDraw.Draw(mask).polygon(points, fill=255)
+    return gaussian_filter(np.asarray(mask, dtype=np.float32) / 255.0, blur)
+
+
+def ellipse_mask(box, blur=2.0):
+    mask = Image.new("L", (WIDTH, HEIGHT), 0)
+    ImageDraw.Draw(mask).ellipse(box, fill=255)
+    return gaussian_filter(np.asarray(mask, dtype=np.float32) / 255.0, blur)
+
+
+def smoothstep(edge0, edge1, value):
+    x = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def sample(image, x, y):
+    channels = [
+        map_coordinates(image[..., c], (y, x), order=1, mode="reflect")
+        for c in range(3)
+    ]
+    return np.stack(channels, axis=-1)
+
+
+def make_masks():
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH].astype(np.float32)
+
+    # The broad silhouette is deliberately a little generous. It is used only
+    # to keep the water motion from bleeding through the subject.
+    body = polygon_mask(
+        [
+            (73, 49), (82, 37), (101, 31), (121, 36), (137, 51),
+            (164, 49), (189, 54), (207, 72), (216, 105), (213, 137),
+            (186, 143), (146, 139), (111, 129), (91, 115), (72, 93),
+            (69, 67),
+        ],
+        blur=1.2,
+    )
+    body = np.maximum(body, ellipse_mask((76, 35, 137, 113), 1.0))
+    body = np.maximum(body, ellipse_mask((126, 48, 216, 142), 1.2))
+    body = np.clip(body, 0.0, 1.0)
+
+    head = polygon_mask(
+        [(76, 55), (80, 39), (94, 32), (116, 34), (130, 46),
+         (134, 65), (129, 88), (116, 103), (94, 103), (80, 89)],
+        blur=2.2,
+    )
+    head = np.maximum(head, ellipse_mask((78, 36, 135, 105), 2.0))
+
+    chest = polygon_mask(
+        [(101, 73), (135, 50), (173, 50), (205, 72), (213, 126),
+         (192, 139), (144, 134), (111, 119), (96, 96)],
+        blur=4.0,
+    )
+    chest = np.minimum(chest, body)
+
+    fish = polygon_mask(
+        [(78, 72), (87, 68), (103, 72), (118, 79), (126, 88),
+         (118, 97), (103, 95), (89, 87), (81, 82)],
+        blur=1.0,
+    )
+    fish *= smoothstep(67.0, 74.0, yy)
+
+    # Keep the shoreline still; only the water and subject move.
+    water = smoothstep(24.0, 30.0, yy) * (1.0 - smoothstep(0.72, 0.98, body))
+    foreground = smoothstep(92.0, 112.0, yy) * water
+
+    return {
+        "body": body,
+        "head": head,
+        "chest": chest,
+        "fish": fish,
+        "water": water,
+        "foreground": foreground,
+        "xx": xx,
+        "yy": yy,
+    }

+
+def water_displacement(xx, yy, t):
+    """Small, coherent surface motion with more travel in the foreground."""
+    depth = smoothstep(25.0, 138.0, yy)
+    dx = (
+        (0.45 + 1.75 * depth) * np.sin(0.060 * xx + 0.155 * yy - 1.35 * t)
+        + 0.55 * np.sin(0.025 * xx - 0.22 * yy + 0.65 * t)
+    )
+    dy = (
+        (0.16 + 0.48 * depth) * np.sin(0.095 * xx - 0.105 * yy + 1.10 * t)
+        + 0.12 * np.sin(0.19 * xx + 0.9 * t)
+    )
+    return dx, dy
+
+
+def subject_displacement(xx, yy, t, masks):
+    body = masks["body"]
+    head = masks["head"]
+    chest = masks["chest"]
+    fish = masks["fish"]
+
+    # A shallow breath, centered below the chest so the upper silhouette barely
+    # moves while the ribcage and wet fur change shape.
+    breath = np.sin(2.0 * np.pi * 0.48 * t)
+    cx, cy = 161.0, 111.0
+    rx = (xx - cx) / 67.0
+    ry = (yy - cy) / 51.0
+    chest_weight = chest * (0.72 + 0.28 * smoothstep(63.0, 112.0, yy))
+    dx = chest_weight * breath * (0.46 * rx + 0.20 * np.sin(0.07 * xx + 0.8 * t))
+    dy = chest_weight * breath * (0.30 * ry - 0.34)
+
+    # The head makes a compact nod and a slight turn. The motion is anchored at
+    # the neck, not applied as a rigid translation to the whole animal.
+    nod = np.sin(2.0 * np.pi * 0.31 * t + 0.35)
+    turn = np.sin(2.0 * np.pi * 0.23 * t - 0.15)
+    hx = xx - 105.0
+    hy = yy - 68.0
+    angle = np.deg2rad(1.55 * turn)
+    # Rotate the sampled coordinates opposite to the apparent head motion.
+    head_dx = (-angle * hy) - 0.52 * turn - 0.12 * hx / 31.0
+    head_dy = (angle * hx) + 0.82 * nod
+    head_weight = head * (0.48 + 0.52 * smoothstep(34.0, 74.0, yy))
+    head_weight *= 1.0 - 0.45 * fish
+    dx += head_weight * head_dx
+    dy += head_weight * head_dy
+
+    # The fish is hinged near the mouth and lags the head by a fraction of a
+    # beat. A second, softer wave travels toward the tail.
+    swing = np.sin(2.0 * np.pi * 0.68 * t - 0.85)
+    fish_hx = xx - 88.0
+    fish_hy = yy - 78.0
+    fish_angle = np.deg2rad(4.8 * swing)
+    fish_weight = fish * (0.35 + 0.65 * smoothstep(75.0, 95.0, yy))
+    dx += fish_weight * ((-fish_angle * fish_hy) + 0.50 * swing)
+    dy += fish_weight * (
+        (fish_angle * fish_hx)
+        + 0.55 * np.sin(2.0 * np.pi * 0.51 * t + 0.6)
+    )
+
+    # Return a soft composite; body prevents head motion from leaking into the
+    # water while head/chest add articulation inside it.
+    inside = np.maximum(head_weight, chest_weight)
+    inside = np.maximum(inside, fish_weight)
+    return dx * body, dy * body, inside
+
+
+def moving_surface_light(base, water, foreground, t):
+    """Add restrained reflected-light bands without repainting the water."""
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH].astype(np.float32)
+    lum = base.astype(np.float32).mean(axis=2)
+    texture = smoothstep(52.0, 112.0, lum)
+
+    phase = 0.082 * xx - 0.245 * yy + 1.55 * t
+    band = np.maximum(0.0, np.sin(phase)) ** 8
+    band2 = np.maximum(0.0, np.sin(0.044 * xx + 0.33 * yy - 0.92 * t + 1.8)) ** 10
+    envelope = water * (0.30 + 0.70 * texture)
+    amount = envelope * (0.055 * band + 0.032 * band2)
+
+    cool = np.array([181.0, 202.0, 209.0], dtype=np.float32)
+    lit = base.astype(np.float32) * (1.0 - amount[..., None]) + cool * amount[..., None]
+
+    # A broken, moving line of disturbed water around the submerged body.
+    line_center = (
+        119.0
+        + 2.2 * np.sin(0.040 * xx + 1.20 * t)
+        + 1.2 * np.sin(0.115 * xx - 0.70 * t)
+    )
+    line = np.exp(-((yy - line_center) / 1.15) ** 2)
+    line *= smoothstep(43.0, 64.0, xx) * (1.0 - smoothstep(215.0, 237.0, xx))
+    line *= foreground * (0.32 + 0.68 * texture)
+    crest = 0.07 * line
+    lit = lit * (1.0 - crest[..., None]) + np.array([194.0, 211.0, 215.0]) * crest[..., None]
+    return np.clip(lit, 0.0, 255.0).astype(np.uint8)
+
+
+def render(base, masks, frame_index):
+    t = frame_index / FPS
+    xx = masks["xx"]
+    yy = masks["yy"]
+    body = masks["body"]
+
+    wdx, wdy = water_displacement(xx, yy, t)
+    sdx, sdy, inside = subject_displacement(xx, yy, t, masks)
+
+    # Fade between independently moving regions over a couple of pixels.
+    dx = wdx * (1.0 - body) + sdx
+    dy = wdy * (1.0 - body) + sdy
+    dx *= 1.0 - inside * 0.08
+    dy *= 1.0 - inside * 0.08
+
+    frame = sample(base, xx + dx, yy + dy)
+    frame = moving_surface_light(frame, masks["water"], masks["foreground"], t)
+
+    # Preserve the exact first frame. For later frames, a very small temporal
+    # easing avoids a one-frame jump at the start.
+    if frame_index == 0:
+        return base.copy()
+    return frame
+
+
+def main():
+    source = Image.open(ROOT / "first_frame.png").convert("RGB")
+    base = np.asarray(source, dtype=np.uint8)
+    assert base.shape == (HEIGHT, WIDTH, 3)
+
+    masks = make_masks()
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    for index in range(FRAME_COUNT):
+        frames[index] = render(base, masks, index)
+
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()
