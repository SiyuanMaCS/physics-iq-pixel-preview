+"""Generate a liquid-paint camera pan from the supplied first frame."""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+WIDTH = 256
+HEIGHT = 144
+FRAMES = 81
+FPS = 16.0
+PAD_X = 56
+PAD_Y = 8
+PAN_PIXELS = 42.0
+
+
+def build_scene(frame):
+    """Extend the painting sideways while preserving the first-frame crop."""
+    scene = cv2.copyMakeBorder(
+        frame, PAD_Y, PAD_Y, PAD_X, PAD_X, cv2.BORDER_REFLECT_101
+    )
+    height, width = scene.shape[:2]
+    yy, xx = np.mgrid[0:height, 0:width].astype(np.float32)
+
+    # Replace the reflected side panels with a vertically undulated continuation.
+    # It agrees exactly with the source at each seam and avoids a visible mirror
+    # axis when the camera reveals the newly exposed paint.
+    for left_side in (True, False):
+        if left_side:
+            distance = np.clip((PAD_X - xx) / PAD_X, 0.0, 1.0)
+            active = xx < PAD_X
+            source_x = distance * 82.0
+        else:
+            distance = np.clip((xx - (PAD_X + WIDTH - 1)) / PAD_X, 0.0, 1.0)
+            active = xx > PAD_X + WIDTH - 1
+            source_x = WIDTH - 1 - distance * 82.0
+
+        vertical_warp = (
+            distance**1.45 * 10.5 * np.sin(2.7 * distance + 1.8 * yy / HEIGHT)
+            + distance**2.0 * 3.2 * np.sin(2.0 * np.pi * yy / 39.0 + 0.7)
+        )
+        source_y = np.clip(yy - PAD_Y + vertical_warp, 0, HEIGHT - 1)
+        source_x = np.clip(source_x, 0, WIDTH - 1)
+        warped = cv2.remap(
+            frame,
+            source_x,
+            source_y,
+            interpolation=cv2.INTER_CUBIC,
+            borderMode=cv2.BORDER_REFLECT_101,
+        )
+
+        # Feather only the last few pixels at the seam.
+        if left_side:
+            seam_weight = np.clip((PAD_X - xx) / 5.0, 0.0, 1.0)
+        else:
+            seam_weight = np.clip(
+                (xx - (PAD_X + WIDTH - 1)) / 5.0, 0.0, 1.0
+            )
+        alpha = (active * seam_weight)[..., None]
+        scene = (
+            scene.astype(np.float32) * (1.0 - alpha)
+            + warped.astype(np.float32) * alpha
+        ).astype(np.uint8)
+
+    return scene
+
+
+def displacement_field(x, y, t):
+    """A smooth, zero-at-rest liquid deformation in output coordinates."""
+    u = np.zeros_like(x, dtype=np.float32)
+    v = np.zeros_like(y, dtype=np.float32)
+
+    # Broad traveling currents. Subtracting the t=0 phase preserves the exact
+    # first frame while giving the paint a gentle non-periodic-looking churn.
+    u += 0.72 * (
+        np.sin(0.041 * y + 1.45 * t)
+        - np.sin(0.041 * y)
+    )
+    u += 0.52 * (
+        np.sin(0.021 * x + 0.061 * y - 0.98 * t + 0.6)
+        - np.sin(0.021 * x + 0.061 * y + 0.6)
+    )
+    v += 0.50 * (
+        np.sin(0.047 * x - 1.18 * t + 1.2)
+        - np.sin(0.047 * x + 1.2)
+    )
+    v += 0.36 * (
+        np.sin(0.032 * y + 0.017 * x + 0.77 * t)
+        - np.sin(0.032 * y + 0.017 * x)
+    )
+
+    # Slow vortices make the bright pools curl rather than merely translate.
+    vortices = (
+        (214.0, 123.0, 51.0, 0.155, 0.5),
+        (209.0, 24.0, 43.0, -0.105, 0.8),
+        (25.0, 104.0, 37.0, 0.070, 1.2),
+    )
+    for cx, cy, sigma, angle_at_end, delay in vortices:
+        dx = x - cx
+        dy = y - cy
+        weight = np.exp(-(dx * dx + dy * dy) / (2.0 * sigma * sigma))
+        ramp = np.clip((t - delay * 0.12) / 4.4, 0.0, 1.0)
+        angle = angle_at_end * ramp
+        u += weight * ((np.cos(angle) - 1.0) * dx - np.sin(angle) * dy)
+        v += weight * (np.sin(angle) * dx + (np.cos(angle) - 1.0) * dy)
+
+    return u, v
+
+
+def render_frame(scene, frame_index, supersample=2):
+    t = frame_index / FPS
+    high_w = WIDTH * supersample
+    high_h = HEIGHT * supersample
+
+    high_scene = cv2.resize(
+        scene, None, fx=supersample, fy=supersample,
+        interpolation=cv2.INTER_CUBIC,
+    )
+    yy, xx = np.mgrid[0:high_h, 0:high_w].astype(np.float32)
+    x = xx / supersample
+    y = yy / supersample
+    du, dv = displacement_field(x, y, t)
+
+    # A leftward camera move shifts image content to the right on screen.
+    pan = PAN_PIXELS * (t / ((FRAMES - 1) / FPS))
+    map_x = (x + PAD_X - pan + du) * supersample
+    map_y = (y + PAD_Y + dv) * supersample
+    warped = cv2.remap(
+        high_scene,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    return cv2.resize(
+        warped, (WIDTH, HEIGHT), interpolation=cv2.INTER_AREA
+    )
+
+
+def main():
+    frame = cv2.cvtColor(
+        cv2.imread(str(ROOT / "first_frame.png"), cv2.IMREAD_COLOR),
+        cv2.COLOR_BGR2RGB,
+    )
+    if frame.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"Unexpected first-frame shape: {frame.shape}")
+
+    scene = build_scene(frame)
+    frames = np.empty((FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = frame
+    for index in range(1, FRAMES):
+        frames[index] = render_frame(scene, index)
+
+    artifact_io.save(frames, ROOT)
+    print(f"Saved {FRAMES} frames at {WIDTH}x{HEIGHT}, {FPS:g} fps")
+
+
+if __name__ == "__main__":
+    main()
