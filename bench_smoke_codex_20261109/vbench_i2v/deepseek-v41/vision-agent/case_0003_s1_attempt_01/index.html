+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+WIDTH = 256
+HEIGHT = 144
+FRAMES = 81
+FPS = 16.0
+TOP_EXTENSION = 28
+
+
+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def build_scene(image):
+    """Extend the marbling above the original frame without a visible seam."""
+    top = image[:TOP_EXTENSION][::-1].copy()
+    x = np.arange(WIDTH, dtype=np.float32)
+
+    # Break up the mirror symmetry while keeping every row's colors continuous.
+    for row in range(TOP_EXTENSION - 1):
+        distance = (TOP_EXTENSION - 1 - row) / TOP_EXTENSION
+        shift = distance * (
+            2.2 * np.sin(2.0 * np.pi * (row / 13.0 + x / 83.0))
+            + 0.8 * np.sin(2.0 * np.pi * (row / 31.0 - x / 37.0))
+        )
+        top[row] = np.stack(
+            [np.interp(x + shift, x, top[row, :, channel]) for channel in range(3)],
+            axis=-1,
+        )
+
+    return np.concatenate((top, image), axis=0).astype(np.float32)
+
+
+def liquid_displacement(x, y, seconds, motion):
+    """Return a smooth, mostly divergence-free inverse displacement field."""
+    xx = x
+    yy = y
+    dx = np.zeros((HEIGHT, WIDTH), dtype=np.float32)
+    dy = np.zeros((HEIGHT, WIDTH), dtype=np.float32)
+
+    # Curl of three broad waves produces slow rolling, not uniform texture drift.
+    waves = (
+        (6.8, 2.0 * np.pi / 79.0, 2.0 * np.pi / 46.0, 0.43, 0.10),
+        (4.5, 2.0 * np.pi / 37.0, 2.0 * np.pi / 73.0, -0.31, 1.70),
+        (2.7, 2.0 * np.pi / 131.0, 2.0 * np.pi / 61.0, 0.24, 4.00),
+    )
+    for amplitude, kx, ky, omega, phase in waves:
+        argument = kx * xx + ky * yy + omega * seconds + phase
+        cosine = np.cos(argument)
+        dx += amplitude * ky * cosine
+        dy -= amplitude * kx * cosine
+
+    # Coherent eddies follow the strongest visible swirls in the source frame.
+    eddies = (
+        (206.0, 25.0, 1.8, 54.0, 1.00),
+        (224.0, 122.0, -1.3, 66.0, -0.55),
+        (17.0, 91.0, 1.0, 48.0, 0.30),
+    )
+    for cx, cy, strength, radius, phase in eddies:
+        ox = xx - cx
+        oy = yy - cy
+        radius_sq = ox * ox + oy * oy
+        distance = np.sqrt(radius_sq) + 1e-5
+        falloff = np.exp(-radius_sq / (radius * radius))
+        pulse = strength * falloff * np.cos(0.36 * seconds + phase)
+        dx += -oy / distance * pulse
+        dy += ox / distance * pulse
+
+    scale = 2.05 * motion
+    return dx * scale, dy * scale
+
+
+def bilinear_sample(scene, sx, sy):
+    x0 = np.floor(sx).astype(np.int32)
+    y0 = np.floor(sy).astype(np.int32)
+    x1 = np.minimum(x0 + 1, WIDTH - 1)
+    y1 = np.minimum(y0 + 1, scene.shape[0] - 1)
+    x0 = np.clip(x0, 0, WIDTH - 1)
+    y0 = np.clip(y0, 0, scene.shape[0] - 1)
+    fx = (sx - np.floor(sx))[..., None]
+    fy = (sy - np.floor(sy))[..., None]
+
+    upper = scene[y0, x0] * (1.0 - fx) + scene[y0, x1] * fx
+    lower = scene[y1, x0] * (1.0 - fx) + scene[y1, x1] * fx
+    return upper * (1.0 - fy) + lower * fy
+
+
+def render_frames(image):
+    scene = build_scene(image)
+    x = np.arange(WIDTH, dtype=np.float32)
+    y = np.arange(HEIGHT, dtype=np.float32)
+    xx, yy = np.meshgrid(x, y)
+    frames = np.empty((FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = image.astype(np.uint8)
+
+    duration = (FRAMES - 1) / FPS
+    for index in range(1, FRAMES):
+        seconds = index / FPS
+        progress = index / (FRAMES - 1)
+        eased = smoothstep(progress)
+        motion = smoothstep(progress / 0.34)
+
+        # The image travels down while new liquid enters from the top: a tilt up.
+        tilt_y = 20.0 * eased
+        tilt_x = 0.35 * np.sin(np.pi * progress) * eased
+        flow_x, flow_y = liquid_displacement(xx, yy, seconds, motion)
+
+        sx = xx - tilt_x - flow_x
+        sy = TOP_EXTENSION + yy - tilt_y - flow_y
+        sx = np.clip(sx, 0.0, WIDTH - 1.0)
+        sy = np.clip(sy, 0.0, scene.shape[0] - 1.0)
+        warped = bilinear_sample(scene, sx, sy)
+        frames[index] = np.clip(np.rint(warped), 0, 255).astype(np.uint8)
+
+    return frames
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    image = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"))
+    assert image.shape == (HEIGHT, WIDTH, 3)
+    frames = render_frames(image)
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
