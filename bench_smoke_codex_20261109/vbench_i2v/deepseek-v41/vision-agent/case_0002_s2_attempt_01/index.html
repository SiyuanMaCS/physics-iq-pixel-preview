+"""Generate a short rightward camera pan across the supplied liquid close-up."""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter, map_coordinates
+
+from artifact_io import save
+
+
+CASE_DIR = Path(__file__).resolve().parent
+WIDTH = 256
+HEIGHT = 144
+FRAME_COUNT = 81
+FPS = 16.0
+MARGIN = 112
+PAN_PIXELS = 61.0
+
+
+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def extend_liquid(image):
+    """Reflect and gently shear the right edge into a stable pan margin."""
+    canvas_width = WIDTH + MARGIN
+    canvas = np.empty((HEIGHT, canvas_width, 3), dtype=np.float32)
+    canvas[:, :WIDTH] = image
+
+    extension_x = np.arange(MARGIN, dtype=np.float32)
+    source_x = WIDTH - 1.0 - extension_x
+
+    # Let the reflected edge unwind into a second, vertically shifted part of
+    # the same marbling. The shift starts at zero, preserving a clean seam.
+    drift = 18.0 * smoothstep(extension_x / 58.0)
+    source_y = np.arange(HEIGHT, dtype=np.float32)[:, None] + drift[None, :]
+    source_y = np.mod(source_y, HEIGHT - 1)
+
+    coords = np.stack(
+        np.broadcast_arrays(source_y, source_x[None, :]), axis=0
+    )
+    reflected = np.empty((HEIGHT, MARGIN, 3), dtype=np.float32)
+    for channel in range(3):
+        reflected[:, :, channel] = map_coordinates(
+            image[:, :, channel], coords, order=3, mode="mirror"
+        )
+
+    # A broad, nearly transparent recolor suppresses the visual repetition of
+    # a pure mirror while keeping local paint striations intact.
+    weight = 0.16 * smoothstep(extension_x / MARGIN)[None, :, None]
+    broad = gaussian_filter(image, sigma=(8.0, 14.0, 0.0))
+    broad = broad[:, -1:, :]
+    color_target = broad + np.array([5.0, 2.0, -2.0], dtype=np.float32)
+    reflected = reflected * (1.0 - weight) + color_target * weight
+
+    canvas[:, WIDTH:] = reflected
+    return canvas
+
+
+def fluid_offsets(x, y, time, ramp):
+    """A small divergence-free-looking warp for flowing marbled paint."""
+    phase = 2.0 * np.pi * time
+    dx = (
+        1.35 * np.sin(0.132 * y + 0.72 * phase)
+        + 0.85 * np.sin(0.061 * x - 0.048 * y - 0.54 * phase)
+    )
+    dy = (
+        1.10 * np.sin(0.103 * x + 0.51 * phase)
+        + 0.70 * np.cos(0.047 * x + 0.083 * y - 0.43 * phase)
+    )
+
+    # Two slow eddies make the existing curls feel viscous rather than rigid.
+    for cx, cy, radius, strength, direction in (
+        (205.0, 48.0, 43.0, 2.8, 1.0),
+        (212.0, 126.0, 38.0, 2.2, -1.0),
+        (82.0, 31.0, 34.0, 1.7, -1.0),
+    ):
+        ox = x - cx
+        oy = y - cy
+        envelope = np.exp(-(ox * ox + oy * oy) / (2.0 * radius * radius))
+        angle = 0.20 * time * direction + 0.09 * np.sin(phase)
+        rx = ox * np.cos(angle) - oy * np.sin(angle)
+        ry = ox * np.sin(angle) + oy * np.cos(angle)
+        dx += strength * envelope * (rx - ox) / radius
+        dy += strength * envelope * (ry - oy) / radius
+
+    return dx * ramp, dy * ramp
+
+
+def generate():
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    assert source.shape == (HEIGHT, WIDTH, 3)
+    image = source.astype(np.float32)
+    canvas = extend_liquid(image)
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = source
+
+    duration = (FRAME_COUNT - 1) / FPS
+    for index in range(1, FRAME_COUNT):
+        time = index / FPS
+        progress = index / (FRAME_COUNT - 1)
+
+        # Ease into the pan so the first few frames remain gentle.
+        pan = PAN_PIXELS * (0.22 * progress + 0.78 * smoothstep(progress))
+        ramp = smoothstep(index / 12.0)
+        dx, dy = fluid_offsets(xx, yy, time, ramp)
+
+        sample_x = np.clip(xx + pan + dx, 0.0, canvas.shape[1] - 1.0)
+        sample_y = np.clip(yy + dy, 0.0, HEIGHT - 1.0)
+        warped = np.empty((HEIGHT, WIDTH, 3), dtype=np.float32)
+        for channel in range(3):
+            warped[:, :, channel] = map_coordinates(
+                canvas[:, :, channel],
+                (sample_y, sample_x),
+                order=3,
+                mode="nearest",
+            )
+
+        # Very restrained moving sheen on the pale paint, plus a touch of
+        # grain that is fixed to the surface rather than the camera.
+        highlight = np.exp(
+            -((xx - (125.0 + 15.0 * np.sin(0.22 * time))) / 76.0) ** 2
+            -((yy - (31.0 + 4.0 * np.cos(0.31 * time))) / 23.0) ** 2
+        )
+        pale = np.clip((warped.mean(axis=2) - 130.0) / 70.0, 0.0, 1.0)
+        warped += (2.8 * highlight * pale * ramp)[:, :, None]
+
+        texture = gaussian_filter(source[:, :, 0], sigma=1.0)[:, :, None] - image[:, :, 0].mean()
+        texture_x = np.mod(xx.astype(np.int32) + int(round(pan)), WIDTH)
+        texture_y = yy.astype(np.int32)
+        grain = texture[texture_y, texture_x, 0] * 0.035 * ramp
+        warped += grain[:, :, None]
+
+        frames[index] = np.clip(np.rint(warped), 0.0, 255.0).astype(np.uint8)
+
+    save(frames, CASE_DIR)
+    return frames
+
+
+if __name__ == "__main__":
+    generate()
