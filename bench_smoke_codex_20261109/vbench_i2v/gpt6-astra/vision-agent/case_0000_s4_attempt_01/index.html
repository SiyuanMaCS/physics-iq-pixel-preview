+"""Advect the original pigments through a smooth incompressible liquid flow.
+
+All rendered detail comes from the supplied photograph.  Backwards integrated
+characteristics, rather than repeated image resampling, retain its fine pigment
+striations throughout the shot.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from PIL import Image
+from scipy.ndimage import map_coordinates, spline_filter
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def velocity(x, y):
+    # Streamfunction eddies on several scales: broad paint currents, the two
+    # orange pools, and smaller eddies along the blue/cream interfaces.
+    eddies = [
+        (202., 9., 28., 30., 185.),
+        (211., 129., 42., 25., -185.),
+        (124., 13., 29., 23., -86.),
+        (148., 73., 54., 30., 146.),
+        (35., 32., 35., 30., -103.),
+        (15., 89., 22., 28., 51.),
+        (70., 83., 19., 15., -23.),
+        (113., 86., 13., 12., 14.),
+        (172., 88., 18., 12., -18.),
+        (241., 81., 13., 14., 25.),
+        (72., 22., 13., 13., 12.),
+        (169., 39., 16., 12., -13.),
+    ]
+    p = np.zeros_like(x)
+    px = np.zeros_like(x)
+    py = np.zeros_like(x)
+    for cx, cy, sx, sy, strength in eddies:
+        dx, dy = x-cx, y-cy
+        g = strength*np.exp(-0.5*((dx/sx)**2+(dy/sy)**2))
+        p += g
+        px -= g*dx/(sx*sx)
+        py -= g*dy/(sy*sy)
+
+    # A gentle long wavelength current gives the central pale-blue ribbon a
+    # flowing trajectory rather than turning the entire image like a camera.
+    phase = (x-25.)/66.
+    envelope = np.exp(-0.5*((y-62.)/39.)**2)
+    wave = 24.*np.sin(phase)*envelope
+    p += wave
+    px += (24./66.)*np.cos(phase)*envelope
+    py -= wave*(y-62.)/(39.*39.)
+
+    # No material needs to be invented outside the reference.  This smooth
+    # streamfunction boundary lets pigment slide tangentially along the crop,
+    # with a zero normal flux, and is identically incompressible.
+    margin = 12.
+    lx, rx = np.exp(-(x+0.5)/margin), np.exp(-(255.5-x)/margin)
+    ly, ry = np.exp(-(y+0.5)/margin), np.exp(-(143.5-y)/margin)
+    bx, by = (1.-lx)*(1.-rx), (1.-ly)*(1.-ry)
+    dbx, dby = (lx-rx)/margin, (ly-ry)/margin
+    vx = bx*(by*py+dby*p)
+    vy = -by*(bx*px+dbx*p)
+    return 0.77*vx, 0.77*vy
+
+
+def main():
+    spec = json.loads((ROOT/'task_spec.json').read_text())
+    image = np.asarray(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    height, width = image.shape[:2]
+    frames = np.empty((spec['frames'], height, width, 3), dtype=np.uint8)
+    frames[0] = image
+    # A single cubic interpolant of the original image is used in every frame.
+    coeff = [spline_filter(image[..., c].astype(np.float64), order=3,
+                           mode='reflect') for c in range(3)]
+    y, x = np.mgrid[0:height, 0:width].astype(np.float64)
+    dt = 1./spec['fps']/4.
+    for frame in range(1, spec['frames']):
+        for _ in range(4):
+            vx, vy = velocity(x, y)
+            mx, my = x-0.5*dt*vx, y-0.5*dt*vy
+            vx, vy = velocity(mx, my)
+            x -= dt*vx
+            y -= dt*vy
+        rgb = np.stack([map_coordinates(c, [y, x], order=3,
+                                        mode='reflect', prefilter=False)
+                        for c in coeff], axis=-1)
+        frames[frame] = np.clip(np.rint(rgb), 0, 255).astype(np.uint8)
+    assert np.array_equal(frames[0], image)
+    artifact_io.save(frames, ROOT)
+    print(f'Saved {len(frames)} RGB frames at {width}x{height}.')
+
+
+if __name__ == '__main__':
+    main()
