+    # a few smaller eddies stirring the boundaries between colors.
+    vortices = [

+        (211., 130., 50., 27., -270.),
+        (129., 63., 64., 27., 165.),
+        (47., 16., 43., 28., -105.),
+        (18., 84., 24., 31., 90.),
+        (106., 94., 19., 14., -30.),
+        (166., 86., 21., 13., 27.),
+        (244., 76., 21., 19., 46.),
+    ]
+    psi = np.zeros_like(x)
+    dx = np.zeros_like(x)
+    dy = np.zeros_like(x)
+    for cx, cy, sx, sy, strength in vortices:
+        a = (x - cx) / sx
+        b = (y - cy) / sy
+        g = strength * np.exp(-.5 * (a*a + b*b))
+        psi += g
+        dx -= g * a / sx
+        dy -= g * b / sy
+    # These windows impose no normal flow at the picture edge but permit
+    # tangential motion. This avoids invented pixels or mirror seams.
+    ax = np.pi * (x + .5) / 256.
+    ay = np.pi * (y + .5) / 144.

+    dex, dey = np.pi / 256. * np.cos(ax), np.pi / 144. * np.cos(ay)
+    # A broad circulating current, plus the local vortices.
+    psi += 25.

+    v = -ey * (ex * dx + dex * psi)
+    return u, v
