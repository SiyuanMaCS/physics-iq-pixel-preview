          + .44*t)
    wy = DEPTH*(.48*(np.sin(.31*Y+.029*X-2.18*t)-np.sin(.31*Y+.029*X))
          + .23*(np.sin(.74*Y-.013*X-3.02*t)-np.sin(.74*Y-.013*X)))
    # Restrained concentric disturbances where the animal contacts the water.
    for cx,cy in [(87,130),(180,133)]:
        rx,ry = X-cx, (Y-cy)*3.0
        radius = np.sqrt(rx*rx+ry*ry)+.01
        amp = .28*np.exp(-radius/43)*smooth(0,.7,t)
        wave = amp*np.sin(radius*.39-t*5.1)
        wx += wave*rx/radius
        wy += wave*.40*ry/radius
    dx += WATER*wx
    dy += WATER*wy
    # A submerged reflection shifts less rigidly than the body itself.
    reflection = smooth(133,146,Y)*SOLID
    dx += reflection*.4*(np.sin(.7*Y-2.4*t)-np.sin(.7*Y))
    dy += reflection*.2*np.sin(2.8*t+.03*X)*smooth(0,.6,t)
    return np.stack([dx,dy],axis=-1).astype(np.float32)


def render(im, t):
    flow = forward(t)
    # A restrained photographic push-in, 1.3% across the shot.
    scale = 1+.013*smooth(0,5,t)
    qx = (X-128)/scale+128
    qy = (Y-84)/scale+84
    sx,sy = qx.copy(),qy.copy()
    for _ in range(5):
        f = linear(flow,sx,sy)
        sx,sy = qx-f[...,0],qy-f[...,1]
    rgb = cubic(im,sx,sy)
    water = linear(WATER,sx,sy)
    # Tiny glancing luminance changes, limited strictly to the river.
    glint = (np.sin(.71*sx+.83*sy-2.8*t)*np.sin(.047*sx-.23*sy-1.5*t)
             - np.sin(.71*sx+.83*sy)*np.sin(.047*sx-.23*sy))
    rgb += (water*DEPTH*1.25*glint)[...,None]*np.array([.88,.96,1.0])

    # A few small drops slipping off the catch; no particle layer at frame 0.
    for k,start in enumerate((.68,1.12,2.46,3.04,4.06,4.53)):
        age=t-start
        if 0 < age < .70:
            # Start in the image plane of the lower end of the fish.
            ox=121.5+.65*np.sin(k*2.3)
            oy=86.5
            of=linear(forward(start),np.array(ox),np.array(oy))
            cx=(ox+of[0]+(1.0+.25*np.sin(k))*age-128)*scale+128
            cy=(oy+of[1]+5*age+62*age*age-84)*scale+84
            envelope=smooth(0,.07,age)*(1-smooth(.52,.70,age))
            droplet=.44*envelope*np.exp(-((X-cx)/.37)**2-((Y-cy)/(.5+age*.60))**2)
            rgb=rgb*(1-droplet[...,None])+np.array([190,191,181])*droplet[...,None]
    return np.uint8(np.clip(np.rint(rgb),0,255))


def main():
    src=np.asarray(Image.open(ROOT/'first_frame.png').convert('RGB'))
    assert src.shape==(H,W,3)
    frames=np.empty((81,H,W,3),np.uint8)
    frames[0]=src
    image=src.astype(np.float32)
    for i in range(1,81):
        frames[i]=render(image,i/16)
    assert np.array_equal(frames[0],src)
    artifact_io.save(frames,ROOT)
    print('Saved 81 RGB frames, 256x144, 16 fps.')
    print('Mean adjacent-frame difference:',float(np.mean(np.abs(np.diff(frames.astype(np.float32),axis=0)))))

if __name__=='__main__':
    main()
