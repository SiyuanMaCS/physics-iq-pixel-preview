+"""Animate the supplied liquid photograph using continuous material flow.
+
+No additional imagery is used. A divergence-free velocity field transports the
+original RGB material, while several dispersive waves provide surface motion.
+The original is sampled afresh each frame, avoiding cumulative image diffusion.
+"""
+from pathlib import Path
+import numpy as np
+from PIL import Image
+from artifact_io import save
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def bilinear(a, x, y):
+    h, w = a.shape[:2]
+    x = np.clip(x, 0, w - 1.00001)
+    y = np.clip(y, 0, h - 1.00001)
+    ix, iy = x.astype(np.int32), y.astype(np.int32)
+    fx, fy = x - ix, y - iy
+    if a.ndim == 3:
+        fx, fy = fx[..., None], fy[..., None]
+    return ((a[iy, ix] * (1-fx) + a[iy, ix+1] * fx) * (1-fy)
+            + (a[iy+1, ix] * (1-fx) + a[iy+1, ix+1] * fx) * fy)
+
+
+def cubic_weights(f):
+    f2, f3 = f*f, f*f*f
+    return (-0.5*f + f2 - 0.5*f3,
+            1 - 2.5*f2 + 1.5*f3,
+            0.5*f + 2*f2 - 1.5*f3,
+            -0.5*f2 + 0.5*f3)
+
+
+def sample_rgb(a, x, y):
+    """Interpolating cubic reconstruction, with a bounded local color range."""
+    h, w = a.shape[:2]
+    x, y = np.clip(x, 0, w-1), np.clip(y, 0, h-1)
+    ix, iy = np.floor(x).astype(int), np.floor(y).astype(int)
+    wx, wy = cubic_weights(x-ix), cubic_weights(y-iy)
+    out = np.zeros((*x.shape, 3), dtype=np.float64)
+    for j in range(4):
+        yy = np.clip(iy+j-1, 0, h-1)
+        for i in range(4):
+            xx = np.clip(ix+i-1, 0, w-1)
+            out += a[yy, xx] * (wx[i]*wy[j])[..., None]
+    # Do not introduce ringing into the tiny reflective details.
+    corners = [a[np.minimum(iy+j, h-1), np.minimum(ix+i, w-1)]
+               for j in (0, 1) for i in (0, 1)]
+    lo, hi = np.minimum.reduce(corners), np.maximum.reduce(corners)
+    return np.minimum(np.maximum(out, lo), hi)
+
+
+def make_velocity(x, y, w, h):
+    # Streamfunction sine modes give smooth incompressible flow with no outward
+    # velocity at the crop boundary. Unequal modes avoid a symmetric whirlpool.
+    modes = [(1, 1, 151.), (2, 1, -104.), (1, 2, 66.),
+             (2, 2, 37.), (3, 1, 30.), (3, 2, -27.),
+             (2, 3, 20.), (4, 2, 11.), (3, 4, -8.),
+             (5, 3, 4.5), (4, 5, 3.)]
+    u, v = np.zeros_like(x), np.zeros_like(y)
+    for m, n, a in modes:
+        kx, ky = m*np.pi/(w-1), n*np.pi/(h-1)
+        u += a*ky*np.sin(kx*x)*np.cos(ky*y)
+        v -= a*kx*np.cos(kx*x)*np.sin(ky*y)
+    return np.stack((u, v), axis=-1)
+
+
+def generate():
+    source = np.array(Image.open(ROOT / 'first_frame.png').convert('RGB'))
+    h, w = source.shape[:2]
+    y, x = np.mgrid[:h, :w].astype(np.float64)
+    field = make_velocity(x, y, w, h)
+    qx, qy = x.copy(), y.copy()
+    frames = np.empty((81, h, w, 3), np.uint8)
+    frames[0] = source
+    src = source.astype(np.float64)
+
+    # The capillary waves have unequal wavelengths, orientations and periods.
+    # Their initial phase is subtracted so frame zero is precisely undistorted.
+    waves = [(92., 0.36, 3.8, 1.50, 0.4),
+             (57., -0.92, 2.7, 1.05, 2.1),
+             (35., 1.22, 1.9, 0.52, 4.4),
+             (21., -0.44, 1.45, 0.20, 1.3)]
+    edge_x = np.sin(np.pi*x/(w-1))**0.65
+    edge_y = np.sin(np.pi*y/(h-1))**0.65
+    dt = 1.0/64.0
+    metrics = []
+    for index in range(1, 81):
+        t = index/16.0
+        # Integrate the inverse flow in four midpoint steps per video frame.
+        # Only coordinates, not RGB values, are transported between frames.
+        for _ in range(4):
+            vel = bilinear(field, qx, qy)
+            mid = bilinear(field, qx-0.5*dt*vel[..., 0],
+                           qy-0.5*dt*vel[..., 1])
+            qx -= dt*mid[..., 0]
+            qy -= dt*mid[..., 1]
+            qx = np.clip(qx, 0, w-1)
+            qy = np.clip(qy, 0, h-1)
+
+        dx, dy, tilt = np.zeros_like(x), np.zeros_like(y), np.zeros_like(x)
+        for wavelength, angle, period, amplitude, phase in waves:
+            ca, sa = np.cos(angle), np.sin(angle)
+            k = 2*np.pi/wavelength
+            p0 = k*(ca*x+sa*y)+phase
+            p = p0-2*np.pi*t/period
+            wave = amplitude*(np.sin(p)-np.sin(p0))
+            dx += ca*wave
+            dy += sa*wave
+            tilt += amplitude*k*(np.cos(p)-np.cos(p0))
+        # Smooth surface swell on top of the smaller, faster capillary waves.
+        swell = 1.35*(np.sin(0.043*x+0.025*y-0.83*t)
+                      - np.sin(0.043*x+0.025*y))
+        dx += 0.42*swell
+        dy += swell
+        sx = qx + edge_x*dx
+        sy = qy + edge_y*dy
+        rgb = sample_rgb(src, sx, sy)
+
+        # Very small tilt-dependent reflection changes, stronger on the glossy
+        # light-blue ridges; the pigment colors and deep-blue pools are retained.
+        lightness = rgb.mean(axis=-1)/255.
+        reflective = np.clip((lightness-0.22)/0.55, 0., 1.)
+        exposure = 1.0 + 0.055*np.tanh(tilt)*reflective
+        rgb *= exposure[..., None]
+        frames[index] = np.clip(np.rint(rgb), 0, 255).astype(np.uint8)
+        metrics.append(float(np.abs(frames[index].astype(float)
+                                    -frames[index-1].astype(float)).mean()))
+
+    assert np.array_equal(frames[0], source)
+    assert frames.shape == (81, 144, 256, 3)
+    save(frames, ROOT)
+    print('Saved 81 RGB frames, 256x144, 16 fps.')
+    print('First frame exact:', np.array_equal(frames[0], source))
+    print('Mean adjacent RGB change:', round(float(np.mean(metrics)), 3))
+    print('Adjacent change range:', round(min(metrics), 3), round(max(metrics), 3))
+    print('Final RGB change:', round(float(np.abs(frames[-1].astype(float)
+                                                -source.astype(float)).mean()), 3))
+
+
+if __name__ == '__main__':
+    generate()
