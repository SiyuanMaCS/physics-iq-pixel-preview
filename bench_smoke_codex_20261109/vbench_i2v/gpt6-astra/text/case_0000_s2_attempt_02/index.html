+"""Texture-preserving, incompressible liquid motion from the supplied RGB frame.
+
+Only the supplied image is used as an appearance source.  The material-coordinate
+field is transported rather than the RGB image, so fine reflections are sampled
+from the original on every frame and never acquire recursive blur.
+"""
+from pathlib import Path
+import json
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def bilinear(a, x, y):
+    h, w = a.shape[:2]
+    x = np.clip(x, 0, w - 1.00001)
+    y = np.clip(y, 0, h - 1.00001)
+    ix = x.astype(np.int32)
+    iy = y.astype(np.int32)
+    fx = x - ix
+    fy = y - iy
+    if a.ndim == 3:
+        fx = fx[..., None]
+        fy = fy[..., None]
+    return ((a[iy, ix] * (1-fx) + a[iy, ix+1] * fx) * (1-fy)
+            + (a[iy+1, ix] * (1-fx) + a[iy+1, ix+1] * fx) * fy)
+
+
+def mirror(x, size):
+    x = np.mod(x, 2*(size-1))
+    return np.where(x <= size-1, x, 2*(size-1)-x)
+
+
+def cubic_weights(t):
+    return (-.5*t + t*t - .5*t*t*t,
+            1 - 2.5*t*t + 1.5*t*t*t,
+            .5*t + 2*t*t - 1.5*t*t*t,
+            -.5*t*t + .5*t*t*t)
+
+
+def sample_rgb(a, x, y):
+    """Interpolating cubic; retain the small, sharp photographed reflections."""
+    h, w, _ = a.shape
+    x, y = mirror(x, w), mirror(y, h)
+    ix, iy = np.floor(x).astype(int), np.floor(y).astype(int)
+    wx, wy = cubic_weights(x-ix), cubic_weights(y-iy)
+    result = np.zeros(x.shape + (3,), np.float32)
+    for j in range(4):
+        yy = mirror(iy+j-1, h).astype(int)
+        for i in range(4):
+            xx = mirror(ix+i-1, w).astype(int)
+            result += a[yy, xx] * (wy[j]*wx[i])[..., None]
+    return result
+
+
+def velocity(x, y, t, w, h):
+    """Analytical curl of a smooth streamfunction, in pixels/second.
+
+    A large, slow recirculation carries the blue and amber regions past one
+    another. Smaller counter-rotating cells stretch their edges. The sine
+    envelope enforces no through-flow at the frame edge without a frozen rim.
+    """
+    kx, ky = np.pi/(w-1), np.pi/(h-1)
+    sx, sy = np.sin(kx*x), np.sin(ky*y)
+    cx, cy = np.cos(kx*x), np.cos(ky*y)
+    amplitude = 405.0 * (1 + .10*np.sin(.69*t))
+    u = amplitude * sx * ky*cy
+    v = -amplitude * kx*cx * sy
+
+    eddies = (
+        (73, 48, 41, 31, -305., .51, .3),
+        (179, 109, 42, 33, 335., .63, 1.6),
+        (196, 28, 30, 25, -170., .72, 3.0),
+        (109, 95, 26, 23, 100., .81, 4.5),
+    )
+    for ex, ey, rx, ry, strength, frequency, phase in eddies:
+        ex += 3.5 * (np.sin(.34*t + phase)-np.sin(phase))
+        ey += 2.0 * (np.cos(.39*t + phase)-np.cos(phase))
+        dx, dy = x-ex, y-ey
+        g = strength * (1 + .17*np.sin(frequency*t+phase))
+        g = g * np.exp(-.5*((dx/rx)**2+(dy/ry)**2))
+        u += g*sx*(ky*cy-sy*dy/(ry*ry))
+        v -= g*sy*(kx*cx-sx*dx/(rx*rx))
+
+    # Long waves add changing strain without sources or sinks.
+    a = 25*np.sin(.91*t+.5)
+    u += a*np.sin(2*kx*x)*2*ky*np.cos(2*ky*y)
+    v -= a*2*kx*np.cos(2*kx*x)*np.sin(2*ky*y)
+    return u, v
+
+
+def surface(x, y, t):
+    """Low-amplitude capillary refraction, distinct from bulk material flow."""
+    fade = 1-np.exp(-(t/.72)**2)
+    p = .071*x + .058*y - 1.64*t + .38*np.sin(.026*x-.038*y)
+    q = -.046*x + .109*y + 1.91*t + 1.7
+    r = .169*x + .131*y - 3.17*t + .32*np.sin(.6*t)
+    s = .281*x - .174*y + 3.83*t + .7
+    dx = fade*(1.60*np.sin(p) - .86*np.sin(q) + .29*np.sin(r) + .12*np.sin(s))
+    dy = fade*(1.02*np.sin(p) + 1.13*np.sin(q) + .23*np.sin(r) - .09*np.sin(s))
+    shine = fade*(.70*np.cos(p)+.56*np.cos(q)+.22*np.cos(r))
+    return dx, dy, shine
+
+
+def generate():
+    spec = json.loads((ROOT/'task_spec.json').read_text())
+    source = np.array(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    h, w = source.shape[:2]
+    n, fps = spec['frames'], float(spec['fps'])
+    assert (w, h) == (spec['width'], spec['height'])
+    image = source.astype(np.float32)
+
+    # Half-pixel material-coordinate mesh. The mesh is advected, not the image.
+    gy, gx = np.mgrid[0:2*h-1, 0:2*w-1].astype(np.float32) / 2
+    material = np.stack((gx, gy), axis=-1)
+    py, px = np.mgrid[:h, :w].astype(np.float32)
+    frames = np.empty((n, h, w, 3), dtype=np.uint8)
+    frames[0] = source
+    dt = 1/(fps*3)
+    diagnostics = []
+    for f in range(1, n):
+        for sub in range(3):
+            time = (f-1)/fps + sub*dt
+            # Midpoint characteristic tracing is stable and nearly reversible.
+            u0, v0 = velocity(gx, gy, time+dt, w, h)
+            xm, ym = gx-.5*dt*u0, gy-.5*dt*v0
+            um, vm = velocity(xm, ym, time+.5*dt, w, h)
+            bx, by = gx-dt*um, gy-dt*vm
+            material = bilinear(material, 2*bx, 2*by).astype(np.float32)
+
+        t = f/fps
+        dx, dy, shine = surface(px, py, t)
+        # Reflect only the subpixel wave overshoot, not the bulk colour pattern.
+        query_x = mirror(px+dx, w)
+        query_y = mirror(py+dy, h)
+        mapped = bilinear(material, 2*query_x, 2*query_y)
+        rgb = sample_rgb(image, mapped[..., 0], mapped[..., 1])
+
+        # Very restrained travelling changes in the existing reflected light.
+        # No new objects, random bubbles, or painted-on highlight contours.
+        lum = .2126*rgb[..., 0]+.7152*rgb[..., 1]+.0722*rgb[..., 2]
+        glint_weight = np.clip((lum-45)/165, 0, 1)
+        delta = shine * (1.1+2.1*glint_weight)
+        rgb += delta[..., None]*np.array([.91, .98, 1.0], np.float32)
+        frames[f] = np.clip(np.rint(rgb), 0, 255).astype(np.uint8)
+
+        if f in (1, 20, 40, 60, 80):
+            # Numerical inspection without opening or visually interpreting images.
+            xm, ym = material[::2, ::2, 0], material[::2, ::2, 1]
+            x_y, x_x = np.gradient(xm)
+            y_y, y_x = np.gradient(ym)
+            jacobian = x_x*y_y-x_y*y_x
+            diagnostics.append({
+                'frame': f,
+                'mean_step_rgb_change': float(np.abs(frames[f].astype(float)-frames[f-1]).mean()),
+                'mean_original_rgb_change': float(np.abs(frames[f].astype(float)-source).mean()),
+                'material_jacobian_quantiles': np.percentile(jacobian, [0, 1, 50, 99, 100]).tolist(),
+                'mean_material_displacement': float(np.mean(np.hypot(xm-px, ym-py))),
+            })
+
+    assert np.array_equal(frames[0], source)
+    (ROOT/'generation_checks.json').write_text(json.dumps(diagnostics, indent=2))
+    artifact_io.save(frames, ROOT)
+    print(json.dumps(diagnostics, indent=2))
+
+
+if __name__ == '__main__':
+    generate()
