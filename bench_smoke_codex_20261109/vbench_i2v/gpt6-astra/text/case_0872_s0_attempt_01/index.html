+ROOT = Path(__file__).resolve().parent
+H, W = 144, 256
+Y, X = np.mgrid[:H, :W].astype(np.float32)

+def smooth(a, b, v):
+    z = np.clip((v-a)/(b-a), 0, 1)
+    return z*z*(3-2*z)
+
+def sample(a, x, y):
+    return cv2.remap(a.astype(np.float32), x.astype(np.float32), y.astype(np.float32),
+                     cv2.INTER_LINEAR, borderMode=cv2.BORDER_REFLECT_101)

+def preparation(im):
+    # The fur is warm even in shadow, the river cool. This isolates only source
+    # pixels; the water between the feet remains part of the water layer.
+    warm = im[..., 0] - im[..., 2]
+    gray = im.mean(2)
+    rough = ((warm > 3) | (gray < 65)) & (Y >= 40) & (X > 70) & (X < 228)
+    labs, _ = label(rough)
+    counts = np.bincount(labs.ravel()); counts[0] = 0
+    core = labs == counts.argmax()
+    core = binary_fill_holes(core)
+    gap = (X>100) & (X<115) & (Y>120) & (Y<136) & (warm < 6) & (gray>115)
+    core[gap] = False
+    # Treat very bottom as the reflected/contact band, rather than rigid fur.
+    alpha = gaussian_filter(binary_dilation(core, iterations=1).astype(np.float32), .48)
+    support = alpha > .0001
+    # A river plate made exclusively by continuing same-row water texture.
+    # Only the narrow newly uncovered border of this plate will be visible.
+    plate = im.copy()
+    for y in range(H):
+        ids = np.flatnonzero(support[y])
+        if not len(ids): continue
+        # Distinguish disconnected ear tips and the exposed gap beneath the leg.
+        changes=np.flatnonzero(np.diff(ids)>1)+1
+        for group in np.split(ids, changes):
+            left, right = group[0], group[-1]
+            n=right-left+1
+            xx=np.arange(left,right+1)
+            lbound=max(0,left-3)
+            rbound=min(W-1,right+3)
+            # Reflection keeps the continuation locally plausible at both edges.
+            lc=np.clip(left-2-(xx-left)%max(6,min(44,left-2)),0,W-1)
+            rc=np.clip(right+2+(right-xx)%max(6,min(38,W-right-2)),0,W-1)
+            mix=smooth(.1,.9,(xx-left)/max(1,n-1))[:,None]
+            texture=im[y,lc]*(1-mix)+im[y,rc]*mix
+            plate[y,xx]=texture
+    # A light spatial filter applies only underneath the matte, never to the
+    # original visible river. This removes seams in the revealed river slivers.
+    blur=gaussian_filter(plate,(.65,1.2,0))
+    plate=plate*(1-alpha[...,None])+blur*alpha[...,None]
+    # Premultiplied color reconstruction makes the identity composite exact.
+    fg=im-plate*(1-alpha[...,None])
+    return plate, fg, alpha
+
+def pose(t):
+    u=t/5.
+    enter=smooth(0,.65,t)
+    stride=2*np.pi*t/1.85
+    dx=-6.2*smooth(0,5,t)+.55*np.sin(stride)*enter
+    dy=1.1*u + .72*np.sin(stride)*enter

+    # Two gentle looks/nods; the fish and muzzle stay attached to the skull.
+    hx=-1.8*np.sin(2*np.pi*t/4.8)*enter-.65*smooth(.4,3.8,t)
+    hy=(-2.7*smooth(0,1.5,t)+1.0*np.sin(stride+.3)*enter

+    return dx,dy,scale,hx,hy,theta,enter,stride
+
+def bear_coordinates(t):

+    qx=(X-139-dx)/scale+139

+        headrot_y=np.sin(theta)*(sx-116)+np.cos(theta)*(sy-88)+88-sy

+        wet=smooth(130,143,sy)
+        ux+=wet*.7*enter*np.sin(sy*.72-t*4.1)
+        sx=qx-ux
+        sy=qy-uy
+    return sx,sy

+    river=smooth(10,23,Y)

+def contact_waves(t, dx, dy):

+        phase=(rr-radius)*1.0
+        displace+=.55*np.sin(phase)*env
+        shine+=4*np.cos(phase-.6)*env
+    return displace,shine
+
+def droplets(frame, alpha, t, dx, dy):
+    # Tiny glints only, matching the size of the original shallow-water spray.
+    out=frame.copy()
+    for j,launch in enumerate([.25,1.18,2.10,3.03,3.95,4.88]):
+        age=t-launch
+        if age<0 or age>.52: continue
+        side=j%2
+        cx=(84 if side==0 else 129)+dx
+        cy=(132 if side==0 else 140)+dy
+        for k in range(5):
+            vx=(-1 if side==0 else 1)*(3.1+k*3.4)
+            vy=-13-k*3.5
+            px=cx+vx*age; py=cy+vy*age+39*age*age
+            amp=.32*np.sin(np.pi*age/.52)**.7
+            spot=np.exp(-((X-px)/(.31+.065*k))**2-((Y-py)/(.46+.07*k))**2)
+            a=(spot*amp*(1-.9*alpha))[...,None]
+            out=out*(1-a)+np.array([215,220,220])*a
+    return out
+
+def main():
+    first=np.array(Image.open(ROOT/'first_frame.png').convert('RGB'),dtype=np.uint8)
+    im=first.astype(np.float32)
+    plate,fg,alpha=preparation(im)
+    frames=np.empty((81,H,W,3),np.uint8)
+    frames[0]=first
+    for i in range(1,81):

+        subject=sample(fg,sx,sy)
+        frame=water*(1-a[...,None])+subject

+        frame=droplets(frame,a,t,dx,dy)
+        frames[i]=np.clip(np.rint(frame),0,255).astype(np.uint8)
+    np.save(ROOT/'.tmp'/'generated_rgb.npy',frames)
+    artifact_io.save(frames,ROOT)
+    print('Saved',frames.shape,frames.dtype,'exact first frame:',np.array_equal(frames[0],first))
+    d=np.abs(np.diff(frames.astype(np.float32),axis=0))
+    print('Frame difference mean:',float(d.mean()),'first:',float(d[0].mean()),'max step:',float(d.mean((1,2,3)).max()))

+if __name__=='__main__':
+    main()

    out=frame.copy()
    for j,launch in enumerate([.25,1.18,2.10,3.03,3.95,4.88]):
        age=t-launch
        if age<0 or age>.52: continue
        side=j%2
        cx=(84 if side==0 else 129)+dx
        cy=(132 if side==0 else 140)+dy
        for k in range(5):
            vx=(-1 if side==0 else 1)*(3.1+k*3.4)
            vy=-13-k*3.5
            px=cx+vx*age; py=cy+vy*age+39*age*age
            amp=.32*np.sin(np.pi*age/.52)**.7
            spot=np.exp(-((X-px)/(.31+.065*k))**2-((Y-py)/(.46+.07*k))**2)
            a=(spot*amp*(1-.9*alpha))[...,None]
            out=out*(1-a)+np.array([215,220,220])*a
    return out

        dx,dy,*_=pose(t)

        water+=shine[...,None]*np.array([.84,.94,1.0],np.float32)

        subject=sample(fg,sx,sy)
        frame=water*(1-a[...,None])+subject
        # Subpixel breathing through the shoulders, with no synthetic relighting.
        frame=droplets(frame,a,t,dx,dy)
        frames[i]=np.clip(np.rint(frame),0,255).astype(np.uint8)
    np.save(ROOT/'.tmp'/'generated_rgb.npy',frames)
    artifact_io.save(frames,ROOT)
    print('Saved',frames.shape,frames.dtype,'exact first frame:',np.array_equal(frames[0],first))
    d=np.abs(np.diff(frames.astype(np.float32),axis=0))
    print('Frame difference mean:',float(d.mean()),'first:',float(d[0].mean()),'max step:',float(d.mean((1,2,3)).max()))
