+"""Generate the requested 81-frame RGB image-to-video sequence.
+
+The shot is intentionally restrained: a smooth optical push-in is combined
+with a very small, coherent coordinate drift.  This keeps the photographed
+liquid texture intact while avoiding the rigid look of a plain crop/resize.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+WIDTH = 256
+HEIGHT = 144
+FRAME_COUNT = 81
+
+
+def smoothstep(t: float) -> float:
+    """Cubic ease with zero velocity at both ends."""
+    return t * t * (3.0 - 2.0 * t)
+
+
+def make_frames(first: np.ndarray) -> np.ndarray:
+    if first.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"unexpected first-frame shape: {first.shape}")
+
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first  # contract: exact, with no resampling of frame zero
+
+    # Output pixel grid.  All motion is expressed as an inverse map into the
+    # original, so every output frame is formed only from first-frame pixels.
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+
+    # Aim slightly right of image centre, near the blue/orange transition.
+    focus_x, focus_y = 137.0, 75.0
+
+    for index in range(1, FRAME_COUNT):
+        t = index / (FRAME_COUNT - 1)
+        e = smoothstep(t)
+
+        # A 1.62x push-in is clearly legible over the five-second shot but
+        # still retains both main colours and sufficient recognizable detail.
+        scale = 1.0 + 0.62 * e
+
+        # Tiny camera drift toward the colour interface gives the push-in a
+        # physical, operated feel.  The drift is in source-image pixels.
+        cx = focus_x + 2.5 * e
+        cy = focus_y + 1.0 * e
+        sx = cx + (xx - WIDTH * 0.5) / scale
+        sy = cy + (yy - HEIGHT * 0.5) / scale
+
+        # Sub-pixel, low-frequency fluid advection.  It is deliberately much
+        # smaller than the camera motion and fades in/out without a jump.
+        envelope = np.sin(np.pi * t) ** 2
+        phase = 2.0 * np.pi * (0.22 * t)
+        flow_x = 0.62 * envelope * np.sin(sy * 0.031 + phase)
+        flow_x += 0.20 * envelope * np.sin((sx + sy) * 0.019 - phase * 0.7)
+        flow_y = 0.40 * envelope * np.sin(sx * 0.027 - phase * 0.8)
+        flow_y += 0.16 * envelope * np.cos((sx - sy) * 0.023 + phase)
+
+        map_x = (sx + flow_x).astype(np.float32)
+        map_y = (sy + flow_y).astype(np.float32)
+        frames[index] = cv2.remap(
+            first,
+            map_x,
+            map_y,
+            interpolation=cv2.INTER_LANCZOS4,
+            borderMode=cv2.BORDER_REFLECT_101,
+        )
+
+    return frames
+
+
+def main() -> None:
+    case_dir = Path(__file__).resolve().parent
+    first = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"))
+    frames = make_frames(first)
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
