+"""Generate a short macro shot of flowing blue and orange liquid.
+
+The supplied still is treated as dye carried by a smooth, incompressible
+two-dimensional flow.  The program advects a coordinate map rather than the
+image itself, so repeated interpolation does not progressively soften the
+source artwork.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 256, 144
+N = 81
+
+
+def reflect_coordinates(a: np.ndarray, limit: int) -> np.ndarray:
+    """Reflect real-valued coordinates into [0, limit-1]."""
+    period = 2.0 * (limit - 1)
+    a = np.mod(a, period)
+    return np.where(a <= limit - 1, a, period - a).astype(np.float32)
+
+
+def velocity(x: np.ndarray, y: np.ndarray, t: float) -> tuple[np.ndarray, np.ndarray]:
+    """A slowly changing divergence-free, multi-scale liquid flow.
+
+    x and y are current pixel coordinates and t is measured in frames.  The
+    vortex strengths are deliberately modest: individual colored folds move
+    visibly while the shot remains a faithful continuation of frame zero.
+    """
+    vx = np.full_like(x, 0.045, dtype=np.float32)
+    vy = np.full_like(y, -0.012, dtype=np.float32)
+
+    # Broad eddies.  Moving their centers slightly keeps the flow from looking
+    # like a rigid image rotation.  Positive strength rotates clockwise in
+    # image coordinates.
+    vortices = (
+        (0.68, 0.47, 0.39, 1.05, 0.00),
+        (0.25, 0.62, 0.30, -0.72, 1.85),
+        (0.88, 0.82, 0.23, -0.54, 3.35),
+        (0.48, 0.12, 0.22, 0.43, 4.70),
+    )
+    scale = float(min(W, H))
+    for cx0, cy0, sigma0, strength, phase in vortices:
+        cx = (cx0 + 0.020 * np.sin(t * 0.020 + phase)) * W
+        cy = (cy0 + 0.027 * np.cos(t * 0.017 + phase * 0.73)) * H
+        sig = sigma0 * scale
+        dx, dy = x - cx, y - cy
+        q = np.exp(-(dx * dx + dy * dy) / (2.0 * sig * sig))
+        pulse = 0.90 + 0.10 * np.sin(t * 0.030 + phase)
+        vx += strength * pulse * (-dy / sig) * q
+        vy += strength * pulse * ( dx / sig) * q
+
+    # Long, soft waves are the curl of analytic stream functions and thus do
+    # not make the apparent liquid expand or contract.
+    modes = (
+        (118.0, 88.0, 0.105, 0.6, 1.7),
+        (73.0, 151.0, 0.070, 2.2, 4.1),
+        (181.0, 59.0, 0.055, 4.7, 0.3),
+    )
+    for lx, ly, amount, px, py in modes:
+        kx, ky = 2.0 * np.pi / lx, 2.0 * np.pi / ly
+        temporal = t * (0.016 + 0.003 * amount / 0.1)
+        sx = np.sin(kx * x + px + temporal * 0.35)
+        cx = np.cos(kx * x + px + temporal * 0.35)
+        sy = np.sin(ky * y + py - temporal)
+        cy = np.cos(ky * y + py - temporal)
+        # 'amount' is the approximate pixel velocity contributed per frame.
+        vx += amount * sx * cy
+        vy -= amount * (kx / ky) * cx * sy
+
+    return vx, vy
+
+
+def remap(a: np.ndarray, mx: np.ndarray, my: np.ndarray, interpolation: int) -> np.ndarray:
+    return cv2.remap(
+        a, mx.astype(np.float32), my.astype(np.float32), interpolation,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+
+def main() -> None:
+    case_dir = Path(__file__).resolve().parent
+    source = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    if source.shape != (H, W, 3):
+        raise ValueError(f"unexpected first-frame shape: {source.shape}")
+
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    map_x, map_y = xx.copy(), yy.copy()
+    frames = np.empty((N, H, W, 3), dtype=np.uint8)
+    frames[0] = source
+
+    # Two half-steps per video frame make coordinate advection smooth even in
+    # the tighter eddies.  Only coordinate maps are interpolated repeatedly;
+    # every displayed frame is sampled directly from the pristine still.
+    for i in range(1, N):
+        for substep in (0.25, 0.75):
+            tau = (i - 1) + substep
+            vx, vy = velocity(xx, yy, tau)
+            back_x = xx - 0.5 * vx
+            back_y = yy - 0.5 * vy
+            map_x = remap(map_x, back_x, back_y, cv2.INTER_CUBIC)
+            map_y = remap(map_y, back_x, back_y, cv2.INTER_CUBIC)
+            map_x = reflect_coordinates(map_x, W)
+            map_y = reflect_coordinates(map_y, H)
+
+        image = remap(source, map_x, map_y, cv2.INTER_CUBIC).astype(np.float32)
+
+        # Very restrained moving illumination gives the glossy surface life.
+        # It is modulated by existing chroma so it enriches the two dyes rather
+        # than painting new objects onto the image.
+        r, g, b = image[..., 0], image[..., 1], image[..., 2]
+        warm = np.clip((r - b - 5.0) / 80.0, 0.0, 1.0)
+        cool = np.clip((b - r + 3.0) / 85.0, 0.0, 1.0)
+        wave = (
+            np.sin(xx * 0.027 + yy * 0.019 - i * 0.045)
+            + 0.55 * np.sin(xx * 0.049 - yy * 0.031 + i * 0.030 + 1.2)
+        ) / 1.55
+        envelope = np.sin(min(i, 16) / 16.0 * np.pi / 2.0)
+        image[..., 0] += envelope * (1.5 * warm * wave - 0.35 * cool * wave)
+        image[..., 1] += envelope * (0.45 * warm * wave + 0.25 * cool * wave)
+        image[..., 2] += envelope * (1.55 * cool * wave - 0.25 * warm * wave)
+
+        frames[i] = np.clip(np.rint(image), 0, 255).astype(np.uint8)
+
+    # Contract-critical: do not let any processing touch the supplied first
+    # frame, even by a one-level interpolation or rounding difference.
+    frames[0] = source
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
