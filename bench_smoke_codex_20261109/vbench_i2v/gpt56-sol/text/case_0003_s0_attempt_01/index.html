+"""Generate a close macro shot of moving blue/orange liquid.
+
+The supplied still is treated as one view into a larger liquid surface.  A
+reflected, smoothly corrected border provides the small amount of off-screen
+material exposed by the camera move.  Each output frame is sampled directly
+from that canvas (rather than from the preceding frame), avoiding cumulative
+resampling blur.  Several very low frequency displacement fields provide the
+slow, cohesive movement expected from viscous liquid.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 256, 144
+N = 81
+FPS = 16.0
+
+
+def smootherstep(x):
+    """A camera easing with zero speed and acceleration at both ends."""
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * x * (x * (x * 6.0 - 15.0) + 10.0)
+
+
+def make_canvas(src):
+    """Extend the macro texture while keeping its edge exactly continuous."""
+    # Enough room for the pan and all displacement. REFLECT_101 works well for
+    # an abstract fluid surface: no constant-colored edge enters the picture.
+    py, px = 72, 28
+    canvas = cv2.copyMakeBorder(src, py, py, px, px, cv2.BORDER_REFLECT_101)
+
+    # Reflection by itself can create an obvious fold. In the upper extension,
+    # increasingly bend the reflected texture sideways, while the bend tends
+    # exactly to zero at the join. This retains perfect continuity at y=py and
+    # turns the repetition into another broad liquid fold.
+    ch, cw = canvas.shape[:2]
+    yy, xx = np.mgrid[0:ch, 0:cw].astype(np.float32)
+    d = np.maximum(0.0, (py - yy) / py)
+    bend = d * d * (8.0 * np.sin(xx / 49.0 + 0.35) + 3.0 * np.sin(xx / 17.0))
+    mx = xx + bend.astype(np.float32)
+    my = yy.copy()
+    bent = cv2.remap(canvas, mx, my, cv2.INTER_CUBIC,
+                     borderMode=cv2.BORDER_REFLECT_101)
+
+    # Feather only within the extension. The final source image area is never
+    # altered, which is important both for fidelity and for exact frame zero.
+    alpha = np.clip(d * 1.7, 0.0, 1.0)[..., None]
+    canvas = canvas.astype(np.float32) * (1.0 - alpha) + bent.astype(np.float32) * alpha
+    return np.clip(canvas, 0, 255).astype(np.uint8), px, py
+
+
+def fluid_maps(t, progress, px, py):
+    """Backward maps for camera motion plus smooth viscous advection."""
+    y, x = np.mgrid[0:H, 0:W].astype(np.float32)
+
+    # A clear but gentle upward tilt: scene features travel downward while new
+    # liquid enters along the top edge. 35 px over five seconds is readable at
+    # this close scale without turning the shot into a rapid pan.
+    camera_y = 35.0 * progress
+
+    # Broad coupled waves. Subtracting the t=0 value makes frame 0's map the
+    # identity. Frequencies are deliberately low to preserve viscous cohesion.
+    p1 = x / 54.0 + y / 31.0 + 0.25 * np.sin(y / 43.0)
+    p2 = x / 27.0 - y / 63.0 + 0.40 * np.sin(x / 71.0)
+    p3 = y / 22.0 - x / 92.0
+    dx = 2.10 * (np.sin(p1 + 0.47 * t) - np.sin(p1))
+    dx += 1.15 * (np.sin(p3 - 0.31 * t) - np.sin(p3))
+    dy = 1.40 * (np.sin(p2 - 0.37 * t) - np.sin(p2))
+    dy += 0.72 * (np.sin(p1 + 0.23 * t) - np.sin(p1))
+
+    # Two wide vortical drifts break up the purely sinusoidal motion. Their
+    # tiny angles are sufficient in macro view and remain sub-pixel per frame.
+    for cx, cy, radius, rate, sign in (
+            (82.0, 59.0, 92.0, 0.0100, 1.0),
+            (191.0, 105.0, 84.0, 0.0080, -1.0)):
+        rx, ry = x - cx, y - cy
+        fall = np.exp(-(rx * rx + ry * ry) / (2.0 * radius * radius))
+        angle = sign * rate * np.sin(np.pi * t / 10.125) * fall
+        dx += -ry * angle
+        dy += rx * angle
+
+    # A mild projective term suggests an actual tilt rather than an entirely
+    # flat vertical slide. It is zero on the first frame.
+    tilt = 0.014 * progress
+    projective_y = tilt * ((y - H * 0.48) ** 2 / H - 0.10 * (y - H * 0.48))
+
+    map_x = x + px + dx
+    map_y = y + py - camera_y + dy + projective_y
+    return map_x.astype(np.float32), map_y.astype(np.float32)
+
+
+def caustic_grade(frame, t, progress):
+    """Very subtle moving highlights tied to the sampled liquid texture."""
+    if progress <= 0.0:
+        return frame
+    f = frame.astype(np.float32)
+    y, x = np.mgrid[0:H, 0:W].astype(np.float32)
+
+    # Two interference waves form large, soft caustics rather than a synthetic
+    # fine checker. Restrict them to naturally mid/bright areas of the image.
+    wave = np.sin(x / 34.0 + y / 23.0 + 0.42 * t)
+    wave += 0.65 * np.sin(x / 71.0 - y / 29.0 - 0.29 * t + 1.1)
+    wave0 = np.sin(x / 34.0 + y / 23.0)
+    wave0 += 0.65 * np.sin(x / 71.0 - y / 29.0 + 1.1)
+    change = (wave - wave0) * progress
+    lum = 0.2126 * f[..., 0] + 0.7152 * f[..., 1] + 0.0722 * f[..., 2]
+    mask = np.clip((lum - 52.0) / 145.0, 0.0, 1.0)
+    delta = (1.55 * change * mask)[..., None]
+    # Highlights in liquid are slightly cool, helping the existing blues while
+    # changing the orange region only in luminance.
+    f += delta * np.array([0.80, 1.00, 1.08], np.float32)
+    return np.clip(np.rint(f), 0, 255).astype(np.uint8)
+
+
+def generate(case_dir):
+    case_dir = Path(case_dir)
+    src = np.array(Image.open(case_dir / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    if src.shape != (H, W, 3):
+        raise ValueError(f"unexpected first frame shape: {src.shape}")
+
+    canvas, px, py = make_canvas(src)
+    frames = np.empty((N, H, W, 3), dtype=np.uint8)
+    frames[0] = src  # contractual pixel identity, bypassing all interpolation
+
+    for i in range(1, N):
+        u = i / (N - 1)
+        progress = smootherstep(u)
+        t = i / FPS
+        mx, my = fluid_maps(t, progress, px, py)
+        frame = cv2.remap(canvas, mx, my, cv2.INTER_CUBIC,
+                          borderMode=cv2.BORDER_REFLECT_101)
+        frames[i] = caustic_grade(frame, t, progress)
+
+    artifact_io.save(frames, case_dir)
+    return frames
+
+
+if __name__ == "__main__":
+    generate(Path(__file__).resolve().parent)
