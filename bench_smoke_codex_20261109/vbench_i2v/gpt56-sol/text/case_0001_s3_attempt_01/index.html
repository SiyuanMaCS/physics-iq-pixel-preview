+"""Generate a gentle leftward camera pan over the supplied liquid macro image.
+
+The source is treated as a small window onto a larger, continuous liquid
+surface.  Moving the camera to the left makes features in the source travel
+to the right.  The newly revealed area is continued from the left edge by a
+reflection (which is value-continuous at the join), with a very small organic
+coordinate deformation to keep the continuation from looking rigid.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+FRAME_COUNT = 81
+FPS = 16.0
+PAN_DISTANCE = 44.0
+
+
+def reflect_coordinate(v: np.ndarray, limit: int) -> np.ndarray:
+    """Reflect arbitrary floating coordinates into [0, limit - 1]."""
+    period = 2.0 * (limit - 1)
+    m = np.mod(v, period)
+    return np.where(m <= limit - 1, m, period - m)
+
+
+def bilinear_rgb(image: np.ndarray, xx: np.ndarray, yy: np.ndarray) -> np.ndarray:
+    """Sample an RGB image at floating point coordinates."""
+    h, w = image.shape[:2]
+    xx = reflect_coordinate(xx, w)
+    yy = reflect_coordinate(yy, h)
+
+    x0 = np.floor(xx).astype(np.int32)
+    y0 = np.floor(yy).astype(np.int32)
+    x1 = np.minimum(x0 + 1, w - 1)
+    y1 = np.minimum(y0 + 1, h - 1)
+    fx = (xx - x0)[..., None]
+    fy = (yy - y0)[..., None]
+
+    a = image[y0, x0].astype(np.float32)
+    b = image[y0, x1].astype(np.float32)
+    c = image[y1, x0].astype(np.float32)
+    d = image[y1, x1].astype(np.float32)
+    top = a + (b - a) * fx
+    bottom = c + (d - c) * fx
+    return top + (bottom - top) * fy
+
+
+def make_frames() -> np.ndarray:
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    h, w = source.shape[:2]
+    if (w, h) != (256, 144):
+        raise ValueError(f"Unexpected source size: {w}x{h}")
+
+    yy, xx = np.mgrid[0:h, 0:w].astype(np.float32)
+    result = np.empty((FRAME_COUNT, h, w, 3), dtype=np.uint8)
+    result[0] = source  # Contract: this must remain pixel-identical.
+
+    duration = (FRAME_COUNT - 1) / FPS
+    for index in range(1, FRAME_COUNT):
+        elapsed = index / FPS
+        u = index / (FRAME_COUNT - 1)
+
+        # Predominantly constant camera speed, with only a very short, gentle
+        # ease at both ends.  Its integral is normalized to finish at 44 px.
+        # The cubic correction has zero endpoints and preserves monotonicity.
+        travel = PAN_DISTANCE * (u - 0.055 * np.sin(2.0 * np.pi * u) / (2.0 * np.pi))
+        world_x = xx - travel
+
+        # Subtle coherent motion appropriate to a macro shot of liquid.  Each
+        # term is differenced against t=0 so the initial frame has no warp.
+        ramp = min(1.0, elapsed / 0.75)
+        phase_a = 2.0 * np.pi * (yy / 176.0 + xx / 510.0)
+        phase_b = 2.0 * np.pi * (yy / 91.0 - xx / 390.0)
+        dx = ramp * (
+            0.52 * (np.sin(phase_a + 0.31 * elapsed) - np.sin(phase_a))
+            + 0.22 * (np.sin(phase_b - 0.23 * elapsed) - np.sin(phase_b))
+        )
+        dy = ramp * (
+            0.58 * (np.sin(phase_b + 0.27 * elapsed) - np.sin(phase_b))
+            + 0.18 * (np.sin(phase_a - 0.19 * elapsed) - np.sin(phase_a))
+        )
+
+        # Reflection supplies the part of the liquid surface revealed to the
+        # left.  A deformation that grows away from the join avoids a visibly
+        # exact mirror while remaining perfectly continuous at world_x == 0.
+        outside = np.clip(-world_x / PAN_DISTANCE, 0.0, 1.25)
+        continuation_dy = outside * outside * (
+            0.85 * np.sin(yy / 19.0 + 0.4)
+            + 0.35 * np.sin(yy / 8.5 - 0.7)
+        )
+
+        sample_x = world_x + dx
+        sample_y = yy + dy + continuation_dy
+        frame = bilinear_rgb(source, sample_x, sample_y)
+        result[index] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    return result
+
+
+if __name__ == "__main__":
+    frames = make_frames()
+    artifact_io.save(frames, CASE_DIR)
