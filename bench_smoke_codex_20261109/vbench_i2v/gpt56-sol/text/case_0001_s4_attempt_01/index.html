+"""Generate the requested left-panning macro shot from the supplied first frame.
+
+The picture is an abstract liquid macro, so a 2-D advected texture is a good
+model of the scene.  The original image is extended to camera-left with a
+smooth, warped continuation of its own texture.  A sub-pixel camera pan and a
+small, long-wavelength flow deformation are then applied.  All operations are
+deterministic RGB/numeric operations; no learned model or external asset is
+used.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+HERE = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 144
+N_FRAMES = 81
+EXTENT = 96
+
+
+def reflect_coordinate(a: np.ndarray, length: int) -> np.ndarray:
+    """Reflect arbitrary floating point coordinates into [0, length-1]."""
+    if length <= 1:
+        return np.zeros_like(a)
+    period = 2.0 * (length - 1)
+    b = np.mod(a, period)
+    return np.where(b <= length - 1, b, period - b)
+
+
+def bilinear(image: np.ndarray, x: np.ndarray, y: np.ndarray,
+             reflect_y: bool = True) -> np.ndarray:
+    """Vectorized floating point RGB sampler."""
+    h, w = image.shape[:2]
+    x = np.clip(x, 0.0, w - 1.00001)
+    if reflect_y:
+        y = reflect_coordinate(y, h)
+    else:
+        y = np.clip(y, 0.0, h - 1.00001)
+
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = np.minimum(x0 + 1, w - 1)
+    y1 = np.minimum(y0 + 1, h - 1)
+    fx = (x - x0)[..., None]
+    fy = (y - y0)[..., None]
+
+    top = image[y0, x0] * (1.0 - fx) + image[y0, x1] * fx
+    bot = image[y1, x0] * (1.0 - fx) + image[y1, x1] * fx
+    return top * (1.0 - fy) + bot * fy
+
+
+def make_left_extension(first: np.ndarray) -> np.ndarray:
+    """Create extra scene area to the left for the camera to reveal.
+
+    Sampling bends a broad strip of the existing liquid back around the left
+    border.  Increasing vertical advection and a nonlinear horizontal trace
+    prevent the result from reading as a simple mirror image.  The deformation
+    fades to zero at the join, so the continuation has no colour seam.
+    """
+    yy, qq = np.meshgrid(
+        np.arange(HEIGHT, dtype=np.float32),
+        np.arange(EXTENT, 0, -1, dtype=np.float32),
+        indexing="ij",
+    )
+    gain = 1.0 - np.exp(-qq / 10.0)
+    source_x = (
+        0.76 * qq
+        + gain * (3.4 * np.sin(qq / 14.0 + yy / 39.0)
+                  + 1.7 * np.sin(qq / 7.5 - yy / 23.0))
+    )
+    source_y = (
+        yy
+        + gain * (7.5 * np.sin(qq / 21.0 + yy / 35.0)
+                  + 3.0 * np.sin(qq / 10.5 - yy / 17.0))
+    )
+    extension = bilinear(first, source_x, source_y)
+
+    # Force the final few columns into a very gentle edge continuation.  This
+    # retains detail while avoiding a conspicuous crease where world x == 0.
+    edge = first[:, :1, :]
+    for k in range(1, 7):
+        column = EXTENT - k
+        weight = (1.0 - k / 7.0) ** 2
+        extension[:, column:column + 1, :] = (
+            extension[:, column:column + 1, :] * (1.0 - weight)
+            + edge * weight
+        )
+    return np.concatenate([extension, first], axis=1)
+
+
+def render() -> np.ndarray:
+    first_u8 = np.asarray(Image.open(HERE / "first_frame.png").convert("RGB"))
+    if first_u8.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"unexpected first frame shape: {first_u8.shape}")
+    first = first_u8.astype(np.float32)
+    scene = make_left_extension(first)
+
+    screen_y, screen_x = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+    frames = np.empty((N_FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first_u8  # Contract: exact, not a resampled approximation.
+
+    for index in range(1, N_FRAMES):
+        progress = index / (N_FRAMES - 1)
+        seconds = index / 16.0
+
+        # Looking farther camera-left makes scene features travel to the right.
+        # 52 pixels over five seconds gives a clear but unhurried macro pan.
+        pan = 52.0 * progress
+        world_x = screen_x - pan
+
+        # Coherent, low-amplitude displacement conveys viscous liquid motion
+        # without fighting the requested camera move.  The exponential onset
+        # prevents a sudden first-frame deformation.
+        flow_on = 1.0 - np.exp(-seconds / 0.65)
+        dx = flow_on * (
+            1.15 * np.sin(world_x / 42.0 + screen_y / 27.0 + 0.36 * seconds)
+            + 0.55 * np.sin(screen_y / 13.0 - world_x / 67.0 - 0.24 * seconds)
+        )
+        dy = flow_on * (
+            1.55 * np.sin(world_x / 34.0 - 0.31 * seconds)
+            + 0.65 * np.sin(world_x / 19.0 + screen_y / 31.0
+                            + 0.22 * seconds)
+        )
+
+        sample_x = EXTENT + world_x + dx
+        sample_y = screen_y + dy
+        frame = bilinear(scene, sample_x, sample_y)
+
+        # A restrained moving caustic (well below the image's native contrast)
+        # makes highlights breathe rather than leaving the liquid as a rigid
+        # photograph.  It also fades in from the supplied frame.
+        caustic = flow_on * (
+            0.85 * np.sin(world_x / 31.0 + screen_y / 45.0 + 0.42 * seconds)
+            + 0.40 * np.sin(screen_y / 18.0 - 0.27 * seconds)
+        )
+        frame = frame + caustic[..., None]
+        frames[index] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    return frames
+
+
+if __name__ == "__main__":
+    artifact_io.save(render(), HERE)
