+"""Generate the requested upward-tilting macro shot of coloured liquid.
+
+The supplied frame is used as the photographed material.  A small, continuous
+texture extension is made above it, then a smooth vertical camera move and
+very gentle, divergence-free deformations give the surface a liquid feel.
+Everything is deterministic and frame zero is copied without resampling.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+from scipy.ndimage import gaussian_filter, map_coordinates
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+W, H, N = 256, 144, 81
+
+
+def sample_rgb(image, yy, xx, order=1):
+    """Sample an RGB image at floating point coordinates."""
+    out = np.empty(yy.shape + (3,), dtype=np.float32)
+    for channel in range(3):
+        out[..., channel] = map_coordinates(
+            image[..., channel], (yy, xx), order=order, mode="reflect",
+            prefilter=(order > 1),
+        )
+    return out
+
+
+def normalised_smooth_noise(rng, shape, sigma):
+    field = gaussian_filter(rng.standard_normal(shape).astype(np.float32), sigma,
+                            mode="reflect")
+    field -= field.mean()
+    field /= field.std() + 1e-6
+    return field
+
+
+def build_world(source):
+    """Continue the abstract surface upward without introducing a hard seam."""
+    top = 96
+    bottom = 8
+    world_h = top + H + bottom
+    world = np.empty((world_h, W, 3), dtype=np.float32)
+    world[top:top + H] = source
+    world[top + H:] = source[-bottom:]
+
+    # Above the photograph, follow its top edge with a softly distorted
+    # reflection.  Distortion is zero at the join, so both colour and shape are
+    # continuous, and increases gradually only in newly revealed material.
+    y, x = np.mgrid[0:top, 0:W].astype(np.float32)
+    d = top - y
+    sy = np.clip(d - 1.0 + 0.0018 * d * d, 0, H - 1)
+    bend = (d / top) ** 1.35
+    sx = x + bend * (
+        8.0 * np.sin((x + 13.0) / 31.0)
+        + 3.5 * np.sin((x - 7.0) / 11.5)
+    )
+    primary = sample_rgb(source, sy, sx, order=1)
+
+    # A weak second scale breaks obvious mirror repetition far from the join.
+    sy2 = np.clip(7.0 + 1.38 * d + 2.5 * np.sin(x / 46.0), 0, H - 1)
+    sx2 = x + 24.0 * np.sin((x + 0.7 * d) / 49.0) - 0.16 * d
+    secondary = sample_rgb(source, sy2, sx2, order=1)
+    mix = (0.28 * np.clip((d - 15.0) / 70.0, 0, 1) ** 1.25)[..., None]
+    extension = primary * (1.0 - mix) + secondary * mix
+
+    # Keep the extension's grain comparable to the source while retaining a
+    # seamless ten-pixel band at the boundary.
+    rng = np.random.default_rng(44107)
+    grain = gaussian_filter(rng.standard_normal((top, W)).astype(np.float32),
+                            0.65, mode="reflect")
+    grain /= grain.std() + 1e-6
+    grain_gain = 1.15 * np.clip((d - 7.0) / 28.0, 0, 1)
+    extension += (grain * grain_gain)[..., None]
+    world[:top] = np.clip(extension, 0, 255)
+    return world, top
+
+
+def generate():
+    source_u8 = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"),
+                           dtype=np.uint8)
+    assert source_u8.shape == (H, W, 3)
+    source = source_u8.astype(np.float32)
+    world, anchor = build_world(source)
+    wh = world.shape[0]
+
+    rng = np.random.default_rng(73621)
+    # Stream-function derivatives produce coherent, incompressible-looking
+    # eddies rather than independent x/y wobble.
+    psi1 = normalised_smooth_noise(rng, (wh, W), 18.0)
+    psi2 = normalised_smooth_noise(rng, (wh, W), 9.0)
+    g1y, g1x = np.gradient(psi1)
+    g2y, g2x = np.gradient(psi2)
+    flow1x, flow1y = g1y, -g1x
+    flow2x, flow2y = g2y, -g2x
+    for fx, fy in ((flow1x, flow1y), (flow2x, flow2y)):
+        scale = np.sqrt(np.mean(fx * fx + fy * fy)) + 1e-6
+        fx /= scale
+        fy /= scale
+
+    # Broad illumination variation, sampled in surface/world coordinates.
+    sheen = normalised_smooth_noise(rng, (wh, W), 25.0)
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    frames = np.empty((N, H, W, 3), dtype=np.uint8)
+    frames[0] = source_u8  # contract: exactly pixel-identical
+
+    for index in range(1, N):
+        t = index / (N - 1)
+        # Smootherstep-like camera tilt with a little linear motion at each end.
+        smooth = t * t * (3.0 - 2.0 * t)
+        pan = 64.0 * (0.075 * t + 0.925 * smooth)
+        world_y = yy + anchor - pan
+
+        # Two slow eddy components. Both are exactly at rest at t=0.
+        a = np.sin(1.12 * np.pi * t)
+        b = 1.0 - np.cos(0.78 * np.pi * t)
+        iy = np.clip(np.rint(world_y).astype(np.int32), 0, wh - 1)
+        ix = np.broadcast_to(np.arange(W, dtype=np.int32), (H, W))
+        dx = 1.15 * a * flow1x[iy, ix] + 0.47 * b * flow2x[iy, ix]
+        dy = 0.82 * a * flow1y[iy, ix] + 0.35 * b * flow2y[iy, ix]
+
+        sampled = sample_rgb(world, world_y + dy, xx + dx, order=1)
+
+        # Moving, extremely restrained caustic sheen helps the coloured masses
+        # read as liquid without repainting or obscuring the supplied image.
+        sheen_x = np.mod(xx + 5.0 * t, W)
+        local_sheen = map_coordinates(
+            sheen, (np.clip(world_y, 0, wh - 1), sheen_x), order=1,
+            mode="reflect", prefilter=False,
+        )
+        light = (1.35 * np.sin(np.pi * t) * local_sheen)[..., None]
+        # Highlights respond a little more than dark blue areas.
+        luma = sampled.mean(axis=2, keepdims=True)
+        sampled += light * (0.50 + 0.50 * luma / 255.0)
+        frames[index] = np.clip(np.rint(sampled), 0, 255).astype(np.uint8)
+
+    return frames
+
+
+if __name__ == "__main__":
+    artifact_io.save(generate(), ROOT)
