+#!/usr/bin/env python3
+"""Deterministic image-to-video animation for this case.
+
+The animation treats the photograph as a collection of softly coupled depth
+planes.  A very small push-in, articulated motion around the bear's head and
+fish, and several coherent water-wave fields provide motion without repainting
+or inventing a second version of the animal.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+W, H, N = 256, 144, 81
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def make_mattes(rgb):
+    """Estimate a conservative bear matte from the connected warm foreground."""
+    f = rgb.astype(np.int16)
+    r, g, b = f[..., 0], f[..., 1], f[..., 2]
+    yy, xx = np.mgrid[:H, :W]
+
+    # The bear/fish is the large warm connected region below the water horizon.
+    warm = ((r - b > 7) & (r - g > -7) & (yy > 35) & (xx > 64)).astype(np.uint8)
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(warm, 8)
+    choices = [(stats[i, cv2.CC_STAT_AREA], i) for i in range(1, count)
+               if stats[i, cv2.CC_STAT_TOP] > 30]
+    component = labels == max(choices)[1]
+
+    # Join dark fur enclosed by the warm highlights, but retain a feathered edge.
+    core = cv2.morphologyEx(component.astype(np.uint8), cv2.MORPH_CLOSE,
+                            cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (19, 19)))
+    core = cv2.dilate(core, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (7, 7)))
+    subject = cv2.GaussianBlur(core.astype(np.float32), (0, 0), 4.0)
+    subject = np.clip(subject, 0, 1)
+
+    # Broad articulated influence fields. They are intentionally soft: this is
+    # a dense photographic warp, not a cutout translation.
+    head = np.exp(-(((xx - 108.0) / 46.0) ** 2 + ((yy - 79.0) / 43.0) ** 2) * 2.0)
+    head *= np.clip(subject * 1.35, 0, 1)
+    fish = np.exp(-(((xx - 108.0) / 22.0) ** 2 + ((yy - 128.0) / 16.0) ** 2) * 2.0)
+    fish *= np.clip(subject * 1.5, 0, 1)
+
+    # Sky and distant water are kept nearly rigid; open foreground water moves.
+    water_depth = np.clip((yy - 35.0) / 109.0, 0, 1).astype(np.float32)
+    water = np.clip((1.0 - subject) * water_depth, 0, 1)
+    return subject.astype(np.float32), head.astype(np.float32), fish.astype(np.float32), water
+
+
+def ripple_overlay(frame, subject, t):
+    """Add restrained, antialiased wake rings in the open water."""
+    hi = 3
+    layer = np.zeros((H * hi, W * hi), np.uint8)
+    # Three continuously renewed rings; vertically compressed by perspective.
+    for k in range(3):
+        q = (t * 0.25 + k / 3.0) % 1.0
+        rx = 25.0 + 102.0 * q
+        ry = 3.5 + 14.0 * q
+        strength = (np.sin(np.pi * q) ** 1.5) * 5.0
+        if strength < .2:
+            continue
+        cv2.ellipse(layer, (138 * hi, 132 * hi),
+                    (int(rx * hi), int(ry * hi)), 0, 188, 352,
+                    int(round(strength * 20)), max(1, hi), cv2.LINE_AA)
+    layer = cv2.resize(layer, (W, H), interpolation=cv2.INTER_AREA).astype(np.float32) / 255.0
+    # Keep the generated wake off fur/fish and fade near the horizon.
+    a = layer * (1.0 - subject) * 0.75
+    out = frame.astype(np.float32)
+    out += a[..., None] * np.array([15.0, 18.0, 20.0], np.float32)
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def generate():
+    source_u8 = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    source = source_u8.astype(np.float32)
+    subject, head, fish, water = make_mattes(source_u8)
+    yy, xx = np.mgrid[:H, :W].astype(np.float32)
+
+    frames = []
+    for i in range(N):
+        if i == 0:
+            frames.append(source_u8.copy())
+            continue
+
+        p = i / (N - 1.0)
+        e = smoothstep(p)
+        t = i / 16.0
+
+        # A quiet documentary push-in with a slight leftward reframe.
+        zoom = 1.0 + 0.018 * e
+        cx, cy = 134.0, 82.0
+        sx = cx + (xx - cx) / zoom + 0.55 * e
+        sy = cy + (yy - cy) / zoom - 0.18 * e
+
+        # The swimming bear advances left.  Neck motion lags its body slightly,
+        # producing a natural bob rather than moving the photo as one card.
+        body_dx = -3.8 * e
+        body_dy = 0.55 * e + 0.42 * (1.0 - np.cos(2.0 * np.pi * t / 2.8))
+        neck_dx = 1.05 * np.sin(2.0 * np.pi * t / 3.15)
+        neck_dy = -1.45 * np.sin(2.0 * np.pi * t / 2.55)
+        disp_x = subject * body_dx + head * neck_dx
+        disp_y = subject * body_dy + head * neck_dy
+
+        # Water refraction: multiple long waves, phase-anchored to frame zero.
+        w1 = np.sin(yy * 0.235 + xx * 0.019 - t * 2.15) - np.sin(yy * 0.235 + xx * 0.019)
+        w2 = np.sin(yy * 0.098 - xx * 0.033 - t * 1.17) - np.sin(yy * 0.098 - xx * 0.033)
+        disp_x += water * (0.55 * w1 + 0.32 * w2)
+        disp_y += water * 0.22 * (np.sin(xx * .085 - t * 1.45) - np.sin(xx * .085))
+
+        # Inverse-map the desired displacement.  A localized tiny rotation gives
+        # the fish a single soft tail/body twitch near mid-shot.
+        sx -= disp_x
+        sy -= disp_y
+        twitch_env = np.exp(-((t - 2.75) / .62) ** 2)
+        angle = 0.035 * twitch_env * np.sin((t - 2.15) * 9.0)
+        fx, fy = 109.0, 129.0
+        rx, ry = sx - fx, sy - fy
+        ca, sa = np.cos(-angle), np.sin(-angle)
+        rsx = fx + ca * rx - sa * ry
+        rsy = fy + sa * rx + ca * ry
+        sx = sx * (1.0 - fish) + rsx * fish
+        sy = sy * (1.0 - fish) + rsy * fish
+
+        warped = cv2.remap(source, sx.astype(np.float32), sy.astype(np.float32),
+                           cv2.INTER_CUBIC, borderMode=cv2.BORDER_REFLECT_101)
+
+        # Low-amplitude coherent glints give the water life without frame noise.
+        glint = ((np.sin(xx * .105 + yy * .31 - t * 1.8) -
+                  np.sin(xx * .105 + yy * .31)) * 1.25 +
+                 (np.sin(xx * .043 - yy * .17 - t * .83) -
+                  np.sin(xx * .043 - yy * .17)) * .7)
+        warped += (glint * water)[..., None] * np.array([.72, .91, 1.0], np.float32)
+        frame = np.clip(warped, 0, 255).astype(np.uint8)
+        frame = ripple_overlay(frame, subject, t)
+        frames.append(frame)
+
+    result = np.stack(frames).astype(np.uint8)
+    # Contract-critical identity check.
+    assert np.array_equal(result[0], source_u8)
+    return result
+
+
+if __name__ == "__main__":
+    artifact_io.save(generate(), ROOT)
+
