+"""Generate a restrained upward camera tilt across the supplied liquid artwork.
+
+The shot is treated as a close-up of a mostly planar marbled-liquid surface.  A
+larger, deterministic texture is first extrapolated above the supplied frame;
+the camera then moves upward over it (so features travel down through frame).
+Only tiny, low-frequency refraction is added to keep the liquid organic without
+obscuring the requested camera move.
+"""
+
+from pathlib import Path
+import json
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+
+
+def _upper_extension(image: np.ndarray, pad: int) -> np.ndarray:
+    """Extrapolate a textured strip above *image* while matching its top edge."""
+    h, w = image.shape[:2]
+    src = image.astype(np.float32)
+
+    # Separate broad colour fields from fine pigment detail.  Broad fields are
+    # continued using the vertical trend at the top edge, while detail is
+    # borrowed from nearby rows with an increasing sideways shear.  This avoids
+    # the conspicuous bilateral seam of plain reflection padding.
+    broad = cv2.GaussianBlur(src, (0, 0), sigmaX=8.0, sigmaY=8.0,
+                             borderType=cv2.BORDER_REFLECT_101)
+    detail = src - broad
+    grad = (broad[0] - broad[12]) / 12.0
+
+    upper = np.empty((pad, w, 3), np.float32)
+    x = np.arange(w, dtype=np.float32)
+    for row in range(pad):
+        d = float(pad - row)  # distance above the original image
+        # Damped linear continuation of the low-frequency colour field.
+        effective_d = 22.0 * (1.0 - np.exp(-d / 22.0))
+        low = broad[0] + effective_d * grad
+
+        # Diagonally folded nearby pigment creates plausible new marbling.  The
+        # displacement is exactly zero at the join and grows smoothly upward.
+        sy = min(h - 1.001, d * 0.92)
+        y0 = int(sy)
+        y1 = min(h - 1, y0 + 1)
+        fy = sy - y0
+        shear = 0.72 * d + 5.0 * (d / pad) * np.sin(x / 31.0 + d / 19.0)
+        sx = np.mod(x + shear, w)
+        x0 = np.floor(sx).astype(np.int32)
+        x1 = (x0 + 1) % w
+        fx = (sx - x0)[:, None]
+        r0 = detail[y0, x0] * (1.0 - fx) + detail[y0, x1] * fx
+        r1 = detail[y1, x0] * (1.0 - fx) + detail[y1, x1] * fx
+        fine = r0 * (1.0 - fy) + r1 * fy
+
+        # Preserve pigment texture but gently attenuate its highest contrast in
+        # the furthest extrapolated area.
+        fine_gain = 1.0 - 0.16 * (d / pad)
+        upper[row] = np.clip(low + fine_gain * fine, 0.0, 255.0)
+
+    canvas = np.concatenate([upper, src], axis=0)
+    # Explicitly preserve the source region byte-for-byte before remapping.
+    canvas[pad:pad + h] = src
+    return canvas.astype(np.uint8)
+
+
+def generate() -> np.ndarray:
+    spec = json.loads((ROOT / "task_spec.json").read_text())
+    count = int(spec["frames"])
+    h, w = int(spec["height"]), int(spec["width"])
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"),
+                       dtype=np.uint8)
+    if first.shape != (h, w, 3):
+        raise ValueError(f"unexpected first-frame shape: {first.shape}")
+
+    pad = 72
+    canvas = _upper_extension(first, pad)
+    yy, xx = np.mgrid[0:h, 0:w].astype(np.float32)
+    frames = np.empty((count, h, w, 3), dtype=np.uint8)
+    frames[0] = first  # contract: frame zero is pixel-identical
+
+    for i in range(1, count):
+        u = i / (count - 1)
+        ease = 0.5 - 0.5 * np.cos(np.pi * u)
+
+        # A 44 px rise gives a clear but unhurried upward tilt over five seconds.
+        rise = 44.0 * ease
+
+        # Sub-pixel coherent refraction.  Its amplitude is deliberately much
+        # smaller than camera travel, keeping the requested tilt unmistakable.
+        phase = 0.55 * np.pi * u
+        flow_x = ease * (
+            0.55 * np.sin((yy + pad - rise) / 23.0 + phase)
+            + 0.24 * np.sin((xx + yy) / 39.0 - phase)
+        )
+        flow_y = ease * (
+            0.34 * np.sin(xx / 29.0 - 0.7 * phase)
+            + 0.16 * np.sin((xx - 1.7 * yy) / 51.0 + phase)
+        )
+
+        map_x = (xx + 1.25 * ease + flow_x).astype(np.float32)
+        map_y = (yy + pad - rise + flow_y).astype(np.float32)
+        frames[i] = cv2.remap(
+            canvas, map_x, map_y,
+            interpolation=cv2.INTER_CUBIC,
+            borderMode=cv2.BORDER_REFLECT_101,
+        )
+
+    return frames
+
+
+if __name__ == "__main__":
+    artifact_io.save(generate(), ROOT)
