+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 256, 144
+N = 81
+TOP_PAD = 64
+
+
+def bilinear_sample(image, xs, ys):
+    """Bilinear RGB sampling with wrapped x and clamped y."""
+    h, w, _ = image.shape
+    xs = np.mod(xs, w)
+    ys = np.clip(ys, 0.0, h - 1.0001)
+
+    x0 = np.floor(xs).astype(np.int32)
+    y0 = np.floor(ys).astype(np.int32)
+    x1 = (x0 + 1) % w
+    y1 = np.minimum(y0 + 1, h - 1)
+    fx = (xs - x0)[..., None]
+    fy = (ys - y0)[..., None]
+
+    a = image[y0, x0].astype(np.float32)
+    b = image[y0, x1].astype(np.float32)
+    c = image[y1, x0].astype(np.float32)
+    d = image[y1, x1].astype(np.float32)
+    return (a * (1 - fx) + b * fx) * (1 - fy) + (c * (1 - fx) + d * fx) * fy
+
+
+def make_upper_outpaint(source):
+    """Continue the cropped marbling upward using only warped source pixels.
+
+    A reflection naturally completes all the shapes cut by the upper frame edge.
+    Increasing, multi-scale lateral displacement away from that edge removes the
+    mechanical mirror look while preserving a seamless join.
+    """
+    canvas = np.empty((TOP_PAD + H, W, 3), dtype=np.float32)
+    canvas[TOP_PAD:] = source
+
+    x = np.arange(W, dtype=np.float32)[None, :]
+    # d=1 is immediately above the original top edge.
+    d = np.arange(TOP_PAD, 0, -1, dtype=np.float32)[:, None]
+    strength = np.clip((d - 1.0) / 55.0, 0.0, 1.0)
+
+    # Reflect progressively deeper source rows, but bend the strata into broad
+    # fluid arcs. All displacement tends to zero at the join.
+    src_y = (d - 1.0)
+    src_y = src_y + strength * (
+        3.2 * np.sin(x / 31.0 + d / 12.0)
+        + 1.35 * np.sin(x / 10.5 - d / 8.0)
+    )
+    src_y = np.clip(src_y, 0.0, 68.0)
+
+    src_x = x + strength * (
+        7.0 * np.sin(x / 48.0 + d / 15.0)
+        + 3.0 * np.sin(x / 17.0 - d / 10.0)
+        + 1.2 * np.sin(x / 5.3 + d / 7.0)
+    )
+    canvas[:TOP_PAD] = bilinear_sample(source, src_x, src_y)
+    return canvas
+
+
+def crop_subpixel(canvas, top):
+    yy = top + np.arange(H, dtype=np.float32)[:, None]
+    xx = np.arange(W, dtype=np.float32)[None, :]
+    # Broadcast both coordinate fields to the complete image plane.
+    yy = np.broadcast_to(yy, (H, W))
+    xx = np.broadcast_to(xx, (H, W))
+    return bilinear_sample(canvas, xx, yy)
+
+
+def generate_frames(source):
+    canvas = make_upper_outpaint(source.astype(np.float32))
+    frames = np.empty((N, H, W, 3), dtype=np.uint8)
+    frames[0] = source  # generation contract: exactly pixel-identical
+
+    for i in range(1, N):
+        t = i / (N - 1)
+        # Quintic easing gives a deliberate camera move with gentle starts and
+        # stops. 48 px over five seconds is clear without becoming a fast pan.
+        ease = t * t * t * (t * (t * 6.0 - 15.0) + 10.0)
+        rise = 48.0 * ease
+        crop = crop_subpixel(canvas, TOP_PAD - rise)
+        frames[i] = np.clip(np.rint(crop), 0, 255).astype(np.uint8)
+    return frames
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    source = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    assert source.shape == (H, W, 3)
+    frames = generate_frames(source)
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
