+    cv2.rectangle(m, (92, 96), (119, 116), cv2.GC_FGD, -1)
+    m[:34] = cv2.GC_BGD

+    fg_model = np.zeros((1, 65), np.float64)

+def transform_matrix(scale, dx, dy):
+    ax, ay = 130.0, 130.0             # feet stay planted in the shallows
+    return np.array([[scale, 0, ax + dx - scale*ax],
+                     [0, scale, ay + dy - scale*ay]], np.float32)
+

+def water_background(bg, k):
+    """Sub-pixel water flow below the far bank; the horizon itself stays steady."""
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    fade = np.clip((yy - 35.0) / 18.0, 0, 1)
+    # Layered phase motion resembles small horizontal wavelets rather than camera shake.
+    sx = fade * (0.38*np.sin(yy*0.43 + k*0.17) +
+                 0.22*np.sin(yy*0.16 - k*0.095))
+    sy = fade * 0.14*np.sin(xx*0.075 + yy*0.12 + k*0.11)
+    return cv2.remap(bg, xx + sx, yy + sy, cv2.INTER_LINEAR,
+                     borderMode=cv2.BORDER_REFLECT_101)
+

+def add_distant_glints(im, k):
+    """Short, dim highlights integrated into existing water texture."""
+    layer = np.zeros_like(im)
+    a = np.zeros((H,W), np.uint8)
+    rng = np.random.default_rng(872)
+    for i in range(22):

+        x0 = float(rng.integers(-15, 252))
+        speed = float(rng.uniform(0.12, 0.42))
+        x = int((x0 + speed*k) % 280 - 12)

+        # Highlights breathe with wave phase, never flashing abruptly.
+        op = int(5 + 12*(0.5 + 0.5*math.sin(k*.13 + i*1.71)))
+        cv2.line(layer, (x,y), (x+length,y), (185,194,191), 1, cv2.LINE_AA)
+        cv2.line(a, (x,y), (x+length,y), op, 1, cv2.LINE_AA)
+    af = a.astype(np.float32)[...,None]/255
+    return (im*(1-af)+layer*af).astype(np.uint8)
+

+def draw_ripples(frame, k, front=False):
+    """Elliptical wakes. Back arcs go beneath the bear; front arcs wash over paws."""
+    over = frame.copy()
+    # three gently expanding, phase-offset wake rings
+    for j in range(3):

+    strength = 0.18 if front else 0.13
+    return cv2.addWeighted(over, strength, frame, 1-strength, 0)
+
+
+def add_splashes(frame, k):
+    """Two alternating low wading splashes, timed to the bear's steps."""
+    out = frame.copy()
+    for center, peak, side in [((87,127), 17, -1), ((199,128), 52, 1)]:

+    return cv2.addWeighted(out, .42, frame, .58, 0)
+

+def locally_flex_fish(fg_pm, alpha, k):
+    """A very small elastic flex at the hanging fish, fixed at the bear's mouth."""
+    yy, xx = np.mgrid[0:H,0:W].astype(np.float32)
+    # ellipse around the visible salmon; deformation vanishes at the mouth (101,78)
+    e = ((xx-101)/25.0)**2 + ((yy-91)/24.0)**2
+    w = np.clip(1-e,0,1) * np.clip((yy-77)/25.0,0,1)
+    wiggle = math.sin(k*.26) * 0.72
+    mapx = xx - wiggle*w
+    mapy = yy - 0.22*math.sin(k*.26+1.2)*w

+                  borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    a = cv2.remap(alpha, mapx, mapy, cv2.INTER_LINEAR,
+                  borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+    return f, a
+

+def main():

+    bgr = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)

+    # Premultiplication prevents a dark halo at transformed fur edges.
+    fg_pm = bgr.astype(np.float32) * alpha[...,None]
+
+    frames = []
+    for k in range(N):
+        if k == 0:
+            frames.append(rgb.copy())  # contract: pixel-identical first frame
+            continue
+        p = k/(N-1)
+        # Two slow steps toward the camera: subtle scale, lateral weight shift, head/body bob.
+        gait = math.sin(2*math.pi*k/38.0)
+        lift = -1.18*(math.sin(math.pi*k/19.0)**2)
+        scale = 1.0 + .026*p + .0012*math.sin(2*math.pi*k/38.0)
+        dx = 2.5*p + .48*gait
+        dy = lift
+        M = transform_matrix(scale, dx, dy)
+
+        base = water_background(bg, k)
+        base = add_distant_glints(base, k)
+        base = draw_ripples(base, k, front=False)
+
+        flex_fg, flex_a = locally_flex_fish(fg_pm, alpha, k)
+        wf = cv2.warpAffine(flex_fg, M, (W,H), flags=cv2.INTER_LINEAR,
+                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+        wa = cv2.warpAffine(flex_a, M, (W,H), flags=cv2.INTER_LINEAR,
+                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+        wa3 = np.clip(wa[...,None],0,1)
+        comp = base.astype(np.float32)*(1-wa3) + wf
+        comp = np.clip(comp,0,255).astype(np.uint8)
+
+        comp = draw_ripples(comp, k, front=True)
+        comp = add_splashes(comp, k)
+        frames.append(cv2.cvtColor(comp, cv2.COLOR_BGR2RGB))
+
+    arr = np.stack(frames).astype(np.uint8)
+    artifact_io.save(arr, CASE)
+
+if __name__ == '__main__':
+    main()

    mapy = yy - 0.22*math.sin(k*.26+1.2)*w
    f = cv2.remap(fg_pm, mapx, mapy, cv2.INTER_LINEAR,
                  borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    a = cv2.remap(alpha, mapx, mapy, cv2.INTER_LINEAR,
                  borderMode=cv2.BORDER_CONSTANT, borderValue=0)
    return f, a


def main():
    rgb = np.asarray(Image.open(CASE/'first_frame.png').convert('RGB'))
    bgr = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)
    hard, alpha = isolate_bear(bgr)

    # Remove subject to obtain clean pixels for the slight forward movement.
    hole = cv2.dilate(hard, np.ones((9,9),np.uint8), iterations=1)
    bg = cv2.inpaint(bgr, hole, 5, cv2.INPAINT_TELEA)
    # Premultiplication prevents a dark halo at transformed fur edges.
    fg_pm = bgr.astype(np.float32) * alpha[...,None]

    frames = []
    for k in range(N):
        if k == 0:
            frames.append(rgb.copy())  # contract: pixel-identical first frame
            continue
        p = k/(N-1)
        # Two slow steps toward the camera: subtle scale, lateral weight shift, head/body bob.
        gait = math.sin(2*math.pi*k/38.0)
        lift = -1.18*(math.sin(math.pi*k/19.0)**2)
        scale = 1.0 + .026*p + .0012*math.sin(2*math.pi*k/38.0)
        dx = 2.5*p + .48*gait
        dy = lift
        M = transform_matrix(scale, dx, dy)

        base = water_background(bg, k)
        base = add_distant_glints(base, k)
        base = draw_ripples(base, k, front=False)

        flex_fg, flex_a = locally_flex_fish(fg_pm, alpha, k)
        wf = cv2.warpAffine(flex_fg, M, (W,H), flags=cv2.INTER_LINEAR,
                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
        wa = cv2.warpAffine(flex_a, M, (W,H), flags=cv2.INTER_LINEAR,
                            borderMode=cv2.BORDER_CONSTANT, borderValue=0)
        wa3 = np.clip(wa[...,None],0,1)
        comp = base.astype(np.float32)*(1-wa3) + wf
        comp = np.clip(comp,0,255).astype(np.uint8)

        comp = draw_ripples(comp, k, front=True)
        comp = add_splashes(comp, k)
        frames.append(cv2.cvtColor(comp, cv2.COLOR_BGR2RGB))

    arr = np.stack(frames).astype(np.uint8)
    artifact_io.save(arr, CASE)

if __name__ == '__main__':
    main()
