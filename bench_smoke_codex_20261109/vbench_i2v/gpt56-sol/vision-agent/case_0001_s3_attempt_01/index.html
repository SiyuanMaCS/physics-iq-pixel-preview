+"""Generate a slow leftward camera pan across the supplied liquid artwork.
+
+The only scene source is first_frame.png.  A coherent continuation is made to
+the left of the photograph by bending a reflected strip of the marbling.  The
+camera then samples this larger, static canvas with sub-pixel interpolation;
+this produces true parallax-free camera motion rather than a cross-fade.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+def make_left_continuation(image: np.ndarray, width: int) -> np.ndarray:
+    """Extrapolate the artwork leftward with a smoothly warped reflection.
+
+    Reflection gives an exact color boundary.  Slowly varying coordinate
+    bends break the conspicuous straight mirror symmetry while retaining the
+    long, flowing contours of the liquid paint.
+    """
+    h, w = image.shape[:2]
+    yy, dd = np.mgrid[0:h, width:0:-1].astype(np.float32)
+
+    # dd is distance into the continuation: width at its far left, 1 at seam.
+    ramp = 1.0 - np.exp(-dd / 15.0)
+    # Preserve the edge exactly, then progressively curl the reflected paint.
+    bend_y = ramp * (
+        7.0 * np.sin(dd / 19.0 + yy / 43.0)
+        + 2.4 * np.sin(dd / 7.3 - yy / 27.0)
+    )
+    bend_x = ramp * (
+        2.8 * np.sin(yy / 22.0 + dd / 31.0)
+        + 1.2 * np.sin(yy / 8.5 - dd / 17.0)
+    )
+    map_x = np.clip(dd - 1.0 + bend_x, 0.0, w - 1.001).astype(np.float32)
+    map_y = np.clip(yy + bend_y, 0.0, h - 1.001).astype(np.float32)
+
+    continuation = cv2.remap(
+        image,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+    # In the first few pixels, blend toward a pure reflection.  This makes the
+    # join imperceptible even along the very thin coral-colored contour lines.
+    pure_x = np.clip(dd - 1.0, 0.0, w - 1.001).astype(np.float32)
+    pure = cv2.remap(
+        image,
+        pure_x,
+        yy.astype(np.float32),
+        interpolation=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    seam_weight = np.exp(-dd / 5.0)[..., None]
+    continuation = continuation * (1.0 - seam_weight) + pure * seam_weight
+    return np.clip(continuation + 0.5, 0, 255).astype(np.uint8)
+
+
+def generate(case_dir: Path) -> np.ndarray:
+    image = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"))
+    h, w = image.shape[:2]
+    frame_count = 81
+
+    extension = 96
+    left = make_left_continuation(image, extension)
+    world = np.concatenate([left, image], axis=1)
+
+    # Continue a little past the right edge only as numerical padding. It is
+    # not exposed by this leftward pan.
+    world = cv2.copyMakeBorder(world, 3, 3, 0, 3, cv2.BORDER_REFLECT_101)
+    base_y, base_x = np.mgrid[0:h, 0:w].astype(np.float32)
+    base_y += 3.0
+
+    frames = np.empty((frame_count, h, w, 3), dtype=np.uint8)
+    frames[0] = image  # Contract: byte-for-byte identical first frame.
+
+    total_pan = 68.0
+    # A short, physically natural acceleration into an otherwise steady pan.
+    tau = 0.075
+    normalizer = 1.0 - tau * (1.0 - np.exp(-1.0 / tau))
+    for i in range(1, frame_count):
+        u = i / (frame_count - 1)
+        progress = (u - tau * (1.0 - np.exp(-u / tau))) / normalizer
+        offset = total_pan * progress
+
+        # Moving camera left makes the painted scene travel right on screen.
+        map_x = base_x + extension - offset
+        frame = cv2.remap(
+            world,
+            map_x.astype(np.float32),
+            base_y,
+            interpolation=cv2.INTER_CUBIC,
+            borderMode=cv2.BORDER_REFLECT_101,
+        )
+        frames[i] = frame
+
+    return frames
+
+
+if __name__ == "__main__":
+    root = Path(__file__).resolve().parent
+    artifact_io.save(generate(root), root)
