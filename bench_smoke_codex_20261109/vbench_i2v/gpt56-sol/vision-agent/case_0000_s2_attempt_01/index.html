+"""Generate a gently moving macro shot of marbled blue and orange liquid.
+
+The still is treated as pigment carried by a smooth, nearly incompressible
+two-dimensional flow.  A cumulative back-map is advected through several
+overlapping eddies and the original is sampled only once per output frame;
+this keeps the fine marble filaments substantially sharper than repeatedly
+warping the already-rendered image.
+"""
+
+from pathlib import Path
+import json
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+def edge_window(x: np.ndarray, y: np.ndarray, w: int, h: int) -> np.ndarray:
+    """Softly anchor the outermost image border to prevent exposed edges."""
+    dx = np.minimum(x, (w - 1) - x)
+    dy = np.minimum(y, (h - 1) - y)
+    # Zero at the border, smoothly reaching full motion 13 pixels inward.
+    wx = np.sin(np.clip(dx / 13.0, 0.0, 1.0) * np.pi / 2.0) ** 2
+    wy = np.sin(np.clip(dy / 13.0, 0.0, 1.0) * np.pi / 2.0) ** 2
+    return wx * wy
+
+
+def velocity(x: np.ndarray, y: np.ndarray, t: float, w: int, h: int):
+    """A time-varying, multi-scale velocity field in pixels per frame."""
+    vx = np.zeros_like(x, dtype=np.float32)
+    vy = np.zeros_like(y, dtype=np.float32)
+
+    # (centre x/y, radius, angular velocity, temporal phase).  The centres
+    # follow the visible pools and curls in the supplied marble image.
+    eddies = (
+        (202.0, 113.0, 53.0, -0.0084, 0.2),  # broad orange whirl, lower-right
+        (190.0, 20.0, 34.0, 0.0090, 1.5),    # curled orange ribbon at the top
+        (115.0, 102.0, 58.0, 0.0049, 2.8),   # deep-blue central basin
+        (40.0, 53.0, 39.0, -0.0046, 4.1),    # pale turbulent current at left
+        (147.0, 66.0, 31.0, -0.0040, 0.8),   # small counter-current at interface
+    )
+    for cx, cy, radius, omega, phase in eddies:
+        dx = x - cx
+        dy = y - cy
+        r2 = dx * dx + dy * dy
+        envelope = np.exp(-0.5 * r2 / (radius * radius)).astype(np.float32)
+        pulse = 1.0 + 0.14 * np.sin(2.0 * np.pi * t / 96.0 + phase)
+        vx += (-dy * omega * pulse) * envelope
+        vy += ( dx * omega * pulse) * envelope
+
+    # Long, very gentle currents stop the motion from looking like isolated
+    # rotating discs.  Their phases evolve slowly over the full shot.
+    phase = 2.0 * np.pi * t / 118.0
+    vx += 0.075 * np.sin(2.0 * np.pi * y / h + phase)
+    vx += 0.038 * np.sin(4.0 * np.pi * y / h - 0.65 * phase)
+    vy += 0.045 * np.sin(2.0 * np.pi * x / w - 0.52 * phase)
+
+    taper = edge_window(x, y, w, h).astype(np.float32)
+    return vx * taper, vy * taper
+
+
+def generate(case_dir: Path) -> np.ndarray:
+    spec = json.loads((case_dir / "task_spec.json").read_text())
+    n = int(spec["frames"])
+    h = int(spec["height"])
+    w = int(spec["width"])
+
+    source = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"),
+                        dtype=np.uint8)
+    if source.shape != (h, w, 3):
+        raise ValueError(f"Unexpected source shape: {source.shape}")
+
+    yy, xx = np.mgrid[0:h, 0:w].astype(np.float32)
+    # map_x/map_y tell where in frame zero each current output pixel originated.
+    map_x = xx.copy()
+    map_y = yy.copy()
+    frames = np.empty((n, h, w, 3), dtype=np.uint8)
+    frames[0] = source  # required to be pixel-identical
+
+    for frame_index in range(1, n):
+        t = frame_index - 0.5
+        vx, vy = velocity(xx, yy, t, w, h)
+
+        # Semi-Lagrangian transport of the cumulative coordinate map.  A pair
+        # of half steps gives stable curved trajectories without visible jitter.
+        for sub in (0.25, 0.75):
+            st = (frame_index - 1) + sub
+            svx, svy = velocity(xx, yy, st, w, h)
+            sx = (xx - 0.5 * svx).astype(np.float32)
+            sy = (yy - 0.5 * svy).astype(np.float32)
+            map_x = cv2.remap(map_x, sx, sy, cv2.INTER_LINEAR,
+                              borderMode=cv2.BORDER_REFLECT_101)
+            map_y = cv2.remap(map_y, sx, sy, cv2.INTER_LINEAR,
+                              borderMode=cv2.BORDER_REFLECT_101)
+
+        rendered = cv2.remap(source, map_x, map_y, cv2.INTER_CUBIC,
+                             borderMode=cv2.BORDER_REFLECT_101)
+
+        # Sub-pixel caustic breathing, derived from the flow rather than random
+        # noise.  It is deliberately tiny so the richly coloured source remains
+        # natural and temporally stable.
+        speed = np.sqrt(vx * vx + vy * vy)
+        sheen = 0.75 * np.sin(0.075 * xx + 0.052 * yy + frame_index * 0.035)
+        sheen *= np.clip(speed / 0.45, 0.0, 1.0)
+        rendered = np.clip(rendered.astype(np.float32) + sheen[..., None],
+                           0, 255).astype(np.uint8)
+        frames[frame_index] = rendered
+
+    return frames
+
+
+if __name__ == "__main__":
+    root = Path(__file__).resolve().parent
+    result = generate(root)
+    artifact_io.save(result, root)
+
