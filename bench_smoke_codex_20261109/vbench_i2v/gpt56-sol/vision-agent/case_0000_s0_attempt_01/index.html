+"""Generate a short, seamless-looking motion study from the supplied marble frame.
+
+The source already resembles a close macro shot of two immiscible liquids.  The
+animation therefore uses a stack of smooth inverse coordinate transforms: local
+vortices make the colored pools curl, while broad low-amplitude waves carry the
+blue channel across the shot.  Every frame is sampled directly from the original
+image (rather than from the preceding frame), so fine marbling remains sharp.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 256, 144
+N_FRAMES = 81
+
+
+def smoothstep(x: float) -> float:
+    return x * x * (3.0 - 2.0 * x)
+
+
+def twist(x, y, cx, cy, radius, angle):
+    """Inverse-map a soft local rotation, fading to zero at its perimeter."""
+    dx = x - cx
+    dy = y - cy
+    r2 = dx * dx + dy * dy
+    # Gaussian falloff has no visible ring at the edge of a vortex.
+    a = angle * np.exp(-r2 / (2.0 * radius * radius))
+    ca = np.cos(a)
+    sa = np.sin(a)
+    return cx + ca * dx - sa * dy, cy + sa * dx + ca * dy
+
+
+def make_frames(source: np.ndarray) -> np.ndarray:
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    frames = np.empty((N_FRAMES, H, W, 3), dtype=np.uint8)
+    frames[0] = source  # contractual exact first frame
+
+    # A blurred luminance field is used only for a very faint moving reflection.
+    # It follows broad liquid forms and does not introduce foreign structures.
+    lum = cv2.cvtColor(source, cv2.COLOR_RGB2GRAY).astype(np.float32) / 255.0
+    lum = cv2.GaussianBlur(lum, (0, 0), 7.0)
+
+    for i in range(1, N_FRAMES):
+        p = i / (N_FRAMES - 1)
+        ease = smoothstep(p)
+        # Keep the motion alive throughout while avoiding an abrupt initial jump.
+        travel = 0.84 * p + 0.16 * ease
+
+        # Gentle macro-camera creep: 1.5% push-in and a tiny diagonal slide.
+        scale = 1.0 - 0.015 * ease
+        x = (xx - 127.5) * scale + 127.5 - 1.25 * ease
+        y = (yy - 71.5) * scale + 71.5 + 0.70 * ease
+
+        # Broad currents first.  Their envelopes keep the outer border nearly
+        # stationary and prevent conspicuous reflected seams.
+        ex = np.sin(np.pi * xx / (W - 1))
+        ey = np.sin(np.pi * yy / (H - 1))
+        x += (3.25 * travel * np.sin((yy - 24.0) * 0.038 + 0.55 * p)
+              + 0.75 * np.sin(yy * 0.085 - 1.7 * p)) * ex
+        y += (1.65 * travel * np.sin(xx * 0.031 - yy * 0.012 - 0.8 * p)
+              + 0.35 * np.sin(xx * 0.073 + 1.1 * p)) * ey
+
+        # Local counter-rotating eddies align with the principal pools in the
+        # photograph: amber at the top and lower right, coral at left, and the
+        # large navy basin in the middle.
+        wobble = np.sin(np.pi * p) * np.sin(2.0 * np.pi * p)
+        x, y = twist(x, y, 194.0 + 1.5 * np.sin(np.pi * p), 14.0,
+                     39.0, -0.60 * travel - 0.035 * wobble)
+        x, y = twist(x, y, 205.0 - 2.0 * np.sin(np.pi * p), 124.0,
+                     54.0, 0.53 * travel + 0.040 * wobble)
+        x, y = twist(x, y, 19.0, 76.0 + 1.5 * np.sin(np.pi * p),
+                     35.0, -0.38 * travel)
+        x, y = twist(x, y, 121.0, 89.0, 73.0,
+                     0.19 * travel + 0.020 * np.sin(2.0 * np.pi * p))
+
+        # A final fine undulation bends the thin cream/coral filaments without
+        # making the whole frame look rubbery.
+        x += 0.60 * np.sin(yy * 0.105 + xx * 0.015 + 2.1 * p) * np.sin(np.pi * p)
+        y += 0.42 * np.sin(xx * 0.092 - yy * 0.018 - 1.6 * p) * np.sin(np.pi * p)
+
+        warped = cv2.remap(
+            source,
+            x.astype(np.float32),
+            y.astype(np.float32),
+            interpolation=cv2.INTER_CUBIC,
+            borderMode=cv2.BORDER_REFLECT_101,
+        ).astype(np.float32)
+
+        # Restrained caustic shimmer (under one percent), strongest in pale
+        # swirls.  It gently changes phase over the five-second shot.
+        wave = np.sin(xx * 0.035 + yy * 0.050 - 2.4 * p)
+        shimmer = 1.0 + 0.0055 * np.sin(np.pi * p) * wave * (0.25 + 0.75 * lum)
+        warped *= shimmer[..., None]
+        frames[i] = np.clip(warped, 0, 255).astype(np.uint8)
+
+    return frames
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    source = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    if source.shape != (H, W, 3):
+        raise ValueError(f"expected {(H, W, 3)}, got {source.shape}")
+    frames = make_frames(source)
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
