+"""Generate a coherent animation of the supplied blue/orange marbled liquid.
+
+The still is treated as pigment carried by a smooth, nearly incompressible
+two-dimensional flow.  We advect a coordinate map rather than repeatedly
+resampling the RGB image, which keeps the many hairline details crisp.
+"""
+
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 256, 144
+N_FRAMES, FPS = 81, 16.0
+
+
+def bilinear(field, x, y):
+    """Bilinearly sample an HxW or HxWxC array at floating coordinates."""
+    x = np.clip(x, 0.0, W - 1.0001)
+    y = np.clip(y, 0.0, H - 1.0001)
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = np.minimum(x0 + 1, W - 1)
+    y1 = np.minimum(y0 + 1, H - 1)
+    fx = x - x0
+    fy = y - y0
+    if field.ndim == 3:
+        fx = fx[..., None]
+        fy = fy[..., None]
+    return ((1.0 - fy) * ((1.0 - fx) * field[y0, x0] + fx * field[y0, x1])
+            + fy * ((1.0 - fx) * field[y1, x0] + fx * field[y1, x1]))
+
+
+def velocity(x, y, t):
+    """A smooth liquid-like velocity field in pixels per second."""
+    u = np.zeros_like(x, dtype=np.float32)
+    v = np.zeros_like(y, dtype=np.float32)
+
+    # Local eddies are positioned on the visible curls in the painting.  The
+    # centers wander by a few pixels so the result does not resemble rigid
+    # rotation of a flat photograph.
+    eddies = (
+        # cx, cy, angular speed, Gaussian radius, phase
+        (204.0, 116.0,  0.255, 54.0, 0.1),  # broad lower-right orange curl
+        (119.0,  73.0, -0.125, 78.0, 1.7),  # large central blue circulation
+        (169.0,  17.0,  0.185, 37.0, 3.1),  # upper ochre curl
+        ( 24.0,  75.0, -0.205, 34.0, 4.4),  # left orange filament
+        ( 79.0,  18.0,  0.105, 43.0, 2.3),  # pale upper-left pool
+    )
+    pulse = 0.91 + 0.09 * np.sin(0.74 * t + 0.25)
+    for cx, cy, omega, radius, phase in eddies:
+        cx = cx + 2.0 * np.sin(0.43 * t + phase)
+        cy = cy + 1.5 * np.cos(0.37 * t + phase * 0.8)
+        dx, dy = x - cx, y - cy
+        falloff = np.exp(-(dx * dx + dy * dy) / (2.0 * radius * radius))
+        u += (-omega * dy) * falloff * pulse
+        v += ( omega * dx) * falloff * pulse
+
+    # Long, low-amplitude currents bend the interfaces instead of merely
+    # spinning them.  These are curls of smooth stream functions.
+    ax = 2.0 * np.pi * (x / W)
+    ay = 2.0 * np.pi * (y / H)
+    phase = 0.34 * t
+    u += 1.70 * np.cos(ay * 1.08 + phase) * np.sin(ax * 0.70 - 0.18 * t)
+    v += -1.10 * np.cos(ax * 0.70 - 0.18 * t) * np.sin(ay * 1.08 + phase)
+    u += 0.62 * np.sin(2.0 * ay - 0.28 * t)
+    v += 0.48 * np.cos(1.45 * ax + 0.22 * t)
+
+    # Ease the flow to zero at the image boundary.  This avoids exposed edges
+    # while retaining strong movement throughout the important central area.
+    edge = (np.sin(np.pi * np.clip(x, 0, W - 1) / (W - 1))
+            * np.sin(np.pi * np.clip(y, 0, H - 1) / (H - 1)))
+    edge = np.sqrt(np.maximum(edge, 0.0)).astype(np.float32)
+    return u * edge, v * edge
+
+
+def render():
+    source_u8 = np.asarray(Image.open("first_frame.png").convert("RGB"), dtype=np.uint8)
+    if source_u8.shape != (H, W, 3):
+        raise ValueError(f"unexpected input shape: {source_u8.shape}")
+    source = source_u8.astype(np.float32)
+
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    # map_x/map_y answer: which point in the original pigment occupies this
+    # point now?  They are themselves advected with two midpoint substeps.
+    map_x = xx.copy()
+    map_y = yy.copy()
+    frames = np.empty((N_FRAMES, H, W, 3), dtype=np.uint8)
+    frames[0] = source_u8  # contract: byte-for-byte identical first frame
+
+    substeps = 2
+    dt = 1.0 / (FPS * substeps)
+    elapsed = 0.0
+    for frame_idx in range(1, N_FRAMES):
+        for _ in range(substeps):
+            tm = elapsed + 0.5 * dt
+            # Midpoint backtrace of the current Eulerian pixel grid.
+            u0, v0 = velocity(xx, yy, tm)
+            hx = xx - 0.5 * dt * u0
+            hy = yy - 0.5 * dt * v0
+            um, vm = velocity(hx, hy, tm)
+            prev_x = xx - dt * um
+            prev_y = yy - dt * vm
+            map_x = bilinear(map_x, prev_x, prev_y).astype(np.float32)
+            map_y = bilinear(map_y, prev_x, prev_y).astype(np.float32)
+            elapsed += dt
+
+        # A restrained macro push-in makes the close-up feel alive without
+        # competing with the pigment flow.  It starts at the identity.
+        t = frame_idx / FPS
+        scale = 1.0 + 0.0035 * t
+        pan_x = 0.45 * (1.0 - np.cos(0.58 * t))
+        pan_y = -0.30 * (1.0 - np.cos(0.47 * t))
+        qx = (xx - 127.5) / scale + 127.5 + pan_x
+        qy = (yy - 71.5) / scale + 71.5 + pan_y
+        sx = bilinear(map_x, qx, qy)
+        sy = bilinear(map_y, qx, qy)
+        image = bilinear(source, sx, sy)
+
+        # Very slight moving illumination, strongest on already pale paint,
+        # gives the surface a wet sheen.  The amplitude remains below 2%.
+        lum = image[..., 0] * 0.2126 + image[..., 1] * 0.7152 + image[..., 2] * 0.0722
+        pale = np.clip((lum - 95.0) / 120.0, 0.0, 1.0)
+        sheen = (0.009 * np.sin(0.035 * xx - 0.052 * yy + 0.82 * t)
+                 + 0.005 * np.sin(0.071 * xx + 0.029 * yy - 0.61 * t))
+        ramp = min(1.0, t / 0.7)
+        image *= 1.0 + (ramp * pale * sheen)[..., None]
+        frames[frame_idx] = np.clip(np.rint(image), 0, 255).astype(np.uint8)
+
+    return frames
+
+
+if __name__ == "__main__":
+    case_dir = Path(__file__).resolve().parent
+    frames = render()
+    artifact_io.save(frames, case_dir)
+    print(f"saved {len(frames)} RGB frames at {W}x{H}")
+
