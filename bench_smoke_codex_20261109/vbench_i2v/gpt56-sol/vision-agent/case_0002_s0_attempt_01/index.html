+"""Generate a smooth rightward pan across the supplied liquid-marbling image.
+
+The source is already a convincing macro photograph/painting.  Rather than
+animating the paint itself, this script builds a static, wider piece of the
+same marbling and moves a virtual camera over it.  The added area is made by a
+seam-constrained, nonlinear reflection of the source.  The nonlinear map is
+important: it preserves all of the tiny pigment detail while preventing the
+new area from looking like a simple mirror tile.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+HERE = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 144
+FRAME_COUNT = 81
+
+
+def remap_rgb(image, map_x, map_y, interpolation=cv2.INTER_CUBIC):
+    """cv2.remap wrapper which keeps RGB channel order and float precision."""
+    return cv2.remap(
+        image,
+        map_x.astype(np.float32),
+        map_y.astype(np.float32),
+        interpolation,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+
+
+def make_panorama(source):
+    """Extend the marbling to the right with a coherent nonlinear reflection."""
+    extension = 96
+    yy, qq = np.mgrid[0:HEIGHT, 1 : extension + 1].astype(np.float32)
+
+    # q -> u turns gently at the edge, then travels back through the richly
+    # detailed right side of the source.  Its zero-ish initial slope makes the
+    # join substantially less conspicuous than an ordinary mirror padding.
+    retreat = 0.11 * qq + 0.0099 * qq * qq
+    envelope = qq / extension
+
+    # Curl the reflected material.  Every displacement vanishes at the join,
+    # but farther right the pigment streams bend at different spatial scales.
+    curl_y = envelope * (
+        8.0 * np.sin(yy / 25.0 + qq / 30.0)
+        + 3.0 * np.sin(yy / 9.5 - qq / 17.0)
+        - 2.0 * np.sin(qq / 20.0)
+    )
+    curl_x = envelope * (
+        2.2 * np.sin(yy / 19.0 + qq / 23.0)
+        + 1.1 * np.sin(yy / 7.0 - qq / 15.0)
+    )
+    map_x = (WIDTH - 1) - retreat + curl_x
+    map_y = yy + curl_y
+
+    src = source.astype(np.float32)
+    detailed = remap_rgb(src, map_x, map_y)
+
+    # Blend a broad-scale version of the same coordinate map underneath the
+    # sharp pigment detail.  This suppresses any high-contrast crease at the
+    # original edge without making the generated material look blurry.
+    low = cv2.GaussianBlur(src, (0, 0), 3.2, borderType=cv2.BORDER_REFLECT_101)
+    low_warped = remap_rgb(low, map_x, map_y)
+    detail = detailed - remap_rgb(
+        cv2.GaussianBlur(src, (0, 0), 1.25, borderType=cv2.BORDER_REFLECT_101),
+        map_x,
+        map_y,
+    )
+    generated = low_warped + detail
+
+    # Keep the first portion maximally faithful to the real edge, then very
+    # gradually introduce the de-correlated variant.  This is a spatial blend
+    # on a static canvas, so the camera motion remains globally rigid.
+    direct_x = (WIDTH - 1) - retreat
+    direct_y = yy + envelope * 2.0 * np.sin(yy / 31.0)
+    direct = remap_rgb(src, direct_x, direct_y)
+    blend = np.clip((qq - 5.0) / 25.0, 0.0, 1.0)[..., None]
+    generated = direct * (1.0 - blend) + generated * blend
+
+    # A tiny low-frequency exposure variation is characteristic of the broad
+    # swirls already in the image and helps avoid a repeated-tile impression.
+    gain = 1.0 + envelope[..., None] * (
+        0.018 * np.sin(yy[..., None] / 37.0 + qq[..., None] / 41.0)
+    )
+    generated = np.clip(generated * gain, 0, 255)
+
+    panorama = np.empty((HEIGHT, WIDTH + extension, 3), np.float32)
+    panorama[:, :WIDTH] = src
+    panorama[:, WIDTH:] = generated
+    return panorama
+
+
+def render_frames(source, panorama):
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = source  # strict contract: byte/pixel-identical first frame
+
+    grid_y, grid_x = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+    for index in range(1, FRAME_COUNT):
+        t = index / (FRAME_COUNT - 1)
+
+        # 72 pixels in 5 seconds: a measured macro-camera pan.  A very mild
+        # sinusoidal velocity modulation avoids robotic motion while retaining
+        # a clearly rightward camera move at every instant.
+        pan_x = 72.0 * (t - 0.075 * np.sin(2.0 * np.pi * t) / (2.0 * np.pi))
+
+        # Sub-pixel interpolation gives stable 16 fps motion.  The minuscule
+        # vertical arc feels hand-operated, but always returns to level.
+        pan_y = 0.65 * np.sin(np.pi * t)
+        image = remap_rgb(panorama, grid_x + pan_x, grid_y + pan_y)
+        frames[index] = np.clip(np.rint(image), 0, 255).astype(np.uint8)
+    return frames
+
+
+def main():
+    source = np.asarray(Image.open(HERE / "first_frame.png").convert("RGB"))
+    assert source.shape == (HEIGHT, WIDTH, 3)
+    panorama = make_panorama(source)
+    frames = render_frames(source, panorama)
+    artifact_io.save(frames, HERE)
+
+
+if __name__ == "__main__":
+    main()
