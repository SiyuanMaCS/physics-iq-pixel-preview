+"""Generate a gently evolving close-up of blue and orange marbled liquid.
+
+The animation is made entirely by advecting the supplied first frame through a
+smooth, incompressible-looking velocity field.  Broad vortices bend the large
+colour plumes, while two very small wave fields suggest motion on the liquid's
+surface.  No external assets are used.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 256, 144
+N_FRAMES = 81
+FPS = 16.0
+
+
+def velocity(x: np.ndarray, y: np.ndarray, t: float):
+    """A smooth velocity field in pixels/second.
+
+    Gaussian rotational fields are approximately divergence free and work
+    especially well for the existing curls in the painting.  Their centres
+    drift by a fraction of a pixel per second so the motion never feels like a
+    rigid global twirl.
+    """
+    u = np.full_like(x, 0.20, dtype=np.float32)
+    v = np.full_like(y, -0.035, dtype=np.float32)
+
+    # (centre x, centre y, radius, angular velocity, x drift, y drift, phase)
+    vortices = (
+        (111.0, 79.0, 72.0, -0.024, 0.42, -0.10, 0.2),
+        (216.0, 119.0, 47.0, -0.047, -0.24, -0.20, 1.8),
+        (191.0, 23.0, 36.0, 0.049, 0.17, 0.09, 2.7),
+        (64.0, 22.0, 44.0, 0.033, -0.10, 0.17, 4.0),
+        (23.0, 105.0, 31.0, 0.045, 0.14, -0.12, 5.1),
+    )
+    for cx, cy, radius, omega, drift_x, drift_y, phase in vortices:
+        cx += drift_x * t + 1.1 * np.sin(0.31 * t + phase)
+        cy += drift_y * t + 0.8 * np.cos(0.27 * t + phase)
+        dx = x - cx
+        dy = y - cy
+        g = np.exp(-(dx * dx + dy * dy) / (2.0 * radius * radius))
+        pulse = 1.0 + 0.10 * np.sin(0.55 * t + phase)
+        u += (-omega * pulse) * dy * g
+        v += (omega * pulse) * dx * g
+
+    # A pair of broad stream-function waves: curl-like flow without visible
+    # straight translation or abrupt changes at the edges.
+    p1 = 0.025 * x + 0.043 * y + 0.34 * t
+    p2 = 0.018 * x - 0.051 * y - 0.27 * t + 1.3
+    u += 0.42 * np.cos(p1) + 0.29 * np.cos(p2)
+    v += -0.25 * np.cos(p1) + 0.18 * np.cos(p2)
+    return u, v
+
+
+def reflect_coordinates(a: np.ndarray, limit: int):
+    """Reflect floating-point coordinates into [0, limit-1]."""
+    period = 2.0 * (limit - 1)
+    a = np.mod(a, period)
+    return np.where(a <= limit - 1, a, period - a).astype(np.float32)
+
+
+def generate():
+    case_dir = Path(__file__).resolve().parent
+    first = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"),
+                       dtype=np.uint8)
+    if first.shape != (H, W, 3):
+        raise ValueError(f"unexpected first-frame shape: {first.shape}")
+
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    map_x = xx.copy()
+    map_y = yy.copy()
+    frames = np.empty((N_FRAMES, H, W, 3), dtype=np.uint8)
+    frames[0] = first  # Contract: byte/pixel-identical first frame.
+
+    dt = 1.0 / FPS
+    for index in range(1, N_FRAMES):
+        t0 = (index - 1) * dt
+
+        # Midpoint integration of the backward characteristic.  Integrating
+        # texture coordinates backwards gives each output pixel its source in
+        # frame zero and avoids repeated resampling blur.
+        u0, v0 = velocity(map_x, map_y, t0)
+        mx = map_x - 0.5 * dt * u0
+        my = map_y - 0.5 * dt * v0
+        um, vm = velocity(mx, my, t0 + 0.5 * dt)
+        map_x -= dt * um
+        map_y -= dt * vm
+
+        t = index * dt
+        # Sub-pixel capillary undulations.  Subtracting the t=0 phase keeps the
+        # initial map exactly at identity; amplitude remains deliberately low.
+        wave_x = 0.46 * (
+            np.sin(0.071 * yy + 0.029 * xx + 0.73 * t)
+            - np.sin(0.071 * yy + 0.029 * xx)
+        )
+        wave_y = 0.36 * (
+            np.sin(0.052 * xx - 0.037 * yy - 0.59 * t + 0.8)
+            - np.sin(0.052 * xx - 0.037 * yy + 0.8)
+        )
+        render_x = reflect_coordinates(map_x + wave_x, W)
+        render_y = reflect_coordinates(map_y + wave_y, H)
+
+        warped = cv2.remap(
+            first,
+            render_x,
+            render_y,
+            interpolation=cv2.INTER_CUBIC,
+            borderMode=cv2.BORDER_REFLECT_101,
+        )
+
+        # Extremely restrained travelling highlight: it gives the glossy pale
+        # regions a changing sheen without repainting the original colours.
+        luminance = (0.2126 * warped[..., 0] + 0.7152 * warped[..., 1]
+                     + 0.0722 * warped[..., 2]) / 255.0
+        sheen = np.sin(0.031 * xx + 0.062 * yy - 0.92 * t)
+        sheen -= np.sin(0.031 * xx + 0.062 * yy)
+        delta = (1.45 * sheen * np.clip((luminance - 0.46) * 2.0, 0.0, 1.0))[..., None]
+        frames[index] = np.clip(warped.astype(np.float32) + delta, 0, 255).astype(np.uint8)
+
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    generate()
