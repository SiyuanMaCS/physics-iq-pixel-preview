+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 256, 144
+N, FPS = 81, 16.0
+
+
+def reflect_coordinates(a, limit):
+    """Reflect coordinates at image boundaries without introducing hard seams."""
+    period = 2.0 * (limit - 1)
+    a = np.mod(a, period)
+    return np.where(a <= limit - 1, a, period - a)
+
+
+def sample_bilinear(field, x, y):
+    """Bilinear sampling for either an HxW or HxWxC numpy array."""
+    x = reflect_coordinates(x, field.shape[1])
+    y = reflect_coordinates(y, field.shape[0])
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = np.minimum(x0 + 1, field.shape[1] - 1)
+    y1 = np.minimum(y0 + 1, field.shape[0] - 1)
+    fx = x - x0
+    fy = y - y0
+
+    if field.ndim == 3:
+        fx = fx[..., None]
+        fy = fy[..., None]
+    a = field[y0, x0] * (1.0 - fx) + field[y0, x1] * fx
+    b = field[y1, x0] * (1.0 - fx) + field[y1, x1] * fx
+    return a * (1.0 - fy) + b * fy
+
+
+def velocity(x, y, t):
+    """A smooth, mostly divergence-free flow tailored to the marble swirls."""
+    # A quiet broad drift keeps the central navy pool flowing as one body.
+    vx = 0.38 + 0.12 * np.sin(y * (2.0 * np.pi / 73.0) - 0.45 * t)
+    vy = 0.08 * np.sin(x * (2.0 * np.pi / 101.0) + 0.35 * t)
+
+    # (center x, center y, radius, angular speed).  Opposing rotations are
+    # characteristic of oil/paint marbling and avoid a rigid global spin.
+    vortices = (
+        (199.0, 18.0, 39.0, 0.125),   # gold curl at the top
+        (207.0, 132.0, 66.0, -0.067), # broad orange curl at lower right
+        (247.0, 78.0, 34.0, -0.115),  # small gold island at right
+        (8.0, 73.0, 43.0, 0.100),     # coral/blue fold entering at left
+        (78.0, 10.0, 63.0, 0.044),    # pale blue eddies, upper left
+        (132.0, 29.0, 48.0, -0.046),  # blue crest through upper middle
+    )
+    for i, (cx0, cy0, radius, omega) in enumerate(vortices):
+        # Sub-pixel center wander makes the liquid feel organic, not geared.
+        cx = cx0 + 1.2 * np.sin(0.31 * t + i * 1.37)
+        cy = cy0 + 0.8 * np.cos(0.27 * t + i * 0.91)
+        dx = x - cx
+        dy = y - cy
+        falloff = np.exp(-(dx * dx + dy * dy) / (radius * radius))
+        pulse = 1.0 + 0.07 * np.sin(0.48 * t + i * 0.83)
+        spin = omega * pulse * falloff
+        vx += -dy * spin
+        vy += dx * spin
+
+    # Very gentle long-wave undulation, well below the speed of the curls.
+    vx += 0.11 * np.sin((x + 1.7 * y) * (2.0 * np.pi / 137.0) - 0.22 * t)
+    vy += 0.10 * np.sin((1.2 * x - y) * (2.0 * np.pi / 119.0) + 0.25 * t)
+    return vx, vy
+
+
+def make_frames(first):
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    # Coordinates in the original image carried by the flowing material.
+    map_x = xx.copy()
+    map_y = yy.copy()
+    original = first.astype(np.float32)
+    frames = np.empty((N, H, W, 3), dtype=np.uint8)
+    frames[0] = first
+
+    dt = 1.0 / FPS
+    for frame_index in range(1, N):
+        t = (frame_index - 0.5) * dt
+
+        # Midpoint backtrace of the output grid, followed by semi-Lagrangian
+        # transport of the original-coordinate map. This preserves fine paint
+        # detail because pixels are always sampled from frame zero only once.
+        v0x, v0y = velocity(xx, yy, t)
+        mid_x = xx - 0.5 * dt * v0x
+        mid_y = yy - 0.5 * dt * v0y
+        vmx, vmy = velocity(mid_x, mid_y, t - 0.5 * dt)
+        departure_x = xx - dt * vmx
+        departure_y = yy - dt * vmy
+        map_x = sample_bilinear(map_x, departure_x, departure_y).astype(np.float32)
+        map_y = sample_bilinear(map_y, departure_x, departure_y).astype(np.float32)
+
+        rgb = sample_bilinear(original, map_x, map_y)
+
+        # Subtle travelling glints, strongest only on already-light paint.
+        # Their low amplitude gives wet depth without recoloring the artwork.
+        phase = 0.035 * map_x + 0.061 * map_y - 0.82 * frame_index * dt
+        phase2 = -0.052 * map_x + 0.028 * map_y + 0.47 * frame_index * dt
+        wave = 0.58 * np.sin(phase) + 0.42 * np.sin(phase2)
+        luma = rgb[..., 0] * 0.2126 + rgb[..., 1] * 0.7152 + rgb[..., 2] * 0.0722
+        sheen = (1.0 - np.exp(-frame_index / 4.0)) * wave * np.clip((luma - 75.0) / 145.0, 0.0, 1.0)
+        rgb += sheen[..., None] * np.array([2.3, 2.8, 3.1], dtype=np.float32)
+
+        frames[frame_index] = np.clip(np.rint(rgb), 0, 255).astype(np.uint8)
+    return frames
+
+
+def main():
+    case_dir = Path(__file__).resolve().parent
+    first = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    if first.shape != (H, W, 3):
+        raise ValueError(f"unexpected first-frame shape: {first.shape}")
+    frames = make_frames(first)
+    assert np.array_equal(frames[0], first)
+    artifact_io.save(frames, case_dir)
+
+
+if __name__ == "__main__":
+    main()
