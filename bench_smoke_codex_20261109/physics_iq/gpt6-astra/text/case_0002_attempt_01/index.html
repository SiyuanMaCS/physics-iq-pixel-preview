    ease = u*u*(3-2*u)
    theta = .09+(1.49-.09)*ease
    if s > .47:
        theta += .024*np.sin((s-.47)*19)*np.exp(-(s-.47)*8)
    indent = 2.1*(1-np.exp(-s*10)) + 2.0*np.exp(-((s-.045)/.038)**2)
    # Small final pulse when the broad face meets the pillow.
    indent += .55*np.exp(-((s-.42)/.075)**2)
    pivot_x = BLOCK_X+.8*block_hit+block_halfwidth
    pivot_y = block_ground+indent
    x = pivot_x-block_halfwidth*np.cos(theta)+block_halfheight*np.sin(theta)
    y = pivot_y-block_halfwidth*np.sin(theta)-block_halfheight*np.cos(theta)
    return x,y,theta,indent


def remap_layer(color, alpha, center, state):
    x,y,theta,_ = state
    cs, sn = np.cos(theta), np.sin(theta)
    dx, dy = X-x, Y-y
    sx = center[0]+cs*dx+sn*dy
    sy = center[1]-sn*dx+cs*dy
    a = map_coordinates(alpha,[sy,sx],order=1,mode='constant',cval=0)
    premul = color*alpha[:,:,None]
    rgb = np.stack([map_coordinates(premul[:,:,c],[sy,sx],order=1,mode='constant',cval=0) for c in range(3)],-1)
    return rgb,a


def pillow_background(t):
    b = ball_state(t)
    k = block_state(t)
    sb, sk = max(0,t-ball_hit), max(0,t-block_hit)
    depth_b = b[3] if t >= ball_hit else 0
    depth_k = k[3] if t >= block_hit else 0
    # A smoothly localized warp bends the existing pillow texture downward.
    gb = np.exp(-.5*((X-74.0)/15.5)**2-.5*((Y-107.5)/10.0)**2)
    gk = np.exp(-.5*((X-214.0)/17.5)**2-.5*((Y-108.5)/10.0)**2)
    pillow_gate = np.clip((Y-88)/9,0,1)
    displacement = pillow_gate*(depth_b*gb+depth_k*.75*gk)
    coords = [Y-displacement, X]
    out = np.stack([map_coordinates(clean[:,:,c],coords,order=1,mode='nearest') for c in range(3)],-1)
    # Gentle transient crease shading accompanies compression, not rigid table motion.
    fabric_shadow = pillow_gate*(.045*depth_b*gb+.028*depth_k*gk)
    out *= (1-fabric_shadow)[:,:,None]
    return out


def shadow(out, x, y, rx, ry, strength):
    sh = np.exp(-.5*((X-x)/rx)**2-.5*((Y-y)/ry)**2)*strength
    return out*(1-sh[:,:,None])


def render(t):
    b, k = ball_state(t),block_state(t)
    out = pillow_background(t)
    # Shadows sharpen naturally as the objects approach the cushions.
    bh = max(0,ball_ground-(b[1]+ball_radius))
    kh = max(0,block_ground-(k[1]+block_halfheight)) if t<block_hit else 0
    bs = .28*np.exp(-bh/29)
    ks = .23*np.exp(-kh/29)
    out = shadow(out,b[0]+1,ball_ground+1.0+b[3]*.65,8.5+bh*.055,1.8+bh*.025,bs)
    out = shadow(out,k[0],block_ground+1.0+k[3]*.65,5.0+7.0*abs(np.sin(k[2]))+kh*.04,1.65+kh*.023,ks)
    # Short exposure integration produces correct directional motion blur at
    # 4 fps without changing the fixed-camera background.
    exposure = 1/80
    samples = 7
    times = np.linspace(max(0,t-exposure/2),t+exposure/2,samples)
    for color,rest_alpha,init_alpha,center,trajectory in [
        (ball_color,ball_rest_alpha,ball_original_alpha,(BALL_X,BALL_Y),ball_state),
        (block_color,block_rest_alpha,block_original_alpha,(BLOCK_X,BLOCK_Y),block_state)]:
        rgb_sum = np.zeros_like(out)
        alpha_sum = np.zeros((H,W))
        # The brief initial softness gives way to the stationary object edge.
        for tt in times:
            p,a = remap_layer(color,rest_alpha,center,trajectory(tt))
            rgb_sum += p/samples
            alpha_sum += a/samples
        out = out*(1-alpha_sum[:,:,None])+rgb_sum
    return np.clip(np.rint(out),0,255).astype(np.uint8)


def main():
    frames = np.empty((20,H,W,3),np.uint8)
    frames[0] = original
    for i in range(1,20):
        frames[i] = render(i/4)
    save(frames,ROOT)
    # Numeric checks also ensure the exact initial image is retained.
    assert np.array_equal(frames[0],original)
    assert frames.shape == (20,144,256,3)
    print('Saved 20 RGB frames at 256x144, 4 fps.')
    print('Impact times (s):',round(ball_hit,3),round(block_hit,3))
    print('Ball final:',np.round(ball_state(4.75),2),'Block final:',np.round(block_state(4.75),2))

if __name__ == '__main__':
    main()
