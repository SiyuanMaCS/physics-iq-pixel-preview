+import cv2
+import numpy as np
+from PIL import Image

+import artifact_io

+ROOT = Path(__file__).resolve().parent
+WIDTH = 256
+HEIGHT = 144
+FRAME_COUNT = 20
+FPS = 4

+def _smoothstep(edge0, edge1, value):
+    value = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+

+def _ball_patch(cx, cy, radius, roll):
+    """Render an anti-aliased, textured tennis ball into a local RGBA patch."""
+    scale = 4
+    pad = 2.0
+    x0 = max(0, int(np.floor(cx - radius - pad)))
+    y0 = max(0, int(np.floor(cy - radius - pad)))
+    x1 = min(WIDTH, int(np.ceil(cx + radius + pad)))
+    y1 = min(HEIGHT, int(np.ceil(cy + radius + pad)))
+
+    hx = (x1 - x0) * scale
+    hy = (y1 - y0) * scale
+    xs = x0 + (np.arange(hx, dtype=np.float32) + 0.5) / scale
+    ys = y0 + (np.arange(hy, dtype=np.float32) + 0.5) / scale
+    xx, yy = np.meshgrid(xs, ys)
+
+    u = (xx - cx) / radius
+    v = (yy - cy) / radius
+    rr = u * u + v * v
+    inside = rr <= 1.0
+    z = np.sqrt(np.maximum(0.0, 1.0 - rr))
+
+    # The ball rolls toward screen-right, so rotate its texture about the
+    # screen-vertical axis and keep the seam moving with the surface.
+    ca, sa = np.cos(roll), np.sin(roll)
+    tx = u * ca + z * sa
+    tz = -u * sa + z * ca
+    ty = v
+
+    lon = np.arctan2(tx, tz)
+    lat = np.arcsin(np.clip(ty, -1.0, 1.0))
+
+    # Two softly curving tennis seams. The secondary term keeps the seam from
+    # reading as a printed straight stripe.
+    seam_wave = np.sin(2.0 * lon + 0.62 * np.sin(2.4 * lat) + 0.22)
+    seam_core = 1.0 - _smoothstep(0.055, 0.115, np.abs(seam_wave))
+    seam_edge = _smoothstep(0.055, 0.095, np.abs(seam_wave)) * (
+        1.0 - _smoothstep(0.095, 0.145, np.abs(seam_wave))
+    )
+
+    # Fine nap texture in ball-local coordinates. Several incommensurate
+    # frequencies avoid a regular cloth or golf-ball pattern.
+    nap = (
+        0.48 * np.sin(31.0 * tx + 19.0 * ty - 23.0 * tz)
+        + 0.31 * np.sin(47.0 * ty - 29.0 * tz + 8.0 * tx)
+        + 0.21 * np.sin(71.0 * (tx + tz) + 37.0 * ty)
+    )
+    nap = np.clip(nap, -1.0, 1.0)
+
+    light = np.array([-0.48, -0.58, 0.66], dtype=np.float32)
+    light /= np.linalg.norm(light)
+    nx, ny, nz = u, v, z
+    diffuse = np.maximum(nx * light[0] + ny * light[1] + nz * light[2], 0.0)
+
+    # Soft key light plus a little warm bounce from the tabletop.
+    shade = 0.43 + 0.67 * diffuse + 0.075 * (1.0 - nz)
+    shade *= 1.0 - 0.10 * _smoothstep(0.72, 1.0, rr)
+
+    base = np.empty((hy, hx, 3), dtype=np.float32)
+    base[..., 0] = 154.0
+    base[..., 1] = 155.0
+    base[..., 2] = 150.0
+
+    fuzz = nap * (3.6 + 2.2 * (1.0 - nz))
+    base += fuzz[..., None]
+    base[..., 0] += 4.0 * (1.0 - nz) + 1.8 * diffuse
+    base[..., 1] += 2.3 * (1.0 - nz) + 1.4 * diffuse
+    base[..., 2] -= 2.2 * (1.0 - nz)
+
+    seam_color = np.array([191.0, 192.0, 184.0], dtype=np.float32)
+    base = base * (1.0 - 0.78 * seam_core[..., None])
+    base += seam_color * (0.78 * seam_core[..., None])
+    base -= 18.0 * seam_edge[..., None]
+
+    color = base * shade[..., None]
+
+    half = light + np.array([0.0, 0.0, 1.0], dtype=np.float32)
+    half /= np.linalg.norm(half)
+    specular = np.maximum(nx * half[0] + ny * half[1] + nz * half[2], 0.0) ** 34
+    color += (29.0 * specular)[..., None]
+
+    # A narrow, slightly uneven silhouette keeps the sphere from looking like
+    # a clean vector shape.
+    theta = np.arctan2(v, u)
+    silhouette = radius * (1.0 + 0.012 * np.sin(17.0 * theta + 0.7))
+    dist = np.sqrt(rr) * radius
+    coverage = _smoothstep(silhouette + 0.32, silhouette - 0.32, dist)
+    coverage *= inside
+
+    rgba = np.dstack((np.clip(color, 0.0, 255.0), coverage * 255.0)).astype(np.uint8)
+    rgba = cv2.resize(rgba, (x1 - x0, y1 - y0), interpolation=cv2.INTER_AREA)
+    return rgba, (x0, y0, x1, y1)
+

+def _composite_ball(frame, cx, cy, radius, roll):
+    patch, (x0, y0, x1, y1) = _ball_patch(cx, cy, radius, roll)
+    rgb = patch[..., :3].astype(np.float32)
+    alpha = patch[..., 3:4].astype(np.float32) / 255.0
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    frame[y0:y1, x0:x1] = np.clip(
+        rgb * alpha + dst * (1.0 - alpha) + 0.5, 0.0, 255.0
+    ).astype(np.uint8)
+
+
+def _shadow_mask(cx, cy, radius):
+    mask = np.zeros((HEIGHT, WIDTH), dtype=np.float32)
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    ex = 1.48 * radius
+    ey = 0.38 * radius
+    sx = cx + 0.30 * radius
+    sy = cy + 0.84 * radius
+    d = ((xx - sx) / ex) ** 2 + ((yy - sy) / ey) ** 2
+    mask += 0.27 * np.exp(-1.7 * d)
+
+    # A tighter ambient-occlusion core just beneath the contact point.
+    d2 = ((xx - cx) / (0.72 * radius)) ** 2 + (
+        (yy - (cy + radius * 0.92)) / (0.18 * radius)
+    ) ** 2
+    mask += 0.20 * np.exp(-2.2 * d2)
+    mask = cv2.GaussianBlur(mask, (0, 0), 1.35)
+    return np.clip(mask, 0.0, 0.42)
+
+
+def _apply_shadow(frame, cx, cy, radius):
+    shadow = _shadow_mask(cx, cy, radius)
+    table = np.zeros((HEIGHT, WIDTH), dtype=bool)
+    table[63:116, :221] = True
+    shadow *= table
+    frame[:] = np.clip(frame.astype(np.float32) * (1.0 - shadow[..., None]), 0, 255).astype(
+        np.uint8
+    )
+
+
+def _pipe_foreground(base):
+    """Return a conservative mask for the dark pipe edge at the spawn point."""
+    lum = base.astype(np.float32).mean(axis=2)
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    region = (
+        (xx < 73)
+        & (yy > 28)
+        & (yy < 82)
+        & (lum < 83)
+        & ((xx + 0.30 * yy) < 91)
+    )
+    mask = region.astype(np.uint8)
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    mask = cv2.GaussianBlur(mask * 255, (0, 0), 0.65)
+    return mask.astype(np.float32) / 255.0
+
+
+def generate():
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    if first.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"Unexpected first frame shape: {first.shape}")
+
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first
+    pipe_mask = _pipe_foreground(first)[..., None]
+
+    start_time = 0.24
+    travel_time = 4.66
+    x_start, x_end = 64.5, 210.5
+    y_start, y_end = 71.8, 83.1
+    r_start, r_end = 9.1, 10.4
+
+    for index in range(1, FRAME_COUNT):
+        t = index / FPS
+        progress = np.clip((t - start_time) / travel_time, 0.0, 1.0)
+
+        # A short acceleration out of the pipe, then an almost constant roll
+        # with a small amount of felt friction near the far end.
+        eased = 0.12 * progress + 0.88 * (
+            0.5 - 0.5 * np.cos(np.pi * progress)
+        ) ** 0.64
+        eased = np.clip(eased, 0.0, 1.0)
+
+        cx = x_start + (x_end - x_start) * eased
+        cy = y_start + (y_end - y_start) * eased
+        radius = r_start + (r_end - r_start) * eased
+        roll = 5.55 * (cx - x_start) / max(radius, 1.0)
+
+        frame = first.copy()
+        _apply_shadow(frame, cx, cy, radius)
+        _composite_ball(frame, cx, cy, radius, roll)
+
+        # Preserve the real pipe lip in front of the ball for the first
+        # fraction of a second, so the ball visibly emerges rather than fades.
+        if cx < 79.0:
+            frame = (
+                pipe_mask * first.astype(np.float32)
+                + (1.0 - pipe_mask) * frame.astype(np.float32)
+            ).astype(np.uint8)
+
+        frames[index] = frame
+
+    artifact_io.save(frames, ROOT)
+
+    # Small local QA sheet for numeric inspection without relying on image tools.
+    strip = np.concatenate([frames[i] for i in (0, 2, 6, 10, 14, 19)], axis=1)
+    Image.fromarray(strip).save(ROOT / ".tmp" / "motion_strip.png")
+    return frames
+
+
+if __name__ == "__main__":
+    generate()
