+def smoothstep(edge0, edge1, value):
+    t = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)

+def make_clean_plate(source):
+    """Remove the held ball by continuing the smooth wall behind it."""
+    clean = source.astype(np.float32).copy()
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]

+    x_left, x_right = 104, 166
+    t = (np.arange(x_left, x_right + 1) - x_left) / (x_right - x_left)
+    left = source[:, x_left].astype(np.float32)
+    right = source[:, x_right].astype(np.float32)
+    fill = (
+        left[:, None, :] * (1.0 - t[None, :, None])
+        + right[:, None, :] * t[None, :, None]
+    )

+    radius = np.hypot(xx - BALL_CX, yy - BALL_CY)

+    alpha[(yy < 18) | (yy > 63)] = 0.0

+def make_sand_layer(source):
+    """Return the visible sand pile as an RGBA layer for local deformation."""
+    x0, y0, x1, y1 = 70, 74, 202, 132
+    crop = source[y0:y1, x0:x1].astype(np.float32)

+    green = (

+def transform_sprite(rgba, angle, scale_x=1.0, scale_y=1.0):
+    h, w = rgba.shape[:2]
+    cx, cy = (w - 1) * 0.5, (h - 1) * 0.5
+    theta = np.deg2rad(angle)
+    c, s = np.cos(theta), np.sin(theta)
+    m = np.array(
+        [
+            [
+                scale_x * c,
+                -scale_y * s,
+                cx - scale_x * c * cx + scale_y * s * cy,

+            [
+                scale_x * s,
+                scale_y * c,
+                cy - scale_x * s * cx - scale_y * c * cy,
+            ],

+        rgba,
