+"""Generate the rolling-tennis-ball demonstration from the supplied first frame."""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 144
+FRAME_COUNT = 20
+
+
+def smoothstep(edge0, edge1, values):
+    x = np.clip((values - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def make_background(first_frame):
+    """Remove the already-visible ball while retaining the pipe and table."""
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    hole = (((xx - 62.5) / 10.2) ** 2 + ((yy - 44.5) / 9.2) ** 2) <= 1.0
+    mask = hole.astype(np.uint8) * 255
+    return cv2.inpaint(first_frame, mask, 5, cv2.INPAINT_TELEA)
+
+
+def ball_rgba(radius_x=8.8, radius_y=8.0, phase=0.0):
+    """A small, antialiased tennis ball with directional shading and seams."""
+    pad = 3
+    cx = radius_x + pad
+    cy = radius_y + pad
+    height = int(np.ceil(2 * cy + 1))
+    width = int(np.ceil(2 * cx + 1))
+    yy, xx = np.mgrid[:height, :width].astype(np.float32)
+    nx = (xx - cx) / radius_x
+    ny = (yy - cy) / radius_y
+    rho2 = nx * nx + ny * ny
+    coverage = 1.0 - smoothstep(0.91, 1.01, np.sqrt(rho2))
+    inside = np.maximum(1.0 - rho2, 0.0)
+    nz = np.sqrt(inside)
+
+    # Key light from upper left, with a cooler, dimmer lower edge.
+    light = np.clip(0.50 + 0.42 * nz - 0.22 * nx - 0.12 * ny, 0.0, 1.0)
+    rim = np.clip((-0.52 * nx - 0.28 * ny) * (1.0 - nz), 0.0, 1.0)
+    base = np.empty((height, width, 3), dtype=np.float32)
+    base[..., 0] = 116.0 + 49.0 * light + 12.0 * rim
+    base[..., 1] = 146.0 + 62.0 * light + 10.0 * rim
+    base[..., 2] = 38.0 + 24.0 * light + 20.0 * rim
+
+    # Felt grain: restrained, deterministic texture so the tiny ball stays photographic.
+    grain = (
+        1.7 * np.sin(2.7 * xx + 1.3 * yy + phase * 3.0)
+        + 1.2 * np.sin(5.1 * xx - 2.2 * yy + phase * 5.0)
+    )
+    base += grain[..., None] * np.array([0.55, 0.62, 0.35], dtype=np.float32)
+
+    # Two pale curved seams, rotating as the ball rolls.
+    seam = np.zeros_like(nx)
+    for offset in (0.0, np.pi):
+        curve = 0.48 * np.sin(2.05 * nx + phase + offset)
+        distance = np.abs(ny - curve)
+        visible = (np.abs(nx) < 0.91) & (nz > 0.10)
+        seam = np.maximum(seam, (1.0 - smoothstep(0.045, 0.105, distance)) * visible)
+    seam_color = np.array([214.0, 222.0, 174.0], dtype=np.float32)
+    seam_mix = (0.44 * seam * (0.45 + 0.55 * light))[..., None]
+    base = base * (1.0 - seam_mix) + seam_color * seam_mix
+
+    # Slightly warm, soft outline catches the table lighting.
+    edge_darken = smoothstep(0.76, 1.0, np.sqrt(rho2))[..., None]
+    base *= 1.0 - 0.12 * edge_darken
+    alpha = coverage
+    rgba = np.dstack([np.clip(base, 0, 255), alpha])
+    return rgba.astype(np.float32)
+
+
+def ramp_surface_y(x):
+    return 75.0 - 0.335 * (x - 123.0)
+
+
+def trajectory():
+    """Ball centers in pixel coordinates for the twenty output frames."""
+    centers = np.zeros((FRAME_COUNT, 2), dtype=np.float32)
+    # Straight exit from the pipe.
+    for i in range(10):
+        t = i / 9.0
+        centers[i, 0] = 62.5 + 111.0 * t
+        centers[i, 1] = 44.2 + 10.2 * t * t
+
+    # Short elastic rebound at the raised right end, then a roll down the cardboard.
+    centers[10] = (176.0, 55.4)
+    centers[11] = (181.5, 50.4)
+    centers[12] = (178.0, 51.3)
+    centers[13] = (165.0, 53.8)
+    centers[14] = (152.0, 57.7)
+    centers[15] = (140.0, 61.7)
+    centers[16] = (128.0, 65.5)
+    centers[17] = (118.5, 68.0)
+    centers[18] = (111.0, 70.5)
+    centers[19] = (106.0, 71.6)
+    return centers
+
+
+def blend_ellipse(frame, cx, cy, rx, ry, opacity):
+    """Composite a soft elliptical shadow."""
+    x0 = max(0, int(np.floor(cx - rx - 2)))
+    x1 = min(WIDTH, int(np.ceil(cx + rx + 2)))
+    y0 = max(0, int(np.floor(cy - ry - 2)))
+    y1 = min(HEIGHT, int(np.ceil(cy + ry + 2)))
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    q = ((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2
+    weight = (1.0 - smoothstep(0.20, 1.0, q)) * opacity
+    shade = np.array([55.0, 45.0, 34.0], dtype=np.float32)
+    patch = frame[y0:y1, x0:x1].astype(np.float32)
+    frame[y0:y1, x0:x1] = (
+        patch * (1.0 - weight[..., None]) + shade * weight[..., None]
+    ).clip(0, 255).astype(np.uint8)
+
+
+def composite_sprite(frame, rgba, center_x, center_y):
+    h, w = rgba.shape[:2]
+    x0 = int(np.floor(center_x - (w - 1) / 2.0))
+    y0 = int(np.floor(center_y - (h - 1) / 2.0))
+    x1, y1 = x0 + w, y0 + h
+    sx0, sy0 = max(0, -x0), max(0, -y0)
+    sx1 = w - max(0, x1 - WIDTH)
+    sy1 = h - max(0, y1 - HEIGHT)
+    x0, y0 = max(0, x0), max(0, y0)
+    x1, y1 = min(WIDTH, x1), min(HEIGHT, y1)
+    sprite = rgba[sy0:sy1, sx0:sx1]
+    alpha = sprite[..., 3:4]
+    dst = frame[y0:y1, x0:x1].astype(np.float32)
+    frame[y0:y1, x0:x1] = (
+        dst * (1.0 - alpha) + sprite[..., :3] * alpha
+    ).clip(0, 255).astype(np.uint8)
+
+
+def render():
+    first = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    if first.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"Unexpected first-frame shape: {first.shape}")
+
+    background = make_background(first)
+    centers = trajectory()
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first
+
+    for i in range(1, FRAME_COUNT):
+        frame = background.copy()
+        cx, cy = centers[i]
+
+        # Broad table shadow while airborne, tightening into a contact shadow on the ramp.
+        if i <= 9:
+            blend_ellipse(frame, cx + 1.5, 82.0, 11.0, 3.0, 0.09)
+        elif i <= 16:
+            surface_y = ramp_surface_y(cx)
+            blend_ellipse(frame, cx + 1.2, surface_y + 0.6, 10.5, 2.4, 0.23)
+        else:
+            blend_ellipse(frame, cx + 1.0, 78.8, 10.0, 2.8, 0.19)
+
+        roll = (cx - centers[0, 0]) / 8.2
+        sprite = ball_rgba(phase=roll)
+        composite_sprite(frame, sprite, cx, cy)
+        frames[i] = frame
+
+    return frames
+
+
+def main():
+    frames = render()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {len(frames)} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
