+ROOT = Path(__file__).resolve().parent
+WIDTH = 256
+HEIGHT = 144
+FRAMES = 20
+FPS = 4

+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def fruit_masks(image):
+    """Return the two orange lobes without including the pale central blade."""
+    hsv = cv2.cvtColor(image, cv2.COLOR_RGB2HSV)
+    hue, saturation, value = cv2.split(hsv)
+
+    allowed = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    cv2.ellipse(allowed, (126, 65), (36, 35), 0, 0, 360, 1, -1)
+    color = (
+        (hue < 32)
+        & (saturation > 42)
+        & (value > 58)
+        & (allowed > 0)
+    ).astype(np.uint8)
+
+    # Bridge small highlights inside the peel, then retain the main fruit body.
+    color = cv2.morphologyEx(
+        color, cv2.MORPH_CLOSE, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
+    )
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(color, 8)
+    if count > 1:
+        main = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+        color = (labels == main).astype(np.uint8)
+
+    contours, _ = cv2.findContours(color, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
+    filled = np.zeros_like(color)
+    if contours:
+        cv2.drawContours(filled, [max(contours, key=cv2.contourArea)], -1, 1, -1)
+    filled = cv2.dilate(
+        filled, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3)), iterations=1
+    )
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    cut_x = 127.7 + 0.025 * (yy - 65.0)
+    left = (filled > 0) & (xx < cut_x)
+    right = (filled > 0) & (xx >= cut_x)
+    return left, right
+
+
+def clean_plate(image, fruit):
+    """Inpaint the fruit while retaining the existing blade and board texture."""
+    mask = cv2.dilate(
+        fruit.astype(np.uint8),
+        cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5)),
+        iterations=1,
+    )
+
+    # Keep the central steel strip from the source frame.
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    cut_x = 127.7 + 0.025 * (yy - 65.0)
+    mask[(np.abs(xx - cut_x) < 2.1) & (yy > 18) & (yy < 92)] = 0
+
+    return cv2.inpaint(image, mask, 4, cv2.INPAINT_TELEA)
+
+
+def shifted_layer(image, mask, dx, dy):
+    source = np.dstack((image, (mask.astype(np.uint8) * 255)))
+    matrix = np.float32([[1.0, 0.0, dx], [0.0, 1.0, dy]])
+    return cv2.warpAffine(
+        source,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+

+def composite(frame, layer, amount=1.0):
+    alpha = (layer[:, :, 3:4].astype(np.float32) / 255.0) * amount
+    rgb = layer[:, :, :3].astype(np.float32)
+    frame[:] = np.clip(
+        rgb * alpha + frame.astype(np.float32) * (1.0 - alpha), 0.0, 255.0
+    ).astype(np.uint8)
+

+def add_cut_surfaces(frame, left, right, separation, progress):
+    """Add a fine pith rim and a wet inner edge where the lobes separate."""
+    if progress <= 0.01:
+        return
+
+    left_shift = int(round(-separation * 0.95))
+    right_shift = int(round(separation))
+    cream = np.array([232, 203, 159], dtype=np.float32)
+    warm = np.array([247, 174, 82], dtype=np.float32)
+
+    for y in range(38, 91):
+        left_xs = np.flatnonzero(left[y])
+        right_xs = np.flatnonzero(right[y])
+        if left_xs.size:
+            x = int(left_xs.max() + left_shift)
+            if 0 <= x < WIDTH - 1:
+                t = smoothstep((y - 38.0) / 52.0)
+                shade = 0.70 + 0.30 * np.sin(np.pi * t)
+                col = cream * shade + warm * (1.0 - shade)
+                frame[y, x] = np.clip(0.72 * frame[y, x] + 0.28 * col, 0, 255)
+                if 45 <= y <= 84:
+                    frame[y, min(x + 1, WIDTH - 1)] = np.clip(
+                        0.88 * frame[y, min(x + 1, WIDTH - 1)] + 0.12 * col, 0, 255
+                    )
+
+        if right_xs.size:
+            x = int(right_xs.min() + right_shift)
+            if 1 <= x < WIDTH:
+                t = smoothstep((y - 38.0) / 52.0)
+                shade = 0.72 + 0.28 * np.sin(np.pi * t)
+                col = cream * shade + warm * (1.0 - shade)
+                frame[y, x] = np.clip(0.72 * frame[y, x] + 0.28 * col, 0, 255)
+                if 45 <= y <= 84:
+                    frame[y, max(x - 1, 0)] = np.clip(
+                        0.88 * frame[y, max(x - 1, 0)] + 0.12 * col, 0, 255
+                    )

+def add_juice_and_shadow(frame, separation, progress, frame_index):
+    if progress <= 0.01:
+        return
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+
+    # Soft contact shadow below the opening seam.
+    shadow_alpha = 0.16 * smoothstep(progress)
+    shadow = np.exp(
+        -0.5
+        * (
+            ((xx - 127.5) / (3.5 + 2.0 * progress)) ** 2
+            + ((yy - 91.0) / 3.0) ** 2
+        )
+    )
+    dark = np.array([74, 58, 49], dtype=np.float32)
+    frame[:] = np.clip(
+        frame.astype(np.float32) * (1.0 - shadow_alpha * shadow[:, :, None])
+        + dark * (shadow_alpha * shadow[:, :, None]),
+        0,
+        255,
+    ).astype(np.uint8)
+
+    # Juice is deliberately sparse at this scale: a bead, a short trail, and glints.
+    bead_y = 83.0 + 2.8 * smoothstep(progress) + 0.35 * np.sin(frame_index * 0.8)
+    bead_x = 127.2 + 0.28 * np.sin(frame_index * 0.55)
+    cv2.circle(frame, (int(round(bead_x)), int(round(bead_y))), 2, (151, 60, 13), -1)
+    cv2.circle(frame, (int(round(bead_x)), int(round(bead_y - 1))), 1, (255, 186, 92), -1)
+
+    if progress > 0.45:
+        cv2.line(
+            frame,
+            (int(round(bead_x)), int(round(bead_y + 2))),
+            (int(round(bead_x + 0.4)), int(round(bead_y + 4))),
+            (142, 61, 22),
+            1,
+            cv2.LINE_AA,
+        )
+
+    # Tiny highlights along the two freshly opened faces.
+    for offset, y in ((-1, 58), (1, 66), (-1, 75)):
+        x = int(round(127.5 + offset * (1.7 + separation * 0.35)))
+        if 0 <= x < WIDTH:
+            frame[y, x] = np.clip(
+                frame[y, x].astype(np.float32) + np.array([22, 20, 12]), 0, 255
+            ).astype(np.uint8)
+
+
+def add_blade_glint(frame, progress, frame_index):
+    if progress <= 0.01:
+        return
+
+    # The knife is already present in the source. A moving specular line makes the
+    # cut feel active without replacing the photographed blade.
+    x0 = 131.0 + 0.22 * np.sin(frame_index * 0.7)
+    y0 = 34.0 + 3.0 * progress
+    y1 = 82.0 + 5.0 * progress
+    for y in range(int(y0), int(y1)):
+        t = (y - y0) / max(1.0, y1 - y0)
+        x = x0 + 0.7 * t
+        strength = 22.0 * np.sin(np.pi * t) * (0.55 + 0.45 * progress)
+        ix = int(round(x))
+        if 0 <= ix < WIDTH:
+            frame[y, ix] = np.clip(
+                frame[y, ix].astype(np.float32) + np.array([strength, strength, strength * 1.05]),
+                0,
+                255,
+            ).astype(np.uint8)
+
+
+def render():
+    image = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB")).copy()
+    if image.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"unexpected first-frame shape: {image.shape}")
+
+    left, right = fruit_masks(image)
+    fruit = left | right
+    background = clean_plate(image, fruit)
+
+    frames = []
+    for index in range(FRAMES):
+        if index == 0:
+            frames.append(image.copy())
+            continue
+
+        t = (index - 1) / (FRAMES - 2)
+        progress = smoothstep(t / 0.74)
+        settle = max(0.0, (t - 0.74) / 0.26)
+        separation = 3.35 * progress - 0.22 * np.sin(np.pi * settle)
+        if index == FRAMES - 1:
+            separation += 0.06
+
+        frame = background.copy()
+        left_layer = shifted_layer(image, left, -0.95 * separation, -0.12 * progress)
+        right_layer = shifted_layer(image, right, 1.00 * separation, 0.16 * progress)
+        composite(frame, left_layer)
+        composite(frame, right_layer)
+
+        add_cut_surfaces(frame, left, right, separation, progress)
+        add_juice_and_shadow(frame, separation, progress, index)
+        add_blade_glint(frame, progress, index)
+        frames.append(frame)
+
+    frames = np.stack(frames).astype(np.uint8)
+    artifact_io.save(frames, ROOT)
+    return frames
+
+
+if __name__ == "__main__":
+    render()
