+"""Generate a short, locked-off demonstration of a kettlebell crushing a cup."""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+WIDTH = 256
+HEIGHT = 144
+FRAMES = 20
+FPS = 4
+
+
+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def make_bell_alpha(image):
+    """Recover the dark kettlebell silhouette with a soft antialiased edge."""
+    gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY).astype(np.float32)
+    saturation = image.max(axis=2).astype(np.float32) - image.min(axis=2)
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    roi = (xx >= 88) & (xx <= 181) & (yy <= 84)
+
+    # The bell is nearly neutral and much darker than the wall/table around it.
+    core = (gray < 112) & ((saturation < 22) | (gray < 68)) & roi
+    core = cv2.morphologyEx(
+        core.astype(np.uint8), cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8)
+    )
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(core, 8)
+    main = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+    core = (labels == main).astype(np.uint8)
+
+    # Include the lighter rim while excluding the open space inside the handle.
+    near = cv2.dilate(core, np.ones((7, 7), np.uint8))
+    edge_alpha = np.clip((137.0 - gray) / 38.0, 0.0, 1.0)
+    alpha = edge_alpha * near
+    alpha = np.maximum(alpha, core.astype(np.float32))
+    alpha[85:] = 0.0
+
+    # Slightly expand the silhouette to keep the original dark fringe attached.
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.55)
+    alpha[85:] = 0.0
+    return np.clip(alpha, 0.0, 1.0)
+
+
+def make_cup_alpha(image):
+    """Recover the bright cup, including its shaded and transparent-looking sides."""
+    gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY).astype(np.float32)
+    saturation = image.max(axis=2).astype(np.float32) - image.min(axis=2)
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    roi = (xx >= 119) & (xx <= 169) & (yy >= 79) & (yy <= 103)
+
+    bright = (gray > 126) & (saturation < 58) & roi
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(
+        bright.astype(np.uint8), 8
+    )
+    main = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
+    bright = labels == main
+
+    # Fill the rim interior and keep the full tapered cup silhouette.
+    points = np.column_stack(np.where(bright)[::-1]).astype(np.int32)
+    hull = cv2.convexHull(points)
+    alpha = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.fillPoly(alpha, [hull], 255)
+    alpha = alpha.astype(np.float32) / 255.0
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.45)
+    alpha[103:] = 0.0
+    return np.clip(alpha, 0.0, 1.0)
+
+
+def make_plate(image, bell_alpha, cup_alpha):
+    """Inpaint the two movable objects while retaining the photographed set."""
+    combined = np.maximum(bell_alpha, cup_alpha)
+    mask = (combined > 0.025).astype(np.uint8) * 255
+    mask = cv2.dilate(mask, np.ones((5, 5), np.uint8))
+    bgr = cv2.cvtColor(image, cv2.COLOR_RGB2BGR)
+    filled = cv2.inpaint(bgr, mask, 4.0, cv2.INPAINT_TELEA)
+    filled = cv2.cvtColor(filled, cv2.COLOR_BGR2RGB)
+
+    # Give the large removed body a smooth continuation, then restore texture
+    # from nearby pixels so the plate does not look like a flat patch.
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    bell_region = (yy < 86) & (xx > 82) & (xx < 188)
+    blurred = cv2.GaussianBlur(filled, (0, 0), 6.0)
+    weights = cv2.GaussianBlur(
+        (mask > 0).astype(np.float32), (0, 0), 5.0
+    )
+    weights = np.clip(weights * 1.35, 0.0, 1.0)
+    blend = filled.astype(np.float32)
+    blend = blend * (1.0 - weights[..., None]) + blurred.astype(np.float32) * weights[
+        ..., None
+    ]
+    blend[~bell_region] = filled[~bell_region]
+    return np.clip(blend, 0, 255).astype(np.uint8)
+
+
+def shift_alpha(alpha, dx, dy):
+    matrix = np.float32([[1, 0, dx], [0, 1, dy]])
+    return cv2.warpAffine(
+        alpha,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+
+
+def transform_rgba(rgb, alpha, dx, dy, angle=0.0, scale=1.0):
+    center = (145.5, 52.0)
+    matrix = cv2.getRotationMatrix2D(center, angle, scale)
+    matrix[0, 2] += dx
+    matrix[1, 2] += dy
+    warped_rgb = cv2.warpAffine(
+        rgb,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    )
+    warped_alpha = cv2.warpAffine(
+        alpha,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    return warped_rgb, np.clip(warped_alpha, 0, 1)
+
+
+def cup_motion(progress):
+    """Return cup top height and rigid tilt for the current progress."""
+    # A short hold gives the setup time to register before the foam yields.
+    p = smoothstep((progress - 0.15) / 0.80)
+    compression = 0.72 * p
+    tilt = -2.2 * p
+    return compression, tilt
+
+
+def render_cup(image, cup_alpha, compression, tilt):
+    """Warp the original cup into a widening, buckling foam silhouette."""
+    cup_height = 21.0 * (1.0 - compression)
+    top_y = 101.0 - cup_height
+    base_y = 101.0
+
+    out = np.zeros((HEIGHT, WIDTH, 3), np.float32)
+    out_alpha = np.zeros((HEIGHT, WIDTH), np.float32)
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    inside_y = (yy >= top_y - 1.0) & (yy <= base_y + 1.0)
+    inside_x = (xx >= 118) & (xx <= 174)
+    region = inside_y & inside_x
+
+    # Map the compressed cup back into the source photograph.
+    t = np.clip((yy - top_y) / max(cup_height, 1.0), 0.0, 1.0)
+    source_y = 81.0 + t * 20.0
+    top_center = 145.5 + tilt * (1.0 - t)
+    center = 145.5 + tilt * (1.0 - t) * 0.35
+    width_factor = 1.0 + 0.30 * smoothstep(compression / 0.72)
+    half_width = (18.4 + 1.8 * t) * width_factor
+    source_half_width = 18.4 + 1.8 * t
+
+    # Irregular folds make the foam buckle rather than shrink like elastic.
+    pleat = (
+        0.75 * np.sin(t * 7.2 + 0.5)
+        + 0.40 * np.sin(t * 12.7 - 0.7)
+    ) * smoothstep(compression / 0.55)
+    boundary_ripple = 0.45 * np.sin(t * 11.0 + 1.2) * smoothstep(
+        compression / 0.60
+    )
+    x_norm = (xx - center) / np.maximum(half_width, 0.1)
+    source_x = (
+        145.5
+        + x_norm * source_half_width
+        + pleat * (0.45 + 0.55 * t)
+        + boundary_ripple * x_norm
+    )
+
+    map_x = source_x.astype(np.float32)
+    map_y = source_y.astype(np.float32)
+    sampled = cv2.remap(
+        image,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    sampled_alpha = cv2.remap(
+        cup_alpha,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+
+    # A narrow shadowed underside appears as the cup is crushed.
+    contact = smoothstep(compression / 0.55)
+    shade = 1.0 - 0.10 * contact * (0.45 + 0.55 * t)
+    sampled = sampled.astype(np.float32) * shade[..., None]
+    out[region] = sampled[region]
+    out_alpha[region] = np.clip(sampled_alpha[region], 0.0, 1.0)
+
+    return out.astype(np.uint8), out_alpha
+
+
+def alpha_composite(base, layer, alpha):
+    a = alpha[..., None]
+    result = layer.astype(np.float32) * a + base.astype(np.float32) * (1.0 - a)
+    return np.clip(result, 0, 255).astype(np.uint8)
+
+
+def add_contact_shadow(frame, center_x, center_y, strength, width):
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    shadow = np.exp(
+        -(
+            ((xx - center_x) / max(width, 1.0)) ** 2
+            + ((yy - center_y) / 3.1) ** 2
+        )
+    )
+    shadow *= strength
+    out = frame.astype(np.float32)
+    out *= (1.0 - 0.24 * shadow)[..., None]
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def add_sensor_grain(frame, rng, strength):
+    """A very small luminance-only noise field keeps the locked shot photographic."""
+    noise = rng.normal(0.0, strength, (HEIGHT, WIDTH, 1))
+    out = frame.astype(np.float32) + noise
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def generate(case_dir):
+    case_dir = Path(case_dir)
+    first = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"))
+    assert first.shape == (HEIGHT, WIDTH, 3)
+
+    bell_alpha = make_bell_alpha(first)
+    cup_alpha = make_cup_alpha(first)
+    plate = make_plate(first, bell_alpha, cup_alpha)
+    rng = np.random.default_rng(191)
+
+    frames = []
+    for index in range(FRAMES):
+        progress = index / (FRAMES - 1)
+        compression, tilt = cup_motion(progress)
+        yield_amount = 0.72 * smoothstep((progress - 0.15) / 0.80)
+
+        frame = plate.copy()
+
+        # The cup stays planted on the table and crushes under the fixed load.
+        cup_rgb, cup_a = render_cup(first, cup_alpha, compression, tilt)
+        frame = alpha_composite(frame, cup_rgb, cup_a)
+
+        # The rigid bell follows the cup's top while acquiring a small tilt.
+        bell_drop = 0.72 * 21.0 * smoothstep((progress - 0.15) / 0.80)
+        bell_tilt = -1.55 * smoothstep((progress - 0.22) / 0.70)
+        bell_rgb, bell_a = transform_rgba(
+            first,
+            bell_alpha,
+            dx=1.6 * smoothstep((progress - 0.22) / 0.70),
+            dy=bell_drop,
+            angle=bell_tilt,
+            scale=1.0,
+        )
+
+        # A soft cast shadow on the tabletop follows the bell base.
+        frame = add_contact_shadow(
+            frame,
+            center_x=145.0 + 1.2 * yield_amount,
+            center_y=82.0 + bell_drop,
+            strength=0.38 + 0.28 * yield_amount,
+            width=27.0 + 5.0 * yield_amount,
+        )
+        frame = alpha_composite(frame, bell_rgb, bell_a)
+
+        # Locked camera, but a whisper of sensor noise avoids plastic stillness.
+        if index > 0:
+            frame = add_sensor_grain(frame, rng, 0.42)
+
+        # The supplied frame is the exact starting observation.
+        if index == 0:
+            frame = first.copy()
+        frames.append(frame)
+
+    return np.stack(frames)
+
+
+if __name__ == "__main__":
+    case_dir = Path(__file__).resolve().parent
+    frames = generate(case_dir)
+    artifact_io.save(frames, case_dir)
