+from pathlib import Path

+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+WIDTH, HEIGHT, FRAME_COUNT = 256, 144, 20
+
+
+def polygon_mask(points, shape, dilation=0):
+    mask = np.zeros(shape, dtype=np.uint8)
+    cv2.fillPoly(mask, [np.asarray(points, dtype=np.int32)], 255)
+    if dilation:
+        mask = cv2.dilate(mask, np.ones((3, 3), np.uint8), iterations=dilation)
+    return mask
+
+
+def translate_rgba(layer, dx, dy):
+    matrix = np.float32([[1, 0, dx], [0, 1, dy]])
+    return cv2.warpAffine(
+        layer,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_NEAREST,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+
+
+def alpha_over(base, layer):
+    alpha = layer[:, :, 3:4].astype(np.float32) / 255.0
+    return (
+        layer[:, :, :3].astype(np.float32) * alpha
+        + base.astype(np.float32) * (1.0 - alpha)
+    ).round().clip(0, 255).astype(np.uint8)
+
+
+def build_knife_extension(source, knife_mask):
+    """Extrapolate the visible blade along its long axis for off-frame travel."""
+    direction = np.array([75.0, 108.0])
+    direction /= np.linalg.norm(direction)
+    pad_top = 32
+    height = HEIGHT + pad_top + 20
+    layer = np.zeros((height, WIDTH, 4), dtype=np.uint8)
+    layer[pad_top : pad_top + HEIGHT, :, :3] = source
+    layer[pad_top : pad_top + HEIGHT, :, 3] = knife_mask
+
+    filled = layer[:, :, 3].copy()
+    for step in range(1, 40):
+        dx = int(round(-direction[0] * step))
+        dy = int(round(-direction[1] * step))
+        matrix = np.float32([[1, 0, dx], [0, 1, pad_top + dy]])
+        shifted = cv2.warpAffine(
+            np.dstack((source, knife_mask)),
+            matrix,
+            (WIDTH, height),
+            flags=cv2.INTER_NEAREST,
+            borderMode=cv2.BORDER_CONSTANT,
+            borderValue=(0, 0, 0, 0),
+        )
+        add = (shifted[:, :, 3] > 0) & (filled == 0)
+        layer[add] = shifted[add]
+        filled[add] = 255
+    return layer, pad_top
+
+
+def crop_extended(layer, pad_top, dx, dy):
+    matrix = np.float32([[1, 0, dx], [0, 1, dy - pad_top]])
+    return cv2.warpAffine(
+        layer,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_NEAREST,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+
+
+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def make_scene(source):
+    knife_points = [
+        (108, 0),
+        (131, 0),
+        (203, 108),
+        (187, 108),
+        (106, 10),
+    ]
+    fruit_points = [
+        (145, 70),
+        (145, 63),
+        (147, 58),
+        (150, 53),
+        (155, 48),
+        (161, 45),
+        (168, 42),
+        (175, 42),
+        (183, 44),

+        (194, 53),
+        (197, 59),
+        (199, 66),

+        (191, 83),
+        (185, 87),
+        (178, 90),
+        (170, 91),
+        (163, 89),
+        (157, 85),
+        (152, 80),
+        (148, 75),
+    ]
+
+    knife_mask = polygon_mask(knife_points, (HEIGHT, WIDTH), dilation=1)
+    fruit_mask = polygon_mask(fruit_points, (HEIGHT, WIDTH))
+
+    # Reconstruct the board behind every moving element.
+    removal_mask = cv2.bitwise_or(knife_mask, fruit_mask)
+    removal_mask = cv2.dilate(removal_mask, np.ones((3, 3), np.uint8), iterations=1)
+    plate = cv2.inpaint(source, removal_mask, 7, cv2.INPAINT_TELEA)
+
+    # Recover fruit texture with no blade running through it. The opening cut
+    # gap later hides the narrow reconstructed strip.
+    fruit_source = cv2.inpaint(source, knife_mask, 4, cv2.INPAINT_TELEA)
+    fruit_layer = np.zeros((HEIGHT, WIDTH, 4), dtype=np.uint8)
+    fruit_layer[:, :, :3] = fruit_source
+    fruit_layer[:, :, 3] = fruit_mask
+
+    knife_layer = np.zeros((HEIGHT, WIDTH, 4), dtype=np.uint8)
+    knife_layer[:, :, :3] = source
+    knife_layer[:, :, 3] = knife_mask
+    knife_ext, knife_pad = build_knife_extension(source, knife_mask)
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    cut_origin = np.array([161.0, 60.0])
+    cut_direction = np.array([75.0, 108.0])
+    cut_direction /= np.linalg.norm(cut_direction)
+    cut_normal = np.array([-cut_direction[1], cut_direction[0]])
+    signed = (xx - cut_origin[0]) * cut_normal[0] + (
+        yy - cut_origin[1]
+    ) * cut_normal[1]
+    left_mask = ((signed <= 0) & (fruit_mask > 0)).astype(np.uint8) * 255
+    right_mask = ((signed > 0) & (fruit_mask > 0)).astype(np.uint8) * 255
+
+    return {
+        "plate": plate,
+        "fruit_layer": fruit_layer,
+        "knife_layer": knife_layer,
+        "knife_ext": knife_ext,
+        "knife_pad": knife_pad,
+        "left_mask": left_mask,
+        "right_mask": right_mask,
+        "cut_origin": cut_origin,
+        "cut_direction": cut_direction,
+        "cut_normal": cut_normal,
+    }
+
+
+def shifted_fruit(layer, mask, dx, dy):
+    shifted_mask = cv2.warpAffine(
+        mask,
+        np.float32([[1, 0, dx], [0, 1, dy]]),
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_NEAREST,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    result = layer.copy()
+    result[:, :, 3] = cv2.bitwise_and(layer[:, :, 3], shifted_mask)
+    return translate_rgba(result, dx, dy)
+
+
+def add_cut_highlight(canvas, scene, offset, gap):
+    if gap < 0.7:
+        return canvas
+    origin = scene["cut_origin"] + offset
+    direction = scene["cut_direction"]
+    normal = scene["cut_normal"]
+    half_length = 23
+    center = origin + normal * (gap * 0.5)
+    p1 = center - direction * half_length
+    p2 = center + direction * half_length
+    overlay = canvas.copy()
+    cv2.line(
+        overlay,
+        tuple(np.round(p1).astype(int)),
+        tuple(np.round(p2).astype(int)),
+        (78, 30, 10),
+        1,
+        cv2.LINE_8,
+    )
+    canvas = cv2.addWeighted(overlay, 0.72, canvas, 0.28, 0)
+
+    edge_center = center - normal * max(0.0, gap - 1.0)
+    p1 = edge_center - direction * half_length
+    p2 = edge_center + direction * half_length
+    overlay = canvas.copy()
+    cv2.line(
+        overlay,
+        tuple(np.round(p1).astype(int)),
+        tuple(np.round(p2).astype(int)),
+        (232, 164, 86),
+        1,
+        cv2.LINE_8,
+    )
+    return cv2.addWeighted(overlay, 0.38, canvas, 0.62, 0)
+
+
+def render(source, scene, frame_index):
+    if frame_index == 0:
+        return source.copy()
+
+    progress = smoothstep(frame_index / (FRAME_COUNT - 2))
+    settle = max(0.0, (frame_index - 13) / 6.0)
+    gap = 3.4 * progress + 0.35 * np.sin(np.pi * min(1.0, settle))
+    offset_left = scene["cut_normal"] * (gap * 0.5)
+    offset_right = -scene["cut_normal"] * (gap * 0.5)
+
+    canvas = scene["plate"].copy()
+
+    # A restrained contact shadow keeps both pieces grounded on the glass.
+    for side_mask, offset in (
+        (scene["left_mask"], offset_left),
+        (scene["right_mask"], offset_right),
+    ):
+        shadow = shifted_fruit(
+            scene["fruit_layer"],
+            side_mask,
+            int(round(offset[0])) + 2,
+            int(round(offset[1])) + 3,
+        )
+        alpha = shadow[:, :, 3:4].astype(np.float32) / 255.0 * 0.20
+        canvas = (
+            canvas.astype(np.float32) * (1.0 - alpha)
+            + np.array([27, 17, 10], dtype=np.float32) * alpha
+        ).round().clip(0, 255).astype(np.uint8)
+
+    left = shifted_fruit(
+        scene["fruit_layer"],
+        scene["left_mask"],
+        int(round(offset_left[0])),
+        int(round(offset_left[1])),
+    )
+    right = shifted_fruit(
+        scene["fruit_layer"],
+        scene["right_mask"],
+        int(round(offset_right[0])),
+        int(round(offset_right[1])),
+    )
+    canvas = alpha_over(canvas, left)
+    canvas = alpha_over(canvas, right)
+    canvas = add_cut_highlight(canvas, scene, np.zeros(2), gap)
+
+    travel = progress * 10.0
+    dx = int(round(scene["cut_direction"][0] * travel))
+    dy = int(round(scene["cut_direction"][1] * travel))
+    knife = crop_extended(scene["knife_ext"], scene["knife_pad"], dx, dy)
+    knife_shadow = crop_extended(
+        scene["knife_ext"], scene["knife_pad"], dx + 3, dy + 3
+    )
+    shadow_alpha = knife_shadow[:, :, 3:4].astype(np.float32) / 255.0 * 0.13
+    canvas = (
+        canvas.astype(np.float32) * (1.0 - shadow_alpha)
+        + np.array([24, 20, 15], dtype=np.float32) * shadow_alpha
+    ).round().clip(0, 255).astype(np.uint8)
+    return alpha_over(canvas, knife)
+
+
+def main():
+    source = np.array(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    scene = make_scene(source)
+    frames = np.stack(
+        [render(source, scene, frame_index) for frame_index in range(FRAME_COUNT)]
+    )
+    assert frames.shape == (FRAME_COUNT, HEIGHT, WIDTH, 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[0], source)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()

+    outline_mask = polygon_mask(outline, source.shape[:2])
+    warm = (
+        (red > green * 1.10)
