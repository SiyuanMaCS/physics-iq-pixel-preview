+def top_ramp_position(time_s):
+    """Accelerating roll along the short upper ramp."""
+    u = np.clip((time_s - 0.25) / 1.80, 0.0, 1.0)
+    x = 79.0 + 12.0 * u * u
+    y = 16.0 + 8.5 * u * u + 0.25 * np.sin(np.pi * u)
+    return float(x), float(y), float(u)

+def patterned_ramp_center_y(x):
+    # The long patterned ramp's top edge is y ~= 20.3 + 0.12*x.
+    return 20.3 + 0.12 * x - 3.0

+def marble_state(time_s):
+    """Return center, radius, airborne fraction, and traveled distance."""
+    if time_s < 0.25:
+        return 79.0, 16.0, 3.15, 0.0, 0.0
+
+    if time_s < 2.05:
+        x, y, u = top_ramp_position(time_s)
+        return x, y, 3.15, 0.0, 12.0 * u * u
+
+    # Release from the upper ramp: retain its exit velocity and accelerate downward.
+    dt = time_s - 2.05
+    if dt < 0.50:
+        x = 91.0 + 12.6 * dt
+        y = 24.5 + 8.9 * dt + 0.5 * 46.0 * dt * dt
+        y = min(y, patterned_ramp_center_y(x))
+        airborne = 1.0 - smoothstep(0.32, 0.50, dt)
+        return x, y, 3.15, float(airborne), 12.0 + 12.6 * dt
+
+    # Roll down the long patterned ramp.
+    dt = time_s - 2.55
+    x = 97.3 + 12.6 * dt + 0.5 * 12.4 * dt * dt
+    y = patterned_ramp_center_y(x)
+    return x, y, 3.15, 0.0, 21.4 + 12.6 * dt

+def render_ball(frame, center_x, center_y, radius, airborne, travel):
+    """Composite an antialiased yellow-glass sphere into an RGB frame."""
+    scale = 8
+    half = int(np.ceil(radius + 3.0))
+    x0 = int(np.floor(center_x)) - half
+    y0 = int(np.floor(center_y)) - half
+    size = half * 2 + 1
+
+    ys, xs = np.mgrid[0:size * scale, 0:size * scale]
+    px = x0 + (xs + 0.5) / scale
+    py = y0 + (ys + 0.5) / scale
+
+    # Projected shadow on the ramp.
+    shadow_y = center_y + 2.7 + 2.0 * airborne
+    shadow_rx = 4.0 - 1.1 * airborne
+    shadow_ry = 1.35 - 0.25 * airborne
+    shadow_d = ((px - center_x - 1.15) / shadow_rx) ** 2 + ((py - shadow_y) / shadow_ry) ** 2
+    shadow_a = np.clip(1.0 - shadow_d, 0.0, 1.0) ** 1.5
+    shadow_a *= 0.25 - 0.08 * airborne
+
+    # Soft contact occlusion directly beneath the bead.
+    contact_d = ((px - center_x - 0.3) / 2.2) ** 2 + ((py - center_y - 2.25) / 0.72) ** 2
+    contact_a = np.clip(1.0 - contact_d, 0.0, 1.0) ** 2 * 0.28
+
+    dx = (px - center_x) / radius
+    dy = (py - center_y) / radius
+    rr = np.sqrt(dx * dx + dy * dy)
+    edge_alpha = 1.0 - smoothstep(0.92, 1.04, rr)
+
+    inside = np.clip(1.0 - (dx * dx + dy * dy), 0.0, 1.0)
+    nz = np.sqrt(inside)
+
+    # Key light from the upper left, with a cool reflected edge.
+    light = np.array([-0.45, -0.62, 0.64])
+    light /= np.linalg.norm(light)
+    diffuse = np.clip(dx * light[0] + dy * light[1] + nz * light[2], 0.0, 1.0)
+

+    )
+

+    pin = np.exp(-0.5 * (hx * hx + hy * hy))
+    gx = (px - (center_x + 0.95)) / 0.55
+    gy = (py - (center_y + 0.45)) / 0.48
+    glint = np.exp(-0.5 * (gx * gx + gy * gy))
+
+    rim = smoothstep(0.58, 1.0, rr) * (1.0 - smoothstep(0.98, 1.06, rr))
+    warm_edge = np.stack(
+        (
+            30.0 * rim,
+            22.0 * rim,
+            2.0 * rim,
+        ),
+        axis=-1,
+    )
+    amber -= warm_edge
+
+    highlight = np.stack(
+        (
+            155.0 * pin + 45.0 * glint,
+            145.0 * pin + 43.0 * glint,
+            95.0 * pin + 30.0 * glint,
+        ),
+        axis=-1,
+    )
+    color = np.clip(amber + highlight, 0.0, 255.0)
+
+    # Glass reveals some of the ramp while keeping a distinct yellow body.
+    glass_alpha = edge_alpha * (0.68 + 0.18 * np.clip(nz, 0.0, 1.0))
+
+    patch = frame[y0:y0 + size, x0:x0 + size].astype(np.float32)
+    shadow_small = cv2.resize(shadow_a + contact_a, (size, size), interpolation=cv2.INTER_AREA)
+    patch *= (1.0 - shadow_small[..., None])
+
+    color_small = cv2.resize(color, (size, size), interpolation=cv2.INTER_AREA)
+    alpha_small = cv2.resize(glass_alpha, (size, size), interpolation=cv2.INTER_AREA)
+    patch = patch * (1.0 - alpha_small[..., None]) + color_small * alpha_small[..., None]
+
+    # A faint warm caustic travels with the bead.
+    cy = center_y + 2.6
+    caustic_d = ((px - center_x + 0.8) / 3.5) ** 2 + ((py - cy) / 0.95) ** 2
+    caustic = np.clip(1.0 - caustic_d, 0.0, 1.0) ** 2 * (0.07 + 0.02 * np.sin(travel * 1.7))
+    caustic_small = cv2.resize(caustic, (size, size), interpolation=cv2.INTER_AREA)
+    patch += caustic_small[..., None] * np.array([85.0, 68.0, 15.0], np.float32)
+
+    frame[y0:y0 + size, x0:x0 + size] = np.clip(patch, 0.0, 255.0).astype(np.uint8)
+
+
+def main():
+    first = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    clean = make_clean_plate(first)
+    frames = np.empty((FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+
+    for index in range(FRAMES):
+        time_s = index / 4.0
+        if index == 0:
+            frames[index] = first
+            continue
+
+        frame = clean.copy()
+        x, y, radius, airborne, travel = marble_state(time_s)
+        render_ball(frame, x, y, radius, airborne, travel)
+        frames[index] = frame
+
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
