+def smooth_circle_mask(size, radius, feather=1.2):
+    yy, xx = np.mgrid[:size, :size]
+    center = (size - 1) / 2.0
+    distance = np.hypot(xx - center, yy - center)
+    return np.clip((radius + feather - distance) / (2.0 * feather), 0.0, 1.0)
