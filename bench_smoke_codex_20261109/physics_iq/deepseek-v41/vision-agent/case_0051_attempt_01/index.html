+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 144
+FRAME_COUNT, FPS = 20, 4
+
+
+def polygon_mask(shape, points, scale=4):
+    """Anti-aliased polygon mask."""
+    height, width = shape
+    big = np.zeros((height * scale, width * scale), np.uint8)
+    scaled = np.round(np.asarray(points) * scale).astype(np.int32)
+    cv2.fillPoly(big, [scaled], 255, lineType=cv2.LINE_AA)
+    return cv2.resize(big, (width, height), interpolation=cv2.INTER_AREA)
+
+
+def smoothstep(value):
+    value = float(np.clip(value, 0.0, 1.0))
+    return value * value * (3.0 - 2.0 * value)
+
+
+def fall_angle(time, start, duration):
+    """Accelerating fall with a tiny damped landing wobble."""
+    if time <= start:
+        return 0.0
+    progress = min((time - start) / duration, 1.0)
+    angle = 87.0 * progress ** 1.55
+    if progress >= 1.0:
+        since_landing = time - (start + duration)
+        angle += 2.2 * np.exp(-4.0 * since_landing) * np.sin(15.0 * since_landing)
+    return angle
+
+
+def rotate_sprite(image, alpha, pivot, angle_degrees):
+    """Rotate an RGBA-like sprite about a fixed tabletop point."""
+    height, width = image.shape[:2]
+    theta = np.deg2rad(angle_degrees)
+    cosine, sine = np.cos(theta), np.sin(theta)
+    px, py = pivot
+    matrix = np.array(
+        [[cosine, sine, px - cosine * px - sine * py],
+         [-sine, cosine, py + sine * px - cosine * py]],
+        dtype=np.float32,
+    )
+    warped_rgb = cv2.warpAffine(
+        image, matrix, (width, height), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=(0, 0, 0),
+    )
+    warped_alpha = cv2.warpAffine(
+        alpha, matrix, (width, height), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+    return warped_rgb, warped_alpha
+
+
+def composite(frame, image, alpha):
+    opacity = (alpha.astype(np.float32) / 255.0)[..., None]
+    frame[:] = np.clip(
+        image.astype(np.float32) * opacity + frame.astype(np.float32) * (1.0 - opacity),
+        0, 255,
+    ).astype(np.uint8)
+
+
+def add_contact_shadow(frame, pivot, angle_degrees, length, strength=0.24):
+    """Soft tabletop shadow that grows beneath a domino as it falls."""
+    theta = np.deg2rad(angle_degrees)
+    reach = max(2.0, length * np.sin(theta) * 0.82)
+    end = (
+        int(round(pivot[0] - reach * 0.93)),
+        int(round(pivot[1] + reach * 0.16)),
+    )
+    shadow = np.zeros_like(frame)
+    cv2.line(shadow, (int(round(pivot[0])), int(round(pivot[1]))), end,
+             (10, 8, 4), 3, cv2.LINE_AA)
+    shadow = cv2.GaussianBlur(shadow, (5, 3), 1.2)
+    frame[:] = np.clip(
+        frame.astype(np.float32) * (1.0 - strength) + shadow.astype(np.float32) * strength,
+        0, 255,
+    ).astype(np.uint8)
+
+
+def draw_stick(frame, pivot, angle_degrees):
+    """Draw the thin wooden actuator with a highlight and soft shadow."""
+    length = 77.0
+    theta = np.deg2rad(174.5 + angle_degrees)
+    tip = (pivot[0] + length * np.cos(theta), pivot[1] + length * np.sin(theta))
+
+    shadow = np.zeros_like(frame)
+    cv2.line(shadow, (int(pivot[0]), int(pivot[1] + 2)),
+             (int(tip[0]), int(tip[1] + 2)), (25, 20, 10), 3, cv2.LINE_AA)
+    shadow = cv2.GaussianBlur(shadow, (3, 3), 0.9)
+    frame[:] = np.clip(frame.astype(np.float32) * 0.82 + shadow.astype(np.float32) * 0.38,
+                       0, 255).astype(np.uint8)
+
+    cv2.line(frame, (int(pivot[0]), int(pivot[1])),
+             (int(tip[0]), int(tip[1])), (73, 63, 39), 4, cv2.LINE_AA)
+    cv2.line(frame, (int(pivot[0]), int(pivot[1] - 1)),
+             (int(tip[0]), int(tip[1] - 1)), (177, 159, 108), 2, cv2.LINE_AA)
+    cv2.line(frame, (int(pivot[0]), int(pivot[1] - 1)),
+             (int(tip[0]), int(tip[1] - 1)), (205, 189, 139), 1, cv2.LINE_AA)
+    cv2.circle(frame, (int(round(tip[0])), int(round(tip[1]))), 1,
+               (151, 132, 82), -1, cv2.LINE_AA)
+
+
+def main():
+    source = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    assert source.shape == (HEIGHT, WIDTH, 3)
+
+    dominoes = [
+        {
+            "name": "gap_black",
+            "polygon": [(129, 54), (139, 54), (139, 86), (130, 86)],
+            "pivot": (130.0, 85.5),
+            "height": 32.0,
+            "start": 2.28,
+            "duration": 1.05,
+        },
+        {
+            "name": "gap_white",
+            "polygon": [(139, 56), (150, 56), (150, 91), (140, 90)],
+            "pivot": (140.0, 89.5),
+            "height": 34.0,
+            "start": 1.68,
+            "duration": 1.05,
+        },
+        {
+            "name": "strike_black",
+            "polygon": [(148, 51), (164, 54), (164, 92), (149, 92)],
+            "pivot": (149.0, 91.0),
+            "height": 39.0,
+            "start": 1.04,
+            "duration": 1.15,
+        },
+    ]
+
+    masks = []
+    for item in dominoes:
+        mask = polygon_mask((HEIGHT, WIDTH), item["polygon"])
+        item["mask"] = mask
+        masks.append(mask)
+
+    # The actuator is a narrow capsule. Its source shadow is removed with it.
+    stick_mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.line(stick_mask, (239, 55), (161, 62), 255, 6, cv2.LINE_AA)
+    masks.append(stick_mask)
+
+    moving_mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    for mask in masks:
+        moving_mask = np.maximum(moving_mask, mask)
+    moving_mask = cv2.dilate(moving_mask, np.ones((3, 3), np.uint8), iterations=1)
+    clean_plate = cv2.inpaint(source, moving_mask, 5, cv2.INPAINT_TELEA)
+
+    # Remove the stick before extracting the strike domino's texture.
+    sprite_source = cv2.inpaint(source, stick_mask, 3, cv2.INPAINT_TELEA)
+    for item in dominoes:
+        item["sprite"] = sprite_source
+
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), np.uint8)
+    frames[0] = source
+
+    for frame_index in range(1, FRAME_COUNT):
+        time = frame_index / FPS
+        frame = clean_plate.copy()
+
+        # Stick motion begins gently and reaches the top edge of the nearest domino.
+        stick_progress = smoothstep((time - 0.10) / 1.0)
+        stick_angle = 5.2 * stick_progress
+
+        # Contact shadows sit on the table, below every domino.
+        for item in dominoes:
+            angle = fall_angle(time, item["start"], item["duration"])
+            add_contact_shadow(frame, item["pivot"], angle, item["height"])
+
+        # Back-to-front order keeps the striking domino above the row it topples.
+        for item in dominoes:
+            angle = fall_angle(time, item["start"], item["duration"])
+            warped, alpha = rotate_sprite(
+                item["sprite"], item["mask"], item["pivot"], -angle,
+            )
+            composite(frame, warped, alpha)
+
+        draw_stick(frame, (238.0, 55.0), stick_angle)
+        frames[frame_index] = frame
+
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()
