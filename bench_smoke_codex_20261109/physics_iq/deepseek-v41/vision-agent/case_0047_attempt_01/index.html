+def smoothstep(a, b, x):
+    t = np.clip((x - a) / (b - a), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def alpha_polygon(shape, points, feather=0.65):
+    """Anti-aliased polygon alpha, made at 4x and reduced to the frame size."""
+    h, w = shape
+    scale = 4
+    mask = np.zeros((h * scale, w * scale), np.uint8)
+    pts = (np.asarray(points, np.float32) * scale).round().astype(np.int32)
+    cv2.fillPoly(mask, [pts], 255, cv2.LINE_AA)
+    mask = cv2.resize(mask, (w, h), interpolation=cv2.INTER_AREA).astype(np.float32) / 255.0
+    if feather > 0:
+        mask = cv2.GaussianBlur(mask, (0, 0), feather)
+    return mask

+def build_clean_plate(source):
+    """Remove the held object, preserving wood grain and the mug/liquid body."""
+    h, w = source.shape[:2]
+    mask = np.zeros((h, w), np.uint8)
+    cv2.fillPoly(
+        mask,
+        [
+            np.array(
+                [
+                    [111, 0],
+                    [147, 0],
+                    [147, 19],
+                    [140, 24],
+                    [139, 89],
+                    [126, 89],
+                    [125, 24],
+                    [111, 19],
+                ],
+                np.int32,
+            )
+        ],
+        255,
+    )
+    clean = cv2.inpaint(source, mask, 4, cv2.INPAINT_TELEA)

+    # The wood above the mug is best reconstructed with real neighbouring grain.
+    x0, x1, y0, y1 = 112, 147, 0, 40
+    donor = source[y0:y1, 148:183, :][:, ::-1, :].copy()
+    if donor.shape[1] < x1 - x0:
+        donor = np.pad(donor, ((0, 0), (0, x1 - x0 - donor.shape[1]), (0, 0)), mode="edge")
+    donor = donor[:, : x1 - x0, :]
+    blend = np.ones((y1 - y0, x1 - x0), np.float32)
+    edge = 4
+    ramp = np.linspace(0.0, 1.0, edge, dtype=np.float32)
+    blend[:, :edge] *= ramp[None, :]
+    blend[:, -edge:] *= ramp[::-1][None, :]

+def draw_jaw(frame, sprite, alpha, dx, dy, opening, side):

+def liquid_ripples(frame, time):
+    """Anisotropic damped rings on the visible liquid surface."""
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    cx, cy, rx, ry = 132.5, 65.0, 28.5, 18.0
+    d = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    inside = np.clip((1.0 - d) * 9.0, 0.0, 1.0)
+    impact = 0.85
+    elapsed = time - impact
+    if elapsed <= 0:
+        return
+    rings = np.zeros_like(d)
+    for delay, strength, speed in ((0.00, 1.00, 0.23), (0.16, 0.65, 0.20), (0.31, 0.42, 0.17)):
+        tau = elapsed - delay
+        if tau <= 0:
+            continue
+        radius = 0.16 + speed * tau
+        width = 0.070 + 0.012 * tau
+        envelope = np.exp(-((d - radius) / width) ** 2)
+        envelope *= np.exp(-0.60 * tau)
+        rings += strength * envelope * np.cos((d - radius) / (0.075 + 0.006 * tau))
+    rings = np.clip(rings, -1.0, 1.0) * inside
+    rings *= np.exp(-0.45 * max(elapsed, 0.0))
+    highlight = np.array([116.0, 151.0, 181.0], np.float32)
+    shadow = np.array([20.0, 3.0, 2.0], np.float32)
+    pos = np.clip(rings, 0.0, 1.0)[..., None]
+    neg = np.clip(-rings, 0.0, 1.0)[..., None]
+    frame[:] = frame * (1.0 - 0.18 * pos - 0.14 * neg)
+    frame[:] += highlight * (0.30 * pos) + shadow * (0.20 * neg)
+

+def draw_meniscus(frame, time, domino_dy):
+    if domino_dy < 2 or domino_dy > 62:
+        return
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)

+    cy = 64.0
+    radius = 6.5 + 0.05 * domino_dy
+    ring = ((xx - cx) / radius) ** 2 + ((yy - cy) / 1.7) ** 2
+    band = np.exp(-((ring - 1.0) / 0.30) ** 2)
+    band *= (1.0 - smoothstep(25.0, 62.0, domino_dy))
+    frame[:] = composite(frame, np.array([190, 215, 226], np.float32), 0.25 * band)

+def draw_droplets(frame, time):
+    """Small deterministic crown droplets after the domino breaks the surface."""
+    t = time
+    if t < 1.05 or t > 2.75:
+        return
+    rng = np.random.default_rng(4701)
+    count = 11
+    for i in range(count):
+        angle = -2.65 + i * (5.3 / (count - 1)) + rng.normal(0, 0.08)
+        speed = 10.0 + rng.uniform(-2.0, 5.0)
+        delay = rng.uniform(0.0, 0.16)
+        tau = t - 1.02 - delay
+        if tau <= 0:
+            continue
+        x = 132.0 + np.cos(angle) * (7.5 + speed * tau * 0.34)
+        y = 62.5 + np.sin(angle) * 3.2 - 30.0 * tau + 37.0 * tau * tau
+        if y > 76:
+            continue
+        radius = max(0.45, 1.15 - tau * 0.35 + rng.normal(0, 0.10))
+        ix, iy = int(round(x)), int(round(y))
+        if 0 <= ix < W and 0 <= iy < H:
+            cv2.circle(frame, (ix, iy), max(1, int(round(radius))), (76, 24, 14), -1, cv2.LINE_AA)
+            cv2.circle(frame, (ix, iy - 1), 1, (156, 184, 206), -1, cv2.LINE_AA)
