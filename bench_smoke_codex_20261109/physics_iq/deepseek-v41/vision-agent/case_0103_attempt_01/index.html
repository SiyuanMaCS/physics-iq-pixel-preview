+def make_clean_plate(first_frame):
+    """Remove the original marble while preserving the patterned ramp."""
+    image = cv2.cvtColor(first_frame, cv2.COLOR_RGB2BGR)
+    mask = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    cv2.ellipse(mask, (104, 20), (5, 4), 0, 0, 360, 255, -1)
+    cleaned = cv2.inpaint(image, mask, 2.5, cv2.INPAINT_TELEA)
+    cleaned = cv2.cvtColor(cleaned, cv2.COLOR_BGR2RGB)
+
+    # Telea is intentionally conservative, but a little fine grain at the ramp
+    # edge keeps the repaired patch from looking smoother than the footage.
+    rng = np.random.default_rng(103)
+    noise = rng.normal(0.0, 1.4, (7, 13, 3))
+    patch = cleaned[15:22, 97:110].astype(np.float64) + noise
+    cleaned[15:22, 97:110] = np.clip(patch, 0, 255).astype(np.uint8)
+    return cleaned
+

+def render_ball_patch(center_x, center_y, scale=10, pad=7):
+    """Draw a small yellow glass sphere and its soft contact shadow."""
+    patch_size = 2 * pad
+    size = patch_size * scale
+    yy, xx = np.mgrid[0:size, 0:size].astype(np.float64)
+    xx = (xx + 0.5) / scale - pad
+    yy = (yy + 0.5) / scale - pad
+
+    radius = 3.25
+    distance = np.sqrt(xx * xx + yy * yy)
+    sphere_alpha = np.clip((radius + 0.45 - distance) / 0.90, 0.0, 1.0)
+
+    nx = np.clip(xx / radius, -1.0, 1.0)
+    ny = np.clip(yy / radius, -1.0, 1.0)
+    nz = np.sqrt(np.clip(1.0 - nx * nx - ny * ny, 0.0, 1.0))
+
+    # Yellow glass: pale transmitted light in the upper-left, deep amber at
+    # the lower-right edge, and a narrow bright Fresnel rim.
+    light = np.clip(0.68 - 0.34 * nx - 0.38 * ny + 0.20 * nz, 0.0, 1.0)
+    pale = np.array([244.0, 229.0, 138.0])
+    amber = np.array([105.0, 117.0, 25.0])
+    color = amber[None, None, :] + (pale - amber)[None, None, :] * light[:, :, None]
+
+    transmission = np.exp(-((xx + 0.25) ** 2 + (yy + 0.30) ** 2) / 2.6)
+    color += np.array([24.0, 25.0, 9.0])[None, None, :] * transmission[:, :, None]
+
+    edge = np.clip((distance - 2.28) / 0.78, 0.0, 1.0)
+    color *= (1.0 - 0.24 * edge[:, :, None])
+    rim_light = np.clip(1.0 - np.abs(distance - 2.82) / 0.36, 0.0, 1.0)
+    upper_rim = np.clip(-0.42 * nx - 0.75 * ny, 0.0, 1.0)
+    color += np.array([34.0, 31.0, 13.0])[None, None, :] * (
+        rim_light * upper_rim
+    )[:, :, None]
+
+    primary = np.exp(-(((xx + 1.05) / 0.52) ** 2 + ((yy + 1.10) / 0.48) ** 2))
+    secondary = np.exp(-(((xx - 0.72) / 0.31) ** 2 + ((yy - 1.10) / 0.22) ** 2))
+    color += np.array([255.0, 250.0, 207.0])[None, None, :] * primary[:, :, None]
+    color += np.array([93.0, 84.0, 35.0])[None, None, :] * secondary[:, :, None]
+    color = np.clip(color, 0, 255)
+
+    # A diffuse, slightly right-shifted shadow grounds the marble on the board.
+    shadow_x = xx - 0.85
+    shadow_y = yy - 2.25
+    shadow_metric = (shadow_x / 3.05) ** 2 + (shadow_y / 1.45) ** 2
+    shadow_alpha = 0.14 * np.exp(-2.3 * shadow_metric)
+    shadow_alpha *= np.clip((distance - 1.8) / 1.0, 0.0, 1.0)
+
+    rgba = np.zeros((size, size, 4), dtype=np.uint8)
+    shadow_rgb = np.array([39.0, 45.0, 32.0])
+    rgba[:, :, :3] = shadow_rgb.astype(np.uint8)
+    rgba[:, :, 3] = np.clip(shadow_alpha * 255.0, 0, 255).astype(np.uint8)
+
+    sphere_rgb = np.clip(color, 0, 255).astype(np.uint8)
+    sphere_a = sphere_alpha[:, :, None]
+    rgba[:, :, :3] = (
+        rgba[:, :, :3].astype(np.float64) * (1.0 - sphere_a)
+        + sphere_rgb.astype(np.float64) * sphere_a
+    ).astype(np.uint8)
+    rgba[:, :, 3] = np.maximum(
+        rgba[:, :, 3],
+        np.clip(sphere_alpha * 255.0, 0, 255).astype(np.uint8),
+    )
+
+    return Image.fromarray(rgba, "RGBA").resize(
+        (patch_size, patch_size), Image.Resampling.LANCZOS
+    )
+
+
+def composite_ball(frame, x, y):
+    pad = 7
+    sprite = render_ball_patch(x, y, pad=pad)
+    left = int(np.floor(x)) - pad
+    top = int(np.floor(y)) - pad
+    right = left + 2 * pad
+    bottom = top + 2 * pad
+
+    # All sampled centers stay well inside the board, but clamp defensively.
+    left_c, top_c = max(left, 0), max(top, 0)
+    right_c, bottom_c = min(right, WIDTH), min(bottom, HEIGHT)
+    sprite = sprite.crop(
+        (left_c - left, top_c - top, right_c - left, bottom_c - top)
+    )
+    base = Image.fromarray(frame, "RGB").convert("RGBA")
+    base.alpha_composite(sprite, (left_c, top_c))
+    return np.asarray(base.convert("RGB"), dtype=np.uint8)
+
+
+def generate():
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    assert first.shape == (HEIGHT, WIDTH, 3)
+
+    clean = make_clean_plate(first)
+    frames = np.empty((N_FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first
+    for index in range(1, N_FRAMES):
+        x, y = BALL_POSITIONS[index]
+        frames[index] = composite_ball(clean, x, y)
+    return frames
+
+
+if __name__ == "__main__":
+    artifact_io.save(generate(), ROOT)
