+def _ellipse(x, y, cx, cy, rx, ry):
+    return ((x - cx) / rx) ** 2 + ((y - cy) / ry) ** 2

+def _smoothstep(a, b, x):
+    t = np.clip((x - a) / (b - a), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+        saturated, 8

+def rotate_body(rgb, mask, center, radius, angle):
+    """Cylindrical texture warp with diffuse-light shading and occlusion edges."""
+    yy, xx = np.mgrid[:H, :W]
+    cx, cy = center

+    dy = (yy - cy) / (radius * 0.90)
+    r2 = dx * dx + dy * dy

+

+    # with a slightly darkened wrap so the rear side has continuous pattern.
+    src_x = np.clip(cx + (u - 0.5) * 2.0 * radius, 0, W - 1)
+    src_y = np.clip(cy + (v - 0.5) * 1.90 * radius, 0, H - 1)

+    blended = (
+        warped.astype(float) * (0.42 + 0.58 * front_weight[:, :, None])

+def attachment_sprite(rgb, mask, pivot, direction):
+    """Extract an attachment into a local sprite and soften its silhouette."""
+    ys, xs = np.where(mask)
+    if not len(xs):
+        return None
+    x0, x1 = xs.min(), xs.max() + 1
+    y0, y1 = ys.min(), ys.max() + 1
+    alpha = mask[y0:y1, x0:x1].astype(float)
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.65)
+    return rgb[y0:y1, x0:x1], alpha, (x0, y0, pivot, direction)

+def place_attachment(canvas, sprite, angle):
+    if sprite is None:
+        return
+    pixels, alpha, (x0, y0, pivot, direction) = sprite
+    a = angle * direction
+    # Orthographic yaw: side members narrow while their far ends foreshorten.
+    sx = max(0.22, np.cos(a))
+    h, w = alpha.shape
+    new_w = max(2, int(round(w * sx)))
+    scaled = cv2.resize(
+        pixels, (new_w, h), interpolation=cv2.INTER_AREA if sx < 1 else cv2.INTER_CUBIC
+    )
+    scaled_a = cv2.resize(alpha, (new_w, h), interpolation=cv2.INTER_AREA)
+
+    if direction < 0:
+        anchor = new_w - 1
+        px, py = pivot
+        tx = int(round(px - anchor))
+    else:
+        px, py = pivot
+        tx = int(round(px))
+    ty = int(y0)
+
+    # Depth cue: the side that turns away is dimmer; the near side catches a
+    # little more of the key light.
+    if direction < 0:
+        depth = 0.80 + 0.20 * max(0.0, np.cos(a))
+    else:
+        depth = 0.82 + 0.18 * max(0.0, np.cos(a))
+    scaled = np.clip(scaled.astype(float) * depth, 0, 255)
+
+    x0c = max(0, tx)
+    x1c = min(W, tx + new_w)
+    y0c = max(0, ty)
+    y1c = min(H, ty + h)
+    if x1c <= x0c or y1c <= y0c:
+        return
+    sx0 = x0c - tx
+    sy0 = y0c - ty
+    sx1 = sx0 + (x1c - x0c)
+    sy1 = sy0 + (y1c - y0c)
+    a_crop = scaled_a[sy0:sy1, sx0:sx1, None]
+    canvas[y0c:y1c, x0c:x1c] = (
+        canvas[y0c:y1c, x0c:x1c].astype(float) * (1.0 - a_crop)
+        + scaled[sy0:sy1, sx0:sx1] * a_crop
+    ).astype(np.uint8)
+
+
+def add_contact_shadow(canvas, angle):
+    yy, xx = np.mgrid[:H, :W]
+    # Slight lateral shift mirrors the apparent motion of the asymmetric body.
+    shift = 1.8 * np.sin(angle)
+    shadow = np.exp(
+        -_ellipse(xx, yy, 127.0 + shift, 108.0, 28.0, 5.5) * 2.3
+    )
+    shadow *= 0.32
+    return np.clip(
+        canvas.astype(float) * (1.0 - shadow[:, :, None]), 0, 255
+    ).astype(np.uint8)
+

+def main():
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))

+    spout = attachment_sprite(first, spout_mask, (153.0, 59.0), 1)
+
+    frames = [first.copy()]
+    # 32 degrees over five seconds reads as a deliberate, slow turntable move.
+    for i in range(1, N):
+        theta = np.deg2rad(32.0 * i / (N - 1))
+        body, _ = rotate_body(first, body_mask, (126.0, 62.0), 38.5, theta)
+        frame = stage.copy()

+        place_attachment(frame, handle, theta)
+        place_attachment(frame, spout, theta)
+        alpha = cv2.GaussianBlur(
+            (body_mask.astype(np.uint8) * 255), (3, 3), 0.6
+        ).astype(float)[:, :, None] / 255.0
+        frame = (
+            frame.astype(float) * (1.0 - alpha) + body.astype(float) * alpha
+        ).astype(np.uint8)
+        frames.append(frame)
+
+    frames = np.asarray(frames, dtype=np.uint8)
+    frames[0] = first
+    artifact_io.save(frames, ROOT)
+    print(f"wrote {ROOT / 'index.html'}")
+

+    main()
