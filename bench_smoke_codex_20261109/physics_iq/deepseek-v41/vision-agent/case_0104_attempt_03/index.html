+def glass_marble(

+

+    # then tint and shade it like a warm yellow sphere.

+    bx0 = max(0, int(np.floor(cx)) - pad)
+    bx1 = min(WIDTH, int(np.ceil(cx)) + pad + 1)

+    by1 = min(HEIGHT, int(np.ceil(cy)) + pad + 1)
