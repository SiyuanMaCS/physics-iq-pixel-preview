+WIDTH = 256
+HEIGHT = 144
+FRAMES = 20
+FPS = 4

+def make_clean_plate(frame: np.ndarray) -> np.ndarray:
+    """Remove the foreground ball and its cast shadow from the static shot."""
+    mask = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    cv2.ellipse(mask, (121, 62), (17, 13), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (137, 69), (20, 8), 0, 0, 360, 255, -1)
+    mask = cv2.dilate(mask, np.ones((3, 3), dtype=np.uint8), iterations=1)
+
+    source = frame.astype(np.uint8)
+    plate = cv2.inpaint(source, mask, 7, cv2.INPAINT_TELEA).astype(np.float32)
+
+    # Carry the fine horizontal wood grain into the repaired area so the plate
+    # does not read as a smooth patch when the ball starts moving.
+    blurred = np.empty_like(frame, dtype=np.float32)
+    for channel in range(3):
+        blurred[:, :, channel] = gaussian_filter(
+            frame[:, :, channel].astype(np.float32), sigma=4.0
+        )
+    residual = frame.astype(np.float32) - blurred
+    soft_mask = gaussian_filter(mask.astype(np.float32) / 255.0, sigma=1.1)
+
+    shifted = np.roll(residual, shift=42, axis=1)
+    shifted[:, :42] = residual[:, :42]
+    grain_strength = 0.62
+    plate = plate + shifted * (soft_mask[:, :, None] * grain_strength)
+
+    return np.clip(plate, 0, 255).astype(np.uint8)
+
+
+def make_ball_sprite(frame: np.ndarray) -> tuple[np.ndarray, tuple[int, int]]:
+    """Extract the ball with a soft elliptical matte for compositing."""
+    crop_x0, crop_y0, crop_x1, crop_y1 = 103, 48, 142, 77
+    crop = frame[crop_y0:crop_y1, crop_x0:crop_x1]
+
+    yy, xx = np.mgrid[crop_y0:crop_y1, crop_x0:crop_x1]
+    cx, cy = 121.5, 62.3
+    rx, ry = 15.3, 10.4
+    radius = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    alpha = np.clip((1.04 - radius) / 0.12, 0.0, 1.0)
+    alpha = gaussian_filter(alpha, sigma=0.45)
+
+    rgba = np.dstack((crop, (alpha * 255).round().astype(np.uint8)))
+    return rgba, (crop_x0, crop_y0)
+
+
+def add_shadow(
+    image: np.ndarray, center_x: float, contact_strength: float
+) -> np.ndarray:
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+
+    # The key light comes from the upper left, so the contact shadow trails to
+    # the lower right of the ball.
+    dx = (xx - (center_x + 8.5)) / 17.0
+    dy = (yy - 70.3) / 5.0
+    density = np.exp(-0.5 * (dx * dx + dy * dy))
+
+    dx2 = (xx - (center_x + 5.0)) / 9.5
+    dy2 = (yy - 69.0) / 3.2
+    core = np.exp(-0.5 * (dx2 * dx2 + dy2 * dy2))
+
+    alpha = np.clip(0.28 * density + 0.20 * core * contact_strength, 0.0, 0.44)
+    shaded = image.astype(np.float32)
+    shadow_color = np.array([47.0, 43.0, 40.0], dtype=np.float32)
+    shaded = shaded * (1.0 - alpha[:, :, None]) + shadow_color * alpha[:, :, None]
+    return np.clip(shaded, 0, 255).astype(np.uint8)
+
+
+def composite_ball(
+    image: np.ndarray,
+    sprite: np.ndarray,
+    center_x: float,
+    squash: float = 1.0,
+) -> np.ndarray:
+    """Place the matted ball while preserving its original fixed lighting."""
+    out = image.astype(np.float32)
+    sx = 1.0 / squash
+    sy = squash
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+
+    sprite_h, sprite_w = sprite.shape[:2]
+    local_x = (xx - center_x) / sx + sprite_w * 0.5
+    local_y = (yy - 62.3) / sy + sprite_h * 0.5
+
+    valid = (
+        (local_x >= 0)
+        & (local_x < sprite_w)
+        & (local_y >= 0)
+        & (local_y < sprite_h)
+    )
+    map_x = np.clip(local_x, 0, sprite_w - 1).astype(np.float32)
+    map_y = np.clip(local_y, 0, sprite_h - 1).astype(np.float32)
+
+    warped = cv2.remap(
+        sprite,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    alpha = (warped[:, :, 3:4].astype(np.float32) / 255.0) * valid[:, :, None]
+    out = (
+        warped[:, :, :3].astype(np.float32) * alpha
+        + out * (1.0 - alpha)
+    )
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def ball_state(time_s: float) -> tuple[float, float, float]:
+    start_x = 122.5
+    contact_x = 192.5
+    speed = (contact_x - start_x) / 3.70
+    contact_time = 3.70
+    rebound_speed = 7.2
+
+    if time_s < contact_time:
+        return start_x + speed * time_s, 1.0, 0.0
+
+    elapsed = time_s - contact_time
+    center_x = contact_x - rebound_speed * elapsed
+
+    # A one-frame compression at contact, followed by a tiny recovery.
+    if elapsed < 0.12:
+        squash = 0.91
+    elif elapsed < 0.32:
+        squash = 1.035
+    else:
+        squash = 1.0
+    return center_x, squash, elapsed
+
+
+def main() -> None:
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    if first.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"Unexpected first-frame shape: {first.shape}")
+
+    plate = make_clean_plate(first)
+    sprite, _ = make_ball_sprite(first)
+    frames = np.empty((FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first
+
+    for index in range(1, FRAMES):
+        time_s = index / FPS
+        center_x, squash, since_impact = ball_state(time_s)
+        contact_strength = 1.0
+        if since_impact > 0:
+            contact_strength = float(np.exp(-since_impact / 0.28))
+        canvas = add_shadow(plate, center_x, contact_strength)
+        frames[index] = composite_ball(canvas, sprite, center_x, squash)
+
+    artifact_io.save(frames, ROOT)
+
+    # Keep a directly playable video alongside the required frame artifacts.
+    imageio.mimsave(
+        ROOT / "video.mp4",
+        frames,
+        fps=FPS,
+        codec="libx264",
+        quality=8,
+        macro_block_size=1,
+    )

+    main()
