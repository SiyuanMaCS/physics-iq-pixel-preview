+def smoothstep(edge0, edge1, values):
+    t = np.clip((values - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def make_background(image):
+    """Remove the initial ball while preserving the photographed table grain."""
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    ball_mask = ((xx - BALL_X0) ** 2 + (yy - BALL_Y0) ** 2 <= 13.0**2).astype(
+        np.uint8
+    )
+    ball_mask = cv2.dilate(ball_mask, np.ones((3, 3), np.uint8), iterations=1)
+    clean_bgr = cv2.inpaint(
+        cv2.cvtColor(image, cv2.COLOR_RGB2BGR),
+        ball_mask * 255,
+        4,
+        cv2.INPAINT_TELEA,
+    )
+    clean = cv2.cvtColor(clean_bgr, cv2.COLOR_BGR2RGB).astype(np.float32)
+
+    # Restore a little of the fine wood texture that inpainting smooths out.
+    donor = image.astype(np.float32)

+def ellipse_alpha(xx, yy, cx, cy, rx, ry, feather):

+def draw_contact_shadow(background, cx, cy, glass):

+def draw_glass_reflection(frame, cx, cy, glass, strength):

+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]

+    )

+def draw_ball(frame, cx, cy, roll_phase):

+def ball_position(time_s):
+    # The screen-space direction follows the table's receding right edge.
+    speed_x = 24.0 - 1.05 * time_s
+    x = BALL_X0 + 24.0 * time_s - 0.525 * time_s * time_s
+    y = BALL_Y0 + 6.9 * time_s - 0.10 * time_s * time_s
+    return x, y, speed_x
