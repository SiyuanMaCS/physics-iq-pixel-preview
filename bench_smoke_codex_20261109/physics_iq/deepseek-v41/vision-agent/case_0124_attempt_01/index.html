+def make_clean_plate(source):

+def make_mug_sprite(source):
+    """Cut the ceramic mug out with a soft elliptical silhouette."""
+    x0, y0, x1, y1 = 109, 39, 140, 74
+    crop = source[y0:y1, x0:x1].copy()
+    alpha = np.zeros(crop.shape[:2], np.float32)
+    center = (int(round(124.0 - x0)), int(round(56.0 - y0)))
+    cv2.ellipse(alpha, center, (11, 14), 0, 0, 360, 1.0, -1)

+def soft_shadow(base, center, axes, strength, tint):
+    """Composite a blurred contact shadow without disturbing the fixed plate."""
+    layer = np.zeros((HEIGHT, WIDTH), np.float32)
+    cv2.ellipse(layer, (int(round(center[0])), int(round(center[1]))),
+                axes, 0, 0, 360, 1.0, -1)
+    layer = cv2.GaussianBlur(layer, (0, 0), 3.0) * strength
+    shade = np.asarray(tint, np.float32)[None, None, :]
+    out = base.astype(np.float32)
+    out *= (1.0 - layer[:, :, None] * (1.0 - shade))
+    return out
+

+def warp_mug(base, crop, alpha, source_center, destination, angle):
+    """Place the mug sprite with subpixel translation and a small rotation."""
+    src_h, src_w = alpha.shape
+    matrix = cv2.getRotationMatrix2D(
+        (source_center[0], source_center[1]), angle, 1.0
+    )
+    matrix[0, 2] += destination[0] - source_center[0]
+    matrix[1, 2] += destination[1] - source_center[1]
+
+    color = cv2.warpAffine(
+        crop, matrix, (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    ).astype(np.float32)
+    mask = cv2.warpAffine(
+        alpha, matrix, (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    ).astype(np.float32)
+
+    out = base.astype(np.float32)
+    out = out * (1.0 - mask[:, :, None]) + color * mask[:, :, None]
+    return np.clip(out, 0, 255).astype(np.uint8)
+

+def falling_state(frame):
+    """Return mug offset, tilt, and shadow state for one sampled frame."""
+    release_time = 1.60
+    t = frame / FPS
+    dt = t - release_time
+    contact_offset = 29.0
+    gravity = 620.0
+
+    if frame <= 6 or dt <= 0.0:
+        return 0.0, 0.0, 0.0, "held"
+
+    impact_time = np.sqrt(2.0 * contact_offset / gravity)
+    if dt < impact_time:
+        dy = 0.5 * gravity * dt * dt
+        angle = -0.9 * (dt / impact_time)
+        return dy, angle, 0.0, "fall"
+
+    tau = dt - impact_time
+    impact_velocity = gravity * impact_time
+    bounce_velocity = 0.21 * impact_velocity
+    bounce_period = 2.0 * bounce_velocity / gravity
+    if tau < bounce_period:
+        dy = contact_offset - bounce_velocity * tau + 0.5 * gravity * tau * tau
+        phase = tau / bounce_period
+        angle = 2.0 * np.sin(np.pi * phase) * (1.0 - phase)
+        return dy, angle, 0.38 * (1.0 - phase), "bounce"
+
+    settle_tau = tau - bounce_period
+    wobble = 0.9 * np.exp(-8.0 * settle_tau) * np.cos(21.0 * settle_tau)
+    return contact_offset - 0.35, wobble, 1.0, "settled"
+

+def render():

+    assert source.shape == (HEIGHT, WIDTH, 3)

+    crop, alpha, source_center = make_mug_sprite(source)
+    frames = []
+
+    for frame in range(N_FRAMES):
+        # The first seven samples are the untouched source plate, preserving
+        # frame 00 exactly and giving the release a short held beat.
+        if frame <= 6:
+            frames.append(source.copy())
+            continue
+
+        dy, angle, ground_amount, phase = falling_state(frame)
+        center = (source_center[0], source_center[1] + dy)
+        base = plate.astype(np.float32)
+
+        if phase == "fall":
+            shadow_center = (center[0] - 6.0, center[1] + 6.0 + dy * 0.22)
+            base = soft_shadow(
+                base, shadow_center, (10, 6),
+                0.13 * (1.0 - dy / 29.0), (0.68, 0.75, 0.88),
+            )
+        else:
+            base = soft_shadow(
+                base, (center[0], 99.2), (12, 3),
+                0.25 * ground_amount, (0.52, 0.55, 0.62),
+            )
+
+        image = warp_mug(
+            np.clip(base, 0, 255).astype(np.uint8),
+            crop, alpha, source_center, center, angle,
+        )
+        frames.append(image)
+
+    frames = np.stack(frames).astype(np.uint8)
+    assert np.array_equal(frames[0], source)
+    return frames
+
+
+if __name__ == "__main__":
+    frames = render()
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {len(frames)} frames to {CASE_DIR}")
