+def fit_wall_background(image):

+    h, w = image.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+    gray = image.mean(axis=2)
+    mug = (xx >= 88) & (xx <= 158) & (yy >= 84)
+    shadow = (xx >= 118) & (xx <= 218) & (yy >= 37) & (yy <= 119)

+        (xx >= 73)
+        & (xx <= 240)
+        & (yy >= 14)
+        & (yy <= 119)
+        & (gray > 82)
+        & ~mug
+        & ~shadow
+    )

+    def basis(x, y, degree=5):
+        terms = []
+        for total in range(degree + 1):
+            for power_x in range(total + 1):
+                terms.append(x**power_x * y ** (total - power_x))
+        return np.stack(terms, axis=1)

+    x = ((xx - 157.0) / 67.0).ravel()

+    mask = np.zeros((h4, w4), dtype=np.uint8)

+        cv2.fillPoly(mask, [points], 255, lineType=cv2.LINE_AA)
