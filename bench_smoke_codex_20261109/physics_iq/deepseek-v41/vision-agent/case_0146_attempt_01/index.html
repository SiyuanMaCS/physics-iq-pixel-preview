+def soft_shadow(shape, cx, cy):
+    h, w = shape
+    yy, xx = np.mgrid[:h, :w]
+    # Soft, slightly offset contact shadow. The second, smaller lobe gives
+    # the fuzzy ball a grounded contact point rather than a detached halo.
+    dx = (xx - (cx + 2.8)) / 13.5
+    dy = (yy - (cy + 6.2)) / 4.8
+    broad = np.exp(-1.9 * (dx * dx + dy * dy))
+    dx2 = (xx - (cx + 0.6)) / 6.5
+    dy2 = (yy - (cy + 4.3)) / 2.7
+    core = np.exp(-2.6 * (dx2 * dx2 + dy2 * dy2))
+    return np.clip(0.23 * broad + 0.17 * core, 0.0, 0.34)

+def render_ball(center_x, center_y, roll):
+    """Render a small fuzzy gray ball with a rolling surface pattern."""
+    ss = 4
+    r = BALL_RADIUS
+    pad = 3
+    x0 = int(np.floor(center_x - r - pad))
+    y0 = int(np.floor(center_y - r - pad))
+    x1 = int(np.ceil(center_x + r + pad)) + 1
+    y1 = int(np.ceil(center_y + r + pad)) + 1
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+
+    # Supersampling gives a clean antialiased silhouette at this small size.
+    samples = []
+    for oy in range(ss):
+        for ox in range(ss):
+            sx = xx + (ox + 0.5) / ss - 0.5
+            sy = yy + (oy + 0.5) / ss - 0.5
+            nx = (sx - center_x) / r
+            ny = (sy - center_y) / r
+            rr = nx * nx + ny * ny
+            inside = rr <= 1.0
+            zz = np.sqrt(np.clip(1.0 - rr, 0.0, 1.0))
+
+            light = np.array([-0.52, -0.66, 0.54])
+            light /= np.linalg.norm(light)
+            diffuse = np.clip(nx * light[0] + ny * light[1] + zz * light[2], 0.0, 1.0)
+            rim = np.clip(zz, 0.0, 1.0) ** 0.36
+
+            # Faint curved seam plus irregular fuzz. Both are mapped to the
+            # visible hemisphere and move coherently as the ball rolls.

+                + 1.4 * np.sin(41.0 * nx - 27.0 * ny + 3.0 * roll)
+            )
+
+            brightness = 0.57 + 0.37 * diffuse + 0.10 * rim

+

+    alpha = sum(s[1] for s in samples) / len(samples)
+    return color, alpha, (x0, y0, x1, y1)

+def composite_sprite(image, color, alpha, bounds):
+    x0, y0, x1, y1 = bounds
+    crop = image[y0:y1, x0:x1].astype(np.float32)
+    a = alpha[:, :, None]

+    for i in range(1, N):

+

+        frame *= (1.0 - shadow[:, :, None])

+        # without the ball becoming a translucent streak.

+        for offset, weight in ((-0.55, 0.18), (0.0, 0.64), (0.55, 0.18)):

+            crop = layer[y0:y1, x0:x1]
+            a = alpha[:, :, None]
+            layer[y0:y1, x0:x1] = crop * (1.0 - a) + color * a
+            accum += layer * weight
+        frame = accum
+
+        # The lamp is a true foreground occluder.
+        fg = fg_mask[:, :, None]
+        frame = frame * (1.0 - fg) + first.astype(np.float32) * fg
+        frames[i] = np.clip(frame + 0.5, 0, 255).astype(np.uint8)
+
+    assert np.array_equal(frames[0], first)
+    return frames
