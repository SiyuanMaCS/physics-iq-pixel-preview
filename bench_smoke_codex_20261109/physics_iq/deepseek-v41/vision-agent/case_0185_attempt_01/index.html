+FPS = 4
+SCALE = 4

+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)

+def stream_points(time_s, impact_y):
+    ys = np.linspace(76.3, impact_y, 22)
+    phase = time_s * 5.6
+    xs = (
+        129.1

+def stroke_mask(shape, points, thickness, blur):
+    mask = np.zeros(shape, dtype=np.uint8)
+    coords = np.rint(points * SCALE).astype(np.int32)
+    cv2.polylines(
+        mask,
+        [coords.reshape(-1, 1, 2)],
+        False,
+        255,
+        thickness=max(1, round(thickness * SCALE)),
+        lineType=cv2.LINE_AA,
+    )
+    if blur:
+        kernel = max(1, round(blur * SCALE) * 2 + 1)
+        mask = cv2.GaussianBlur(mask, (kernel, kernel), blur * SCALE)
+    return mask

+def composite_mask(region, mask, color, opacity):
+    if mask.shape != region.shape[:2]:
+        mask = cv2.resize(
+            mask,
+            (region.shape[1], region.shape[0]),
+            interpolation=cv2.INTER_AREA,
+        )
+    alpha = (mask.astype(np.float32) / 255.0) * opacity
+    alpha = alpha[..., None]
+    return region * (1.0 - alpha) + np.asarray(color, dtype=np.float32) * alpha
+
+
+def ellipse_mask(shape, center, axes, thickness=-1):
+    mask = np.zeros(shape, dtype=np.uint8)
+    cv2.ellipse(
+        mask,
+        (round(center[0] * SCALE), round(center[1] * SCALE)),
+        (round(axes[0] * SCALE), max(1, round(axes[1] * SCALE))),
+        0,
+        0,
+        360,
+        255,
+        thickness=-1 if thickness < 0 else max(1, round(thickness * SCALE)),
+        lineType=cv2.LINE_AA,
+    )
+    return mask

+def liquid_mask(base):

+def main():
