+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+W, H = 256, 144
+N_FRAMES = 20
+FPS = 4
+CASE_DIR = Path(__file__).resolve().parent
+
+
+def _circle_mask(center, radius, supersample=4):
+    cx, cy = center
+    yy, xx = np.mgrid[0:H, 0:W]
+    distance = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+    return np.clip(radius + 0.5 - distance, 0.0, 1.0)
+
+
+def _make_plate(image):
+    """Remove the original balls and their contact shadows from the tabletop."""
+    mask = np.zeros((H, W), np.uint8)
+    cv2.ellipse(mask, (106, 61), (11, 11), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (158, 62), (21, 11), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (150, 63), (13, 9), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (166, 64), (13, 9), 0, 0, 360, 255, -1)
+    return cv2.inpaint(image, mask, 5, cv2.INPAINT_TELEA)
+
+
+def _extract_sprite(image, center, radius):
+    cx, cy = center
+    pad = 4
+    x0, x1 = int(cx - radius - pad), int(cx + radius + pad + 1)
+    y0, y1 = int(cy - radius - pad), int(cy + radius + pad + 1)
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    distance = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
+    alpha = np.clip(radius + 0.55 - distance, 0.0, 1.0)
+    return image[y0:y1, x0:x1].copy(), alpha.astype(np.float32), (x0, y0)
+
+
+def _overlay_sprite(frame, sprite, alpha, origin, center):
+    x0, y0 = origin
+    cx, cy = center
+    ox = int(round(cx - (sprite.shape[1] - 1) / 2))
+    oy = int(round(cy - (sprite.shape[0] - 1) / 2))
+    h, w = sprite.shape[:2]
+    dst = frame[oy : oy + h, ox : ox + w].astype(np.float32)
+    a = alpha[..., None]
+    frame[oy : oy + h, ox : ox + w] = np.clip(
+        sprite.astype(np.float32) * a + dst * (1.0 - a), 0, 255
+    ).astype(np.uint8)
+
+
+def _add_shadow(frame, center, radius, strength=0.28):
+    cx, cy = center
+    yy, xx = np.mgrid[0:H, 0:W]
+    sx = radius * 1.35
+    sy = radius * 0.48
+    dx = (xx - (cx + radius * 0.55)) / sx
+    dy = (yy - (cy + radius * 0.70)) / sy
+    falloff = np.clip(1.0 - (dx * dx + dy * dy), 0.0, 1.0) ** 1.7
+    table = np.clip((yy - 34) / 7.0, 0.0, 1.0) * np.clip((109 - yy) / 6.0, 0.0, 1.0)
+    alpha = falloff * table * strength
+    shadow_color = np.array([22, 19, 17], np.float32)
+    frame[:] = (
+        frame.astype(np.float32) * (1.0 - alpha[..., None])
+        + shadow_color * alpha[..., None]
+    ).astype(np.uint8)
+
+
+def _position(t):
+    """One-dimensional rolling and a lightly damped two-ball impact."""
+    gray_start = 106.0
+    gray_contact = 134.0
+    impact = 1.15
+    brown_start = 150.0
+    orange_start = 165.0
+
+    if t < impact:
+        gray_x = gray_start + (gray_contact - gray_start) * t / impact
+        brown_x = brown_start
+        orange_x = orange_start
+    else:
+        dt = t - impact
+
+        # The grey ball is stopped by the first ball; the impulse then
+        # propagates through the touching brown/orange pair.
+        gray_x = gray_contact
+        brown_x = brown_start + min(4.0, 5.0 * dt - 1.1 * dt * dt)
+        orange_x = orange_start + min(16.0, 18.0 * dt - 3.2 * dt * dt)
+
+    return gray_x, brown_x, orange_x
+
+
+def generate():
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    assert source.shape == (H, W, 3)
+    plate = _make_plate(source)
+
+    gray_sprite, gray_alpha, gray_origin = _extract_sprite(source, (106, 57), 8.6)
+    brown_sprite, brown_alpha, brown_origin = _extract_sprite(source, (150, 57), 8.7)
+    orange_sprite, orange_alpha, orange_origin = _extract_sprite(source, (165, 58), 8.8)
+
+    frames = []
+    for index in range(N_FRAMES):
+        if index == 0:
+            frames.append(source.copy())
+            continue
+
+        t = index / FPS
+        gray_x, brown_x, orange_x = _position(t)
+        frame = plate.copy()
+
+        # Draw back-to-front so the small overlap at the pair's contact
+        # retains the original frame's occlusion.
+        _add_shadow(frame, (orange_x, 58), 8.8, 0.24)
+        _add_shadow(frame, (brown_x, 57), 8.7, 0.27)
+        _add_shadow(frame, (gray_x, 57), 8.6, 0.29)
+        _overlay_sprite(frame, orange_sprite, orange_alpha, orange_origin, (orange_x, 58))
+        _overlay_sprite(frame, brown_sprite, brown_alpha, brown_origin, (brown_x, 57))
+        _overlay_sprite(frame, gray_sprite, gray_alpha, gray_origin, (gray_x, 57))
+        frames.append(frame)
+
+    frames = np.stack(frames).astype(np.uint8)
+    artifact_io.save(frames, CASE_DIR)
+    return frames
+
+
+if __name__ == "__main__":
+    generate()
