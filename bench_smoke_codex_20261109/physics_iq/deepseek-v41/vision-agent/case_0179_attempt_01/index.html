+def _sample_bilinear(image, x, y):
+    """Bilinear sample an RGB float image at floating point coordinates."""
+    height, width = image.shape[:2]
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    fx = (x - x0)[..., None]
+    fy = (y - y0)[..., None]
+    x0 = np.clip(x0, 0, width - 1)
+    x1 = np.clip(x0 + 1, 0, width - 1)
+    y0 = np.clip(y0, 0, height - 1)
+    y1 = np.clip(y0 + 1, 0, height - 1)
+    top = image[y0, x0] * (1.0 - fx) + image[y0, x1] * fx
+    bottom = image[y1, x0] * (1.0 - fx) + image[y1, x1] * fx
+    return top * (1.0 - fy) + bottom * fy

+def _composite_shadow(frame, position, radius, strength):
+    """Composite a soft, slightly directional contact shadow."""
+    x, y = position
+    pad_x, pad_y = int(radius * 2.1), int(radius * 1.2)
+    cx, cy = int(round(x)), int(round(y))
+    x0, x1 = max(0, cx - pad_x), min(W, cx + pad_x + 1)
+    y0, y1 = max(0, cy - pad_y), min(H, cy + pad_y + 1)
+    if x0 >= x1 or y0 >= y1:
+        return
+    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
+    sx = x + radius * 0.22
+    sy = y + radius * 0.77

+    b = radius * 0.49
+    alpha = np.exp(-2.2 * (((xx - sx) / a) ** 2 + ((yy - sy) / b) ** 2))
+    alpha *= strength

+def _composite_ball(frame, sprite):
+    color, alpha, bounds = sprite
+    x0, y0, x1, y1 = bounds
+    sx0, sy0 = max(0, x0), max(0, y0)
+    sx1, sy1 = min(W, x1), min(H, y1)
+    if sx0 >= sx1 or sy0 >= sy1:
+        return
+    ax0, ay0 = sx0 - x0, sy0 - y0
+    ax1, ay1 = ax0 + sx1 - sx0, ay0 + sy1 - sy0
+    a = alpha[ay0:ay1, ax0:ax1, None]
+    c = color[ay0:ay1, ax0:ax1]
+    frame[sy0:sy1, sx0:sx1] = frame[sy0:sy1, sx0:sx1] * (1.0 - a) + c * a
+

+def _ball_state(name, time_seconds):
+    spec = BALLS[name]
+    position = spec["center"] + spec["velocity"] * time_seconds
+    distance = float(np.linalg.norm(spec["velocity"] * time_seconds))
+    radius = float(spec["radius"][0])
+    phase = spec["phase"] + distance / radius
+    return position, phase, radius
+

+def _render_sample(plate, source, time_seconds):
+    frame = plate.copy()
+    # Shadows are composited before the balls.
+    for name in ("grey", "brown"):
+        position, _, radius = _ball_state(name, time_seconds)
+        if -30 < position[0] < W + 30:
+            strength = 0.17 if name == "brown" else 0.20
+            _composite_shadow(frame, position, radius, strength)
+    for name in ("grey", "brown"):
+        position, phase, _ = _ball_state(name, time_seconds)
+        if -30 < position[0] < W + 30 and -30 < position[1] < H + 30:
+            _composite_ball(frame, _make_ball_sprite(source, BALLS[name], position, phase))
+
+    # The brown ball disappears into the left pipe rather than rolling over it.
+    if -30 < _ball_state("brown", time_seconds)[0][0] < 48:
+        frame[:, :42] = source[:, :42]
+    return np.clip(frame, 0.0, 255.0)
+

+    frames = [source.astype(np.uint8)]

+    for frame_index in range(1, N_FRAMES):
+        t = frame_index / FPS
+        accumulated = np.zeros((H, W, 3), np.float32)
+        for offset in shutter_offsets:
+            accumulated += _render_sample(plate, source, t + float(offset) / FPS)
+        frame = np.clip(accumulated / len(shutter_offsets) + 0.5, 0, 255).astype(np.uint8)
+        frames.append(frame)
+    return np.stack(frames)
