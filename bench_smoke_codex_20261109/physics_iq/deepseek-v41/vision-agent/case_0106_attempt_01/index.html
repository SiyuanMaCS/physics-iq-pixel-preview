+def ramp_roll(start, end, duration, time, speedup=1.7):
+    """Roll along a ramp with gravity-like acceleration and a soft stop."""
+    u = np.clip(time / duration, 0.0, 1.0)
+    travel = (u**speedup) / (1.0 + (speedup - 1.0) * u)
+    return np.asarray(start) + (np.asarray(end) - np.asarray(start)) * travel

+def composite_shadow(image, center, kind):

+    body += np.array([18.0, 15.0, 8.0]) * (z[..., None] ** 2)
+
+    coverage = np.clip((1.0 - np.sqrt(rr)) * RADIUS + 0.55, 0.0, 1.0)
+    alpha = coverage * opacity * (0.96 - 0.18 * edge)
+    alpha = alpha[..., None]
+    image[y0:y1, x0:x1] = body * alpha + image[y0:y1, x0:x1] * (1.0 - alpha)
+
+
+def generate():
+    source = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    clean = make_clean_plate(source)
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = source

+    previous = marble_pose(0)[0]

+        frames[index] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+        previous = center

+    artifact_io.save(frames, ROOT)
+

+    generate()
