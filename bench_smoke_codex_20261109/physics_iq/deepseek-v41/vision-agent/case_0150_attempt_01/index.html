+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+WIDTH = 256
+HEIGHT = 144
+N_FRAMES = 20
+FPS = 4
+
+# The photographed ball, measured from first_frame.png.
+BALL_CX = 104.5
+BALL_CY = 47.5
+BALL_RX = 11.2
+BALL_RY = 9.6
+
+
+def ellipse_mask(cx, cy, rx, ry, feather=1.0):
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    distance = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    return np.clip((1.0 - distance) * max(rx, ry) / feather + 0.5, 0.0, 1.0)
+
+
+def make_clean_plate(frame):
+    """Remove the photographed ball and its contact shadow."""
+    mask = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    cv2.ellipse(
+        mask,
+        (round(BALL_CX), round(BALL_CY)),
+        (round(BALL_RX + 1.8), round(BALL_RY + 2.0)),
+        0,
+        0,
+        360,
+        255,
+        -1,
+    )
+    cv2.ellipse(mask, (116, 54), (10, 6), 0, 0, 360, 255, -1)
+    mask = cv2.dilate(mask, np.ones((3, 3), dtype=np.uint8), iterations=1)
+    return cv2.inpaint(frame, mask, 7, cv2.INPAINT_NS)
+
+
+def make_ball_sprite(frame):
+    """Extract a clean, rotating copy of the ball from the first frame."""
+    size = 41
+    left = int(round(BALL_CX - (size - 1) / 2))
+    top = int(round(BALL_CY - (size - 1) / 2))
+    crop = frame[top : top + size, left : left + size].copy()
+
+    yy, xx = np.mgrid[0:size, 0:size]
+    local_cx = BALL_CX - left
+    local_cy = BALL_CY - top
+    edge = np.sqrt(
+        ((xx - local_cx) / (BALL_RX + 0.35)) ** 2
+        + ((yy - local_cy) / (BALL_RY + 0.35)) ** 2
+    )
+    source_alpha = np.clip((1.0 - edge) * 7.0 + 0.5, 0.0, 1.0)
+    rgba = np.dstack((crop, (source_alpha * 255).round().astype(np.uint8)))
+    return Image.fromarray(rgba, "RGBA")
+
+
+def add_contact_shadow(image, cx, cy):
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    shadow = np.exp(
+        -0.5
+        * (
+            ((xx - (cx + 4.7)) / 10.0) ** 2
+            + ((yy - (cy + 4.6)) / 3.8) ** 2
+        )
+    )
+    contact = np.exp(
+        -0.5
+        * (
+            ((xx - (cx + 2.8)) / 7.0) ** 2
+            + ((yy - (cy + 4.2)) / 2.5) ** 2
+        )
+    )
+    attenuation = 1.0 - 0.22 * shadow - 0.17 * contact
+    return np.clip(image.astype(np.float32) * attenuation[..., None], 0, 255)
+
+
+def composite_sprite(canvas, sprite, angle, cx, cy):
+    rotated = sprite.rotate(
+        angle,
+        resample=Image.Resampling.BICUBIC,
+        center=((sprite.width - 1) / 2, (sprite.height - 1) / 2),
+    )
+    sprite_rgb = np.asarray(rotated.convert("RGB"), dtype=np.float32)
+    alpha = ellipse_mask(
+        (sprite.width - 1) / 2,
+        (sprite.height - 1) / 2,
+        BALL_RX,
+        BALL_RY,
+        feather=1.15,
+    )
+
+    left = int(np.floor(cx - (sprite.width - 1) / 2))
+    top = int(np.floor(cy - (sprite.height - 1) / 2))
+    x0, y0 = max(0, left), max(0, top)
+    x1 = min(WIDTH, left + sprite.width)
+    y1 = min(HEIGHT, top + sprite.height)
+    if x0 >= x1 or y0 >= y1:
+        return canvas
+
+    sx0, sy0 = x0 - left, y0 - top
+    sx1 = sx0 + (x1 - x0)
+    sy1 = sy0 + (y1 - y0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    source = sprite_rgb[sy0:sy1, sx0:sx1]
+    destination = canvas[y0:y1, x0:x1]
+    canvas[y0:y1, x0:x1] = source * a + destination * (1.0 - a)
+    return canvas
+
+
+def generate():
+    case_dir = Path(__file__).resolve().parent
+    first = np.asarray(Image.open(case_dir / "first_frame.png").convert("RGB"))
+    clean_plate = make_clean_plate(first)
+    sprite = make_ball_sprite(first)
+    frames = np.empty((N_FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+
+    # Frame zero is copied verbatim to satisfy the pixel-identical requirement.
+    frames[0] = first
+    for index in range(1, N_FRAMES):
+        cx = BALL_CX + 8.55 * index
+        cy = BALL_CY + 0.012 * index
+        canvas = add_contact_shadow(clean_plate, cx, cy)
+        # The 2D texture rotation keeps the white seam visibly turning as it rolls.
+        angle = -31.0 * index
+        canvas = composite_sprite(canvas, sprite, angle, cx, cy)
+        frames[index] = np.clip(canvas, 0, 255).round().astype(np.uint8)
+
+    return frames
+
+
+if __name__ == "__main__":
+    case_dir = Path(__file__).resolve().parent
+    frames = generate()
+    artifact_io.save(frames, case_dir)
