+def _smoothstep(value):

+def _row_interpolated_fill(image, mask):
+    """Fill masked pixels between the nearest visible pixels on each row."""
+    filled = image.astype(np.float32).copy()
+    mask_bool = mask.astype(bool)
+    for y in range(HEIGHT):
+        row_mask = mask_bool[y]
+        if not row_mask.any():
+            continue
+        visible = np.flatnonzero(~row_mask)
+        if visible.size == 0:
+            continue
+        for x in np.flatnonzero(row_mask):
+            left = visible[visible < x]
+            right = visible[visible > x]
+            if left.size and right.size:
+                x0, x1 = left[-1], right[0]
+                blend = (x - x0) / max(1, x1 - x0)
+                filled[y, x] = image[y, x0] * (1.0 - blend) + image[y, x1] * blend
+            elif left.size:
+                filled[y, x] = image[y, left[-1]]
+            elif right.size:
+                filled[y, x] = image[y, right[0]]
+    return filled

+def _sphere_patch(cx, cy, radius):
+    pad = 4
+    size = int(np.ceil(radius * 2 + pad * 2))
+    scale = 4
+    s = size * scale
+    yy, xx = np.mgrid[0:s, 0:s].astype(np.float32) / scale

+    nx = (xx - center) / radius
+    ny = (yy - center) / radius
+    rr = nx * nx + ny * ny
+    inside = rr <= 1.0

+    nz = np.sqrt(np.clip(1.0 - rr, 0.0, 1.0))
+    light = np.array([-0.43, -0.57, 0.70], dtype=np.float32)
