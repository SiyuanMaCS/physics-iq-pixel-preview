+def make_plate(frame: np.ndarray) -> np.ndarray:
+    """Remove the starting ball while preserving the table and tube mouth."""
+    plate = frame.copy()
+    mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.ellipse(mask, (50, 50), (14, 13), 0, 0, 360, 255, -1)
+    plate = cv2.inpaint(plate, mask, 4, cv2.INPAINT_TELEA)
+
+    # Rebuild the dark tube opening that the initial ball partly covers.
+    tube = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.ellipse(tube, (47, 48), (10, 8), -8, 0, 360, 255, -1)
+    tube_soft = cv2.GaussianBlur(tube.astype(np.float32) / 255.0, (3, 3), 0.55)
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    rr = np.hypot((xx - 47.0) / 10.0, (yy - 48.0) / 8.0)
+    inside = np.clip((1.0 - rr) * 3.0, 0.0, 1.0)
+    rim = np.exp(-((rr - 0.93) / 0.13) ** 2)
+    mouth_rgb = np.zeros_like(plate, dtype=np.float32)
+    mouth_rgb[:] = (17.0, 13.0, 10.0)
+    mouth_rgb += rim[..., None] * np.array((34.0, 24.0, 18.0))
+    alpha = (inside * tube_soft * 0.86)[..., None]
+    plate = plate.astype(np.float32) * (1.0 - alpha) + mouth_rgb * alpha
+    return np.clip(plate, 0, 255).astype(np.uint8)

+def shade_mask() -> np.ndarray:
+    """Anti-aliased silhouette of the opaque lampshade."""
+    scale = 4
+    mask = Image.new("L", (WIDTH * scale, HEIGHT * scale), 0)
+    draw = ImageDraw.Draw(mask)

+        (76, 20),
+        (139, 20),
+        (160, 110),
+        (57, 110),
+    ]
+    draw.polygon([(x * scale, y * scale) for x, y in polygon], fill=255)
+    mask = mask.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
+    return np.asarray(mask, dtype=np.float32) / 255.0

+def smoothstep(edge0: float, edge1: float, x: np.ndarray) -> np.ndarray:
+    t = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def sphere_sprite(radius: float, roll: float) -> tuple[np.ndarray, np.ndarray]:
+    """Render a small grey tennis ball at 4x resolution."""
+    scale = 4
+    size = int(np.ceil(2.0 * (radius + 1.0) * scale))
+    yy, xx = np.mgrid[:size, :size].astype(np.float32)
+    center = (size - 1.0) * 0.5
+    nx = (xx - center) / (radius * scale)
+    ny = (yy - center) / (radius * scale)
+    rr2 = nx * nx + ny * ny
+    inside = rr2 <= 1.0
+    nz = np.sqrt(np.clip(1.0 - rr2, 0.0, 1.0))
+
+    light = np.array((-0.53, -0.60, 0.60), dtype=np.float32)
+    light /= np.linalg.norm(light)
+    diffuse = np.clip(nx * light[0] + ny * light[1] + nz * light[2], 0.0, 1.0)
+    ambient = 0.30 + 0.13 * nz
+    shade = 0.39 * diffuse + ambient
+
+    # Equirectangular felt pattern; the offset makes the ball visibly roll.
+    lon = np.arctan2(nx, nz)
+    lat = np.arcsin(np.clip(ny, -1.0, 1.0))
+    u = lon + roll
+    fuzz = (
+        0.055 * np.sin(19.0 * u + 8.0 * lat)
+        + 0.035 * np.sin(31.0 * u - 13.0 * lat + 1.7)
+        + 0.022 * np.sin(47.0 * u + 21.0 * lat)
+    )
+
+    # A faint curved tennis seam, kept subtle because the ball is only ~18 px wide.
+    seam_a = np.abs(np.sin(1.65 * u + 0.65 * np.sin(2.0 * lat) + 0.35))
+    seam_b = np.abs(np.sin(1.65 * u - 0.65 * np.sin(2.0 * lat) + np.pi + 0.35))
+    seam = np.maximum(np.exp(-(seam_a / 0.075) ** 2), np.exp(-(seam_b / 0.075) ** 2))
+    seam *= smoothstep(0.15, 0.55, nz)
+
+    base = np.array((153.0, 149.0, 139.0), dtype=np.float32)
+    rgb = base[None, None, :] * shade[..., None]
+    rgb *= (1.0 + fuzz)[..., None]
+    rgb += seam[..., None] * np.array((43.0, 40.0, 34.0))
+
+    # Soft edge and a cool sky reflection around the upper-left.
+    edge = smoothstep(0.91, 1.0, np.sqrt(rr2))
+    alpha = (1.0 - edge) * inside.astype(np.float32)
+    spec = np.clip(-nx * 0.42 - ny * 0.55 + nz * 0.72, 0.0, 1.0) ** 22
+    rgb += spec[..., None] * np.array((24.0, 25.0, 25.0))
+
+    rgb = cv2.resize(rgb, (size // scale, size // scale), interpolation=cv2.INTER_AREA)
+    alpha = cv2.resize(alpha, (size // scale, size // scale), interpolation=cv2.INTER_AREA)
+    return np.clip(rgb, 0, 255), np.clip(alpha, 0, 1)
+

+def add_contact_shadow(
+    image: np.ndarray, cx: float, cy: float, radius: float, strength: float
+) -> np.ndarray:
+    """Darken the table with a soft contact shadow beneath the ball."""
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH].astype(np.float32)
+    sx = radius * 1.55
+    sy = radius * 0.52
+    px = cx + radius * 0.48
+    py = cy + radius * 1.02
+    core = np.exp(-(((xx - px) / sx) ** 2 + ((yy - py) / sy) ** 2) * 2.3)
+    penumbra = np.exp(-(((xx - (px + 1.0)) / (sx * 1.55)) ** 2
+                       + ((yy - (py + 1.0)) / (sy * 1.8)) ** 2) * 2.0)
+    shadow = np.clip(core * 0.72 + penumbra * 0.28, 0.0, 1.0) * strength
+    shadow *= smoothstep(55.0, 72.0, xx)
+    tint = np.array((48.0, 38.0, 27.0), dtype=np.float32)
+    return image.astype(np.float32) * (1.0 - shadow[..., None] * 0.52) + tint * (
+        shadow[..., None] * 0.18
+    )
+
+
+def composite_rgba(
+    image: np.ndarray, rgb: np.ndarray, alpha: np.ndarray, left: int, top: int
+) -> np.ndarray:
+    h, w = alpha.shape
+    x0 = max(0, left)
+    y0 = max(0, top)
+    x1 = min(WIDTH, left + w)
+    y1 = min(HEIGHT, top + h)
+    if x0 >= x1 or y0 >= y1:
+        return image
+    sx0 = x0 - left
+    sy0 = y0 - top
+    sx1 = sx0 + (x1 - x0)
+    sy1 = sy0 + (y1 - y0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    out = image.copy()
+    out[y0:y1, x0:x1] = (
+        out[y0:y1, x0:x1].astype(np.float32) * (1.0 - a)
+        + rgb[sy0:sy1, sx0:sx1].astype(np.float32) * a
+    )
+    return out
+
+
+def make_frames() -> np.ndarray:
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    plate = make_plate(source)
+    occluder = shade_mask()
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = source
+
+    radius = 9.1
+    start_x = 50.0
+    start_y = 48.5
+    speed = 9.6
+    slope = 0.102
+
+    for index in range(1, FRAME_COUNT):
+        # A very short acceleration as the ball clears the tube, then steady rolling.
+        eased = index - 0.72 * np.exp(-index / 1.35)
+        cx = start_x + speed * eased
+        cy = start_y + slope * (cx - start_x)
+
+        image = add_contact_shadow(plate, cx, cy, radius, 0.72)
+        roll = (cx - start_x) / radius
+        rgb, alpha = sphere_sprite(radius, roll)
+        top = int(round(cy - (alpha.shape[0] - 1) * 0.5))
+        left = int(round(cx - (alpha.shape[1] - 1) * 0.5))
+        image = composite_rgba(image, rgb, alpha, left, top)
+
+        # The ball is behind the lampshade; restore the shade silhouette over it.
+        a = occluder[..., None]
+        image = image.astype(np.float32) * (1.0 - a) + source.astype(np.float32) * a
+        frames[index] = np.clip(image, 0, 255).astype(np.uint8)
+
+    return frames
+
+
+def main() -> None:
+    frames = make_frames()
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
