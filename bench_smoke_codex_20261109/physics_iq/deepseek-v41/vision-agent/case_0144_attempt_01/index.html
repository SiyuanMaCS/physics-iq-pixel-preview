+def smoothstep(edge0: float, edge1: float, x: np.ndarray | float):
+    t = np.clip((np.asarray(x) - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def ellipse_alpha(
+    height: int,
+    width: int,
+    center: tuple[float, float],
+    radii: tuple[float, float],
+    feather: float = 0.9,
+) -> np.ndarray:
+    yy, xx = np.mgrid[:height, :width].astype(np.float32)
+    r = np.sqrt(
+        ((xx - center[0]) / radii[0]) ** 2
+        + ((yy - center[1]) / radii[1]) ** 2
+    )
+    return np.clip((1.02 - r) / feather, 0.0, 1.0)

+def physics_state(t: float) -> tuple[np.ndarray, float, bool]:
+    """Return potato center, rotation, and whether it is fully underwater."""
+    if t <= RELEASE_T:
+        return POTATO_START.copy(), 0.0, False
+
+    dt = t - RELEASE_T
+    fall_distance = max(0.0, CONTACT_CENTER_Y - POTATO_START[1])
+    entry_center_y = SURFACE_Y
+    air_distance = entry_center_y - POTATO_START[1]
+    air_accel = 174.0
+    t_air = np.sqrt(2.0 * air_distance / air_accel)
+    v_entry = air_accel * t_air
+
+    if dt < t_air:
+        y = POTATO_START[1] + 0.5 * air_accel * dt * dt
+        vy = air_accel * dt
+    else:
+        tw = dt - t_air
+        # Water drag plus buoyancy slows the potato without making it float.
+        water_accel = -34.0
+        y = entry_center_y + v_entry * tw + 0.5 * water_accel * tw * tw
+        vy = v_entry + water_accel * tw
+        if y >= CONTACT_CENTER_Y:
+            # Inelastic contact with the glass base, followed by a tiny rebound.
+            over = y - CONTACT_CENTER_Y
+            y = CONTACT_CENTER_Y - min(3.2, over * 0.18)
+            if tw > 0.0:
+                vy = -min(8.0, abs(vy) * 0.045)
+
+    x = POTATO_START[0] - 1.55 * smoothstep(0.0, 0.95, dt)
+    if t > 1.75:
+        x += 0.18 * np.sin((t - 1.75) * 7.0) * np.exp(-(t - 1.75) * 2.2)
+
+    settle = smoothstep(1.45, 2.35, t)
+    rotation = (
+        -2.2 * smoothstep(0.0, 0.35, dt)
+        + 8.2 * smoothstep(0.0, 1.0, dt)
+        - 6.3 * settle
+    )
+    if t > 1.5:
+        rotation += 0.45 * np.sin((t - 1.5) * 5.0) * np.exp(-(t - 1.5) * 2.4)
+    fully_under = y - POTATO_AXES[1] > SURFACE_Y - 2.0
+    return np.array([x, y]), float(rotation), fully_under

+def draw_potato_shadow(
+    canvas: np.ndarray, center: np.ndarray, alpha: np.ndarray

+    # A compact contact shadow follows the potato and fades once it is deep.
+    depth = max(0.0, center[1] - SURFACE_Y)
+    strength = 0.18 * max(0.15, 1.0 - depth / 55.0)
+    h, w = 10, 30
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    blob = np.exp(
+        -(((xx - w * 0.5) / 11.0) ** 2 + ((yy - h * 0.55) / 2.5) ** 2)
+    )
+    blob = cv2.GaussianBlur(blob, (0, 0), 1.1)
+    cx, cy = int(round(center[0])), int(round(center[1] + 27.5))
+    x0, x1 = cx - w // 2, cx - w // 2 + w
+    y0, y1 = cy - h // 2, cy - h // 2 + h
+    sx0, sy0 = max(0, x0), max(0, y0)
+    sx1, sy1 = min(W, x1), min(H, y1)
+    if sx0 < sx1 and sy0 < sy1:
+        lx0, ly0 = sx0 - x0, sy0 - y0
+        lx1, ly1 = lx0 + sx1 - sx0, ly0 + sy1 - sy0
+        a = blob[ly0:ly1, lx0:lx1, None] * strength
+        a *= (1.0 - alpha[sy0:sy1, sx0:sx1, None])
+        dst = canvas[sy0:sy1, sx0:sx1]
+        canvas[sy0:sy1, sx0:sx1] = dst * (1.0 - a) + np.array(
+            [24.0, 29.0, 21.0]
+        ) * a
+    return canvas

+    alpha = np.zeros((H, W), dtype=np.float32)

+    right_x = 128.5 + 4.3 * opening + 3.0 * retract
+    bottom = 22.5 - 21.0 * retract

+def warp_water_surface(base: np.ndarray, t: float) -> np.ndarray:
+    if t < IMPACT_T - 0.04:
+        return base
+    elapsed = max(0.0, t - IMPACT_T)
+    yy, xx = np.mgrid[:H, :W].astype(np.float32)
+    distance = xx - 119.0
+    depression = (
+        5.3
+        * np.exp(-elapsed / 0.33)
+        * np.exp(-(distance / 7.0) ** 2)
+        + 0.95
+        * np.exp(-elapsed / 1.35)
+        * np.sin(2.0 * np.pi * (distance / 15.0 - elapsed * 2.25))
+        * np.exp(-(distance / 21.0) ** 2)
+    )
+    band = smoothstep(69.0, 73.0, yy) * (1.0 - smoothstep(84.0, 89.0, yy))
+    band *= smoothstep(91.0, 97.0, xx) * (1.0 - smoothstep(140.0, 146.0, xx))
+    map_x = np.ascontiguousarray(xx, dtype=np.float32)
+    map_y = np.ascontiguousarray(yy - depression * band, dtype=np.float32)
+    warped = cv2.remap(
+        base,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT,
+    )
+    weight = band * (1.0 - smoothstep(83.0, 87.0, yy))
+    return base * (1.0 - weight[..., None]) + warped * weight[..., None]

+def add_caustic_highlights(canvas: np.ndarray, t: float) -> np.ndarray:
+    # Two extremely soft moving bands keep the still liquid from looking frozen.
+    yy, xx = np.mgrid[:H, :W].astype(np.float32)
+    in_water = (yy > SURFACE_Y + 1.0) & (yy < 132.0) & (xx > 95.0) & (xx < 141.0)
+    wave = (
+        np.sin(xx * 0.27 + t * 1.7)
+        + np.sin(yy * 0.18 - t * 1.15)
+        + 0.45 * np.sin((xx + yy) * 0.35 + t * 2.3)

+    glow = np.clip((wave - 1.65) / 1.2, 0.0, 1.0) * in_water
+    glow *= np.clip((yy - SURFACE_Y) / 20.0, 0.0, 1.0)
+    glow = cv2.GaussianBlur(glow.astype(np.float32), (0, 0), 1.0)
+    color = np.array([18.0, 23.0, 18.0], dtype=np.float32)
+    return canvas + glow[..., None] * color

+def make_frame(
+    index: int,
+    first_frame: np.ndarray,
+    base: np.ndarray,
+    sprite_rgb: np.ndarray,
+    sprite_alpha: np.ndarray,
+) -> np.ndarray:
+    if index == 0:
+        return first_frame.copy()
+
+    t = index / FPS
+    canvas = warp_water_surface(base.astype(np.float32), t)
+    center, angle, _ = physics_state(t)
+
+    # Draw the tool before the object so the fingers visibly open behind it.
+    canvas = draw_gripper(canvas, t)
+    canvas = draw_potato_shadow(canvas, center, np.zeros((H, W), dtype=np.float32))
+    canvas, object_alpha = paste_sprite(
+        canvas, sprite_rgb, sprite_alpha, center, angle, t
+    )
+    canvas = draw_bubbles(canvas, t)
+    canvas = draw_splash(canvas, t)
+    canvas = add_caustic_highlights(canvas, t)
+
+    # Very faint temporal sensor variation; the locked-off geometry stays fixed.
+    rng = np.random.default_rng(1400 + index)
+    noise = rng.normal(0.0, 0.16, (H, W, 1)).astype(np.float32)
+    canvas = canvas + noise
+    return np.clip(np.rint(canvas), 0, 255).astype(np.uint8)
