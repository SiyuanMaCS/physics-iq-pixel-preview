+def build_tool_mask(frame):
+    """Recover the blue-black gripper from its known upper-frame region."""
+    region = frame[0:33, 120:160]
+    r, g, b = region[:, :, 0], region[:, :, 1], region[:, :, 2]
+    dark = np.maximum.reduce((r, g, b)) < 78
+    blue = (b > r + 11) & (b > 54) & (g > 38)
+    candidate = dark | blue
+
+    # The tool silhouette is contiguous on every row. Filling between its
+    # extreme colored pixels also captures the black body between blue rails.
+    mask = np.zeros_like(candidate)
+    for y in range(candidate.shape[0]):
+        xs = np.flatnonzero(candidate[y])
+        if xs.size:
+            mask[y, xs.min() : xs.max() + 1] = True
+
+    mask = cv2.morphologyEx(mask.astype(np.uint8), cv2.MORPH_CLOSE, np.ones((2, 2), np.uint8))
+    mask = cv2.dilate(mask, np.ones((2, 2), np.uint8), iterations=1)
+
+    alpha = cv2.GaussianBlur(mask.astype(np.float32), (3, 3), 0.42)
+    alpha = np.clip((alpha - 0.10) / 0.82, 0.0, 1.0)
+
+    full_mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    full_mask[0:33, 120:160] = mask
+    full_alpha = np.zeros((HEIGHT, WIDTH), np.float32)
+    full_alpha[0:33, 120:160] = alpha
+    return full_mask, full_alpha
+
+
+def reconstruct_background(frame, tool_mask):
+    """Remove the gripper while retaining the original wall, blocks and grain."""
+    inpaint_mask = cv2.dilate(tool_mask, np.ones((5, 5), np.uint8), iterations=1)
+    cleaned = cv2.inpaint(frame, inpaint_mask, 4, cv2.INPAINT_TELEA)
+
+    # Inpainting tends to flatten the fine sensor grain. Restore a restrained
+    # amount of the neighboring texture only inside the repaired wall area.
+    ys, xs = np.where((inpaint_mask > 0) & (np.indices(inpaint_mask.shape)[0] < 28))
+    if ys.size:
+        grain = np.zeros_like(frame, dtype=np.int16)
+        noise = np.random.default_rng(173).normal(0.0, 1.25, size=(ys.size, 3))
+        grain[ys, xs] = np.rint(noise).astype(np.int16)
+        cleaned = np.clip(cleaned.astype(np.int16) + grain, 0, 255).astype(np.uint8)
+
+    return cleaned
+

+def restore_block_top(frame):
+    """Rebuild the few pink pixels that were hidden by the closed gripper."""
+    result = frame.copy()
+    x0, x1 = 134, 144
+    # Copy the stem's stable shading and grain upward, with a rounded cap.
+    source = result[35:41, x0:x1].astype(np.float32)
+    widths = {
+        28: (137, 141),
+        29: (135, 143),
+        30: (134, 144),
+        31: (134, 144),
+        32: (134, 144),
+    }
+    for y, (left, right) in widths.items():
+        source_y = min(3, y - 28)
+        row = source[source_y].copy()
+        if y in (28, 29):
+            row *= 0.92
+        result[y, left:right] = np.clip(row[: right - left], 0, 255).astype(np.uint8)
+
+    # Re-establish the darker left edge and warm highlight seen on the stem.
+    result[28:33, 134] = (result[34:39, 134].astype(np.float32) * 0.84).astype(np.uint8)
+    result[28:33, 143] = (result[34:39, 143].astype(np.float32) * 0.96).astype(np.uint8)
+    return result

+def warp_half(rgb, alpha, shift_x, shift_y):
+    """Move one jaw with bilinear sampling and matched alpha."""
+    yy, xx = np.indices((HEIGHT, WIDTH), dtype=np.float32)
+    map_x = xx - shift_x
+    map_y = yy + shift_y
+    moved_rgb = cv2.remap(
+        rgb,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    )
+    moved_alpha = cv2.remap(
+        alpha,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    return moved_rgb, moved_alpha
+

+def composite_tool(base, rgb, alpha, lift, jaw_open):
+    """Open the lower jaw, then lift both fingers out of the fixed shot."""
+    out = base.astype(np.float32)
+
+    # The split only spreads toward the tips; the tool body remains rigid.
+    yy, xx = np.indices((HEIGHT, WIDTH), dtype=np.float32)
+    ramp = np.clip((yy - 16.0) / 14.0, 0.0, 1.0)
+    left = (xx <= 139.0).astype(np.float32)
+    right = (xx >= 140.0).astype(np.float32)
+
+    left_rgb, left_a = warp_half(rgb, alpha * left, -jaw_open * ramp, lift)
+    right_rgb, right_a = warp_half(rgb, alpha * right, jaw_open * ramp, lift)
+
+    for part_rgb, part_a in ((left_rgb, left_a), (right_rgb, right_a)):
+        a = np.clip(part_a, 0.0, 1.0)[:, :, None]
+        out = out * (1.0 - a) + part_rgb.astype(np.float32) * a
+
+    return np.clip(out, 0, 255).astype(np.uint8)
+

+        dtype=np.float32,
+    )
+
+    frames = np.empty((N_FRAMES, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first
