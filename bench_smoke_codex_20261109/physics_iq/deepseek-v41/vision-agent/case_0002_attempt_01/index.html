+def _fill_between_rows(image, x0, x1, y0, y1, left, right):
+    """Replace a narrow region with the smooth background between its sides."""
+    image = image.copy()
+    span = np.arange(x0, x1 + 1, dtype=np.float32)
+    weight = ((span - left) / float(right - left))[:, None]
+    for y in range(y0, y1 + 1):
+        a = image[y, left].astype(np.float32)
+        b = image[y, right].astype(np.float32)
+        image[y, x0 : x1 + 1] = a * (1.0 - weight) + b * weight
+    return image

+def make_clean_plate(source):

+    plate = _fill_between_rows(plate, 51, 91, 5, 23, 50, 92)

+    plate = _fill_between_rows(plate, 182, 221, 5, 21, 181, 222)

+def smoothstep(x):
+    x = np.clip(x, 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)

+def ball_state(t):
+    """Soft-ball drop with one visible rebound, then a damped settle."""
+    release = 0.025
+    gravity = 96.0
+    initial_y = 32.0
+    contact_y = 80.0
+    hit_time = release + np.sqrt(2.0 * (contact_y - initial_y) / gravity)
+    if t <= hit_time:

+            second_duration = 2.0 * second_speed / gravity

+def warp_patch(frame, patch, center, target, angle=0.0, scale=(1.0, 1.0)):
+    rgb, alpha = patch
+    x0, y0, x1, y1 = 0, 0, rgb.shape[1], rgb.shape[0]
+    cx, cy = center
+    matrix = cv2.getRotationMatrix2D(
+        (cx - x0, cy - y0), angle, max(scale[0], scale[1])
+    )
+    matrix[0, 0] *= scale[0] / max(scale[0], scale[1])
+    matrix[0, 1] *= scale[0] / max(scale[0], scale[1])
+    matrix[1, 0] *= scale[1] / max(scale[0], scale[1])
+    matrix[1, 1] *= scale[1] / max(scale[0], scale[1])
+    matrix[0, 2] += target[0] - cx
+    matrix[1, 2] += target[1] - cy
+
+    premultiplied = rgb * alpha[..., None]
+    warped_rgb = cv2.warpAffine(
+        premultiplied,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    )
+    warped_alpha = cv2.warpAffine(
+        alpha,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    warped_alpha = np.clip(warped_alpha, 0.0, 1.0)
+    return frame * (1.0 - warped_alpha[..., None]) + warped_rgb, warped_alpha

+def add_soft_shadow(frame, center_x, surface_y, height, width, strength):
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    height = max(height, 0.35)
+    normalized = ((xx - center_x) / width) ** 2 + ((yy - surface_y) / height) ** 2
+    shadow = np.exp(-2.25 * normalized)
+    shadow *= np.clip(strength, 0.0, 0.55)
+    tint = np.array([0.63, 0.67, 0.70], np.float32)
+    return frame * (1.0 - shadow[..., None] * (1.0 - tint))
+
+
+def add_contact_crease(frame, center_x, surface_y, width, strength):
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    low = np.exp(
+        -2.3
+        * (((xx - center_x) / width) ** 2 + ((yy - (surface_y + 0.5)) / 2.0) ** 2)
+    )
+    high = np.exp(
+        -2.8
+        * (((xx - center_x) / (width * 0.75)) ** 2
+           + ((yy - (surface_y - 2.0)) / 1.1) ** 2)
+    )
+    frame = frame * (1.0 - (0.16 * strength) * low[..., None])
+    return frame + (9.0 * strength) * high[..., None]
+

+def render(source, plate, layers):
+    ball, block, left_tool, right_tool = layers
+    frames = np.empty((FRAMES, HEIGHT, WIDTH, 3), np.uint8)
+    frames[0] = source

+    for index in range(1, FRAMES):

+
+        # Both grippers recoil a fraction of a pixel as the objects leave.

+            frame, left_tool, (70.0, 11.0), (70.0, 11.0 - recoil)
+        )
+        frame, _ = warp_patch(

+        )
+
+        bx, by, b_angle, b_squash, b_speed, b_hit = ball_state(t)
+        ball_height = max(0.0, 80.0 - by)
+        shadow_strength = 0.09 + 0.17 * (1.0 - smoothstep(ball_height / 48.0))
+        frame = add_soft_shadow(
+            frame,
+            bx,
+            92.0,
+            2.4 + 2.0 * (1.0 - smoothstep(ball_height / 48.0)),
+            11.0 + 5.0 * (1.0 - smoothstep(ball_height / 48.0)),
+            shadow_strength,
+        )
+        contact_strength = 0.0
+        if t >= b_hit:
+            contact_strength = np.exp(-2.8 * (t - b_hit))
+            frame = add_contact_crease(frame, bx, 92.0, 14.0, contact_strength)
+        b_scale_x = 1.0 + (1.0 - b_squash) * 0.70
+        frame, _ = warp_patch(
+            frame,
+            ball,
+            (70.0, 32.0),
+            (bx, by),
+            angle=b_angle,
+            scale=(b_scale_x, b_squash),
+        )
+
+        x, y, angle, k_hit = block_state(t)
+        block_height = max(0.0, 74.8 - y)
+        block_shadow = 0.08 + 0.19 * (
+            1.0 - smoothstep(block_height / 36.0)
+        )
+        frame = add_soft_shadow(
+            frame,
+            x + 1.0,
+            93.0,
+            2.1 + 1.6 * (1.0 - smoothstep(block_height / 36.0)),
+            12.0 + 2.5 * (1.0 - smoothstep(block_height / 36.0)),
+            block_shadow,
+        )
+        if t >= k_hit:
+            block_contact = np.exp(-3.2 * (t - k_hit))
+            frame = add_contact_crease(frame, x, 93.0, 11.5, block_contact)
+        frame, _ = warp_patch(
+            frame,
+            block,
+            (204.5, 39.5),
+            (x, y),
+            angle=angle,
+            scale=(1.0, 0.985),
+        )
+
+        # A very light sensor-noise layer keeps the composited patches matched
+        # to the compressed, softly focused source.
+        rng = np.random.default_rng(1000 + index)
+        noise = rng.normal(0.0, 0.55, frame.shape[:2])[..., None]
+        frame = np.clip(frame + noise, 0, 255).astype(np.uint8)
+        frames[index] = frame
+    return frames
+
