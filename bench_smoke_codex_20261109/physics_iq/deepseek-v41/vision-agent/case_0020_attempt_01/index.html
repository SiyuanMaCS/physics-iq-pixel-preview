+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+FRAME_COUNT = 20
+FPS = 4
+WIDTH = 256
+HEIGHT = 144
+
+
+def ball_sprite(frame):
+    """Extract the original tennis ball and a clean antialiased circular matte."""
+    x0, x1 = 76, 98
+    y0, y1 = 42, 63
+    patch = frame[y0:y1, x0:x1].astype(np.float32)
+
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    radius = np.hypot(xx - 86.5, yy - 51.7)
+    alpha = np.clip((9.55 - radius) / 1.25 + 0.5, 0.0, 1.0)
+
+    # The upper-left side catches more light; retain that cue after rotation.
+    light = np.clip(1.07 - 0.010 * (xx - 82) - 0.006 * (yy - 48), 0.88, 1.06)
+    patch *= light[..., None]
+    return np.clip(patch, 0, 255).astype(np.uint8), alpha.astype(np.float32)
+
+
+def background_plate(frame):
+    """Remove the initial ball and its shadow with local image inpainting."""
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    ball_mask = ((xx - 86.5) / 11.5) ** 2 + ((yy - 51.6) / 10.5) ** 2 <= 1.0
+    shadow_mask = ((xx - 94.0) / 14.5) ** 2 + ((yy - 60.0) / 7.5) ** 2 <= 1.0
+    mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    mask[ball_mask | shadow_mask] = 255
+
+    bgr = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
+    repaired = cv2.inpaint(bgr, mask, 4.5, cv2.INPAINT_NS)
+    repaired = cv2.cvtColor(repaired, cv2.COLOR_BGR2RGB)
+
+    # Preserve the untouched first-frame pixels exactly.
+    repaired[~mask.astype(bool)] = frame[~mask.astype(bool)]
+    return repaired
+
+
+def ball_path(t):
+    """A lightly decelerating table roll followed by a slower climb."""
+    entry_t = 2.78
+    entry_x = 146.0
+    if t <= entry_t:
+        x = 86.0 + 24.0 * t - 0.90 * t * t
+        ground_y = 60.2
+    else:
+        dt = t - entry_t
+        x = entry_x + 18.8 * dt - 1.65 * dt * dt
+        ground_y = 60.2 - 0.285 * (x - entry_x)
+    center_y = ground_y - 8.55
+    return x, center_y, ground_y
+
+
+def composite_ball(frame, sprite, alpha, center, angle):
+    """Paste a rotated, antialiased copy of the original ball."""
+    patch_h, patch_w = sprite.shape[:2]
+    center_high = ((patch_w - 1) * 0.5, (patch_h - 1) * 0.5)
+    matrix = cv2.getRotationMatrix2D(center_high, angle, 1.0)
+
+    warped = cv2.warpAffine(
+        sprite,
+        matrix,
+        (patch_w, patch_h),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_REFLECT_101,
+    ).astype(np.float32)
+    warped_alpha = cv2.warpAffine(
+        alpha,
+        matrix,
+        (patch_w, patch_h),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+
+    left = int(round(center[0] - center_high[0]))
+    top = int(round(center[1] - center_high[1]))
+    sx0, sy0 = max(0, -left), max(0, -top)
+    sx1 = min(patch_w, WIDTH - left)
+    sy1 = min(patch_h, HEIGHT - top)
+    if sx1 <= sx0 or sy1 <= sy0:
+        return frame
+
+    dx0, dy0 = left + sx0, top + sy0
+    dx1, dy1 = left + sx1, top + sy1
+    a = warped_alpha[sy0:sy1, sx0:sx1, None]
+    out = frame.astype(np.float32)
+    out[dy0:dy1, dx0:dx1] = (
+        warped[sy0:sy1, sx0:sx1] * a
+        + out[dy0:dy1, dx0:dx1] * (1.0 - a)
+    )
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def composite_shadow(frame, ball_x, ball_y, ground_y, t):
+    """Draw a soft contact shadow that becomes tighter on the ramp."""
+    on_ramp = t > 2.78
+    cx = ball_x + (6.7 if not on_ramp else 5.6)
+    cy = ground_y + 2.1
+    rx = 11.7 if not on_ramp else 9.0
+    ry = 6.1 if not on_ramp else 4.6
+    strength = 0.32 if not on_ramp else 0.20
+
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    q = ((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2
+    soft = np.clip(1.0 - q, 0.0, 1.0) ** 0.72
+    soft = cv2.GaussianBlur(soft.astype(np.float32), (0, 0), 1.25)
+    shade = 1.0 - strength * soft[..., None]
+    return np.clip(frame.astype(np.float32) * shade, 0, 255).astype(np.uint8)
+
+
+def main():
+    first = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    base = background_plate(first)
+    sprite, alpha = ball_sprite(first)
+
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first
+
+    roll_angle = 0.0
+    previous_x, _, _ = ball_path(0.0)
+    for index in range(1, FRAME_COUNT):
+        t = index / FPS
+        x, y, ground_y = ball_path(t)
+        dx = x - previous_x
+        roll_angle = (roll_angle + np.degrees(dx / 9.0)) % 360.0
+
+        image = base.copy()
+        image = composite_shadow(image, x, y, ground_y, t)
+        image = composite_ball(image, sprite, alpha, (x, y), roll_angle)
+        frames[index] = image
+        previous_x = x
+
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
