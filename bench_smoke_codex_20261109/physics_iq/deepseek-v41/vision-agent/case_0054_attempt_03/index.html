+PIVOT_Y = np.array([25.2, 25.7, 26.2, 26.7, 27.2], dtype=np.float64)

+def make_plate(src_bgr):
+    """Remove the source spheres and their strings while retaining the apparatus."""
+    mask = np.zeros((H, W), dtype=np.uint8)
+
+    initial_centers = [
+        (59.5, 96.5),
+        (69.5, 94.5),
+        (110.5, 97.5),
+        (122.0, 96.5),
+        (134.0, 100.0),
+    ]
+    for center, radius in zip(initial_centers, RADII):
+        cv2.circle(mask, tuple(np.round(center).astype(int)), int(math.ceil(radius + 1.4)), 255, -1)
+
+    # Thin V suspensions are reconstructed for every frame. Masking only below
+    # the rails avoids disturbing the photographed chrome top bar.
+    for x, y in zip(REST_X, REST_Y):
+        for anchor in ((x - 1.7, PIVOT_Y[0] + 2.4), (x + 1.7, PIVOT_Y[0] - 1.0)):
+            cv2.line(
+                mask,
+                (int(round(anchor[0])), int(round(anchor[1]))),
+                (int(round(x)), int(round(y))),
+                255,
+                3,
+                cv2.LINE_AA,
+            )
+    for center in initial_centers[:2]:

+            mask,
+            (int(round(center[0] + 1.5)), int(round(PIVOT_Y[0] - 1.0))),
+            (int(round(center[0])), int(round(center[1]))),
+            255,
+            3,
+            cv2.LINE_AA,
+        )
+
+    return cv2.inpaint(src_bgr, mask, 3.2, cv2.INPAINT_TELEA)

+def smoothstep(u):
+    u = float(np.clip(u, 0.0, 1.0))
+    return u * u * (3.0 - 2.0 * u)
+
+
+def pair_angles(t):
+    """Return per-ball pendulum angles for a two-ball Newton cradle transfer."""
+    angles = np.zeros(5, dtype=np.float64)
+    release = 0.25
+    quarter = 0.42
+    if t <= release:
+        angles[:2] = -AMPLITUDE
+        return angles
+
+    tau = t - release
+    segment = int(math.floor(tau / quarter))
+    u = (tau - segment * quarter) / quarter
+    decay = 1.0 - 0.025 * min(t, 5.0)
+
+    if segment == 0:
+        # Released pair falls from the left toward the resting chain.
+        angles[:2] = -AMPLITUDE * decay * math.cos(0.5 * math.pi * u)
+    elif segment % 2 == 1:
+        # The returning pair transfers motion to the rightmost pair.
+        phase = 0.5 * math.pi * u
+        angles[3:] = AMPLITUDE * decay * math.sin(phase)
+    else:
+        # The rightmost pair returns and sends the left pair outward again.
+        phase = 0.5 * math.pi * u
+        angles[:2] = -AMPLITUDE * decay * math.sin(phase)
+    return angles
+
+
+def ball_centers(t):
+    angles = pair_angles(t)
+    x = REST_X + STRING_LENGTH * np.sin(angles)
+    # The camera views the cradle almost edge-on. A reduced vertical rise keeps
+    # the tiny spheres centered correctly while preserving the pendulum arc.
+    y = REST_Y - 0.28 * STRING_LENGTH * (1.0 - np.cos(angles))
+    return np.stack([x, y], axis=1), angles
+
+
+def rgba_upsample(layer, size):
+    return cv2.resize(layer, (size[0] * SS, size[1] * SS), interpolation=cv2.INTER_LINEAR)
+
+
+def composite_rgba(base_bgr, layer_rgba):
+    alpha = layer_rgba[:, :, 3:4].astype(np.float32) / 255.0
+    color = layer_rgba[:, :, :3].astype(np.float32)
+    return np.clip(color * alpha + base_bgr.astype(np.float32) * (1.0 - alpha), 0, 255).astype(np.uint8)
+
+
+def make_wires(centers):
+    layer = np.zeros((H * SS, W * SS, 4), dtype=np.uint8)
+    for i, center in enumerate(centers):
+        pivot_x = REST_X[i]
+        pivot_y = PIVOT_Y[i]
+        anchors = (
+            (pivot_x - 1.8, pivot_y + 1.0),
+            (pivot_x + 1.8, pivot_y - 1.0),
+        )
+        for j, anchor in enumerate(anchors):
+            start = (int(round(anchor[0] * SS)), int(round(anchor[1] * SS)))
+            end = (
+                int(round((center[0] + (-0.35 if j == 0 else 0.35)) * SS)),
+                int(round((center[1] - 0.2) * SS)),
+            )
+            cv2.line(layer, start, end, (55, 59, 62, 185), 5, cv2.LINE_AA)
+            cv2.line(layer, start, end, (182, 187, 187, 210), 2, cv2.LINE_AA)
+    layer = cv2.resize(layer, (W, H), interpolation=cv2.INTER_AREA)
+    return layer
+

+def sphere_rgba(cx, cy, radius, index):
+    """Render a small warm steel sphere with an upper-left key reflection."""
+    pad = 2.5
+    local_w = int(math.ceil((2.0 * radius + 2.0 * pad) * SS))
+    local_h = local_w
+    yy, xx = np.mgrid[0:local_h, 0:local_w].astype(np.float32)
+    center = (local_w - 1.0) / 2.0
+    dx = (xx - center) / (radius * SS)
+    dy = (yy - center) / (radius * SS)
+    rr2 = dx * dx + dy * dy
+    inside = rr2 <= 1.0
+
+    nz = np.sqrt(np.clip(1.0 - rr2, 0.0, 1.0))
+    light = np.array([-0.50, -0.63, 0.60], dtype=np.float32)
+    light /= np.linalg.norm(light)
+    diffuse = np.clip(dx * light[0] + dy * light[1] + nz * light[2], 0.0, 1.0)
+
+    # Broad vertical window reflection plus a compact specular glint.
+    broad = np.exp(-((dx + 0.38) / 0.19) ** 2) * np.exp(-(dy / 0.72) ** 2)
+    glint = np.exp(-((dx + 0.25) / 0.105) ** 2 - ((dy + 0.43) / 0.11) ** 2)
+    rim = np.exp(-((dx - 0.76) / 0.11) ** 2) * np.clip(0.4 - dy, 0.0, 1.0)
+    crevice = np.exp(-((dx + 0.88) / 0.13) ** 2) * np.clip(0.35 + dy, 0.0, 1.0)
+
+    seed = (index - 2) * 0.018
+    steel = 47.0 + 98.0 * diffuse + 24.0 * (0.5 + 0.5 * nz)
+    steel += 95.0 * broad + 205.0 * glint + 35.0 * rim - 20.0 * crevice
+    r = steel * (0.96 + seed) + 10.0
+    g = steel * (0.94 + seed) + 7.0
+    b = steel * (0.87 + seed) + 4.0
+    rgb = np.stack([r, g, b], axis=2)
+    rgb = np.clip(rgb, 0, 255)
+
+    alpha = np.clip((1.0 - rr2) * radius * SS * 0.9 + 0.5, 0.0, 1.0)
+    alpha *= inside
+    rgba = np.zeros((local_h, local_w, 4), dtype=np.uint8)
+    rgba[:, :, :3] = rgb.astype(np.uint8)
+    rgba[:, :, 3] = (alpha * 255.0).astype(np.uint8)
+    rgba = cv2.resize(rgba, (max(1, local_w // SS), max(1, local_h // SS)), interpolation=cv2.INTER_AREA)
+
+    out = np.zeros((H, W, 4), dtype=np.uint8)
+    h, w = rgba.shape[:2]
+    x0 = int(round(cx - (w - 1) / 2.0))
+    y0 = int(round(cy - (h - 1) / 2.0))
+    x1, y1 = x0 + w, y0 + h
+    sx0, sy0 = max(0, x0), max(0, y0)
+    sx1, sy1 = min(W, x1), min(H, y1)
+    out[sy0:sy1, sx0:sx1] = rgba[sy0 - y0:sy1 - y0, sx0 - x0:sx1 - x0]
+    return out
+
+
+def render_frame(plate_bgr, centers, angles):
+    frame = plate_bgr.copy()
+    wires = make_wires(centers)
+    frame = composite_rgba(frame, wires)
+
+    # Draw back-to-front by resting x. The one-pixel overlap produces the dark
+    # contact seams visible between adjacent steel spheres.
+    order = np.argsort(centers[:, 0])
+    for i in order:
+        sprite = sphere_rgba(centers[i, 0], centers[i, 1], RADII[i], i)
+        frame = composite_rgba(frame, sprite)
+    return frame
+
+
+def main():
+    source_rgb = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    source_bgr = cv2.cvtColor(source_rgb, cv2.COLOR_RGB2BGR)
+    plate = make_plate(source_bgr)
+    cv2.imwrite(str(CASE_DIR / ".tmp" / "plate_candidate.png"), plate)
+
+    frames = [source_rgb.copy()]
+    for frame_index in range(1, N):
+        t = frame_index / FPS
+        centers, angles = ball_centers(t)
+        frame_bgr = render_frame(plate, centers, angles)
+        frames.append(cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2RGB))
+    frames = np.stack(frames, axis=0).astype(np.uint8)
+
+    # Keep the required frame-zero contract exact.
+    frames[0] = source_rgb
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
