+def _poly_mask(polygons, scale=S, blur=0.75):
+    mask = np.zeros((H * scale, W * scale), np.uint8)
+    for poly in polygons:
+        pts = np.rint(np.asarray(poly, dtype=np.float32) * scale).astype(np.int32)
+        cv2.fillPoly(mask, [pts], 255)
+    if blur:
+        mask = cv2.GaussianBlur(mask, (0, 0), blur * scale)
+    return mask.astype(np.float32) / 255.0
+
+
+def _row_texture_fill(image, mask):
+    """Fill masked table regions while retaining a little horizontal wood grain."""
+    out = image.astype(np.float32).copy()
+    src = image.astype(np.float32)
+    detail_src = src[:, 5:44]
+    detail_src = detail_src - cv2.GaussianBlur(detail_src, (0, 0), 3.0)
+
+    for y in range(H):
+        row_mask = mask[y] > 0
+        if not np.any(row_mask):
+            continue
+        valid = np.flatnonzero(~row_mask)
+        if len(valid) < 2:
+            continue
+        starts = np.flatnonzero(row_mask & ~np.r_[False, row_mask[:-1]])
+        ends = np.flatnonzero(row_mask & ~np.r_[row_mask[1:], False])
+        for start, end in zip(starts, ends):
+            left = valid[valid < start]
+            right = valid[valid > end]
+            if len(left) and len(right):
+                xl, xr = left[-1], right[0]
+                t = np.linspace(0.0, 1.0, end - start + 1, dtype=np.float32)[:, None]
+                fill = src[y, xl] * (1.0 - t) + src[y, xr] * t
+            elif len(left):
+                fill = np.repeat(src[y, left[-1]][None], end - start + 1, axis=0)
+            elif len(right):
+                fill = np.repeat(src[y, right[0]][None], end - start + 1, axis=0)
+            else:
+                continue
+
+            xs = np.arange(start, end + 1)
+            tex_x = (xs + (y // 3) * 7) % detail_src.shape[1]
+            grain = detail_src[y, tex_x] * 0.55
+            out[y, start : end + 1] = fill + grain
+
+    return np.clip(out, 0, 255).astype(np.uint8)
+

+def _build_clean_plate(frame):
+    block_shadows = [b["shadow"] for b in BLOCKS]
+    block_mask = _poly_mask(block_shadows, scale=1, blur=0.0)
+
+    stick_mask = np.zeros((H, W), np.uint8)
+    cv2.line(stick_mask, (145, 84), (218, 63), 255, 7, cv2.LINE_AA)
+    cv2.circle(stick_mask, (214, 64), 5, 255, -1, cv2.LINE_AA)
+    clean_mask = np.maximum((block_mask * 255).astype(np.uint8), stick_mask)
+
+    inpainted = cv2.inpaint(frame, clean_mask, 4, cv2.INPAINT_TELEA)
+    table_fill = _row_texture_fill(inpainted, clean_mask > 0)
+
+    # Inpaint is stronger around the black platform; the grain fill is better
+    # over the unobstructed table.
+    platform_region = np.zeros((H, W), np.uint8)
+    cv2.ellipse(platform_region, (219, 79), (38, 30), 0, 0, 360, 255, -1)
+    alpha = np.where(platform_region > 0, 0.82, 0.42).astype(np.float32)[..., None]
+    return np.clip(table_fill * alpha + inpainted * (1.0 - alpha), 0, 255).astype(np.uint8)
+

+def _smooth_fall(t, start, duration=2.1):
+    p = float(np.clip((t - start) / duration, 0.0, 1.0))
+    # Gravity-like acceleration, with a gentle landing.
+    p = p**1.55
+    return p * p * (3.0 - 2.0 * p)
+
+
+def _stick_angle(t):
+    if t <= 0:
+        return 0.0
+    # Clockwise sweep from the initial contact position.
+    p = float(np.clip(t / 2.45, 0.0, 1.0))
+    eased = p * p * (3.0 - 2.0 * p)
+    return 3.0 + 55.0 * eased + 1.2 * np.sin(np.pi * min(t / 2.45, 1.0))
+
+
+def _rotate_point(point, pivot, angle_deg):
+    a = np.deg2rad(angle_deg)
+    c, s = np.cos(a), np.sin(a)
+    x, y = point[0] - pivot[0], point[1] - pivot[1]
+    return np.array([pivot[0] + c * x - s * y, pivot[1] + s * x + c * y])
+
+
+def _draw_stick(frame, angle_deg):
+    hi = cv2.resize(frame, (W * S, H * S), interpolation=cv2.INTER_CUBIC)
+    pivot = np.array([214.0, 64.0])
+    base_tip = np.array([149.0, 82.0])
+    tip = _rotate_point(base_tip, pivot, angle_deg)
+    d = tip - pivot
+    length = np.linalg.norm(d)
+    u = d / length
+    n = np.array([-u[1], u[0]])
+
+    # Tapered wooden rod, with a dark lower edge and a narrow highlight.
+    p0 = (pivot + n * 3.4) * S
+    p1 = (tip + n * 1.25) * S
+    p2 = (tip - n * 1.25) * S
+    p3 = (pivot - n * 3.4) * S
+    rod = np.array([p0, p1, p2, p3], np.int32)
+
+    shadow = np.array(
+        [(pivot + n * 4.6 + np.array([1.4, 1.8])) * S,
+         (tip + n * 2.0 + np.array([1.4, 1.8])) * S,
+         (tip - n * 2.0 + np.array([1.4, 1.8])) * S,
+         (pivot - n * 4.6 + np.array([1.4, 1.8])) * S],
+        np.int32,
+    )
+    layer = hi.astype(np.float32)
+    overlay = layer.copy()
+    cv2.fillPoly(overlay, [shadow], (29, 23, 18), cv2.LINE_AA)
+    layer = cv2.addWeighted(overlay, 0.22, layer, 0.78, 0)
+    cv2.fillPoly(layer, [rod], (73, 99, 128), cv2.LINE_AA)
+
+    edge = np.array(
+        [(pivot - n * 2.7) * S, (tip - n * 0.8) * S,
+         (tip - n * 1.15) * S, (pivot - n * 3.35) * S],
+        np.int32,
+    )
+    cv2.fillPoly(layer, [edge], (48, 65, 91), cv2.LINE_AA)
+    highlight = np.array(
+        [(pivot + n * 1.9) * S, (tip + n * 0.55) * S,
+         (tip + n * 1.0) * S, (pivot + n * 3.05) * S],
+        np.int32,
+    )
+    cv2.fillPoly(layer, [highlight], (100, 132, 165), cv2.LINE_AA)
+    cv2.circle(layer, tuple(np.rint(pivot * S).astype(int)), int(3.1 * S), (28, 28, 28), -1, cv2.LINE_AA)
+    cv2.circle(layer, tuple(np.rint((pivot + np.array([-1.0, -0.8])) * S).astype(int)),
+               int(1.1 * S), (74, 76, 80), -1, cv2.LINE_AA)
+    return cv2.resize(layer, (W, H), interpolation=cv2.INTER_AREA)
+
+
+def _warp_sprite(rgb_hi, alpha_hi, pivot, angle_deg):
+    center = (pivot[0] * S, pivot[1] * S)
+    matrix = cv2.getRotationMatrix2D(center, angle_deg, 1.0)
+    rgb = cv2.warpAffine(
+        rgb_hi, matrix, (W * S, H * S),
+        flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT,
+    )
+    alpha = cv2.warpAffine(
+        alpha_hi, matrix, (W * S, H * S),
+        flags=cv2.INTER_LINEAR, borderMode=cv2.BORDER_CONSTANT,
+    )
+    rgb = cv2.resize(rgb, (W, H), interpolation=cv2.INTER_AREA)
+    alpha = cv2.resize(alpha, (W, H), interpolation=cv2.INTER_AREA)
+    return rgb, np.clip(alpha, 0.0, 1.0)
+
+
+def _composite(frame, plate, sprites, stick_angle):
+    t = frame / FPS
+    result = plate.astype(np.float32)
+
+    rotated = []
+    shadow_accum = np.zeros((H, W), np.float32)
+    for block, sprite in zip(BLOCKS, sprites):
+        progress = _smooth_fall(t, block["start"])
+        angle = -block["final"] * progress
+        rgb, alpha = _warp_sprite(sprite["rgb"], sprite["alpha"], block["pivot"], angle)
+        rotated.append((rgb, alpha))
+
+        shifted = np.roll(alpha, (2, -1), axis=(0, 1))
+        shadow = cv2.GaussianBlur(shifted, (0, 0), 2.1)
+        shadow_accum = np.maximum(shadow_accum, shadow * (0.28 + 0.12 * progress))
+
+    result *= (1.0 - 0.42 * shadow_accum[..., None])
+
+    # Right-hand blocks are nearer the camera, so they correctly occlude the
+    # blocks they knock over.
+    for rgb, alpha in rotated:
+        a = alpha[..., None]
+        result = result * (1.0 - a) + rgb * a
+
+    result = np.clip(result, 0, 255).astype(np.uint8)
+    if frame > 0:
+        result = _draw_stick(result, _stick_angle(t))
+    return result
+
+
+def main():
+    first = cv2.cvtColor(cv2.imread(str(ROOT / "first_frame.png")), cv2.COLOR_BGR2RGB)
+    assert first.shape == (H, W, 3)
+
+    plate = _build_clean_plate(cv2.cvtColor(first, cv2.COLOR_RGB2BGR))
+    plate = cv2.cvtColor(plate, cv2.COLOR_BGR2RGB)
+    src_hi = cv2.resize(first, (W * S, H * S), interpolation=cv2.INTER_CUBIC)
+
+    sprites = []
+    for block in BLOCKS:
+        alpha_hi = _poly_mask([block["poly"]], scale=S, blur=0.65)
+        rgb_hi = src_hi.astype(np.float32) * alpha_hi[..., None]
+        sprites.append({"rgb": rgb_hi, "alpha": alpha_hi})
+
+    frames = [first.copy()]
+    for index in range(1, N):
+        frames.append(_composite(index, plate, sprites, _stick_angle(index / FPS)))
+
+    frames = np.asarray(frames, dtype=np.uint8)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()
