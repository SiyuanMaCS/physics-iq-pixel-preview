+def wrench_alpha(rgb):
+    """Build a soft matte for the small silver wrench."""
+    r = rgb[:, :, 0].astype(np.int16)
+    g = rgb[:, :, 1].astype(np.int16)
+    b = rgb[:, :, 2].astype(np.int16)
+    value = np.max(rgb, axis=2).astype(np.int16)
+    saturation = (value - np.min(rgb, axis=2)) / (value + 1e-6)
+
+    region = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.fillPoly(region, [WRENCH_POLYGON], 255)
+    metal = (
+        (value > 52)
+        & (saturation < 0.44)
+        & (b >= g - 5)
+        & (b >= r - 11)
+        & (region > 0)
+    )
+
+    mask = metal.astype(np.uint8)
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((2, 2), np.uint8))
+    mask = ndimage.binary_fill_holes(mask).astype(np.uint8)
+
+    # Retain the connected wrench body, then restore its antialiased edge.
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(mask)
+    kept = np.zeros_like(mask)
+    for index in range(1, count):
+        if stats[index, cv2.CC_STAT_AREA] >= 3:
+            kept[labels == index] = 1
+    kept = cv2.dilate(kept, np.ones((2, 2), np.uint8), iterations=1)
+
+    alpha = cv2.GaussianBlur(kept.astype(np.float32), (0, 0), 0.48)
+    return np.clip((alpha - 0.08) / 0.84, 0.0, 1.0)
+

+def clean_plate(rgb, alpha):
+    """Remove the wrench and its contact shadow from the black disk."""
+    object_mask = (alpha > 0.12).astype(np.uint8) * 255
+    removal = cv2.dilate(object_mask, np.ones((5, 5), np.uint8), iterations=1)
+
+    # The source has a short, dark contact shadow along the lower edge.
+    shadow = np.zeros_like(removal)
+    cv2.ellipse(shadow, (132, 77), (15, 5), 18, 0, 360, 255, -1)
+    removal = cv2.bitwise_or(removal, shadow)
+
+    plate = np.zeros_like(removal)
+    cv2.ellipse(plate, (156, 69), (42, 29), 0, 0, 360, 255, -1)
+    removal = cv2.bitwise_and(removal, plate)
+
+    bgr = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)
+    cleaned = cv2.inpaint(bgr, removal, 4, cv2.INPAINT_TELEA)
+    return cv2.cvtColor(cleaned, cv2.COLOR_BGR2RGB)

+def rotation_matrix(angle, center, compression):
+    """Destination-to-source affine map for rotation on a horizontal plane."""
+    cx, cy = center
+    c, s = np.cos(angle), np.sin(angle)
+    # Forward screen motion is an ellipse: x = cx + u, y = cy + compression * v.
+    a = np.array(
+        [
+            [c, s / compression, cx - c * cx - (s / compression) * cy],
+            [
+                -(compression**2) * s,
+                c,
+                cy + (compression**2) * s * cx - c * cy,
+            ],
+        ],
+        dtype=np.float64,
+    )
+    return a

+def render_turn(base, sprite, alpha, angle, center, compression):
+    """Render one frame with a supersampled local sprite composite."""
+    x0, y0, x1, y1 = 100, 40, 165, 100
+    scale = 4
+    pw, ph = (x1 - x0) * scale, (y1 - y0) * scale
+
+    base_patch = cv2.resize(
+        base[y0:y1, x0:x1], (pw, ph), interpolation=cv2.INTER_CUBIC
+    ).astype(np.float32)
+    sprite_patch = cv2.resize(
+        sprite[y0:y1, x0:x1], (pw, ph), interpolation=cv2.INTER_CUBIC
+    )
+    alpha_patch = cv2.resize(
+        alpha[y0:y1, x0:x1], (pw, ph), interpolation=cv2.INTER_LINEAR
+    )
+
+    m = rotation_matrix(angle, center, compression)
+    local = np.array(
+        [
+            [
+                m[0, 0],
+                m[0, 1],
+                scale * (m[0, 0] * x0 + m[0, 1] * y0 + m[0, 2] - x0),
+            ],
+            [
+                m[1, 0],
+                m[1, 1],
+                scale * (m[1, 0] * x0 + m[1, 1] * y0 + m[1, 2] - y0),
+            ],
+        ],
+        dtype=np.float32,
+    )
+    rotated = cv2.warpAffine(
+        sprite_patch,
+        local,
+        (pw, ph),
+        flags=cv2.INTER_CUBIC | cv2.WARP_INVERSE_MAP,
+        borderMode=cv2.BORDER_CONSTANT,
+    ).astype(np.float32)
+    rotated_alpha = cv2.warpAffine(
+        alpha_patch,
+        local,
+        (pw, ph),
+        flags=cv2.INTER_LINEAR | cv2.WARP_INVERSE_MAP,
+        borderMode=cv2.BORDER_CONSTANT,
+    )
+
+    # The contact shadow follows the wrench with a small light-source offset.
+    shift = np.array([[1, 0, 2.0 * scale], [0, 1, 1.7 * scale]], dtype=np.float32)
+    shadow_alpha = cv2.warpAffine(
+        rotated_alpha,
+        shift,
+        (pw, ph),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+    )
+    shadow_alpha = cv2.GaussianBlur(shadow_alpha, (0, 0), 1.35 * scale)
+
+    out = base_patch * (1.0 - 0.48 * shadow_alpha[:, :, None])
+    out = out * (1.0 - rotated_alpha[:, :, None]) + rotated * rotated_alpha[:, :, None]
+    out = np.clip(out, 0, 255).astype(np.uint8)
+    out = cv2.resize(out, (x1 - x0, y1 - y0), interpolation=cv2.INTER_AREA)
+
+    frame = base.copy()
+    frame[y0:y1, x0:x1] = out
+    return frame

+        raise AssertionError("frame 0 must match first_frame.png exactly")

+    main()
