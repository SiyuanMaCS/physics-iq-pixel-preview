+def build_statue_layers(frame, matte):
+    ys, xs = np.where(matte > 0)
+    x0, x1 = int(xs.min()) - 3, int(xs.max()) + 4
+    y0, y1 = int(ys.min()) - 3, int(ys.max()) + 1
+    x0, y0 = max(0, x0), max(0, y0)
+    x1, y1 = min(W, x1), min(H, y1)

+    crop = frame[y0:y1, x0:x1].astype(np.float32)
+    alpha = matte[y0:y1, x0:x1].astype(np.float32) / 255.0
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.55)

+    # Lift only the ceramic pixels for a faint warm rim. This stays locked to
+    # the material and survives the small yaw warp without looking painted on.
+    luma = cv2.cvtColor(crop.astype(np.uint8), cv2.COLOR_BGR2GRAY).astype(np.float32)
+    ceramic = np.clip((luma - 42.0) / 105.0, 0, 1) * alpha
+    highlight = cv2.GaussianBlur(ceramic, (0, 0), 0.75)
+    crop += highlight[..., None] * np.array([5.0, 7.0, 8.0], np.float32)
+    return crop, alpha, (x0, y0, x1, y1)

+def yaw_warp(layers, angle):
+    """A cylindrical yaw approximation for a small turntable figurine."""
+    crop, alpha = layers
+    h, w = crop.shape[:2]
+    center = (w - 1) * 0.50
+
+    # Interior texture slides across the silhouette as the object turns; the
+    # silhouette breathes slightly rather than becoming a flat card.
+    shift = 2.3 * np.sin(angle)
+    squeeze = 1.0 - 0.055 * (1.0 - np.cos(angle))
+    map_x, map_y = np.meshgrid(
+        np.arange(w, dtype=np.float32),
+        np.arange(h, dtype=np.float32),
+    )
+    map_x = np.ascontiguousarray(map_x, dtype=np.float32)
+    map_y = np.ascontiguousarray(map_y, dtype=np.float32)
+    src_x = (
+        center
+        + (map_x - center) / np.float32(squeeze)
+        - np.float32(shift) * (np.float32(0.35) + np.float32(0.65) * map_y / np.float32(h))
+    )
+    src_x = np.ascontiguousarray(src_x, dtype=np.float32)
+    warped = cv2.remap(
+        crop, src_x, map_y, cv2.INTER_CUBIC,
+        borderMode=cv2.BORDER_REFLECT_101,
+    )
+    warped_alpha = cv2.remap(
+        alpha, src_x, map_y, cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )
+
+    # Simulated key light from the upper right: the bright face crosses from
+    # one side of the figurine to the other over the rotation.
+    nx = (map_x - center) / max(center, 1.0)
+    light = np.clip(0.90 + 0.13 * np.cos(angle) - 0.10 * nx * np.sin(angle), 0.72, 1.10)
+    warped *= light[..., None]
+    return np.clip(warped, 0, 255), warped_alpha
+

+def paste_statue(canvas, layers, angle):
+    crop, alpha, (x0, y0, x1, y1) = layers
+    warped, warped_alpha = yaw_warp((crop, alpha), angle)
+    region = canvas[y0:y1, x0:x1].astype(np.float32)
+    out = warped * warped_alpha[..., None] + region * (1.0 - warped_alpha[..., None])
+    canvas[y0:y1, x0:x1] = np.clip(out, 0, 255).astype(np.uint8)

+def draw_turntable(canvas, angle, statue_alpha):
+    """Rotating black platform with restrained warm rim and radial sheen."""
+    overlay = canvas.astype(np.float32)
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    cx, cy = 108.0, 138.0
+    rx, ry = 35.0, 9.4
+    nx = (xx - cx) / rx
+    ny = (yy - cy) / ry
+    r = np.sqrt(nx * nx + ny * ny)
+    disc = smoothstep(1.06, 0.91, r)
+
+    # Keep the statue and pedestal in front of the disc.
+    disc *= 1.0 - cv2.GaussianBlur(statue_alpha, (0, 0), 1.0)
+    disc *= (yy >= 125)
+
+    theta = np.arctan2(ny, nx)
+    radial = 0.5 + 0.5 * np.cos(4.0 * (theta + angle))
+    sheen = 2.0 + 5.0 * radial * (0.25 + 0.75 * (1.0 - r))
+    front = np.clip((cy + 0.8 - yy) / 2.0, 0.0, 1.0)
+    rim = smoothstep(0.88, 1.02, r) * smoothstep(1.08, 0.98, r)
+    rim_color = np.array([20.0, 24.0, 27.0], np.float32)
+
+    platform = overlay * 0.72 + np.array([2.0, 3.0, 4.0], np.float32)
+    platform += sheen[..., None] * front[..., None] * np.array([0.72, 0.84, 1.0], np.float32)
+    platform = platform * (1.0 - 0.32 * rim[..., None])
+    platform += rim_color * rim[..., None]
+    w = disc[..., None]
+    overlay = platform * w + overlay * (1.0 - w)
+    canvas[:] = np.clip(overlay, 0, 255).astype(np.uint8)

+def draw_cast_shadow(canvas, matte, angle):
+    """Project the rotating silhouette onto the back wall as a soft shadow."""
+    src = matte.astype(np.float32) / 255.0
+
+    # Object-space pitch and yaw alter the projected silhouette. A small
+    # horizontal scale term makes the profile turn rather than merely slide.
+    breathe = 1.0 + 0.055 * np.cos(angle + 0.35)
+    shear = 0.12 * np.sin(angle)
+    src_pts = np.float32([[84, 63], [136, 63], [84, 143], [136, 143]])
+    dst_pts = np.float32([
+        [151 - 6.0 * np.sin(angle), 16],
+        [151 + 25.0 * breathe, 16],
+        [153 - 7.0 * np.sin(angle), 117],
+        [153 + 31.0 * breathe, 117 + 5.0 * shear],
+    ])
+    m = cv2.getPerspectiveTransform(src_pts, dst_pts)
+    shadow = cv2.warpPerspective(src, m, (W, H), flags=cv2.INTER_LINEAR)
+
+    # The wall receives a broad penumbra, not a hard cutout.
+    shadow = cv2.GaussianBlur(shadow, (0, 0), 1.15)
+    shadow *= smoothstep(6.0, 19.0, np.arange(H, dtype=np.float32))[:, None]
+    shadow *= 1.0 - smoothstep(108.0, 126.0, np.arange(H, dtype=np.float32))[:, None]
+
+    canvas_f = canvas.astype(np.float32)
+    # Shadow color retains the warm neutral wall cast instead of becoming pure
+    # black, and the densest core is slightly cooler.
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    core = np.clip(shadow, 0, 1)
+    tint = np.array([0.86, 0.90, 0.96], np.float32)
+    darkened = canvas_f * (1.0 - 0.88 * core[..., None])
+    darkened += np.array([1.5, 1.7, 2.0], np.float32) * core[..., None] * tint
+    canvas[:] = np.clip(darkened, 0, 255).astype(np.uint8)
+
+
+def add_ambient_grain(frame, index, rng):
+    """A fixed, tiny sensor texture keeps the inpainted wall from looking plastic."""
+    f = frame.astype(np.float32)
+    noise = rng.normal(0.0, 0.85, (H, W, 1)).astype(np.float32)
+    # Modulate by scene luminance so the black surround stays clean.
+    lum = f.mean(axis=2, keepdims=True)
+    f += noise * np.clip(lum / 95.0, 0.0, 1.0)
+    return np.clip(f, 0, 255).astype(np.uint8)
+

+def generate():
+    first = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    frame_bgr = cv2.cvtColor(first, cv2.COLOR_RGB2BGR)
+    matte = make_statue_matte()

+    rng = np.random.default_rng(84)
+
+    frames = []
+    for i in range(N_FRAMES):
+        if i == 0:
+            frames.append(first.copy())
+            continue
+
+        # 58 degrees clockwise over the five-second shot. The easing starts and
+        # ends gently while maintaining a continuous, constant-rate turn.
+        t = i / (N_FRAMES - 1)
+        eased = 0.16 * t + 0.84 * (0.5 - 0.5 * np.cos(np.pi * t))
+        angle = np.deg2rad(58.0) * eased
+
+        canvas = background.copy()
+        draw_turntable(canvas, angle, statue_alpha)
+        draw_cast_shadow(canvas, matte, angle)
+        paste_statue(canvas, layers, angle)
+        canvas = add_ambient_grain(canvas, i, rng)
+        frames.append(cv2.cvtColor(canvas, cv2.COLOR_BGR2RGB))
+
+    frames = np.stack(frames).astype(np.uint8)
+    artifact_io.save(frames, CASE_DIR)
+    return frames
+

+if __name__ == "__main__":
+    generate()
