+from pathlib import Path
+import math
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+W, H = 256, 144
+N, FPS = 20, 4
+
+
+def _composite_rgba(dst, rgb, alpha, x, y):
+    """Composite a small sprite with its center at a subpixel position."""
+    sh, sw = rgb.shape[:2]
+    matrix = np.float32([[1.0, 0.0, x - sw * 0.5], [0.0, 1.0, y - sh * 0.5]])
+    warped_rgb = cv2.warpAffine(
+        rgb, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=(0, 0, 0),
+    )
+    warped_a = cv2.warpAffine(
+        alpha, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0,
+    )[..., None]
+    return (dst.astype(np.float32) * (1.0 - warped_a) +
+            warped_rgb.astype(np.float32) * warped_a).clip(0, 255)
+
+
+def _draw_wire(dst, start, end, shade):
+    overlay = dst.copy()
+    line_mask = np.zeros((H, W), np.uint8)
+    start_i = (int(round(start[0])), int(round(start[1])))
+    end_i = (int(round(end[0])), int(round(end[1])))
+    cv2.line(line_mask, start_i, end_i, 255, 1, cv2.LINE_AA)
+    cv2.line(overlay, start_i, end_i, shade, 1, cv2.LINE_AA)
+    alpha = (line_mask.astype(np.float32) / 255.0 * 0.72)[..., None]
+    return (dst.astype(np.float32) * (1.0 - alpha) +
+            overlay.astype(np.float32) * alpha)
+
+
+def main():
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    source_bgr = cv2.cvtColor(source, cv2.COLOR_RGB2BGR)
+
+    # The cradle pivots and the five ball positions measured from the source.
+    pivots = [(97.0, 20.0), (110.0, 20.0), (123.0, 20.0),
+              (136.0, 20.0), (149.0, 20.0)]
+    rest = [(97.0, 96.0), (110.0, 96.0), (123.0, 96.0),
+            (136.0, 96.0), (149.0, 96.0)]
+    held_center = (62.2, 87.5)
+    length = 76.0
+
+    # Remove the balls and their suspensions. The photograph supplies all of
+    # the fixed scenery, while only the moving metal parts are rebuilt.
+    remove = np.zeros((H, W), np.uint8)
+    for center in [held_center, *rest[1:]]:
+        cv2.circle(remove, (int(round(center[0])), int(round(center[1]))),
+                   8, 255, -1, cv2.LINE_AA)
+    for pivot, center in zip(pivots, [held_center, *rest[1:]]):
+        for offset in (-2.0, 2.0):
+            p0 = (int(round(pivot[0] + offset)), 21)
+            p1 = (int(round(center[0] + offset * 0.45)),
+                  int(round(center[1] - 4.0)))
+            cv2.line(remove, p0, p1, 255, 2, cv2.LINE_AA)
+    remove[:23, :] = 0
+    plate = cv2.inpaint(source_bgr, remove, 3, cv2.INPAINT_TELEA)
+    plate = cv2.cvtColor(plate, cv2.COLOR_BGR2RGB).astype(np.float32)
+
+    # A clean sample of one sphere is reused for every ball. A feathered
+    # circular alpha removes the table pixels around the sampled metal.
+    sprite_center = (110, 96)
+    patch_radius = 7
+    sx, sy = sprite_center
+    sprite_rgb = source[sy - patch_radius:sy + patch_radius + 1,
+                        sx - patch_radius:sx + patch_radius + 1].copy()
+    yy, xx = np.mgrid[-patch_radius:patch_radius + 1,
+                      -patch_radius:patch_radius + 1]
+    distance = np.sqrt(xx.astype(np.float32) ** 2 + yy.astype(np.float32) ** 2)
+    alpha = np.clip((6.8 - distance) / 1.05, 0.0, 1.0)
+
+    # Restore the grabber over the moving ball. The mask keeps the black jaws
+    # and blue grips but does not carry the original silver sphere forward.
+    blue = ((source[..., 2].astype(np.int16) -
+             source[..., 0].astype(np.int16) > 20) &
+            (source[..., 2] > 55))
+    roi = np.zeros((H, W), bool)
+    roi[52:103, 0:74] = True
+    gray = source.mean(axis=2)
+    black_jaw = (gray < 82) & roi
+    tool_restore = (blue & roi) | black_jaw
+    tool_restore = cv2.dilate(tool_restore.astype(np.uint8),
+                              np.ones((3, 3), np.uint8)) > 0
+
+    amplitude = math.radians(27.0)
+    quarter = 0.55
+    release_time = 0.25
+    cycle = 4.0 * quarter
+
+    frames = []
+    for frame_index in range(N):
+        if frame_index == 0:
+            frames.append(source.copy())
+            continue
+
+        t = frame_index / FPS
+        theta = [0.0] * 5
+        if t < release_time:
+            theta[0] = -amplitude
+        else:
+            u = (t - release_time) % cycle
+            if u < quarter:
+                theta[0] = -amplitude * math.cos(0.5 * math.pi * u / quarter)
+            elif u < 2.0 * quarter:
+                theta[4] = amplitude * math.sin(
+                    0.5 * math.pi * (u - quarter) / quarter)
+            elif u < 3.0 * quarter:
+                theta[4] = amplitude * math.cos(
+                    0.5 * math.pi * (u - 2.0 * quarter) / quarter)
+            else:
+                theta[0] = -amplitude * math.sin(
+                    0.5 * math.pi * (u - 3.0 * quarter) / quarter)
+
+        centers = []
+        for i, (pivot, neutral) in enumerate(zip(pivots, rest)):
+            if i in (0, 4):
+                cx = pivot[0] + length * math.sin(theta[i])
+                cy = pivot[1] + length * math.cos(theta[i])
+            else:
+                cx, cy = neutral
+            centers.append((cx, cy))
+
+        frame = plate.copy()
+
+        # Two fine suspension threads per ball, drawn before the spheres.
+        for i, ((px, py), (cx, cy)) in enumerate(zip(pivots, centers)):
+            for offset, shade in ((-1.9, (76, 73, 68)),
+                                  (1.9, (105, 102, 96))):
+                start = (px + offset, py + 1.0)
+                end = (cx + offset * 0.45, cy - 4.2)
+                frame = _draw_wire(frame, start, end, shade)
+
+        # Ball zero is behind the grabber at the start of the action. Draw all
+        # balls in order so contact at the bottom of the arc remains crisp.
+        for cx, cy in centers:
+            frame = _composite_rgba(frame, sprite_rgb, alpha, cx, cy)
+
+        # Put the original gripper back on top of the ball during the release.
+        frame[tool_restore] = source[tool_restore]
+
+        frames.append(np.clip(frame, 0, 255).astype(np.uint8))
+
+    frames = np.asarray(frames, dtype=np.uint8)
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
