+def _smoothstep(edge0, edge1, value):
+    t = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def _make_feature_alpha():
+    """Recover the dark smile texture as an antialiased alpha layer."""
+    alpha = np.zeros((HEIGHT, WIDTH), dtype=np.float32)
+
+    # The two eyes and the curved mouth occupy slightly different depths
+    # on the front of the ball.
+    boxes = (
+        (110.6, 112.8, 94.4, 98.6),

+        ya, yb = int(np.floor(y0)), int(np.ceil(y1))
+        for y in range(ya, yb):
+            for x in range(xa, xb):

+                if not (x0 <= px <= x1 and y0 <= py <= y1):

+                # Dark facial ink is well below the bright yellow-green
+                # diffuse surface in this small central region.
+                ink = np.clip((158.0 - green[y, x]) / 78.0, 0.0, 1.0)
+                if ink > 0.06:

+
+    # A restrained blur matches the camera's sub-pixel antialiasing.
+    alpha = cv2.GaussianBlur(alpha, (0, 0), 0.42)
+    return alpha

+def _remove_original_face(image, alpha):
+    mask = (alpha > 0.035).astype(np.uint8) * 255
+    # Inpaint only the printed smile.  The surrounding ball lighting and
+    # silhouette remain the original photograph.
+    return cv2.inpaint(image, mask, 2.0, cv2.INPAINT_TELEA)
+

+def _sample_alpha(alpha, cx, cy, radius, x, y, angle):
+    """Project a sphere point back to the original face texture."""
+    inside = x * x + y * y
+    valid = inside < 1.0
+    z = np.sqrt(np.maximum(0.0, 1.0 - inside))
+
+    # Yaw around the vertical screen axis.  Positive yaw carries the
+    # front face toward the mirror on the right.
+    source_x = np.cos(angle) * x - np.sin(angle) * z
+    source_y = y
+    sample_x = cx + radius * source_x
+    sample_y = cy + radius * source_y
+
+    sampled = map_coordinates(
+        alpha,
+        np.stack((sample_y, sample_x), axis=0),
+        order=1,
+        mode="constant",
+        cval=0.0,
+        prefilter=False,
+    )
+    return sampled * valid
+
+
+def _render_ball(base, alpha, cx, cy, radius, angle, feature_strength):
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    x = (xx + 0.5 - cx) / radius
+    y = (yy + 0.5 - cy) / radius
+    surface = _sample_alpha(alpha, cx, cy, radius, x, y, angle)
+    surface = np.clip(surface * feature_strength, 0.0, 0.94)
+
+    # The ink is a dark olive-brown, not pure black, under the warm light.
+    ink_color = np.array([69.0, 67.0, 28.0], dtype=np.float32)
+    out = base.astype(np.float32)
+    local = out * 0.55 + ink_color * 0.45
+    out = out * (1.0 - surface[..., None]) + local * surface[..., None]
+    return np.clip(out, 0.0, 255.0)
+
+
+def _render_reflection(base, alpha, angle):
+    # The mirror image is a smaller, softer view of the same ball.  Its
+    # visible hemisphere is blank until the rotating face reaches the side
+    # facing the mirror.
+    cx, cy, radius = 187.3, 76.4, 7.75
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    x = (xx + 0.5 - cx) / radius
+    y = (yy + 0.5 - cy) / radius
+    surface = _sample_alpha(alpha, cx, cy, radius, x, y, angle)
+
+    # The reflection is dim and softened by the glass.  Keep the reveal
+    # subtle at this small scale.
+    reveal = _smoothstep(0.06, 0.20, np.sin(angle))

+    surface = np.clip(surface * (0.44 + 0.28 * reveal), 0.0, 0.48)
+
+    out = base.astype(np.float32)
+    ink_color = np.array([47.0, 48.0, 24.0], dtype=np.float32)
+    local = out * 0.48 + ink_color * 0.52
+    out = out * (1.0 - surface[..., None]) + local * surface[..., None]
+    return np.clip(out, 0.0, 255.0)
+

+def generate():

+    alpha = _make_feature_alpha()
+    clean_ball = _remove_original_face(source, alpha)
+
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = source
+
+    # A constant, slow yaw: just under 50 degrees over the five-second shot.
+    total_yaw = np.deg2rad(49.0)
+    for index in range(1, FRAME_COUNT):
+        angle = total_yaw * index / (FRAME_COUNT - 1)
+        frame = _render_ball(
+            clean_ball,
+            alpha,
+            cx=118.1,
+            cy=101.5,
+            radius=13.25,
+            angle=angle,
+            feature_strength=1.0,
+        )
+        frame = _render_reflection(frame, alpha, angle)
+        frames[index] = np.clip(np.rint(frame), 0, 255).astype(np.uint8)
+
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    generate()
