+def make_background(frame):

+def card_texture(frame):
+    size = (176, 160)
+    destination = np.float32(
+        [[0, 0], [size[0] - 1, 0], [size[0] - 1, size[1] - 1], [0, size[1] - 1]]
+    )
+    matrix = cv2.getPerspectiveTransform(CARD_SRC, destination)
+    texture = cv2.warpPerspective(
+        frame, matrix, size, flags=cv2.INTER_CUBIC, borderMode=cv2.BORDER_REFLECT
+    )

+def card_quad(theta):
+    cos_t = np.cos(theta)
+    sin_t = np.sin(theta)
+    points = []
+    for x in (-CARD_W / 2.0, CARD_W / 2.0, CARD_W / 2.0, -CARD_W / 2.0):
+        projected_x = x * cos_t
+        depth = x * sin_t
+        screen_x = CARD_CX + FOCAL * projected_x / (CAMERA_D - depth)
+        points.append(screen_x)

+    # A common horizontal top and bottom keep the card planted in its base.

+def draw_ball(canvas, center, radius, theta):

+def compose_frame(background, texture, time_value):
+    canvas = cv2.resize(
+        background, (OUT_WIDTH, OUT_HEIGHT), interpolation=cv2.INTER_CUBIC
+    )
+
+    phase = np.clip(time_value / 5.0, 0.0, 1.0)
+    theta = np.deg2rad(125.0) * (phase ** 1.28)
+    descend = smoothstep(phase / 0.86)
+    ball_x = 148.0 + 0.7 * np.sin(phase * np.pi * 1.4)
+    ball_y = 58.0 + 55.0 * descend
+    if phase > 0.88:
+        settle = (phase - 0.88) / 0.12
+        ball_y += 1.2 * np.sin(settle * np.pi * 2.0) * (1.0 - settle)
+    radius = 8.8
+    release = smoothstep((phase - 0.72) / 0.20)
+
+    draw_platform(canvas)
+    draw_ball(canvas, (ball_x, ball_y), radius, theta)
+    draw_gripper(canvas, (ball_x, ball_y), radius, release)
+    draw_card(canvas, texture, theta)
+
+    output = cv2.resize(
+        canvas, (WIDTH, HEIGHT), interpolation=cv2.INTER_AREA
+    )
+    return np.clip(output, 0, 255).astype(np.uint8)
+
