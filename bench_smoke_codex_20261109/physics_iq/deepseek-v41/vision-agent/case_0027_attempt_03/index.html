+def make_clean_plate(source):
+    """Remove the source ball and its shadow while retaining tabletop detail."""
+    bgr = cv2.cvtColor(source, cv2.COLOR_RGB2BGR)
+    mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.ellipse(mask, (82, 57), (13, 13), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (99, 65), (20, 7), -11, 0, 360, 255, -1)
+
+    repaired = cv2.inpaint(bgr, mask, 4, cv2.INPAINT_TELEA)
+    repaired = cv2.cvtColor(repaired, cv2.COLOR_BGR2RGB).astype(np.float32)
+
+    # Reinject fine wood grain where the repair is otherwise too smooth.
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    fine = (
+        1.7 * np.sin(0.47 * xx + 2.25 * yy)

+def _unit_sphere_samples(xx, yy, radius):
+    nx = xx / radius
+    ny = yy / radius
+    rr = nx * nx + ny * ny
+    nz = np.sqrt(np.maximum(0.0, 1.0 - rr))

+def render_ball(radius, spin, phase):

+def add_shadow(frame, cx, cy, radius, on_glass):
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    angle = np.deg2rad(-12.0)
+    dx = xx - (cx + 1.55 * radius)
+    dy = yy - (cy + 0.68 * radius)
+    rx, ry = (1.92 * radius, 0.54 * radius)
+    xr = np.cos(angle) * dx - np.sin(angle) * dy
+    yr = np.sin(angle) * dx + np.cos(angle) * dy
+    weight = np.exp(-0.5 * ((xr / rx) ** 2 + (yr / ry) ** 2))
+    weight *= (np.abs(dx) < 4.0 * rx) & (np.abs(dy) < 4.0 * ry)
+
+    strength = 0.30 if not on_glass else 0.18
+    tint = np.array([0.68, 0.72, 0.80], np.float32) if on_glass else np.array([0.62, 0.68, 0.78], np.float32)
+    attenuation = 1.0 - strength * weight[:, :, None] * (1.0 - tint)
+    return np.clip(frame.astype(np.float32) * attenuation, 0, 255)
+
+
+def add_glass_contact(frame, cx, cy, radius, amount):
+    if amount <= 0:
+        return frame
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    dx = (xx - cx) / (1.35 * radius)
+    dy = (yy - (cy + 0.93 * radius)) / (0.20 * radius)
+    glow = np.exp(-0.5 * (dx * dx + dy * dy))
+    glow *= amount
+    return np.clip(frame + glow[:, :, None] * np.array([12.0, 18.0, 20.0]), 0, 255)
+

+def composite(frame, ball, center_x, center_y):

+    top = int(round(center_y - ball.height / 2.0))
+    canvas = Image.fromarray(np.clip(frame, 0, 255).astype(np.uint8), "RGB").convert("RGBA")
+    canvas.alpha_composite(ball, (left, top))
+    return np.asarray(canvas.convert("RGB"), dtype=np.uint8)
+

+def main():

+        spin = 0.93 * (cx - 81.5) / max(radius, 1.0)
+
+        frame = clean.astype(np.float32)
+        # Very light temporal sensor texture keeps the locked-off shot from looking frozen.
+        frame += rng.normal(0.0, 0.38, frame.shape)
+

+        frame = add_glass_contact(frame, cx, cy, radius, np.clip((cx - 141.0) / 16.0, 0.0, 1.0))
+        ball = render_ball(radius, spin, 0.35)
+        frame = composite(frame, ball, cx, cy)
+        frames.append(frame)
+
+    frames = np.asarray(frames, dtype=np.uint8)
+    assert np.array_equal(frames[0], source)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {len(frames)} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
