+def _feather_mask(shape, center, axes):
+    """Return a soft elliptical alpha mask."""
+    yy, xx = np.mgrid[: shape[0], : shape[1]]
+    distance = (
+        ((xx - center[0]) / axes[0]) ** 2
+        + ((yy - center[1]) / axes[1]) ** 2
+    )
+    alpha = np.clip((1.08 - distance) / 0.16, 0.0, 1.0)
+    return alpha.astype(np.float32)
+
+
+def _extract_sprite(image, center, axes):
+    """Extract a ball and a little of its blurred fringe."""
+    x0 = max(0, int(np.floor(center[0] - axes[0] - 3)))
+    x1 = min(WIDTH, int(np.ceil(center[0] + axes[0] + 4)))
+    y0 = max(0, int(np.floor(center[1] - axes[1] - 3)))
+    y1 = min(HEIGHT, int(np.ceil(center[1] + axes[1] + 4)))
+    rgb = image[y0:y1, x0:x1].astype(np.float32)
+    alpha = _feather_mask(rgb.shape[:2], (center[0] - x0, center[1] - y0), axes)
+    return rgb, alpha, (x0, y0)
+
+
+def _clean_plate(image):
+    """Remove the two initial balls while retaining the table and pipes."""
+    mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.ellipse(mask, (142, 50), (15, 12), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (91, 89), (18, 15), 0, 0, 360, 255, -1)
+    mask = cv2.GaussianBlur(mask, (5, 5), 1.1)
+    return cv2.inpaint(image, mask, 4, cv2.INPAINT_TELEA)
+
+
+def _motion_blur(rgb, alpha, direction, length):
+    """Blur premultiplied color and alpha along the table direction."""
+    if length <= 1:
+        return rgb, alpha
+    weights = np.hanning(length + 2)[1:-1].astype(np.float32)
+    weights /= weights.sum()
+    kernel = np.zeros((1, length), np.float32)
+    kernel[0] = weights
+    if direction > 0:
+        kernel = kernel[:, ::-1]
+    premultiplied = rgb * alpha[..., None]
+    rgb_blur = cv2.filter2D(
+        premultiplied, -1, kernel, borderType=cv2.BORDER_REPLICATE
+    )
+    alpha_blur = cv2.filter2D(
+        alpha, -1, kernel, borderType=cv2.BORDER_REPLICATE
+    )
+    return rgb_blur / np.maximum(alpha_blur[..., None], 1e-4), alpha_blur
+
+
+def _scale_sprite(rgb, alpha, scale):
+    if abs(scale - 1.0) < 1e-3:
+        return rgb, alpha
+    height, width = alpha.shape
+    size = (
+        max(1, int(round(width * scale))),
+        max(1, int(round(height * scale))),
+    )
+    rgb_scaled = cv2.resize(rgb, size, interpolation=cv2.INTER_CUBIC)
+    alpha_scaled = cv2.resize(alpha, size, interpolation=cv2.INTER_CUBIC)
+    return rgb_scaled, alpha_scaled
+
+
+def _composite(canvas, rgb, alpha, center):
+    height, width = alpha.shape
+    left = int(round(center[0] - width / 2))
+    top = int(round(center[1] - height / 2))
+    x0, y0 = max(0, left), max(0, top)
+    x1 = min(WIDTH, left + width)
+    y1 = min(HEIGHT, top + height)
+    if x0 >= x1 or y0 >= y1:
+        return
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + x1 - x0, sy0 + y1 - y0
+    local_rgb = rgb[sy0:sy1, sx0:sx1]
+    local_alpha = alpha[sy0:sy1, sx0:sx1, None]
+    destination = canvas[y0:y1, x0:x1].astype(np.float32)
+    canvas[y0:y1, x0:x1] = np.clip(
+        local_rgb * local_alpha + destination * (1.0 - local_alpha),
+        0,
+        255,
+    ).astype(np.uint8)
+
+
+def _add_shadow(canvas, center, axes, opacity, direction):
+    """Darken the wood with a soft, slightly trailing contact shadow."""
+    layer = np.zeros((HEIGHT, WIDTH), np.float32)
+    offset_x = -3.2 if direction < 0 else 3.2
+    offset_y = 4.0
+    cv2.ellipse(
+        layer,
+        (int(round(center[0] + offset_x)), int(round(center[1] + offset_y))),
+        (int(round(axes[0] * 1.05)), max(2, int(round(axes[1] * 0.56)))),
+        0,
+        0,
+        360,
+        1.0,
+        -1,
+    )
+    if direction < 0:
+        tail = cv2.GaussianBlur(layer, (0, 0), 3.0)
+        layer = np.maximum(layer, tail * 0.45)
+    else:
+        layer = cv2.GaussianBlur(layer, (0, 0), 2.1)
+    layer = cv2.GaussianBlur(layer, (0, 0), 1.25)
+    multiplier = 1.0 - opacity * layer[..., None]
+    canvas[:] = np.clip(canvas.astype(np.float32) * multiplier, 0, 255).astype(
+        np.uint8
+    )
+
+

+    plate = _clean_plate(first)

+    # The foreground balls are small because they are still near the far edge
+    # of the table. Their paths run in opposite directions and do not collide.
+    balls = [

+            "axes": (9.0, 8.0),
+            "velocity": (-22.0, 0.42),
+            "acceleration": 0.45,
+            "scale_rate": 0.010,
+            "shadow": 0.16,
+            "blur": 7,
+        },
+        {
+            "center": (91.0, 88.5),
+            "axes": (12.4, 11.6),
+            "velocity": (23.0, 0.63),
+            "acceleration": -0.32,
+            "scale_rate": 0.008,
+            "shadow": 0.25,
+            "blur": 9,
+        },
+    ]
+
+    for ball in balls:
+        ball["sprite"] = _extract_sprite(first, ball["center"], ball["axes"])
+
+    frames = [first.copy()]
+    for frame_index in range(1, FRAME_COUNT):
+        canvas = plate.copy()
+        time = float(frame_index)
+
+        for ball in balls:
+            direction = -1 if ball["velocity"][0] < 0 else 1
+            travel = (
+                ball["velocity"][0] * time
+                + 0.5 * ball["acceleration"] * time * time
+            )
+            x = ball["center"][0] + travel
+            y = ball["center"][1] + ball["velocity"][1] * time
+            scale = 1.0 + ball["scale_rate"] * time
+            axes = (ball["axes"][0] * scale, ball["axes"][1] * scale)
+
+            if x < -axes[0] - 14 or x > WIDTH + axes[0] + 14:
+                continue
+
+            _add_shadow(canvas, (x, y), axes, ball["shadow"], direction)
+            rgb, alpha, _ = ball["sprite"]
+            rgb, alpha = _scale_sprite(rgb, alpha, scale)
+            rgb, alpha = _motion_blur(rgb, alpha, direction, ball["blur"])
+            _composite(canvas, rgb, alpha, (x, y))
+
+        frames.append(canvas)
+
+    return np.stack(frames, axis=0)
+

+    frames = generate_frames()
+    artifact_io.save(frames, case_dir)
