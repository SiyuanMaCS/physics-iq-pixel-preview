+def _row_clean_plate(image):
+    """Remove the mug and its soft shadow from the smooth projection screen."""
+    source = image.astype(np.float32)
+    out = source.copy()
+    x0, x1, y0, y1 = 107, 149, 47, 90
+    xs = np.arange(WIDTH)
+    valid = ((xs >= 20) & (xs <= 105)) | ((xs >= 150) & (xs <= 240))
+    xn = (xs - 128.0) / 128.0
+    design = np.stack((np.ones_like(xn), xn, xn * xn), axis=1)
+
+    for y in range(y0, y1):
+        for channel in range(3):
+            keep = valid.copy()
+            coefficients = None
+            for _ in range(4):
+                coefficients = np.linalg.lstsq(
+                    design[keep], source[y, keep, channel], rcond=None
+                )[0]
+                residual = source[y, keep, channel] - design[keep] @ coefficients
+                center = np.median(residual)
+                spread = np.median(np.abs(residual - center)) + 1e-6
+                next_keep = keep.copy()
+                accepted = np.abs(residual - center) < 3.0 * 1.4826 * spread
+                if accepted.sum() > 40:
+                    next_keep[keep] = accepted
+                keep = next_keep
+            out[y, x0:x1, channel] = (design @ coefficients)[x0:x1]
+
+    alpha = np.ones((y1 - y0, x1 - x0), np.float32)
+    for i in range(4):
+        weight = (i + 1) / 5.0
+        alpha[:, i] = np.minimum(alpha[:, i], weight)
+        alpha[:, -1 - i] = np.minimum(alpha[:, -1 - i], weight)
+        alpha[i, :] = np.minimum(alpha[i, :], weight)
+        alpha[-1 - i, :] = np.minimum(alpha[-1 - i, :], weight)
+    out[y0:y1, x0:x1] = (
+        source[y0:y1, x0:x1] * (1.0 - alpha[..., None])
+        + out[y0:y1, x0:x1] * alpha[..., None]
+    )
+    return out

+def build_mug_sprite(image):
+    """Extract the original mug with a clean feathered alpha matte."""
+    x0, y0, x1, y1 = 114, 50, 144, 82
+    roi = image[y0:y1, x0:x1].copy()

+    ax = xx + x0
+    ay = yy + y0
+    warm = (
+        (r.astype(np.int16) > b.astype(np.int16) + 2)
+        & (r > 24)
+        & (g.astype(np.int16) >= b.astype(np.int16))
+    )

+    body = np.zeros_like(warm, dtype=np.uint8)

+        if len(selected):

+    # the blue-gray screen into the mug silhouette.

+def _compose_mug(up_bg, sprite, offset_x, offset_y, angle, scale, squash):
+    """Composite one premultiplied sprite sample onto the supersampled patch."""
+    x0, y0, x1, y1 = PATCH
+    crop_x, crop_y = 114, 50
+    center = (14.5, 16.0)
+    rotation = cv2.getRotationMatrix2D(center, angle, scale)
+
+    # Apply vertical squash around the sprite center, then map source pixels to
+    # the supersampled patch.
+    rotation[1, :2] *= squash
+    rotation[1, 2] = center[1] + squash * (rotation[1, 2] - center[1])
+    transform = rotation.copy()
+    transform[:, :2] *= SUPERSAMPLE
+    transform[:, 2] = SUPERSAMPLE * (
+        rotation[:, 2]
+        + np.array([crop_x - x0 + offset_x, crop_y - y0 + offset_y])
+    )
+
+    premultiplied = sprite[:, :, :3].astype(np.float32) * (
+        sprite[:, :, 3:4].astype(np.float32) / 255.0
+    )
+    rgba_premultiplied = np.dstack(
+        (premultiplied, sprite[:, :, 3].astype(np.float32))
+    )
+    output_size = (up_bg.shape[1], up_bg.shape[0])
+    warped = cv2.warpAffine(
+        rgba_premultiplied,
+        transform,
+        output_size,
+        flags=cv2.INTER_LANCZOS4,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+    alpha = np.clip(warped[:, :, 3:4] / 255.0, 0.0, 1.0)
+    return warped[:, :, :3] + up_bg * (1.0 - alpha)

+def _draw_shadows(up_bg, offset_y, contact):
+    x0, y0, x1, y1 = PATCH
+    h = (y1 - y0) * SUPERSAMPLE
+    w = (x1 - x0) * SUPERSAMPLE
+    shadow = np.zeros((h, w), np.float32)
+
+    screen_strength = 0.105 * max(0.0, 1.0 - max(0.0, offset_y - 8.0) / 10.0)
+    if screen_strength > 0:
+        center = (
+            int((133.0 - x0) * SUPERSAMPLE),
+            int((78.0 + offset_y * 0.42 - y0) * SUPERSAMPLE),
+        )
+        axes = (int(8.5 * SUPERSAMPLE), int(3.0 * SUPERSAMPLE))
+        cv2.ellipse(shadow, center, axes, 0, 0, 360, screen_strength, -1)
+
+    if contact > 0:
+        center = (int((129.0 - x0) * SUPERSAMPLE), int((93.0 - y0) * SUPERSAMPLE))
+        axes = (int(10.0 * SUPERSAMPLE), int(2.0 * SUPERSAMPLE))
+        cv2.ellipse(shadow, center, axes, 0, 0, 360, 0.22 * contact, -1)
+
+    shadow = cv2.GaussianBlur(shadow, (0, 0), 2.0 * SUPERSAMPLE)
+    return up_bg * (1.0 - np.clip(shadow[..., None], 0.0, 0.35))

+def _add_dust(up_image, frame_index, contact):
+    if frame_index < 5 or frame_index > 9:
+        return up_image
+    x0, y0, _, _ = PATCH
+    phase = frame_index - 5
+    fade = max(0.0, 1.0 - phase / 5.0)
+    rng = np.random.default_rng(126)
+    layer = np.zeros(up_image.shape[:2], np.float32)
+    for _ in range(10):
+        px = 123.0 + rng.uniform(-7.0, 7.0) + phase * rng.uniform(-0.8, 0.8)
+        py = 92.0 + rng.uniform(-1.0, 2.0) - phase * rng.uniform(0.25, 1.0)
+        radius = (0.55 + rng.uniform(0.0, 0.8) + phase * 0.25) * SUPERSAMPLE
+        center = (
+            int((px - x0) * SUPERSAMPLE),
+            int((py - y0) * SUPERSAMPLE),
+        )
+        cv2.circle(layer, center, max(1, int(radius)), 0.09 * fade, -1)
+    layer = cv2.GaussianBlur(layer, (0, 0), 1.1 * SUPERSAMPLE)
+    return up_image * (1.0 - np.clip(layer[..., None], 0.0, 0.12) * contact)
+

+def _interpolate(start, end, count):
+    if count <= 1:
+        return [end]
+    return [
+        tuple(a + (b - a) * t for a, b in zip(start, end))
+        for t in np.linspace(0.0, 1.0, count)
+    ]
+

+def motion_for_frame(index):
+    # (vertical offset, angle, scale, vertical squash, samples)
+    tracks = {
+        1: ((0.0, 0.0, 1.0, 1.0), (0.7, -0.6, 1.0, 1.0), 3),
+        2: ((0.7, -0.6, 1.0, 1.0), (3.4, -2.2, 1.0, 1.0), 6),

+        7: ((12.8, -5.8, 0.99, 1.0), (16.8, -2.0, 0.99, 0.98), 5),
+        8: ((16.8, -2.0, 0.99, 0.98), (15.2, 0.5, 1.0, 1.0), 4),
+        9: ((15.2, 0.5, 1.0, 1.0), (16.55, -0.4, 1.0, 1.0), 3),
+        10: ((16.55, -0.4, 1.0, 1.0), (16.25, 0.15, 1.0, 1.0), 2),
+    }
+    if index in tracks:
+        start, end, count = tracks[index]
+        return _interpolate(start, end, count)
+    return [(16.3, 0.0, 1.0, 1.0)]
+

+def render_frame(index, first, clean, sprite, cable):
+    if index == 0:
+        return first.copy()
+
+    x0, y0, x1, y1 = PATCH
+    base = clean.copy()
+
+    cable_bottom = {1: 39, 2: 26, 3: 13}.get(index, 0)
+    if cable_bottom > 0:
+        cable_height = cable_bottom
+        cable_region = cable[:cable_height]
+        bx0, by0 = 117, 0
+        base[by0 : by0 + cable_height, bx0 : bx0 + cable_region.shape[1]] = (
+            base[by0 : by0 + cable_height, bx0 : bx0 + cable_region.shape[1]]
+            * (1.0 - cable_region[:, :, 3:4].astype(np.float32) / 255.0)
+            + cable_region[:, :, :3].astype(np.float32)
+            * (cable_region[:, :, 3:4].astype(np.float32) / 255.0)
+        ).astype(np.uint8)
+
+    patch = base[y0:y1, x0:x1]
+    up_bg = cv2.resize(
+        patch,
+        None,
+        fx=SUPERSAMPLE,
+        fy=SUPERSAMPLE,
+        interpolation=cv2.INTER_CUBIC,
+    ).astype(np.float32)
+
+    samples = motion_for_frame(index)
+    final_offset = samples[-1][0]
+    contact = np.clip((final_offset - 12.0) / 5.0, 0.0, 1.0)
+    up_bg = _draw_shadows(up_bg, final_offset, contact)
+
+    accumulator = np.zeros_like(up_bg)
+    for offset_y, angle, scale, squash in samples:
+        accumulator += _compose_mug(
+            up_bg,
+            sprite,
+            offset_x=0.0,
+            offset_y=offset_y,
+            angle=angle,
+            scale=scale,
+            squash=squash,
+        )
+    up_bg = accumulator / len(samples)
+    up_bg = _add_dust(up_bg, index, contact)
+
+    rendered_patch = cv2.resize(
+        np.clip(up_bg, 0, 255).astype(np.uint8),
+        (x1 - x0, y1 - y0),
+        interpolation=cv2.INTER_AREA,
+    )
+    frame = base.copy()
+    frame[y0:y1, x0:x1] = rendered_patch
+    return frame
+
+
+def main():
+    first = cv2.imread(str(ROOT / "first_frame.png"), cv2.IMREAD_COLOR)
+    if first is None:
+        raise FileNotFoundError("first_frame.png")
+    clean = build_clean_plate(first)
+    sprite = build_mug_sprite(first)
+    cable = build_cable_overlay(first, clean)
+
+    frames = np.stack(
+        [render_frame(i, first, clean, sprite, cable) for i in range(FRAMES)],
+        axis=0,
+    ).astype(np.uint8)
+    save(frames[:, :, :, ::-1].copy(), ROOT)
+
+
+if __name__ == "__main__":
+    main()
