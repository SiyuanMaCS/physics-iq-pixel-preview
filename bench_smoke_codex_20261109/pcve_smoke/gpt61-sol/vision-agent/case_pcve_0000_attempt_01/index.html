def simulate(tracks):

    start=np.array([to_world(ANCHORS[j]) for j in [1,0,2]])
    # Use the actual incoming path through the moment of contact.

        for _ in range(20):
            dt=0.05;state+=vel*dt
            # Very low air-table drag, preserving the source's almost uniform free flight.
            vel*=np.exp(-0.00065*dt)
            for j in [0,1]:
                gap=r[j]+r[j+1]
                if state[j+1]-state[j]<gap:
                    overlap=gap-(state[j+1]-state[j]);total=mass[j]+mass[j+1]
                    state[j]-=overlap*mass[j+1]/total;state[j+1]+=overlap*mass[j]/total
                    if vel[j]>vel[j+1]:

                        vel[j]=((mass[j]-e*mass[j+1])*u+(1+e)*mass[j+1]*v)/total
                        vel[j+1]=((mass[j+1]-e*mass[j])*v+(1+e)*mass[j]*u)/total
            # The source front bumper strongly damps the white mallet's rebound.
            limits=[(0.085,1.92),(0.085,1.92),(0.06,1.936)]

        q[0,0]=start[0,0]+0.001*(state[0]-start[0,1])

def paste_sprite(canvas,sprite,world):
    rgb,alpha,origin,anchor=sprite;dest=project(world)
    w0=to_world(anchor)
    # Apparent size varies continuously with depth, not with mass.
    scale=(H[2]@np.r_[w0,1])/(H[2]@np.r_[world,1])
    yy,xx=np.mgrid[:canvas.shape[0],:canvas.shape[1]]
    # Small contact shadow and translucent reflection on polished air-table surface.
    cx,cy=dest
    shadow=np.exp(-(((xx-(cx+5*scale))/(27*scale))**2+((yy-(cy+6*scale))/(5.0*scale))**2)*2)
    canvas*=1-0.11*shadow[:,:,None]
    M=np.array([[scale,0,dest[0]+scale*(origin[0]-anchor[0])],[0,scale,dest[1]+scale*(origin[1]-anchor[1])]],float)
    a=cv2.warpAffine(alpha.astype('float32'),M,(1280,720),flags=cv2.INTER_LINEAR)
    premul=cv2.warpAffine((rgb*alpha[:,:,None]).astype('float32'),M,(1280,720),flags=cv2.INTER_LINEAR)
    # Subtle surface reflection derived from the actual source object's colors.
    rm=M.copy();rm[1,1]=-scale*0.8;rm[1,2]=dest[1]+scale*0.8*(anchor[1]-origin[1])+5*scale
    ra=cv2.warpAffine(alpha.astype('float32'),rm,(1280,720));rp=cv2.warpAffine((rgb*alpha[:,:,None]).astype('float32'),rm,(1280,720))
    ra=cv2.GaussianBlur(ra,(0,0),4.0)*0.10;rp=cv2.GaussianBlur(rp,(0,0),4.0)*0.10
    canvas[:]=canvas*(1-ra[:,:,None])+rp
    canvas[:]=canvas*(1-a[:,:,None])+premul
