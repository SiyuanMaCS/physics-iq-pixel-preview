+
+import cv2
+import numpy as np
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+N_FRAMES = 96
+HEIGHT = 144
+WIDTH = 256
+FPS = 24
+
+# The red mallet is stationary until the player's stroke begins at frame 13.
+STROKE_START = 13
+STROKE_END = 31
+HEAVY_SCALE = 0.25
+
+
+def load_frames():
+    data = np.load(CASE_DIR / "source_pixels.npz")
+    return data["rgb"].astype(np.uint8)
+
+
+def red_mask(frame):
+    rgb = frame.astype(np.int16)
+    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+    mask = (r > 92) & (r > g * 1.34) & (r > b * 1.16)
+    mask[:52] = False
+    mask[96:] = False
+    mask[:, :88] = False
+    mask[:, 165:] = False
+    return mask
+
+
+def red_center(frame):
+    ys, xs = np.nonzero(red_mask(frame))
+    if len(xs) == 0:
+        return None
+    return float(xs.mean()), float(ys.mean())
+
+
+def track_red(frames):
+    centers = []
+    last = None
+    for frame in frames:
+        center = red_center(frame)
+        if center is None:
+            center = last
+        centers.append(center)
+        if center is not None:
+            last = center
+    centers = np.asarray(centers, dtype=np.float32)
+    for i in range(1, len(centers)):
+        if not np.isfinite(centers[i]).all():
+            centers[i] = centers[i - 1]
+    return centers
+
+
+def ellipse_mask(shape, center, radius, feather=1.8):
+    yy, xx = np.mgrid[: shape[0], : shape[1]]
+    dx = (xx - center[0]) / radius[0]
+    dy = (yy - center[1]) / radius[1]
+    dist = np.sqrt(dx * dx + dy * dy)
+    alpha = np.clip((1.0 - dist) * radius[0] / feather + 0.5, 0.0, 1.0)
+    return alpha
+
+
+def build_background_plate(frames):
+    """Temporal median with red-mallet pixels excluded per pixel."""
+    h, w = frames.shape[1:3]
+    samples = np.empty((len(frames), h, w, 3), dtype=np.float32)
+    valid = np.empty((len(frames), h, w), dtype=bool)
+    for i, frame in enumerate(frames):
+        samples[i] = frame
+        valid[i] = ~red_mask(frame)
+    plate = np.empty((h, w, 3), dtype=np.float32)
+    for y in range(h):
+        for x in range(w):
+            values = samples[:, y, x][valid[:, y, x]]
+            plate[y, x] = np.median(values, axis=0)
+    return np.clip(plate + 0.5, 0, 255).astype(np.uint8)
+
+
+def heavy_centers(source_centers):
+    start = source_centers[STROKE_START].copy()
+    end = source_centers[STROKE_END].copy()
+    centers = source_centers.copy()
+    for i in range(STROKE_START, len(centers)):
+        if i <= STROKE_END:
+            u = (i - STROKE_START) / (STROKE_END - STROKE_START)
+            u = 0.5 - 0.5 * np.cos(np.pi * np.clip(u, 0.0, 1.0))
+        else:
+            u = 1.0
+        centers[i] = start + HEAVY_SCALE * u * (end - start)
+    return centers
+
+
+def render(frames, source_centers, new_centers, background_plate):
+    h, w = frames.shape[1:3]
+    out = np.empty_like(frames)
+    yy, xx = np.mgrid[:h, :w]
+
+    sprite_radius = (13.0, 10.5)
+    for i, frame in enumerate(frames):
+        base = frame.astype(np.float32)
+        src = source_centers[i]
+        dst = new_centers[i]
+
+        if i >= STROKE_START:
+            old_alpha = ellipse_mask((h, w), src, sprite_radius, feather=1.4)
+            base = base * (1.0 - old_alpha[..., None]) + background_plate * old_alpha[..., None]
+
+        if i >= STROKE_START:
+            half_w, half_h = 16, 13
+            x0 = int(round(src[0])) - half_w
+            x1 = int(round(src[0])) + half_w + 1
+            y0 = int(round(src[1])) - half_h
+            y1 = int(round(src[1])) + half_h + 1
+            x0, x1 = max(0, x0), min(w, x1)
+            y0, y1 = max(0, y0), min(h, y1)
+            sprite = frame[y0:y1, x0:x1].astype(np.float32)
+            local_center = (src[0] - x0, src[1] - y0)
+            alpha = ellipse_mask(sprite.shape[:2], local_center, sprite_radius, feather=1.8)
+
+            dx = int(round(dst[0])) - int(round(src[0]))
+            dy = int(round(dst[1])) - int(round(src[1]))
+            tx0, tx1 = x0 + dx, x1 + dx
+            ty0, ty1 = y0 + dy, y1 + dy
+            cx0, cy0 = max(0, tx0), max(0, ty0)
+            cx1, cy1 = min(w, tx1), min(h, ty1)
+            if cx1 > cx0 and cy1 > cy0:
+                sx0 = cx0 - tx0
+                sy0 = cy0 - ty0
+                sx1 = sx0 + (cx1 - cx0)
+                sy1 = sy0 + (cy1 - cy0)
+                a = alpha[sy0:sy1, sx0:sx1, None]
+                patch = sprite[sy0:sy1, sx0:sx1]
+                target = base[cy0:cy1, cx0:cx1]
+                base[cy0:cy1, cx0:cx1] = target * (1.0 - a) + patch * a
+
+        out[i] = np.clip(base + 0.5, 0, 255).astype(np.uint8)
+    return out
+
+
+def save_preview(frames, path):
+    writer = cv2.VideoWriter(
+        str(path), cv2.VideoWriter_fourcc(*"mp4v"), FPS, (WIDTH, HEIGHT)
+    )
+    for frame in frames:
+        writer.write(cv2.cvtColor(frame, cv2.COLOR_RGB2BGR))
+    writer.release()
+
+
+def main():
+    frames = load_frames()
+    assert frames.shape == (N_FRAMES, HEIGHT, WIDTH, 3)
+    source_centers = track_red(frames)
+    new_centers = heavy_centers(source_centers)
+    background_plate = build_background_plate(frames)
+    edited = render(frames, source_centers, new_centers, background_plate)
+
+    artifact_io.save(edited, CASE_DIR)
+    save_preview(edited, CASE_DIR / "counterfactual.mp4")
+    np.savez_compressed(
+        CASE_DIR / "red_mallet_motion.npz",
+        source_centers=source_centers,
+        heavy_centers=new_centers,
+    )
+
+
+if __name__ == "__main__":
+    main()
