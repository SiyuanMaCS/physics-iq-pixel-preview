def match(frame, patch, mask, guess_x, guess_y, search=6):
    \"\"\"Find integer top-left position minimizing masked color SSD, near guess.\"\"\"
    h, w = mask.shape
    best = None
    gx, gy = int(round(guess_x)), int(round(guess_y))
    for dy in range(gy-search, gy+search+1):
        for dx in range(gx-search, gx+search+1):
            if dx < 0 or dy < 0 or dx+w > W or dy+h > H:
                continue
            region = frame[dy:dy+h, dx:dx+w]
            dist = np.abs(region - patch).sum(axis=2)
            cost = dist[mask].sum() / mask.sum()
            if best is None or cost < best[0]:
                best = (cost, dx, dy)
    return best
