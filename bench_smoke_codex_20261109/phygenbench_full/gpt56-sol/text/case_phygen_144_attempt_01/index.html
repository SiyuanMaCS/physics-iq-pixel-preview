def finish(img,f):
    rgb=np.asarray(img.convert('RGB').resize((W,H),Image.Resampling.LANCZOS),dtype=np.float32)
    yy,xx=np.mgrid[0:H,0:W]
    dx=(xx-(W-1)/2)/(W*.72);dy=(yy-(H-1)/2)/(H*.72)
    vign=np.clip(1-.31*(dx*dx+dy*dy),.72,1.0)
    rgb*=vign[:,:,None]
    # slight warm global lift after ignition, without blowing out flame cores
    if f>=9:
        warm=min(1,(f-9)/9)
        rgb[:,:,0]+=warm*2.2
        rgb[:,:,2]-=warm*.8
    return np.uint8(np.clip(rgb,0,255))

    img=draw_debris(img,f)
    img=draw_foreground_strands(img,f)
    img=draw_ground_fire(img,f)
    img=draw_sparks(img,f)
    return finish(img,f)
