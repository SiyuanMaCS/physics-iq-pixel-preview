+def clean_table(image):
+    """Recover the small table area hidden by the moving ball and its shadow.

+    Local wooden grain supplies the gradient field, while all unoccluded

+    """

+    shadow = (((.96 * (x - 49) - .28 * (y - 47)) / 17) ** 2 +
+              ((.28 * (x - 49) + .96 * (y - 47)) / 6.5) ** 2 < 1)
+    mask = ball | shadow
+    index = np.full((h, w), -1, dtype=int)
+    index[mask] = np.arange(mask.sum())
+    yy, xx = np.where(mask)
+    donor = image[np.clip(y + 6, 0, h - 1), np.clip(x + 65, 0, w - 1)]
+    A = sparse.lil_matrix((len(yy), len(yy)))
+    b = np.zeros((len(yy), 3))
+    for k, (v, u) in enumerate(zip(yy, xx)):
+        A[k, k] = 4
+        for dv, du in ((0, 1), (0, -1), (1, 0), (-1, 0)):
+            v1, u1 = v + dv, u + du
+            b[k] += donor[v, u] - donor[v1, u1]
+            if mask[v1, u1]:
+                A[k, index[v1, u1]] = -1
+            else:
+                b[k] += image[v1, u1]
+    bg = image.copy()
+    bg[mask] = spsolve(A.tocsr(), b)
+    return np.clip(bg, 0, 255)
