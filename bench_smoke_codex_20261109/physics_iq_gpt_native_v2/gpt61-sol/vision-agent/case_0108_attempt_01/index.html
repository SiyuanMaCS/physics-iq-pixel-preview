+def background(image):
+    # The first photograph is also the static plate. Remove only the marble,
+    # retaining the patterned lip of its starting track.
+    yy, xx = np.mgrid[:144, :256]
+    region = (((xx - 130.25) / 4.5) ** 2 +

+def simulate():
+    # Piecewise rolling-contact dynamics with ballistic transfers. Coordinates
+    # describe the marble center, just above each track's top edge.
+    g = 1150.0
+    tracks = [
+        (87.0, 150.0, 33.0, -0.100),
+        (42.0, 100.0, 42.0, 0.176),
+        (90.0, 152.0, 66.0, -0.080),
+        (39.0, 96.0, 74.7, 0.176),
+    ]
+    dt = .0005
+    x, y = 130.25, 28.675
+    vx = vy = 0.0
+    track = 0
+    mode = 'roll'
+    history = []
+    for k in range(10002):
+        t = k * dt
+        history.append((t, x, y, vx, vy))
+        if mode == 'rest':
+            continue
+        if mode == 'catch':
+            # A brief small bounce inside the U-shaped final collector.
+            vx += (-95 * (x - 104.5) - 16 * vx) * dt
+            vy += g * dt
+            x += vx * dt
+            y += vy * dt
+            floor = 109.05 - .175 * (x - 96)
+            if y > floor:
+                y = floor
+                vy = -abs(vy)*.15
+                vx *= .6
+                if abs(vy) < 3:
+                    mode = 'rest'
+            x = np.clip(x, 98, 108.5)
+            continue
+        if mode == 'roll':
+            lo, hi, intercept, slope = tracks[track]
+            acceleration = (5/7) * g*slope/(1+slope*slope)
+            vx += acceleration * dt
+            x += vx * dt
+            y = intercept + slope*(x-lo)
+            vy = vx*slope
+            if (slope < 0 and x <= lo) or (slope > 0 and x >= hi):
+                mode = 'fall'
+                track += 1
+        else:
+            oldy = y
+            vy += g * dt
+            x += vx * dt
+            y += vy * dt
+            if track < len(tracks):
+                lo, hi, intercept, slope = tracks[track]
+                surface = intercept + slope*(x-lo)
+                if lo <= x <= hi and oldy <= surface <= y:
+                    # Energy is dissipated at the joint; retain a little
+                    # horizontal motion, so there is a natural short reversal.
+                    vx *= .13
+                    y = surface
+                    mode = 'roll'
+            elif y >= 109.05 - .175*(x-96):
+                y = 109.05 - .175*(x-96)
+                mode = 'catch'
+                vy = -abs(vy)*.21
+                vx *= .23
+    return np.array(history)
+

+def marble(plate, x, y):
+    # Supersample only a local window, using a softly focused, translucent
+    # amber-yellow glass sphere and its subdued board/contact shadow.
+    ss = 6
+    left, top = int(np.floor(x)) - 7, int(np.floor(y)) - 7
+    right, bottom = left + 15, top + 15
+    left, top = max(0, left), max(0, top)
+    right, bottom = min(256, right), min(144, bottom)
+    h, w = (bottom-top)*ss, (right-left)*ss
+    yy, xx = np.mgrid[:h, :w].astype(float)
+    xx = left + (xx+.5)/ss
+    yy = top + (yy+.5)/ss
+    dx, dy = xx-x, yy-y
+    r = 2.22
+    rho = np.sqrt(dx*dx + dy*dy)/r
+    alpha = np.clip((r-np.sqrt(dx*dx+dy*dy))*ss+.5, 0, 1)
+    normalz = np.sqrt(np.clip(1-rho*rho, 0, 1))
+    light = np.clip(-.25*dx/r - .37*dy/r + .87*normalz, 0, 1)
+    color = np.zeros((h, w, 3), float)
+    color[:] = (92, 72, 11)
+    color += light[..., None]*np.array([28, 23, 9])
+    glint = np.exp(-((dx+.72)**2 + (dy+.95)**2)/.22)
+    color += glint[..., None]*np.array([22, 20, 13])
+    shadow = .13*np.exp(-((dx-.85)**2/6 + (dy-1.35)**2/3))
+    alpha = gaussian_filter(alpha, .53*ss)
+    premult = np.stack([gaussian_filter(color[..., c] *
+                                       np.clip((r-np.sqrt(dx*dx+dy*dy))*ss+.5,0,1),
+                                       .53*ss) for c in range(3)], axis=-1)
+    # Downsample premultiplied layer before compositing so that the static
+    # photograph never undergoes resampling.
+    alpha = alpha.reshape(bottom-top, ss, right-left, ss).mean((1,3))
+    premult = premult.reshape(bottom-top, ss, right-left, ss, 3).mean((1,3))
+    shadow = shadow.reshape(bottom-top, ss, right-left, ss).mean((1,3))
+    out = plate.astype(float).copy()
+    under = out[top:bottom, left:right]*(1-shadow[..., None])
+    out[top:bottom, left:right] = under*(1-alpha[..., None])+premult
+    return out
+
+
+def main():
+    first = np.array(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    plate = background(first)
+    trajectory = simulate()
+    frames = [first]
+    for i in range(1, 20):
+        # Short real-world exposure: only the moving marble is blurred.
+        layers = []
+        for offset in np.linspace(-.010, .010, 5):
+            t = i/4 + offset
+            x = np.interp(t, trajectory[:,0], trajectory[:,1])
+            y = np.interp(t, trajectory[:,0], trajectory[:,2])
+            layers.append(marble(plate, x, y))
+        frames.append(np.clip(np.mean(layers, axis=0),0,255).round().astype(np.uint8))
+    artifact_io.save(np.stack(frames), ROOT)
+    np.savetxt(ROOT/'.tmp/trajectory.txt', trajectory[::500], fmt='%.3f',
+               header='seconds x y vx vy')
+
+
+if __name__ == '__main__':
+    main()
