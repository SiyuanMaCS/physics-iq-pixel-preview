    out[:,1]=CY-PITCH*xy[:,1] + (z*.325 if mirror else -z)
    return out


def get_texture(src):
    # A normalized material texture sampled from the interior of the lit face.
    th,tw=200,80
    vv,uu=np.mgrid[:th,:tw].astype(float)
    uu=(uu+.5)/tw; vv=(vv+.5)/th
    xy=corners(THETA0)
    top=project(xy,BH); bot=project(xy,0)
    # Sampling is inset to exclude the silhouette's mixed wall/blue pixels.
    a=.10+.70*uu
    b=.075+.86*vv
    tx=(top[0,0]*(1-a)+top[1,0]*a)*(1-b)+(bot[0,0]*(1-a)+bot[1,0]*a)*b
    ty=(top[0,1]*(1-a)+top[1,1]*a)*(1-b)+(bot[0,1]*(1-a)+bot[1,1]*a)*b
    texture=cv2.remap(src.astype(np.float32),tx.astype(np.float32),ty.astype(np.float32),cv2.INTER_LINEAR)
    # Suppress isolated camera/edge artifacts while retaining vertical illumination.
    texture=gaussian_filter(texture,(.7,.8,0))
    return texture


def raster_face(canvas, poly, texture, color_scale, face_id, mirror=False):
    # Analytic inverse parallelogram map, antialiased by supersampling.
    poly=np.asarray(poly)*SS
    lo=np.floor(poly.min(0)-2).astype(int); hi=np.ceil(poly.max(0)+2).astype(int)
    lo=np.maximum(lo,[0,0]); hi=np.minimum(hi,[W*SS,H*SS])
    if np.any(hi<=lo):return
    xs=np.arange(lo[0],hi[0])+0.5-SS*.5
    ys=np.arange(lo[1],hi[1])+0.5-SS*.5
    # Pixel coordinates use integer source-image centers.
    gx,gy=np.meshgrid(xs,ys)
    a=poly[1]-poly[0]; b=poly[3]-poly[0]
    mat=np.stack([a,b],axis=1)
    if abs(np.linalg.det(mat))<1e-6:return
    inv=np.linalg.inv(mat)
    dx=gx-poly[0,0]; dy=gy-poly[0,1]
    uu=inv[0,0]*dx+inv[0,1]*dy
    vv=inv[1,0]*dx+inv[1,1]*dy
    mask=(uu>=0)&(uu<=1)&(vv>=0)&(vv<=1)
    if not mask.any():return
    if face_id=='top':
        rgb=np.zeros((*uu.shape,3),np.float32)
        rgb[:]=[33,90,130]
        rgb*=1+.07*(uu-.5)[...,None]
    else:
        # All four sides have the same blue material; their brightness follows
        # the fixed light direction as the face normals rotate.
        mapsx=np.clip(uu,0,1).astype(np.float32)*(texture.shape[1]-1)
        mapsy=np.clip(vv,0,1).astype(np.float32)*(texture.shape[0]-1)
        rgb=cv2.remap(texture,mapsx,mapsy,cv2.INTER_LINEAR)
        rgb*=color_scale
        edge=np.exp(-((uu-.992)/.018)**2)*.20
        if not mirror:
            highlight=np.array([100,223,255],np.float32)*np.minimum(color_scale,1.05)
            rgb=rgb*(1-edge[...,None])+highlight*edge[...,None]
    patch=canvas[lo[1]:hi[1],lo[0]:hi[0]]
    if mirror:
        # Blurred, attenuated reflection in the black glossy turntable.
        strength=.43-.065*(1-np.clip(vv,0,1))
        reflected=rgb*np.array([.62,.92,1.0]) * strength[...,None]
        patch[mask]=reflected[mask]
    else:
        patch[mask]=np.clip(rgb[mask],0,255)


def draw_block(base,theta,texture,reflection=False):
    xy=corners(theta)
    top=project(xy,BH,reflection); bottom=project(xy,0,reflection)
    out=base.copy()
    light=np.array([.65,-.76]);light/=np.linalg.norm(light)
    u=np.array([np.cos(theta),np.sin(theta)])
    v=np.array([-np.sin(theta),np.cos(theta)])
    normals=[-v,u,v,-u]
    n0=np.array([np.sin(THETA0),-np.cos(THETA0)])
    init=.20+.80*max(0,np.dot(n0,light))
    visible=[]
    # Camera is in front of the tabletop, toward negative world Y.
    for j,n in enumerate(normals):
        if n[1]<-1e-5:
            k=(j+1)%4
            ratio=(.20+.80*max(0,np.dot(n,light)))/init
            powers=np.array([.55,.40,.09]) if ratio>=1 else np.array([1.0,.9,.7])
            scale=ratio**powers
            depth=(xy[j,1]+xy[k,1])*.5
            visible.append((depth,j,[top[j],top[k],bottom[k],bottom[j]],scale))
    if not reflection:
        raster_face(out,[top[3],top[2],top[1],top[0]],texture,np.ones(3),'top')
    for _,j,poly,scale in sorted(visible,reverse=True):
        raster_face(out,poly,texture,scale,j,reflection)
    return out


def shadow_extent(theta):
    xy=corners(theta)
    xx=1.50*xy[:,0]+.15*xy[:,1]
    return xx.max()-xx.min()


def main():
    src=np.array(Image.open(ROOT/'first_frame.png').convert('RGB'))
    bg,shadow=background_and_shadow(src)
    texture=get_texture(src)
    bigbg=cv2.resize(bg,(W*SS,H*SS),interpolation=cv2.INTER_CUBIC)
    disk=plate_mask(SS)
    originalwidth=shadow_extent(THETA0)
    yy,xx=np.mgrid[:H,:W].astype(np.float32)
    frames=[src.copy()]
    for index in range(1,20):
        t=index/4
        theta=THETA0-np.deg2rad(30)*t
        ratio=shadow_extent(theta)/originalwidth
        # Rotating a vertical cuboid changes the width of its wall shadow,
        # while the lamp, wall, height and turntable center stay fixed.
        center=110.9 + .22*(np.sin(theta)-np.sin(THETA0))
        mapx=(xx-center)/ratio+110.9
        mapy=yy
        a=cv2.remap(shadow,mapx.astype(np.float32),mapy,cv2.INTER_LINEAR,borderMode=cv2.BORDER_CONSTANT)
        if ratio<.86:
            a=gaussian_filter(a,(0,.5*np.sqrt(1-ratio*ratio)))
        frame=bg*(1-a[...,None])
        # Preserve small warm ambient illumination in the penumbra.
        frame[:,:,0]+=a*1.0
        frame[:,:,2]-=a*1.0
        scene=cv2.resize(frame,(W*SS,H*SS),interpolation=cv2.INTER_CUBIC)
        reflected=draw_block(scene,theta,texture,True)
        # Reflection is constrained to the finite circular platform.
        delta=reflected-scene
        delta=gaussian_filter(delta,(SS*.19,SS*.19,0))
        scene += delta*disk[...,None]
        # A small stable contact shadow makes the rotating block sit on the disc.
        ybig,xbig=np.mgrid[:H*SS,:W*SS].astype(np.float32)
        xf=(xbig+.5)/SS-.5; yf=(ybig+.5)/SS-.5
        footprint=shadow_extent(theta)/1.5
        contact=np.exp(-((xf-CX)/(footprint*.52))**2-((yf-124)/.52)**2)*.11
        scene*=1-contact[...,None]*disk[...,None]
        scene=draw_block(scene,theta,texture)
        frame=cv2.resize(scene,(W,H),interpolation=cv2.INTER_AREA)
        frames.append(np.clip(np.rint(frame),0,255).astype(np.uint8))
    frames=np.stack(frames)
    save(frames,ROOT)
    # Keep a numerical candidate convenient for local quality checks.
    np.save(ROOT/'.tmp'/'candidate_rgb.npy',frames)

if __name__=='__main__':
    main()
