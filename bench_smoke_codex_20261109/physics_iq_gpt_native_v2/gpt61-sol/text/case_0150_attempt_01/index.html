+def surface_texture(im, background):
+    # Recover the visible felt, shading and rounded boundary numerically.
+    Y, X = np.mgrid[:144, :256].astype(float)
+    radius, cx, cy, shear = 8.75, 104.15, 47.95, 0.073
+    n = np.stack(((X - cx - shear*(Y-cy)) / radius, (Y - cy) / 8.92), -1)
+    rr = np.sum(n*n, -1)

+    alpha = gaussian_filter(alpha, .63)
+    # Inverse alpha compositing is well conditioned in the interior.
+    recovered = (im.astype(float) - background * (1 - alpha[...,None])) / np.maximum(alpha[...,None], .12)
+    nz = np.sqrt(np.maximum(0,1-rr))
+    base = np.stack((43.5 - 9*n[...,0] + 1*n[...,1] + 3*nz,
+                     23.4 - 3*n[...,0] + .6*n[...,1] + 3*nz,
+                     15.5 - n[...,0] + .4*n[...,1] + 3*nz), -1)
+    # Remove the seam before making the felt texture. A rotating surface seam is
+    # reconstructed below, rather than a stationary painted stripe.
+    seam_y = 47.8 + .095*(X-cx)
+    body = (rr < .56) & (np.abs(Y-seam_y) > 1.55)
+    felt = np.clip(recovered - base, -9, 9)
+    felt[~body] = 0
+    return felt, body, recovered, alpha

import numpy as np
from PIL import Image
import cv2
from scipy.ndimage import gaussian_filter, map_coordinates
import artifact_io

ROOT = Path(__file__).resolve().parent


def clean_plate(im):
    yy, xx = np.mgrid[:144, :256]
    # Only remove the original sphere and its immediate soft contact shadow.
    region = (((xx - 104.0) / 18.0) ** 2 + ((yy - 49.0) / 12.3) ** 2 < 1) & (yy >= 38)
    return cv2.inpaint(im, region.astype(np.uint8) * 255, 4, cv2.INPAINT_NS).astype(float)


def surface_texture(im, background):
    # Recover the visible felt, shading and rounded boundary numerically.
    Y, X = np.mgrid[:144, :256].astype(float)
    radius, cx, cy, shear = 8.75, 104.15, 47.95, 0.073
    n = np.stack(((X - cx - shear*(Y-cy)) / radius, (Y - cy) / 8.92), -1)
    rr = np.sum(n*n, -1)
    alpha = np.clip((1 - np.sqrt(rr)) * radius + .5, 0, 1)
    alpha = gaussian_filter(alpha, .63)
    # Inverse alpha compositing is well conditioned in the interior.
    recovered = (im.astype(float) - background * (1 - alpha[...,None])) / np.maximum(alpha[...,None], .12)
    nz = np.sqrt(np.maximum(0,1-rr))
    base = np.stack((43.5 - 9*n[...,0] + 1*n[...,1] + 3*nz,
                     23.4 - 3*n[...,0] + .6*n[...,1] + 3*nz,
                     15.5 - n[...,0] + .4*n[...,1] + 3*nz), -1)
    # Remove the seam before making the felt texture. A rotating surface seam is
    # reconstructed below, rather than a stationary painted stripe.
    seam_y = 47.8 + .095*(X-cx)
    body = (rr < .56) & (np.abs(Y-seam_y) > 1.55)
    felt = np.clip(recovered - base, -9, 9)
    felt[~body] = 0
    return felt, body, recovered, alpha

    bg = clean_plate(original)
    felt, body, recovered, original_alpha = surface_texture(original, bg)

    for frame_number in range(1,20):

        cx = cx0 + distance

        ny = (YY-cy)/8.92

        # the image by the table's perspective. Inverse map into sphere texture.
        axis = np.array([-.108, 1., 0.])
        axis /= np.linalg.norm(axis)
        co, si = np.cos(theta), np.sin(theta)
        dot = nx*axis[0]+ny*axis[1]

        qx = nx*co - si*(axis[1]*nz) + axis[0]*dot*(1-co)

            # modeled above; no camera/background dissolve is introduced.
            pass

    assert np.array_equal(video[0],original)
