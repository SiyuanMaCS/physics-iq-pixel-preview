+"""Generate the requested locked-camera rolling-ball demonstration.
+
+The photograph is used as the static plate.  The one moving object is isolated
+as a small difference sprite, which retains the photograph's real blur and
+lighting, and is rolled analytically across the perspective plane of the table.
+"""
+
+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+HERE = Path(__file__).resolve().parent
+
+
+def clean_table(frame: np.ndarray) -> np.ndarray:
+    """Remove the ball in order to recover a static photographic plate."""
+    mask = np.zeros(frame.shape[:2], np.uint8)
+    # Ball, antialiased fringe, and its short contact shadow.
+    cv2.ellipse(mask, (108, 76), (10, 9), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (109, 82), (12, 4), 0, 0, 360, 255, -1)
+    # Telea interpolation works especially well here because the tabletop has
+    # low-contrast, nearly horizontal wood grain.
+    return cv2.inpaint(frame, mask, 5.0, cv2.INPAINT_TELEA)
+
+
+def object_delta(frame: np.ndarray, plate: np.ndarray):
+    """Return the real ball and shadow as a feathered signed difference layer."""
+    x0, y0, x1, y1 = 95, 65, 123, 88
+    raw = frame[y0:y1, x0:x1].astype(np.float32) - plate[y0:y1, x0:x1].astype(np.float32)
+
+    yy, xx = np.mgrid[y0:y1, x0:x1]
+    # Soft ball support, plus a flatter support for the photographed shadow.
+    ball = np.clip(1.0 - (((xx - 108.0) / 9.4) ** 2 + ((yy - 76.0) / 8.2) ** 2), 0, 1)
+    ball = np.clip(ball * 2.0, 0, 1)
+    shadow = np.exp(-0.5 * (((xx - 109.0) / 8.3) ** 2 + ((yy - 82.0) / 2.5) ** 2))
+    support = np.maximum(ball, 0.72 * shadow)
+    support = cv2.GaussianBlur(support.astype(np.float32), (0, 0), 0.65)
+    return raw * support[..., None], (108.0 - x0, 76.0 - y0)
+
+
+def shifted_rotated_delta(delta: np.ndarray, center, angle: float) -> np.ndarray:
+    """Rotate the ball texture while leaving the contact shadow approximately fixed."""
+    h, w = delta.shape[:2]
+    # The sprite is so small that a cubic affine transform gives a natural
+    # subpixel photographic softness rather than a conspicuous drawn edge.
+    matrix = cv2.getRotationMatrix2D(center, angle, 1.0)
+    return cv2.warpAffine(delta, matrix, (w, h), flags=cv2.INTER_CUBIC,
+                          borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+
+
+def composite_delta(canvas: np.ndarray, sprite: np.ndarray, cx: float, cy: float, center):
+    """Add a signed RGB sprite to canvas at a subpixel center."""
+    h, w = sprite.shape[:2]
+    # A fractional translation avoids 4-fps stair stepping.
+    left = cx - center[0]
+    top = cy - center[1]
+    ix, iy = int(np.floor(left)), int(np.floor(top))
+    fx, fy = left - ix, top - iy
+    trans = np.float32([[1, 0, fx], [0, 1, fy]])
+    moved = cv2.warpAffine(sprite, trans, (w, h), flags=cv2.INTER_CUBIC,
+                           borderMode=cv2.BORDER_CONSTANT, borderValue=0)
+
+    xa, ya = max(0, ix), max(0, iy)
+    xb, yb = min(canvas.shape[1], ix + w), min(canvas.shape[0], iy + h)
+    if xa < xb and ya < yb:
+        sx0, sy0 = xa - ix, ya - iy
+        canvas[ya:yb, xa:xb] += moved[sy0:sy0 + (yb - ya), sx0:sx0 + (xb - xa)]
+
+
+def generate() -> np.ndarray:
+    first = np.asarray(Image.open(HERE / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    plate_u8 = clean_table(first)
+    delta, center = object_delta(first, plate_u8)
+
+    frames = [first.copy()]  # Contract: frame zero is byte-for-byte identical.
+    # The rear and front table edges converge.  A rightward line on its plane
+    # therefore descends slightly in image space.  The final center lies on the
+    # sloping right edge of the tabletop.
+    for i in range(1, 20):
+        u = i / 19.0
+        # Smooth-start only in the first half-second, followed by a steady roll.
+        # This prevents a jarring frame-0 jump while still traversing the table.
+        distance = 118.0 * (u - 0.035 * np.sin(np.pi * u))
+        cx = 108.0 + distance
+        cy = 76.0 + 0.045 * distance
+
+        # Approximately 4.2 complete turns across the visible run.  Rotating
+        # the real seam/highlight texture makes the rolling action legible.
+        angle = -1510.0 * (distance / 118.0)
+        sprite = shifted_rotated_delta(delta, center, angle)
+
+        work = plate_u8.astype(np.float32)
+        composite_delta(work, sprite, cx, cy, center)
+        frames.append(np.clip(np.rint(work), 0, 255).astype(np.uint8))
+
+    result = np.stack(frames, axis=0)
+    assert result.shape == (20, 144, 256, 3) and result.dtype == np.uint8
+    assert np.array_equal(result[0], first)
+    return result
+
+
+if __name__ == "__main__":
+    artifact_io.save(generate(), HERE)
