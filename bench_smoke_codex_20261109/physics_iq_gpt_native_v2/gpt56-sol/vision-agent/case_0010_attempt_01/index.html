+from PIL import Image

+def soft_ellipse(h, w, cx, cy, rx, ry, feather):
+    yy, xx = np.mgrid[:h, :w].astype(np.float32)
+    d = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    # Feather is expressed as a fraction of the normalized ellipse radius.
+    return np.clip((1.0 + feather - d) / feather, 0.0, 1.0)
+
+
+def reconstruct_tabletop(rgb):
+    """Remove the initial orange ball while retaining the horizontal wood grain."""
+    h, w = rgb.shape[:2]
+    yy, xx = np.mgrid[:h, :w]
+
+    # This mask covers the orange sphere and its diffuse right-hand shadow only.
+    ball = ((xx - 73.0) / 13.8) ** 2 + ((yy - 51.5) / 13.0) ** 2 < 1.0
+    shadow = ((xx - 84.0) / 19.0) ** 2 + ((yy - 53.0) / 11.8) ** 2 < 1.0
+    hole = ((ball | shadow).astype(np.uint8) * 255)
+
+    # Telea inpainting follows the long, nearly horizontal features of this
+    # tabletop.  A mild bilateral pass only inside the hole removes tiny
+    # propagation artifacts without changing any stationary scene pixels.
+    bgr = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)
+    filled = cv2.inpaint(bgr, hole, 4.0, cv2.INPAINT_TELEA)
+    filled = cv2.cvtColor(filled, cv2.COLOR_BGR2RGB)
+    smooth = cv2.bilateralFilter(filled, 5, 9, 9)
+    core = cv2.GaussianBlur(hole, (0, 0), 1.2).astype(np.float32)[:, :, None] / 255.0
+    clean = rgb.astype(np.float32) * (1.0 - core) + smooth.astype(np.float32) * core
