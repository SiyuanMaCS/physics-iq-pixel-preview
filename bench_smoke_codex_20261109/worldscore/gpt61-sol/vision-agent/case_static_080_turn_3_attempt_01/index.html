+"""A deterministic, image-based dolly retreat of a stationary terrace.
+
+The visible image is retained as the texture.  Border architecture is continued
+with slope-aligned texture sampling; a small, boundary-anchored parallax field
+inside the terrace gives the distant skyline less motion than the doorway.
+No scene element, illumination, or random texture changes with time.
+"""
+from pathlib import Path
+import numpy as np
+import cv2
+from PIL import Image
+import artifact_io
+
+ROOT = Path(__file__).resolve().parent
+
+
+def remap(image, x, y):
+    return cv2.remap(image, x.astype(np.float32), y.astype(np.float32),
+                     cv2.INTER_CUBIC, borderMode=cv2.BORDER_REPLICATE)
+
+
+def make_surround(image, pad=56):
+    h, w = image.shape[:2]
+    yy, xx = np.mgrid[-pad:h+pad, -pad:w+pad].astype(np.float32)
+    x = np.clip(xx, 0, w-1)
+    y = np.clip(yy, 0, h-1)
+    canvas = remap(image, x, y).astype(np.float32)
+
+    # Continue the diagonal stair flight, rather than stretching its last column.
+    right = xx > w-1
+    dx = np.maximum(xx-(w-1), 0)
+    sx = (w-1)-np.minimum(dx, 22)
+    sy = yy + 0.79*(xx-sx)
+    stair = remap(image, sx, np.minimum(sy, 88)).astype(np.float32)
+    # Staircase ends behind the pale foreground parapet.
+    weight = np.clip((90-yy)/5, 0, 1)*right
+    canvas = canvas*(1-weight[..., None])+stair*weight[..., None]
+
+    # The upper wall and the floor are unpatterned surfaces. Continue their
+    # illumination, with a little fixed microtexture, not edge-pixel stripes.
+    rng = np.random.default_rng(810)
+    grain = cv2.GaussianBlur(rng.normal(0, .38, canvas.shape).astype(np.float32),
+                             (3, 3), .55)
+    outside = (xx < 0) | (xx > w-1) | (yy < 0) | (yy > h-1)
+    distance = np.maximum.reduce([np.maximum(-xx,0), np.maximum(xx-w+1,0),
+                                  np.maximum(-yy,0), np.maximum(yy-h+1,0)])
+    if np.any(yy < 0):
+        top_color = np.median(image[3:8].astype(np.float32), axis=0)[None]
+        top = remap(np.repeat(top_color, 8, axis=0), x, np.zeros_like(x))
+        mix = np.clip(-yy/8, 0, 1)
+        canvas = canvas*(1-mix[..., None])+top*mix[..., None]
+    if np.any(yy >= h):
+        # Broad, softly lit concrete and painted panel below the photograph.
+        lower = cv2.GaussianBlur(image[-4:].astype(np.float32), (7, 3), 1.1)
+        low = remap(lower, x, np.full_like(y, 3))
+        mix = np.clip((yy-(h-1))/7, 0, 1)
+        canvas = canvas*(1-mix[..., None])+low*mix[..., None]
+        canvas -= (np.maximum(yy-h+1, 0)*.065)[..., None]
+    canvas += grain * (outside*np.clip(distance/4,0,1))[..., None]
+    canvas = np.clip(canvas, 0, 255).astype(np.uint8)
+    canvas[pad:pad+h, pad:pad+w] = image
+    return canvas
+
+
+def main():
+    source = np.asarray(Image.open(ROOT/'first_frame.png').convert('RGB'))
+    pad = 56
+    canvas = make_surround(source, pad)
+    Image.fromarray(canvas).save(ROOT/'.tmp/surround.png')
+    yy, xx = np.mgrid[0:144, 0:256].astype(np.float32)
+    cx, cy = 125.0, 64.0
+    frames = [source.copy()]
+    for i in range(1, 50):
+        u = i/49
+        # A retreat at almost constant speed, eased at the beginning and end.
+        t = .82*u + .18*(3*u*u-2*u*u*u)
+        scale = 1/(1+.235*t)
+        sx = cx+(xx-cx)/scale
+        sy = cy+(yy-cy)/scale
+
+        # Distant garden is seen through an architectural opening. The
+        # field vanishes at its edges, leaving the straight lintel and jambs
+        # untouched; the skyline has a little less parallax than the floor.
+        ax = np.clip((sx-20)/142, 0, 1)
+        ay = np.clip((sy-13)/125, 0, 1)
+        aperture = np.sin(np.pi*ax)**.8 * np.sin(np.pi*ay)**.8
+        aperture *= ((sx > 20)&(sx < 162)&(sy > 13)&(sy < 138))
+        distance = np.clip((110-sy)/55, 0, 1)
+        p = .035*t*aperture*distance
+        sx -= (sx-102)*p
+        sy -= (sy-65)*p
+        frame = remap(canvas, sx+pad, sy+pad)
+        frames.append(frame)
+    frames = np.stack(frames).astype(np.uint8)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == '__main__':
+    main()
