+def polygon_mask(points, shape=(HEIGHT, WIDTH), blur=0.0):

+def ellipse_mask(center, axes, shape=(HEIGHT, WIDTH), blur=0.0):

+class Sprite:
+    def __init__(self, image, alpha, pivot):
+        self.rgb = image.astype(np.float32)
+        self.alpha = alpha.astype(np.float32) / 255.0
+        self.pivot = np.asarray(pivot, np.float32)

+    def render(self, target, angle=0.0, scale=1.0, opacity=1.0):
+        matrix = cv2.getRotationMatrix2D(tuple(self.pivot), angle, scale)
+        matrix[:, 2] += np.asarray(target, np.float32) - self.pivot
+        rgb = cv2.warpAffine(
+            self.rgb,
+            matrix,

+            matrix,
+            (WIDTH, HEIGHT),
+            flags=cv2.INTER_LINEAR,
+            borderMode=cv2.BORDER_CONSTANT,
+            borderValue=0,

+def color_sprite(image, region, color_mask, pivot):

+def ease(value):

+def interpolate_keyframes(frame, keys):
+    if frame <= keys[0][0]:
+        return np.asarray(keys[0][1], np.float32)
+    if frame >= keys[-1][0]:
+        return np.asarray(keys[-1][1], np.float32)
+    for (a, pa), (b, pb) in zip(keys[:-1], keys[1:]):
+        if a <= frame <= b:
+            t = ease((frame - a) / float(b - a))
+            return np.asarray(pa, np.float32) * (1.0 - t) + np.asarray(pb, np.float32) * t
