+import cv2
+import numpy as np

+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 160
+FRAME_COUNT = 20

+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def bilinear_sample(image, x, y):
+    h, w = image.shape[:2]
+    x = np.clip(x, 0.0, w - 1.001)
+    y = np.clip(y, 0.0, h - 1.001)
+    x0 = np.floor(x).astype(np.int32)
+    y0 = np.floor(y).astype(np.int32)
+    x1 = np.minimum(x0 + 1, w - 1)
+    y1 = np.minimum(y0 + 1, h - 1)
+    fx = (x - x0)[..., None]
+    fy = (y - y0)[..., None]
+    top = image[y0, x0] * (1.0 - fx) + image[y0, x1] * fx
+    bottom = image[y1, x0] * (1.0 - fx) + image[y1, x1] * fx
+    return top * (1.0 - fy) + bottom * fy
+
+
+def cloth_mask():
+    mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    outline = np.array(
+        [
+            (124, 87), (155, 87), (156, 95), (154, 106), (157, 121),
+            (153, 134), (157, 147), (152, 152), (126, 152), (123, 143),
+            (125, 134), (122, 124), (124, 113), (122, 102),
+        ],
+        np.int32,
+    )
+    cv2.fillPoly(mask, [outline], 255)
+    return mask
+
+
+def arm_alpha(image):
+    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
+    hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)
+
+    upper = np.zeros((HEIGHT, WIDTH), np.uint8)
+    upper_parts = [
+        [(151, 91), (159, 90), (163, 101), (169, 108), (167, 120),
+         (162, 128), (156, 123), (152, 114), (150, 103)],
+        [(156, 108), (164, 104), (174, 111), (179, 124), (171, 136),
+         (162, 130), (157, 120)],
+    ]
+    for part in upper_parts:
+        cv2.fillPoly(upper, [np.array(part, np.int32)], 255)
+
+    dark = gray < 86
+    silver = (hsv[:, :, 1] < 31) & (hsv[:, :, 2] > 92)
+    alpha = ((dark | silver) & (upper > 0)).astype(np.uint8) * 255
+
+    wrist = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.fillPoly(
+        wrist,
+        [np.array(
+            [(165, 121), (177, 118), (189, 126), (198, 135),
+             (211, 142), (226, 156), (231, 160), (165, 160),
+             (160, 143), (160, 131)],
+            np.int32,
+        )],
+        255,
+    )
+    alpha = cv2.bitwise_or(alpha, wrist)
+    alpha = cv2.morphologyEx(alpha, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+
+    # Reject isolated table fragments while preserving the hand and the
+    # off-screen forearm.
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(alpha)
+    kept = np.zeros_like(alpha)
+    for index in range(1, count):
+        area = stats[index, cv2.CC_STAT_AREA]
+        top = stats[index, cv2.CC_STAT_TOP]
+        height = stats[index, cv2.CC_STAT_HEIGHT]
+        if area >= 16 or top + height >= HEIGHT - 2:
+            kept[labels == index] = 255
+    return kept
+
+
+def clean_background(image, object_mask):
+    inpainted = cv2.inpaint(image, object_mask, 6, cv2.INPAINT_NS)
+
+    # The cloth hides a broad part of a horizontally grained table. Continue
+    # that grain from the unobstructed strip at the left instead of leaving a
+    # smooth inpaint patch visible when the cloth moves.
+    patch = image[55:160, 78:122].astype(np.float32)
+    background = inpainted.astype(np.float32)
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    sx = np.mod(xx - 78, patch.shape[1])
+    sy = np.clip(yy - 55, 0, patch.shape[0] - 1)
+    grain = patch[sy, sx]
+    region = cv2.dilate(object_mask, np.ones((5, 5), np.uint8)) > 0
+    blend = np.where(region[..., None], 0.56, 0.0)
+    background = background * (1.0 - blend) + grain * blend
+
+    # A gentle horizontal light falloff matches the original table.
+    falloff = np.interp(xx, [0, 255], [1.015, 0.955])[..., None]
+    background *= falloff
+    return np.clip(background, 0, 255).astype(np.uint8)
+
+
+def extract_assets(image):
+    cloth = cloth_mask()
+    arm = arm_alpha(image)
+    object_mask = cv2.bitwise_or(cloth, arm)
+    background = clean_background(image, object_mask)
+
+    # Remove the original gripper from the cloth texture before it is warped.
+    repair_mask = cv2.dilate(arm, np.ones((3, 3), np.uint8))
+    repaired = cv2.inpaint(image, repair_mask, 3, cv2.INPAINT_TELEA)
+    texture = repaired.astype(np.float32)
+
+    # Cloth sprites are stored premultiplied by alpha for clean edges.
+    alpha = (cloth.astype(np.float32) / 255.0)[..., None]
+    cloth_rgba = np.dstack([texture * alpha, alpha[..., 0] * 255.0])
+
+    x0, y0, x1, y1 = 146, 86, WIDTH, HEIGHT
+    arm_roi = image[y0:y1, x0:x1].astype(np.float32)
+    arm_a = (arm[y0:y1, x0:x1].astype(np.float32) / 255.0)[..., None]
+    arm_rgba = np.dstack([arm_roi * arm_a, arm_a[..., 0] * 255.0])
+    return background, cloth_rgba, (x0, y0), arm_rgba
+
+
+def composite_rgba(canvas, rgba, x0, y0):
+    h, w = rgba.shape[:2]
+    sx0 = max(0, -x0)
+    sy0 = max(0, -y0)
+    dx0 = max(0, x0)
+    dy0 = max(0, y0)
+    dx1 = min(WIDTH, x0 + w)
+    dy1 = min(HEIGHT, y0 + h)
+    if dx1 <= dx0 or dy1 <= dy0:
+        return
+    src = rgba[sy0:sy0 + dy1 - dy0, sx0:sx0 + dx1 - dx0]
+    alpha = src[..., 3:4] / 255.0
+    dst = canvas[dy0:dy1, dx0:dx1].astype(np.float32)
+    canvas[dy0:dy1, dx0:dx1] = np.clip(
+        src[..., :3] + dst * (1.0 - alpha), 0, 255
+    ).astype(np.uint8)
+
+
+def draw_left_half(canvas, cloth_rgba, crease_x):
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    alpha = cloth_rgba[..., 3] / 255.0
+    half_alpha = alpha * (xx <= crease_x)
+    rgba = cloth_rgba.copy()
+    rgba[..., 3] = half_alpha * 255.0
+    rgba[..., :3] *= (half_alpha > 0)[..., None]
+    # Re-premultiply the unchanged left side.
+    rgba[..., :3] = cloth_rgba[..., :3] * (xx <= crease_x)[..., None]
+    composite_rgba(canvas, rgba, 0, 0)
+
+
+def folded_half_layer(cloth_rgba, crease_x, angle):
+    h, w = cloth_rgba.shape[:2]
+    source = cloth_rgba[..., :3].astype(np.float32)
+    source_alpha = cloth_rgba[..., 3] / 255.0
+    output = np.zeros((h, w, 4), np.float32)
+
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    c = float(np.cos(angle))
+    s = float(np.sin(angle))
+    if abs(c) < 0.055:
+        # Edge-on cloth still has a visible folded seam.
+        distance = np.abs(xx - crease_x)
+        valid = (distance <= 1.0) & (yy >= 87) & (yy <= 153)
+        shade = np.clip(0.82 + 0.1 * np.exp(-(distance / 1.5) ** 2), 0, 1)
+        output[valid, :3] = np.array([126, 112, 82], np.float32) * shade[valid, None]
+        output[valid, 3] = 235
+        return output
+
+    d_out = xx - crease_x
+    d_source = d_out / c
+    y_source = yy - 0.22 * d_source * s
+    valid = (
+        (d_source >= 0.0)
+        & (d_source <= 34.0)
+        & (y_source >= 0.0)
+        & (y_source < HEIGHT - 0.01)
+        & (xx >= 112)
+        & (xx <= 162)
+    )
+    sample_x = crease_x + d_source
+    sampled = bilinear_sample(source, sample_x, y_source)
+    sampled_alpha = bilinear_sample(
+        source_alpha[..., None], sample_x, y_source
+    )[..., 0]
+    valid &= sampled_alpha > 0.015
+
+    shade = 0.86 + 0.14 * min(1.0, abs(c) + 0.08)
+    sampled *= shade
+
+    # A thin lit edge on the moving cloth helps read the folded panel at low
+    # resolution without changing its silhouette.
+    edge = np.zeros_like(valid, np.uint8)
+    edge[valid] = 255
+    edge = cv2.morphologyEx(edge, cv2.MORPH_GRADIENT, np.ones((2, 2), np.uint8)) > 0
+    edge &= valid
+    sampled[edge] = np.clip(sampled[edge] * 1.11 + 5.0, 0, 255)
+
+    output[valid, :3] = sampled[valid]
+    output[valid, 3] = np.clip(sampled_alpha[valid] * 255.0, 0, 255)
+    return output
+
+
+def cast_shadow(canvas, center_x, center_y, angle):
+    if abs(np.sin(angle)) < 0.04:
+        return canvas
+    shadow = np.zeros((HEIGHT, WIDTH), np.float32)
+    width = max(3, int(14 * abs(np.cos(angle)) + 5))
+    cv2.ellipse(
+        shadow, (int(center_x + 5), int(center_y + 5)),
+        (width, 20), 0, 0, 360, 1.0, -1,
+    )
+    shadow = cv2.GaussianBlur(shadow, (0, 0), 3.2)
+    shadow *= 0.13 * abs(np.sin(angle))
+    darkened = canvas.astype(np.float32) * (1.0 - shadow[..., None] * 0.32)
+    return np.clip(darkened, 0, 255).astype(np.uint8)
+
+
+def transform_arm(arm_rgba, origin, grasp, destination, angle):
+    sx, sy = origin
+    gx, gy = grasp
+    tx, ty = destination
+    theta = np.deg2rad(angle)
+    c, s = np.cos(theta), np.sin(theta)
+
+    # Rotation about the grasp point followed by a rigid translation.
+    matrix = np.array(
+        [
+            [c, -s, tx - c * (gx - sx) + s * (gy - sy)],
+            [s, c, ty - s * (gx - sx) - c * (gy - sy)],
+        ],
+        np.float32,
+    )
+    warped = cv2.warpAffine(
+        arm_rgba, matrix, (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+
+    # Continue the forearm into the right border after the rigid hand has
+    # translated. The hidden part of the arm lies outside the camera frame.
+    alpha = warped[..., 3] > 2
+    for y in range(HEIGHT):
+        columns = np.flatnonzero(alpha[y])
+        if not len(columns):
+            continue
+        right = columns[-1]
+        if right < WIDTH - 1 and right > 190:
+            warped[y, right + 1:, :] = warped[y, right, :]
+    return warped
+
+
+def arm_shadow(canvas, arm_layer):
+    alpha = arm_layer[..., 3] / 255.0
+    shadow = np.zeros((HEIGHT, WIDTH), np.float32)
+    shadow[3:, 2:] = alpha[:-3, :-2]
+    shadow = cv2.GaussianBlur(shadow, (0, 0), 1.4) * 0.28
+    base = canvas.astype(np.float32) * (1.0 - shadow[..., None] * 0.42)
+    return np.clip(base, 0, 255).astype(np.uint8)
+
+
+def render_frame(frame_index, first, assets):
+    if frame_index == 0:
+        return first.copy()
+
+    background, cloth_rgba, arm_origin, arm_rgba = assets
+    canvas = background.copy()
+
+    fold = smoothstep((frame_index - 2.0) / 12.0)
+    release = smoothstep((frame_index - 16.0) / 3.0)
+    angle = np.pi * fold
+    crease_x = 141.0
+
+    draw_left_half(canvas, cloth_rgba, crease_x)
+    canvas = cast_shadow(canvas, crease_x, 119.0, angle)
+    folded = folded_half_layer(cloth_rgba, crease_x, angle)
+    composite_rgba(canvas, folded, 0, 0)
+
+    grasp = (157.0, 113.0)
+    destination = (
+        grasp[0] - 32.0 * fold + 11.0 * release,
+        grasp[1] - 6.0 * np.sin(angle) - 9.0 * release,
+    )
+    rotation = -19.0 * np.sin(angle) * (1.0 - 0.65 * release)
+    arm_layer = transform_arm(
+        arm_rgba, arm_origin, grasp, destination, rotation
+    )
+    canvas = arm_shadow(canvas, arm_layer)
+    composite_rgba(canvas, arm_layer, 0, 0)
+    return canvas
+
+
+def main():
+    first = cv2.imread(str(CASE_DIR / "first_frame.png"))
+    if first is None:
+        raise FileNotFoundError(CASE_DIR / "first_frame.png")
+    first = cv2.cvtColor(first, cv2.COLOR_BGR2RGB)
+    assets = extract_assets(cv2.cvtColor(first, cv2.COLOR_RGB2BGR))
+
+    frames = []
+    for index in range(FRAME_COUNT):
+        bgr = render_frame(index, cv2.cvtColor(first, cv2.COLOR_RGB2BGR), assets)
+        frames.append(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
+    frames = np.stack(frames).astype(np.uint8)
+    assert frames.shape == (FRAME_COUNT, HEIGHT, WIDTH, 3)
+    assert np.array_equal(frames[0], first)
+    artifact_io.save(frames, CASE_DIR)
+
+
+if __name__ == "__main__":
+    main()
