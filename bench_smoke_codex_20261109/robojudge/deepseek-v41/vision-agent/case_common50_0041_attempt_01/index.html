+    mask[inner > 0] = cv2.GC_FGD

+    x, y, w, h = cv2.boundingRect(polygon)

+    x1, y1 = min(WIDTH, x + w + 4), min(HEIGHT, y + h + 4)

+def _transform_layer(layer, source_point, target_point, angle_deg):
+    """Rotate about the grip point, then translate it exactly to the target."""
