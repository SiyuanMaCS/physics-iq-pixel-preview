+def ease(value):
+    value = float(np.clip(value, 0.0, 1.0))
+    return value * value * (3.0 - 2.0 * value)

+def lerp(a, b, amount):
+    return float(a) + (float(b) - float(a)) * float(amount)

+def segment_red_block(image):
+    rgb = image.astype(np.int16)
+    r, g, b = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
+    mask = ((r > 34) & (r > g * 1.55 + 4) & (r > b * 1.30 + 4)).astype(np.uint8)
+
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(mask, 8)
+    if count <= 1:
+        raise RuntimeError("red block was not found")
+    index = 1 + int(np.argmax(stats[1:, cv2.CC_STAT_AREA]))
+    component = (labels == index).astype(np.uint8)
+    component = cv2.morphologyEx(component, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+
+    # Keep a narrow antialiased fringe without pulling in the neutral table.
+    vicinity = cv2.dilate(component, np.ones((5, 5), np.uint8))
+    redness = r.astype(np.float32) - 0.55 * g - 0.45 * b
+    soft = np.clip((redness - 8.0) / 38.0, 0.0, 1.0)
+    alpha = np.maximum(component.astype(np.float32), soft * vicinity).astype(np.float32)
+    alpha = cv2.GaussianBlur(alpha, (3, 3), 0.45)
+    alpha *= vicinity
+
+    ys, xs = np.nonzero(component)
+    x0, x1 = max(0, xs.min() - 3), min(WIDTH, xs.max() + 4)
+    y0, y1 = max(0, ys.min() - 3), min(HEIGHT, ys.max() + 4)
+    return component, alpha, (x0, y0, x1, y1)

+def make_clean_plate(image, block_mask):
+    mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    # The red block's right-facing side casts a soft shadow onto the table.
+    mask[5:100, 20:88] = block_mask[5:100, 20:88] * 255
+    mask[8:99, 66:88] = 255
+    mask = cv2.dilate(mask, np.ones((3, 3), np.uint8))
+
+    bgr = cv2.cvtColor(image, cv2.COLOR_RGB2BGR)
+    filled = cv2.inpaint(bgr, mask, 9, cv2.INPAINT_TELEA)
+    clean = cv2.cvtColor(filled, cv2.COLOR_BGR2RGB).astype(np.float32)
+
+    # Return a little of the tabletop's fine sensor texture to the repaired area.
+    rng = np.random.default_rng(17)
+    noise = rng.normal(0.0, 1.8, (HEIGHT, WIDTH, 1))
+    clean += noise
+
+    # Inpainting a tall object can flatten the broad illumination gradient.
+    # Blend a row-wise plate estimate back in, away from the block boundary.
+    repair = mask > 0
+    for y in range(5, 100):
+        left = image[y, 12:22].mean(axis=0)
+        right = image[y, 91:105].mean(axis=0)
+        if np.any(repair[y, 22:91]):
+            xs = np.arange(22, 91, dtype=np.float32)
+            t = ((xs - 18.0) / (94.0 - 18.0))[:, None]
+            plate = (left[None, :] * (1.0 - t) + right[None, :] * t).astype(np.float32)
+            local = repair[y, 22:91]
+            clean[y, 22:91][local] = (
+                clean[y, 22:91][local] * 0.38 + plate[local] * 0.62
+            )
+    return np.clip(clean, 0, 255).astype(np.uint8)

+def make_block_sprite(image, alpha, bbox):
+    x0, y0, x1, y1 = bbox
+    rgb = image[y0:y1, x0:x1].copy()
+    rgba = np.dstack([rgb, np.round(alpha[y0:y1, x0:x1] * 255.0).astype(np.uint8)])
+    return rgba

+def place_sprite(canvas, sprite, center, scale=1.0, shadow_strength=0.36, shadow_offset=(5, 7)):
+    height, width = sprite.shape[:2]
+    src = np.array(
+        [[0, 0], [width - 1, 0], [0, height - 1], [width - 1, height - 1]],
+        dtype=np.float32,
+    )
+    half_w = width * scale * 0.5
+    half_h = height * scale * 0.5
+    dst = np.array(
+        [
+            [center[0] - half_w, center[1] - half_h],
+            [center[0] + half_w, center[1] - half_h],
+            [center[0] - half_w, center[1] + half_h],
+            [center[0] + half_w, center[1] + half_h],
+        ],
+        dtype=np.float32,
+    )
+    transform = cv2.getPerspectiveTransform(src, dst)
+    warped = cv2.warpPerspective(
+        sprite,
+        transform,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+    rgba = warped.astype(np.float32)
+    a = rgba[:, :, 3:4] / 255.0
+
+    if shadow_strength > 0:
+        shadow_mask = np.zeros((HEIGHT, WIDTH), np.float32)
+        shift = np.float32([[1, 0, shadow_offset[0]], [0, 1, shadow_offset[1]]])
+        shifted_alpha = cv2.warpAffine(
+            warped[:, :, 3], shift, (WIDTH, HEIGHT), flags=cv2.INTER_LINEAR
+        ).astype(np.float32) / 255.0
+        shadow_mask = cv2.GaussianBlur(shifted_alpha, (0, 0), 2.2)
+        shadow_mask = np.clip(shadow_mask * shadow_strength, 0.0, 0.55)
+        shadow_color = np.array([18.0, 22.0, 25.0], dtype=np.float32)
+        out = canvas.astype(np.float32)
+        out = out * (1.0 - shadow_mask[:, :, None]) + shadow_color * shadow_mask[:, :, None]
+        canvas = np.clip(out, 0, 255).astype(np.uint8)
+
+    out = canvas.astype(np.float32)
+    out = out * (1.0 - a) + rgba[:, :, :3] * a
+    return np.clip(out, 0, 255).astype(np.uint8)

+def solve_elbow(anchor, wrist, upper=111.0, lower=103.0):
+    ax, ay = anchor
+    wx, wy = wrist
+    dx, dy = wx - ax, wy - ay
+    distance = math.hypot(dx, dy)
+    distance = float(np.clip(distance, abs(upper - lower) + 1.0, upper + lower - 0.4))
+    theta = math.atan2(dy, dx)
+    along = (upper * upper - lower * lower + distance * distance) / (2.0 * distance)
+    height = math.sqrt(max(0.0, upper * upper - along * along))
+    # Positive branch bends the elbow below the straight anchor-to-wrist line.
+    ex = ax + along * math.cos(theta) - height * math.sin(theta)
+    ey = ay + along * math.sin(theta) + height * math.cos(theta)
+    return (ex, ey)

+def draw_polyline(draw, points, width, fill):
+    points = [(round(x * SS), round(y * SS)) for x, y in points]
+    draw.line(points, fill=fill, width=round(width * SS), joint="curve")
+    radius = width * SS * 0.5
+    for x, y in points:
+        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)

+        draw.ellipse((x - r * 0.74, y - r * 0.74, x + r * 0.74, y + r * 0.74), fill=(116, 124, 126, 255))
+        draw.ellipse((x - r * 0.43, y - r * 0.43, x + r * 0.43, y + r * 0.43), fill=(50, 58, 61, 255))
+        draw.ellipse((x - r * 0.18, y - r * 0.18, x + r * 0.18, y + r * 0.18), fill=(184, 188, 187, 255))

+    # Compact parallel jaw gripper, aligned with the tabletop vertical.

+        if x0 > x1:

+def main():

+    carry_frames = range(6, 12)

+    frames = []

+        if index == 0:
+            frames.append(original.copy())
+            continue

+        if index <= 5:

+        if index in carry_frames:

+                    center,
+                    scale=scale,
+                    shadow_strength=shadow,
+                    shadow_offset=(5 + round(lift * 0.35), 7 + round(lift * 0.25)),
+                )

+                sprite_center[0] + (target_wrist[0] - source_wrist[0]),
+                sprite_center[1],
+            )
+            frame = Image.fromarray(
+                place_sprite(
+                    np.asarray(frame.convert("RGB")),
+                    sprite,
+                    center,
+                    scale=1.0,
+                    shadow_strength=0.33,
+                    shadow_offset=(5, 7),
+                )
+            ).convert("RGBA")
+
+        if index in wrist_path:
+            frame = draw_arm(
+                frame,
+                anchor,
+                wrist_path[index],
+                grip_open=(index in open_grip),
+            )
+
+        frames.append(np.asarray(frame.convert("RGB"), dtype=np.uint8))
+
+    frames = np.stack(frames, axis=0)
+    assert frames.shape == (N_FRAMES, HEIGHT, WIDTH, 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[0], original)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()
