+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 192
+FRAMES = 20
+FPS = 4
+
+
+def smoothstep(value):
+    value = float(np.clip(value, 0.0, 1.0))
+    return value * value * (3.0 - 2.0 * value)
+
+
+def lerp(a, b, amount):
+    return tuple(a[i] + (b[i] - a[i]) * amount for i in range(2))
+
+
+def cup_mask():
+    polygon = np.array(
+        [
+            [95, 78],
+            [114, 75],
+            [125, 78],
+            [135, 84],
+            [142, 92],
+            [145, 102],
+            [142, 111],
+            [138, 118],
+            [134, 128],
+            [126, 136],
+            [104, 136],
+            [92, 127],
+            [87, 114],
+            [87, 94],
+        ],
+        dtype=np.int32,
+    )
+    mask = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    cv2.fillPoly(mask, [polygon], 255)
+    return mask
+
+
+def build_scene():
+    image = cv2.cvtColor(cv2.imread(str(CASE_DIR / "first_frame.png")), cv2.COLOR_BGR2RGB)
+    mask = cup_mask()
+
+    removal = cv2.dilate(mask, np.ones((5, 5), np.uint8), iterations=1)
+    background = cv2.inpaint(
+        cv2.cvtColor(image, cv2.COLOR_RGB2BGR),
+        removal,
+        4,
+        cv2.INPAINT_TELEA,
+    )
+    background = cv2.cvtColor(background, cv2.COLOR_BGR2RGB)
+
+    alpha = cv2.GaussianBlur(mask.astype(np.float32) / 255.0, (3, 3), 0.55)
+    cup = image.copy()
+    return image, background, cup, alpha, mask
+
+
+def draw_soft_shadow(frame, center, size, opacity):
+    if opacity <= 0:
+        return
+    pad = 18
+    x, y = int(round(center[0])), int(round(center[1]))
+    half_w, half_h = int(round(size[0] / 2)), int(round(size[1] / 2))
+    x0, x1 = max(0, x - half_w - pad), min(WIDTH, x + half_w + pad)
+    y0, y1 = max(0, y - half_h - pad), min(HEIGHT, y + half_h + pad)
+    if x1 <= x0 or y1 <= y0:
+        return
+    local = np.zeros((y1 - y0, x1 - x0), dtype=np.float32)
+    cv2.ellipse(
+        local,
+        (x - x0, y - y0),
+        (half_w, half_h),
+        0,
+        0,
+        360,
+        1.0,
+        -1,
+        lineType=cv2.LINE_AA,
+    )
+    local = cv2.GaussianBlur(local, (0, 0), 3.2)
+    region = frame[y0:y1, x0:x1].astype(np.float32)
+    shade = np.array([18, 24, 28], dtype=np.float32)
+    region = region * (1.0 - local[:, :, None] * opacity) + shade * (local[:, :, None] * opacity)
+    frame[y0:y1, x0:x1] = np.clip(region, 0, 255).astype(np.uint8)
+
+
+def draw_gripper(frame, wrist, openness=0.25, alpha=1.0):
+    scale = 4
+    canvas = Image.new("RGBA", (WIDTH * scale, HEIGHT * scale), (0, 0, 0, 0))
+    draw = ImageDraw.Draw(canvas)
+    anchor = (51.0, 132.0)
+
+    # A short forearm joins the animated jaws to the visible left gripper.
+    draw.line(
+        (
+            int(anchor[0] * scale),
+            int(anchor[1] * scale),
+            int(wrist[0] * scale),
+            int(wrist[1] * scale),
+        ),
+        fill=(29, 31, 30, int(235 * alpha)),
+        width=7 * scale,
+    )
+    draw.line(
+        (
+            int((anchor[0] + 1) * scale),
+            int((anchor[1] - 2) * scale),
+            int((wrist[0] + 1) * scale),
+            int((wrist[1] - 1) * scale),
+        ),
+        fill=(70, 72, 69, int(180 * alpha)),
+        width=2 * scale,
+    )
+
+    spread = 4.0 + 5.0 * float(np.clip(openness, 0.0, 1.0))
+    wrist_x, wrist_y = wrist
+    head = [
+        (wrist_x - 7.0, wrist_y - 4.0),
+        (wrist_x + 7.0, wrist_y - 4.0),
+        (wrist_x + 6.0, wrist_y + 5.0),
+        (wrist_x - 6.0, wrist_y + 5.0),
+    ]
+    draw.polygon(
+        [(int(x * scale), int(y * scale)) for x, y in head],
+        fill=(31, 33, 32, int(245 * alpha)),
+        outline=(105, 107, 102, int(190 * alpha)),
+        width=1 * scale,
+    )
+    draw.ellipse(
+        (
+            int((wrist_x - 3.5) * scale),
+            int((wrist_y - 1.0) * scale),
+            int((wrist_x + 3.5) * scale),
+            int((wrist_y + 5.0) * scale),
+        ),
+        fill=(77, 79, 75, int(235 * alpha)),
+    )
+
+    for side in (-1.0, 1.0):
+        inner = wrist_x + side * (spread - 2.0)
+        outer = wrist_x + side * (spread + 1.5)
+        tip_x = wrist_x + side * (spread - 1.2)
+        jaw = [
+            (inner, wrist_y + 3.0),
+            (outer, wrist_y + 3.0),
+            (tip_x + side * 1.5, wrist_y + 17.0),
+            (tip_x - side * 1.0, wrist_y + 18.0),
+        ]
+        draw.polygon(
+            [(int(x * scale), int(y * scale)) for x, y in jaw],
+            fill=(24, 26, 25, int(250 * alpha)),
+            outline=(93, 95, 90, int(190 * alpha)),
+            width=1 * scale,
+        )
+        draw.line(
+            (
+                int((outer - side * 0.5) * scale),
+                int((wrist_y + 5.0) * scale),
+                int((tip_x + side * 0.2) * scale),
+                int((wrist_y + 15.0) * scale),
+            ),
+            fill=(111, 113, 107, int(145 * alpha)),
+            width=1 * scale,
+        )
+
+    overlay = np.asarray(canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS))
+    rgb = overlay[:, :, :3].astype(np.float32)
+    opacity = overlay[:, :, 3:4].astype(np.float32) / 255.0
+    frame[:] = np.clip(frame.astype(np.float32) * (1.0 - opacity) + rgb * opacity, 0, 255).astype(np.uint8)
+
+
+def cup_center_for_frame(index):
+    start = np.array([116.0, 105.0])
+    top = np.array([116.0, 67.0])
+    over_tray = np.array([39.0, 67.0])
+    placed = np.array([39.0, 146.0])
+
+    if index <= 4:
+        return start
+    if index <= 8:
+        amount = smoothstep((index - 4) / 4.0)
+        return start + (top - start) * amount
+    if index <= 15:
+        amount = smoothstep((index - 8) / 7.0)
+        center = top + (over_tray - top) * amount
+        center[1] -= 8.0 * np.sin(np.pi * amount)
+        return center
+    if index <= 18:
+        amount = smoothstep((index - 15) / 3.0)
+        return over_tray + (placed - over_tray) * amount
+    return placed
+
+
+def wrist_for_frame(index, center):
+    if index == 0:
+        return np.array([55.0, 133.0])
+    if index <= 4:
+        amount = smoothstep((index - 0.5) / 3.5)
+        start = np.array([55.0, 133.0])
+        target = center + np.array([0.0, -29.0])
+        return start + (target - start) * amount
+    if index <= 18:
+        return center + np.array([0.0, -29.0])
+    return np.array([43.0, 101.0])
+
+
+def openness_for_frame(index):
+    if index <= 1:
+        return 0.85
+    if index <= 3:
+        return 0.85 - 0.60 * smoothstep((index - 1) / 2.0)
+    if index <= 18:
+        return 0.24
+    return 0.92
+
+
+def paste_cup(frame, cup, alpha, center, source_center):
+    dx = int(round(center[0] - source_center[0]))
+    dy = int(round(center[1] - source_center[1]))
+    for y in range(HEIGHT):
+        source_y = y - dy
+        if source_y < 0 or source_y >= HEIGHT:
+            continue
+        for x in range(WIDTH):
+            source_x = x - dx
+            if source_x < 0 or source_x >= WIDTH:
+                continue
+            a = alpha[source_y, source_x]
+            if a <= 0.002:
+                continue
+            frame[y, x] = np.clip(
+                frame[y, x].astype(np.float32) * (1.0 - a) + cup[source_y, source_x].astype(np.float32) * a,
+                0,
+                255,
+            ).astype(np.uint8)
+
+
+def generate():
+    original, background, cup, alpha, mask = build_scene()
+    ys, xs = np.where(mask > 0)
+    source_center = np.array([(xs.min() + xs.max()) / 2.0, (ys.min() + ys.max()) / 2.0])
+    frames = []
+
+    for index in range(FRAMES):
+        if index == 0:
+            frames.append(original.copy())
+            continue
+
+        frame = background.copy()
+        center = cup_center_for_frame(index)
+        wrist = wrist_for_frame(index, center)
+
+        shadow_center = center + np.array([5.0, 28.0])
+        shadow_scale = 1.0 + 0.16 * smoothstep((center[1] - 67.0) / 79.0)
+        shadow_opacity = 0.34 if index < 19 else 0.26
+        draw_soft_shadow(frame, shadow_center, (34 * shadow_scale, 8 * shadow_scale), shadow_opacity)
+
+        paste_cup(frame, cup, alpha, center, source_center)
+        draw_gripper(frame, wrist, openness_for_frame(index))
+        frames.append(frame)
+
+    return np.stack(frames, axis=0)
+
+
+if __name__ == "__main__":
+    artifact_io.save(generate(), CASE_DIR)
