+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image
+
+import artifact_io
+
+
+ROOT = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 160
+FRAME_COUNT = 20
+
+
+def smoothstep(edge0, edge1, value):
+    x = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)
+
+
+def polygon_mask(points):
+    mask = np.zeros((HEIGHT, WIDTH), np.uint8)
+    cv2.fillPoly(mask, [np.asarray(points, np.int32)], 255)
+    return mask
+
+
+def blur_alpha(mask, sigma=0.55):
+    return cv2.GaussianBlur(mask.astype(np.float32) / 255.0, (0, 0), sigma)
+
+
+def make_masks(image):
+    rgb_max = image.max(axis=2).astype(np.int16)
+    rgb_min = image.min(axis=2).astype(np.int16)
+    mean = image.mean(axis=2)
+
+    hand_polygon = polygon_mask(
+        [
+            (184, 94),
+            (194, 96),
+            (201, 103),
+            (199, 116),
+            (197, 126),
+            (200, 142),
+            (199, 160),
+            (168, 160),
+            (174, 143),
+            (179, 132),
+            (180, 120),
+            (182, 111),
+        ]
+    )
+    neutral = ((rgb_max - rgb_min) < 32) & (mean < 178)
+    very_dark = mean < 58
+    hand_mask = hand_polygon & ((neutral | very_dark).astype(np.uint8) * 255)
+    hand_mask = cv2.morphologyEx(
+        hand_mask, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8), iterations=1
+    )
+    hand_mask = cv2.dilate(hand_mask, np.ones((2, 2), np.uint8), iterations=1)
+
+    mug_body = polygon_mask(
+        [(116, 108), (137, 108), (141, 124), (137, 133), (121, 134), (115, 128)]
+    )
+    handle_polygon = polygon_mask(
+        [(136, 111), (146, 111), (147, 118), (144, 123), (138, 125)]
+    )
+    hsv = cv2.cvtColor(image, cv2.COLOR_RGB2HSV)
+    gray_handle = (hsv[:, :, 1] < 75) & (hsv[:, :, 2] > 65)
+    handle = (handle_polygon > 0) & gray_handle
+    handle = cv2.morphologyEx(
+        handle.astype(np.uint8) * 255,
+        cv2.MORPH_CLOSE,
+        np.ones((2, 2), np.uint8),
+        iterations=1,
+    )
+    mug_mask = cv2.bitwise_or(mug_body, handle)
+
+    shadow_polygon = polygon_mask([(132, 126), (150, 126), (148, 138), (130, 137)])
+    removal_mask = cv2.bitwise_or(hand_mask, mug_mask)
+    removal_mask = cv2.bitwise_or(removal_mask, shadow_polygon)
+    removal_mask = cv2.dilate(removal_mask, np.ones((3, 3), np.uint8), iterations=1)
+    return hand_mask, mug_mask, removal_mask
+
+
+def reconstruct_background(image, removal_mask):
+    bgr = cv2.cvtColor(image, cv2.COLOR_RGB2BGR)
+    filled = cv2.inpaint(bgr, removal_mask, 5, cv2.INPAINT_TELEA)
+    result = cv2.cvtColor(filled, cv2.COLOR_BGR2RGB)
+
+    # The inpainted table can flatten the horizontal grain. Add a restrained,
+    # locally correlated texture only where the source objects were removed.
+    rng = np.random.default_rng(31)
+    noise = rng.normal(0.0, 2.0, (HEIGHT, WIDTH, 1))
+    noise = cv2.GaussianBlur(noise.astype(np.float32), (0, 0), 0.7)
+    noise = cv2.GaussianBlur(noise, (0, 0), 3.0)[:, :, None]
+    soft_mask = blur_alpha(removal_mask, 2.2)[:, :, None]
+    result = np.clip(result.astype(np.float32) + noise * soft_mask, 0, 255).astype(np.uint8)
+    return result
+
+
+def warp_hand(image, alpha, reach, lift):
+    yy = np.arange(HEIGHT, dtype=np.float32)[:, None]
+    weight = np.clip((160.0 - yy) / 58.0, 0.0, 1.0) ** 0.78
+    dx = -56.0 * reach * weight + 3.0 * lift * weight
+    dy = 8.0 * reach * weight - 24.0 * lift * weight
+
+    xx = np.arange(WIDTH, dtype=np.float32)[None, :]
+    map_x = np.broadcast_to(xx - dx, (HEIGHT, WIDTH)).astype(np.float32)
+    map_y = np.broadcast_to(yy - dy, (HEIGHT, WIDTH)).astype(np.float32)
+    warped_rgb = cv2.remap(
+        image,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    )
+    warped_alpha = cv2.remap(
+        alpha,
+        map_x,
+        map_y,
+        interpolation=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    return warped_rgb, np.clip(warped_alpha, 0.0, 1.0)
+
+
+def transform_mug(image, alpha, x_shift, y_shift, angle):
+    center = (128.0, 120.0)
+    matrix = cv2.getRotationMatrix2D(center, angle, 1.0)
+    matrix[0, 2] += x_shift
+    matrix[1, 2] += y_shift
+    rgb = cv2.warpAffine(
+        image,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    )
+    a = cv2.warpAffine(
+        alpha,
+        matrix,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    return rgb, np.clip(a, 0.0, 1.0)
+
+
+def composite(base, rgb, alpha):
+    a = alpha[:, :, None]
+    return np.clip(rgb.astype(np.float32) * a + base.astype(np.float32) * (1.0 - a), 0, 255)
+
+
+def add_contact_shadow(frame, lift):
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    cx = 136.0
+    cy = 132.0
+    sx = 9.0 + 1.5 * lift
+    sy = 3.4 + 0.4 * lift
+    falloff = np.exp(-(((xx - cx) / sx) ** 2 + ((yy - cy) / sy) ** 2) * 1.7)
+    strength = 0.22 * (1.0 - 0.55 * lift)
+    shadow = 1.0 - strength * falloff[:, :, None]
+    return np.clip(frame.astype(np.float32) * shadow, 0, 255)
+
+
+def generate():
+    source = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    assert source.shape == (HEIGHT, WIDTH, 3)
+
+    hand_mask, mug_mask, removal_mask = make_masks(source)
+    background = reconstruct_background(source, removal_mask)
+    hand_alpha = blur_alpha(hand_mask, 0.55)
+    mug_alpha = blur_alpha(mug_mask, 0.45)
+
+    frames = [source.copy()]
+    for index in range(1, FRAME_COUNT):
+        t = index / (FRAME_COUNT - 1)
+        reach = smoothstep(0.03, 0.48, t)
+        lift = smoothstep(0.49, 0.91, t)
+
+        frame = add_contact_shadow(background.copy(), lift)
+
+        mug_x = 3.0 * lift
+        mug_y = -24.0 * lift
+        mug_angle = -3.0 * lift
+        mug_rgb, mug_a = transform_mug(source, mug_alpha, mug_x, mug_y, mug_angle)
+        frame = composite(frame, mug_rgb, mug_a)
+
+        hand_rgb, hand_a = warp_hand(source, hand_alpha, reach, lift)
+        frame = composite(frame, hand_rgb, hand_a)
+
+        frames.append(np.clip(frame, 0, 255).astype(np.uint8))
+
+    frames = np.asarray(frames, dtype=np.uint8)
+    assert np.array_equal(frames[0], source)
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    generate()
