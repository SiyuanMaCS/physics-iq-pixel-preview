+
+The scene is animated with source-pixel sprites: the baguette and the visible
+right hand are lifted from the basket, carried to the plate, and released.
+Only local image operations are used.

+import math

+import cv2
+import numpy as np
+from PIL import Image

+from artifact_io import save

+ROOT = Path(__file__).resolve().parent

+def _poly_mask(points):
+    mask = np.zeros((H, W), np.uint8)
+    cv2.fillPoly(mask, [np.asarray(points, np.int32)], 255)
+    return mask
+
+
+def _feather(mask, sigma=0.8):
+    return cv2.GaussianBlur(mask, (0, 0), sigma).astype(np.float32) / 255.0
+
+
+def _warp_rgba(image, alpha, matrix, size=(W, H)):
+    rgba = np.dstack((image, alpha))
+    warped = cv2.warpAffine(
+        rgba,
+        matrix,
+        size,
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0, 0),
+    )
+    return warped[:, :, :3], warped[:, :, 3].astype(np.float32) / 255.0
+
+
+def _paste(dst, rgb, alpha, opacity=1.0):
+    a = np.clip(alpha * opacity, 0.0, 1.0)[:, :, None]
+    return (dst.astype(np.float32) * (1.0 - a) + rgb.astype(np.float32) * a).astype(np.uint8)
+
+
+def _affine(angle_deg, scale, center, target):
+    """Rotate/scale around a source grip point, then move that point to target."""
+    a = math.radians(angle_deg)
+    c, s = math.cos(a) * scale, math.sin(a) * scale
+    cx, cy = center
+    tx, ty = target
+    return np.array(
+        [
+            [c, -s, tx - c * cx + s * cy],
+            [s, c, ty - s * cx - c * cy],
+        ],
+        np.float32,
+    )
+
+
+def _smoothstep(x):
+    x = float(np.clip(x, 0.0, 1.0))
+    return x * x * (3.0 - 2.0 * x)
+
+
+def _lerp(a, b, t):
+    return float(a) + (float(b) - float(a)) * t
+
+
+def _synthesize_cavity(source):
+    """Build a quiet, woven basket interior to replace the occupied bread area."""
+    h, w = source.shape[:2]
+    yy, xx = np.indices((h, w))
+    base = np.zeros_like(source)
+
+    # Warm dark interior with a vertical shadow gradient.
+    top = np.array([63, 42, 23], np.float32)
+    bottom = np.array([88, 61, 31], np.float32)
+    t = np.clip((yy - 78) / 40.0, 0.0, 1.0)[:, :, None]
+    base = (top * (1.0 - t) + bottom * t).astype(np.float32)
+
+    # Low-frequency color variation keeps the restored cavity from looking flat.
+    rng = np.random.default_rng(18)
+    coarse = rng.normal(0.0, 1.0, (18, 30)).astype(np.float32)
+    coarse = cv2.resize(coarse, (w, h), interpolation=cv2.INTER_CUBIC)
+    coarse = cv2.GaussianBlur(coarse, (0, 0), 3.0)
+    base += coarse[:, :, None] * np.array([5.0, 4.0, 2.5], np.float32)
+
+    # Faint horizontal weave and vertical basket ribs.
+    weave = 3.2 * np.sin((yy + 2.0 * np.sin(xx / 9.0)) * 1.35)
+    ribs = 1.8 * np.sin(xx * 0.78 + 0.45 * np.sin(yy / 4.0))
+    base += (weave + ribs)[:, :, None]
+
+    # Basket shadow near the rim and the back wall.
+    shadow = np.clip((94.0 - yy) / 24.0, 0.0, 1.0) * 0.30
+    base *= (1.0 - shadow[:, :, None])
+    return np.clip(base, 0, 255).astype(np.uint8)
+

+def _make_clean_background(source):
+    """Remove the bread and the visible arm while preserving plate/table cues."""
+    bread = _poly_mask(
+        [
+            [101, 93], [102, 87], [107, 83], [113, 81], [120, 81],
+            [128, 83], [136, 85], [143, 86], [149, 86], [155, 84],
+            [162, 81], [169, 78], [176, 76], [182, 77], [187, 80],
+            [190, 84], [190, 89], [186, 93], [180, 97], [173, 101],
+            [166, 105], [159, 108], [152, 111], [145, 112], [138, 110],
+            [131, 107], [124, 103], [117, 99], [110, 96], [105, 94],
+        ]
+    )
+    hand = _poly_mask(
+        [
+            [158, 77], [181, 74], [190, 78], [198, 87], [202, 98],
+            [207, 106], [214, 109], [222, 109], [231, 112], [238, 118],
+            [239, 144], [212, 144], [207, 127], [201, 119], [193, 115],
+            [183, 112], [173, 111], [166, 106], [161, 99], [158, 91],
+        ]
+    )

+    occupied = cv2.bitwise_or(bread, hand)
+    occupied = cv2.dilate(occupied, np.ones((3, 3), np.uint8), iterations=1)
+    clean = cv2.inpaint(source, occupied, 6, cv2.INPAINT_TELEA)
+
+    # Use a constructed woven interior rather than trusting inpaint inside the bowl.
+    cavity = _synthesize_cavity(source)
+    cavity_mask = _poly_mask(
+        [
+            [98, 95], [103, 86], [111, 81], [121, 79], [132, 82],
+            [143, 86], [153, 85], [164, 81], [175, 76], [184, 77],
+            [191, 83], [192, 91], [187, 98], [178, 104], [168, 109],
+            [157, 113], [145, 114], [134, 111], [123, 105], [113, 100],
+            [104, 98],
+        ]
+    )
+    cavity_mask = cv2.GaussianBlur(cavity_mask, (0, 0), 1.0)
+    clean = _paste(clean, cavity, cavity_mask.astype(np.float32) / 255.0, 0.90)
+
+    # Rebuild the cloth behind the forearm from nearby table texture.
+    cloth = source[116:144, 44:139].copy()
+    cloth = cv2.resize(cloth, (96, 33), interpolation=cv2.INTER_LINEAR)
+    target = clean[111:144, 180:256]
+    donor = cloth[:, : target.shape[1]]
+    donor_mask = np.zeros((target.shape[0], target.shape[1]), np.float32)
+    donor_mask[4:, :] = 1.0
+    donor_mask = cv2.GaussianBlur(donor_mask, (0, 0), 2.2)[:, :, None]
+    target[:] = (
+        target.astype(np.float32) * (1.0 - 0.62 * donor_mask)
+        + donor.astype(np.float32) * (0.62 * donor_mask)
+    ).astype(np.uint8)

+    # Soft shadow under the future plate position.
+    overlay = clean.astype(np.float32)
+    yy, xx = np.indices((H, W))
+    shadow = np.exp(-(((xx - 219.0) / 42.0) ** 2 + ((yy - 132.0) / 11.0) ** 2))
+    overlay *= (1.0 - 0.10 * shadow[:, :, None])
+    clean = np.clip(overlay, 0, 255).astype(np.uint8)
+    return clean, bread, hand

+def _make_bread_sprite(source, bread_mask):
+    """Extract the baguette and replace the hand-covered right end with bread texture."""
+    bgr = source.copy()
+
+    # The fingertips obscure the right end. Borrow nearby crumb/crust texture so
+    # the released baguette does not carry a hand-shaped patch onto the plate.
+    donor = source[82:101, 122:154].copy()
+    donor = cv2.resize(donor, (34, 23), interpolation=cv2.INTER_CUBIC)
+    patch = np.zeros((H, W), np.uint8)
+    patch[79:102, 156:190] = 255
+    patch = cv2.bitwise_and(patch, bread_mask)
+    a = cv2.GaussianBlur(patch, (0, 0), 1.1).astype(np.float32) / 255.0
+    fill = np.zeros_like(source)
+    fill[79:102, 156:190] = donor
+    bgr = _paste(bgr, fill, a)
+
+    # A slightly feathered polygon reads better after rotation than a hard cut.
+    alpha = cv2.GaussianBlur(bread_mask, (0, 0), 0.65).astype(np.float32)
+    return bgr, alpha / 255.0
+
+
+def _make_hand_sprite(source, hand_mask):
+    """Extract the arm/fingers, retaining their actual silhouette and shading."""
+    alpha = cv2.GaussianBlur(hand_mask, (0, 0), 0.55).astype(np.float32) / 255.0
+    return source.copy(), alpha
+

+def _bread_state(t):
+    """Return (angle, scale, center) for bread through the 20-frame sequence."""
+    if t <= 2:
+        p = _smoothstep(t / 2.0)
+        return _lerp(0, -1.0, p), _lerp(1.0, 1.01, p), (_lerp(144, 145, p), _lerp(93, 92, p))
+    if t <= 8:
+        p = _smoothstep((t - 2.0) / 6.0)
+        return _lerp(-1.0, -8.0, p), _lerp(1.01, 1.00, p), (_lerp(145, 164, p), _lerp(92, 84, p))
+    if t <= 13:
+        p = _smoothstep((t - 8.0) / 5.0)
+        return _lerp(-8.0, 8.0, p), 1.0, (_lerp(164, 205, p), _lerp(84, 109, p))
+    if t <= 16:
+        p = _smoothstep((t - 13.0) / 3.0)
+        return _lerp(8.0, 12.0, p), _lerp(1.0, 0.94, p), (_lerp(205, 219, p), _lerp(109, 128, p))
+    return 12.0, 0.94, (219.0, 128.0)
+
+
+def _hand_state(t):
+    """Grip target and angle for the arm; later frames release and withdraw."""
+    if t <= 2:
+        p = _smoothstep(t / 2.0)
+        return (_lerp(178, 174, p), _lerp(91, 94, p)), _lerp(0, 3, p)
+    if t <= 8:
+        p = _smoothstep((t - 2.0) / 6.0)
+        return (_lerp(174, 192, p), _lerp(94, 88, p)), _lerp(3, 11, p)
+    if t <= 13:
+        p = _smoothstep((t - 8.0) / 5.0)
+        return (_lerp(192, 230, p), _lerp(88, 116, p)), _lerp(11, 20, p)
+    if t <= 16:
+        p = _smoothstep((t - 13.0) / 3.0)
+        return (_lerp(230, 245, p), _lerp(116, 132, p)), _lerp(20, 24, p)
+    if t <= 19:
+        p = _smoothstep((t - 16.0) / 3.0)
+        return (_lerp(245, 285, p), _lerp(132, 149, p)), _lerp(24, 31, p)
+    return (285.0, 149.0), 31.0
+
+
+def generate():
+    source = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    source = source[:, :, ::-1].copy()  # RGB -> BGR for the compositing helpers.
+
+    clean, bread_mask, hand_mask = _make_clean_background(source)
+    bread_rgb, bread_alpha = _make_bread_sprite(source, bread_mask)
+    hand_rgb, hand_alpha = _make_hand_sprite(source, hand_mask)
+
+    frames_bgr = []
+    for i in range(N):
+        if i == 0:
+            frames_bgr.append(source.copy())
+            continue
+
+        frame = clean.copy()
+        angle, scale, center = _bread_state(i)
+        bread_m = _affine(angle, scale, (145.0, 93.0), center)
+        b_rgb, b_a = _warp_rgba(bread_rgb, (bread_alpha * 255).astype(np.uint8), bread_m)
+        frame = _paste(frame, b_rgb, b_a)
+
+        grip, hand_angle = _hand_state(i)
+        hand_m = _affine(hand_angle, 1.0, (176.0, 94.0), grip)
+        h_rgb, h_a = _warp_rgba(hand_rgb, (hand_alpha * 255).astype(np.uint8), hand_m)
+        frame = _paste(frame, h_rgb, h_a)
+
+        # A small contact shadow sells the bread's height during the lift.
+        if i >= 3:
+            yy, xx = np.indices((H, W))
+            lift = min(1.0, max(0.0, (i - 2) / 7.0))
+            cx, cy = center
+            sx, sy = 28.0, 6.5
+            sh = np.exp(-(((xx - cx) / sx) ** 2 + ((yy - (cy + 11)) / sy) ** 2))
+            frame = np.clip(frame.astype(np.float32) * (1.0 - 0.09 * lift * sh[:, :, None]), 0, 255).astype(np.uint8)
+
+        # Very slight sensor grain avoids the appearance of a static cutout scene.
+        rng = np.random.default_rng(100 + i)
+        grain = rng.normal(0.0, 0.75, (H, W, 1)).astype(np.float32)
+        frame = np.clip(frame.astype(np.float32) + grain, 0, 255).astype(np.uint8)
+        frames_bgr.append(frame)
+
+    frames_rgb = np.stack([cv2.cvtColor(f, cv2.COLOR_BGR2RGB) for f in frames_bgr])
+    # Guarantee the contract exactly, including no processing of the opening frame.
+    frames_rgb[0] = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    save(frames_rgb, ROOT)
+
+
+if __name__ == "__main__":
+    generate()
