+def polygon_mask(points, height=HEIGHT, width=WIDTH):
+    mask = np.zeros((height, width), np.uint8)

+def ease_in_out(value):
+    value = np.clip(value, 0.0, 1.0)
+    return float(value * value * (3.0 - 2.0 * value))

+def eased_segment(frame, start, end, a, b):

+def make_foreground_alpha(image, points, background, wrist_rect=None):
+    polygon = polygon_mask(points) > 0
+    hsv = cv2.cvtColor(image, cv2.COLOR_RGB2HSV)
+    saturation = hsv[:, :, 1].astype(np.float32)
+    value = hsv[:, :, 2].astype(np.float32)
+
+    # Keep the neutral/dark robot surface and reject table-colored gaps
+    # enclosed by the hand silhouette.
+    color_distance = np.linalg.norm(
+        image.astype(np.float32) - background.astype(np.float32), axis=2
+    )
+    foreground = polygon & ((saturation < 108.0) | (value < 48.0) | (color_distance > 25.0))
