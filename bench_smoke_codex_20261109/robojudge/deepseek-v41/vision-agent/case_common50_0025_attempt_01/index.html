+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter

+import artifact_io

+ROOT = Path(__file__).resolve().parent
+FRAME_COUNT = 20
+WIDTH = HEIGHT = 256

+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def polygon_alpha(points, feather):
+    mask = Image.new("L", (WIDTH, HEIGHT), 0)
+    ImageDraw.Draw(mask).polygon(points, fill=255)
+    if feather:
+        mask = mask.filter(ImageFilter.GaussianBlur(feather))
+    return np.asarray(mask, dtype=np.float32) / 255.0
+
+
+def blur_float(array, radius):
+    image = Image.fromarray(np.clip(array * 255.0, 0, 255).astype(np.uint8))
+    image = image.filter(ImageFilter.GaussianBlur(radius))
+    return np.asarray(image, dtype=np.float32) / 255.0
+
+
+def bilinear(image, source_x, source_y):
+    x0 = np.floor(source_x).astype(np.int32)
+    y0 = np.floor(source_y).astype(np.int32)
+    x1 = x0 + 1
+    y1 = y0 + 1
+    wx = source_x - x0
+    wy = source_y - y0
+
+    x0 = np.clip(x0, 0, WIDTH - 1)
+    x1 = np.clip(x1, 0, WIDTH - 1)
+    y0 = np.clip(y0, 0, HEIGHT - 1)
+    y1 = np.clip(y1, 0, HEIGHT - 1)
+    wx = wx[..., None]
+    wy = wy[..., None]
+
+    top = image[y0, x0] * (1.0 - wx) + image[y0, x1] * wx
+    bottom = image[y1, x0] * (1.0 - wx) + image[y1, x1] * wx
+    return top * (1.0 - wy) + bottom * wy
+
+
+def inverse_rotation(x, y, center, angle):
+    cosine = np.cos(angle)
+    sine = np.sin(angle)
+    offset_x = x - center[0]
+    offset_y = y - center[1]
+    rotated_x = center[0] + cosine * offset_x + sine * offset_y
+    rotated_y = center[1] - sine * offset_x + cosine * offset_y
+    return rotated_x, rotated_y
+
+
+def make_hand_weight():
+    # The silhouette follows the exposed right hand and lower forearm. The
+    # feather hides the boundary while the lower forearm stays nearly fixed.
+    points = [
+        (100, 106), (99, 96), (100, 84), (102, 73), (106, 63),
+        (111, 57), (116, 53), (122, 52), (128, 56), (132, 63),
+        (134, 72), (136, 81), (135, 87), (131, 90), (126, 89),

+        (91, 235), (96, 213), (101, 191), (104, 169), (105, 149),
+        (102, 130), (97, 118), (94, 110),
+    ]
+    return polygon_alpha(points, 1.35)
+

+def make_tap_weight(first_frame):
+    y_grid, x_grid = np.mgrid[0:HEIGHT, 0:WIDTH]
+    region = (
+        (x_grid >= 113) & (x_grid <= 141) &
+        (y_grid >= 61) & (y_grid <= 96)
+    )
+    rgb = first_frame.astype(np.float32)
+    maximum = rgb.max(axis=2)
+    minimum = rgb.min(axis=2)
+    saturation = (maximum - minimum) / np.maximum(maximum, 1.0)
+    metal = region & (maximum > 72.0) & (saturation < 0.34)
+    weight = blur_float(metal.astype(np.float32), 1.15)
+
+    # Bias the metal mask toward the lever, whose pivot is beside the neck.
+    lever_gate = np.exp(
+        -(((x_grid - 126.0) / 12.0) ** 2 + ((y_grid - 79.0) / 10.0) ** 2)
+    )
+    return np.clip(weight * (0.42 + 0.58 * lever_gate), 0.0, 1.0)
+
+
+def add_water(frame, hand_weight, amount, frame_index):
+    if amount <= 0.0:
+        return frame
+
+    y_grid, x_grid = np.mgrid[0:HEIGHT, 0:WIDTH]
+    output = frame.astype(np.float32)
+    start_y = 91.0
+    end_y = 148.0
+    center_x = 123.4 + 0.75 * np.sin(frame_index * 0.62)
+    width = 1.25 + 0.20 * np.sin(y_grid * 0.55 + frame_index * 0.45)
+    cross = np.exp(-((x_grid - center_x) / width) ** 2)
+    vertical = smoothstep((y_grid - start_y) / 5.0) * (
+        1.0 - smoothstep((y_grid - (end_y - 4.0)) / 4.0)
+    )
+    clear_of_hand = 1.0 - 0.93 * hand_weight
+    water_alpha = amount * 0.23 * cross * vertical * clear_of_hand
+
+    # Moving highlights make the narrow stream legible at 256 px.
+    streak_phase = (y_grid - start_y + frame_index * 15.0) % 24.0
+    streak = np.exp(-((streak_phase - 6.0) / 3.2) ** 2)
+    highlight_alpha = amount * 0.16 * cross * vertical * streak * clear_of_hand
+
+    water_color = np.array([176.0, 211.0, 214.0], dtype=np.float32)
+    highlight_color = np.array([224.0, 239.0, 238.0], dtype=np.float32)
+    output = output * (1.0 - water_alpha[..., None]) + water_color * water_alpha[..., None]
+    output = (
+        output * (1.0 - highlight_alpha[..., None])
+        + highlight_color * highlight_alpha[..., None]
+    )
+
+    # A faint ripple at the sink surface appears with the flow.
+    ripple = np.exp(
+        -(((x_grid - center_x) / (4.5 + frame_index * 0.07)) ** 2 +
+          ((y_grid - 149.0) / 1.7) ** 2)
+    )
+    ripple_alpha = amount * 0.09 * ripple
+    output = output * (1.0 - ripple_alpha[..., None]) + water_color * ripple_alpha[..., None]
+    return np.clip(output, 0.0, 255.0)
+
+
+def generate_frames():
+    first = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"))
+    base = first.astype(np.float32)
+    hand_weight = make_hand_weight()
+    tap_weight = make_tap_weight(first)
+    y_grid, x_grid = np.mgrid[0:HEIGHT, 0:WIDTH]
+    x_grid = x_grid.astype(np.float32)
+    y_grid = y_grid.astype(np.float32)
+
+    frames = []
+    for index in range(FRAME_COUNT):
+        if index == 0:
+            frames.append(first.copy())
+            continue
+
+        # Hold the grip briefly, execute the turn, then settle with the water on.
+        turn = smoothstep((index - 3.0) / 11.5)
+        grip = smoothstep((index - 2.0) / 5.5)
+        settle = 0.12 * smoothstep((index - 14.0) / 4.0)
+
+        hand_angle = np.deg2rad(8.2 * turn + 0.8 * grip - settle)
+        hand_sx, hand_sy = inverse_rotation(
+            x_grid, y_grid, (103.5, 112.0), hand_angle
+        )
+        top_gate = 1.0 - smoothstep((y_grid - 92.0) / 43.0)
+        curl_x = (1.8 * grip + 0.7 * turn) * top_gate
+        curl_y = 0.9 * grip * top_gate
+        source_x = x_grid + hand_weight * (hand_sx - x_grid + curl_x)
+        source_y = y_grid + hand_weight * (hand_sy - y_grid + curl_y)
+
+        tap_angle = np.deg2rad(27.0 * turn)
+        tap_sx, tap_sy = inverse_rotation(
+            x_grid, y_grid, (124.5, 81.0), tap_angle
+        )
+        source_x = source_x + tap_weight * (tap_sx - source_x)
+        source_y = source_y + tap_weight * (tap_sy - source_y)
+
+        frame = bilinear(base, source_x, source_y)
+
+        # The turned control catches a brief highlight as it moves.
+        glint = tap_weight * turn * np.exp(
+            -(((x_grid - 130.0) / 6.5) ** 2 + ((y_grid - 78.0) / 4.5) ** 2)
+        )
+        frame += glint[..., None] * np.array([20.0, 19.0, 16.0], dtype=np.float32)
+
+        water_amount = smoothstep((index - 10.0) / 6.0)
+        frame = add_water(frame, hand_weight, water_amount, index)
+        frames.append(np.clip(np.rint(frame), 0, 255).astype(np.uint8))
+
+    return np.stack(frames, axis=0)
+
+
+def main():
+    frames = generate_frames()
+    assert frames.shape == (FRAME_COUNT, HEIGHT, WIDTH, 3)
+    assert frames.dtype == np.uint8
+    assert np.array_equal(frames[0], np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB")))
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()
