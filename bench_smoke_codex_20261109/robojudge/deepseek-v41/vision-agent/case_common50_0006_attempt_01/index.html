+def inverse_kinematics(
+    base: tuple[float, float],
+    target: tuple[float, float],
+    l1: float,
+    l2: float,
+    bend: float = -1.0,
+) -> tuple[float, float]:
+    bx, by = base
+    tx, ty = target
+    dx, dy = tx - bx, ty - by
+    distance = max(1e-3, float(np.hypot(dx, dy)))
+    distance = min(distance, l1 + l2 - 0.05)
+    angle = float(np.arctan2(dy, dx))
+    cosine = (l1 * l1 + distance * distance - l2 * l2) / (2.0 * l1 * distance)
+    offset = float(np.arccos(np.clip(cosine, -1.0, 1.0)))
+    elbow_angle = angle + bend * offset
+    return bx + l1 * np.cos(elbow_angle), by + l1 * np.sin(elbow_angle)
