+from pathlib import Path

+import cv2
+import numpy as np
+from PIL import Image

+import artifact_io

+CASE_DIR = Path(__file__).resolve().parent
+FRAME_COUNT = 20
+HEIGHT = 144
+WIDTH = 256

+def _polygon(points):
+    mask = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    cv2.fillPoly(mask, [np.asarray(points, dtype=np.int32)], 255)
+    return mask

+def _object_alpha(frame):
+    """Isolate the right-hand roller assembly from the blue fabric."""
+    hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV)
+    hue, saturation, value = cv2.split(hsv)
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+
+    search = _polygon(
+        [
+            (128, 80),
+            (176, 76),
+            (194, 91),
+            (194, 101),
+            (213, 99),
+            (231, 97),
+            (246, 104),
+            (256, 100),
+            (256, 143),
+            (194, 143),
+            (189, 129),
+            (179, 120),
+            (159, 120),
+            (142, 119),
+            (132, 112),
+        ]
+    )
+
+    colored = (
+        ((hue < 38) & (saturation > 34) & (value > 32))
+        | ((saturation < 82) & (value > 38))
+    )
+    mask = ((search > 0) & colored).astype(np.uint8) * 255
+    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
+    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, np.ones((2, 2), np.uint8))
+
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(mask)
+    clean = np.zeros_like(mask)
+    for index in range(1, count):
+        if stats[index, cv2.CC_STAT_AREA] >= 18:
+            clean[labels == index] = 255
+
+    # The two largest pieces are the roller and the hand/handle assembly.
+    alpha = cv2.GaussianBlur(clean, (3, 3), 0.55).astype(np.float32) / 255.0
+    alpha[yy < 76] = 0.0
+    alpha[xx < 128] = 0.0
+    return np.clip(alpha, 0.0, 1.0)
+
+
+def _make_cloth_plate(frame, alpha):
+    """Remove the movable assembly and reconstruct the blue fabric beneath it."""
+    removal = (alpha > 0.08).astype(np.uint8) * 255
+    removal = cv2.dilate(removal, np.ones((5, 5), np.uint8))
+
+    # Include the contact shadow immediately below the roller.
+    roller_shadow = _polygon(
+        [(133, 108), (190, 101), (198, 116), (191, 124), (142, 124)]
+    )
+    removal = np.maximum(removal, roller_shadow)
+    removal = cv2.GaussianBlur(removal, (5, 5), 1.1)
+    removal = (removal > 20).astype(np.uint8) * 255
+
+    bgr = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
+    plate = cv2.inpaint(bgr, removal, 5, cv2.INPAINT_TELEA)
+    plate = cv2.cvtColor(plate, cv2.COLOR_BGR2RGB).astype(np.float32)
+
+    # Restore a little high-frequency cloth texture to the otherwise smooth fill.
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    fine = (
+        1.7 * np.sin(0.69 * xx + 0.11 * yy)
+        + 1.2 * np.sin(1.31 * yy - 0.09 * xx)
+        + 0.8 * np.cos(2.17 * xx + 0.71 * yy)
+    )
+    blend = cv2.GaussianBlur((removal > 0).astype(np.float32), (7, 7), 1.8)
+    plate += (fine * 0.62 * blend)[..., None]
+
+    # Preserve the source exactly outside the reconstruction region.
+    source = frame.astype(np.float32)
+    keep = cv2.GaussianBlur((removal == 0).astype(np.float32), (5, 5), 1.1)
+    plate = plate * (1.0 - keep[..., None]) + source * keep[..., None]
+    return np.clip(plate, 0, 255).astype(np.uint8), removal
+
+
+def _warp_rgba(rgb, alpha, dx, dy):
+    """Subpixel translation using premultiplied alpha."""
+    transform = np.array([[1.0, 0.0, dx], [0.0, 1.0, dy]], dtype=np.float32)
+    premultiplied = rgb.astype(np.float32) * alpha[..., None]
+    moved_premultiplied = cv2.warpAffine(
+        premultiplied,
+        transform,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=(0, 0, 0),
+    )
+    moved_alpha = cv2.warpAffine(
+        alpha,
+        transform,
+        (WIDTH, HEIGHT),
+        flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT,
+        borderValue=0,
+    )
+    moved_rgb = moved_premultiplied / np.maximum(moved_alpha[..., None], 1e-5)
+    return np.clip(moved_rgb, 0, 255), np.clip(moved_alpha, 0, 1)
+
+
+def _add_contact_shadow(image, dx, dy, progress):
+    if progress <= 0.0:
+        return image
+    yy, xx = np.mgrid[:HEIGHT, :WIDTH]
+    cx = 163.0 + dx
+    cy = 117.0 + dy * 0.28
+    ellipse = ((xx - cx) / 27.0) ** 2 + ((yy - cy) / 5.2) ** 2
+    shadow = np.exp(-2.2 * ellipse).astype(np.float32)
+    strength = 0.13 * min(1.0, progress * 5.0)
+    return np.clip(
+        image.astype(np.float32) * (1.0 - strength * shadow[..., None]), 0, 255
+    ).astype(np.uint8)
+
+
+def _composite_assembly(plate, source, alpha, dx, dy):
+    shadowed = _add_contact_shadow(plate, dx, dy, -dx / 24.0)
+    moved_rgb, moved_alpha = _warp_rgba(source, alpha, dx, dy)
+
+    output = shadowed.astype(np.float32)
+    output = output * (1.0 - moved_alpha[..., None]) + moved_rgb * moved_alpha[..., None]
+
+    # The original hand runs off the right edge. Keep that wrist region anchored
+    # while the roller strokes left, so no background is exposed at the boundary.
+    if dx < -2.0:
+        strip_start = max(232, int(round(256 + dx)))
+        feather = np.ones((HEIGHT, WIDTH), dtype=np.float32)
+        edge = max(1, min(7, strip_start - 230))
+        for offset in range(edge):
+            x = strip_start + offset
+            if x < WIDTH:
+                feather[:, x] = (offset + 1) / (edge + 1)
+        anchor = np.zeros((HEIGHT, WIDTH), dtype=np.float32)
+        anchor[:, strip_start:] = feather[:, strip_start:]
+        output = output * (1.0 - anchor[..., None]) + source.astype(np.float32) * anchor[
+            ..., None
+        ]
+
+    return np.clip(output, 0, 255).astype(np.uint8)
+
+
+def generate():
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"))
+    if source.shape != (HEIGHT, WIDTH, 3):
+        raise ValueError(f"unexpected first-frame shape: {source.shape}")
+
+    alpha = _object_alpha(source)
+    plate, _ = _make_cloth_plate(source, alpha)
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = source
+
+    for index in range(1, FRAME_COUNT):
+        u = index / (FRAME_COUNT - 1)
+        # A single slow cleaning stroke: easy in, steady across the shirt.
+        progress = u * u * (3.0 - 2.0 * u)
+        dx = -24.0 * progress
+        dy = 7.0 * progress
+        # Small hand tremor keeps the motion from looking mechanically linear.
+        dx += 0.22 * np.sin(index * 1.35)
+        dy += 0.16 * np.sin(index * 1.07 + 0.8)
+        frames[index] = _composite_assembly(plate, source, alpha, dx, dy)
+
+    artifact_io.save(frames, CASE_DIR)
+    return frames
+
+
+if __name__ == "__main__":
+    generate()
