+def smoothstep(value: float) -> float:
+    value = max(0.0, min(1.0, value))
+    return value * value * (3.0 - 2.0 * value)

+def lerp(a, b, amount):
+    return np.asarray(a, dtype=np.float64) * (1.0 - amount) + np.asarray(b, dtype=np.float64) * amount

+def lerp_angle(a, b, amount):
+    delta = (b - a + math.pi) % (2.0 * math.pi) - math.pi
+    return a + delta * amount

+def ellipse_mask(shape, center, radii, feather=0.12):
+    yy, xx = np.mgrid[: shape[0], : shape[1]]
+    nx = (xx - center[0]) / radii[0]
+    ny = (yy - center[1]) / radii[1]
+    radius = np.sqrt(nx * nx + ny * ny)
+    return np.clip((1.0 - radius) / feather, 0.0, 1.0)

+def inverse_kinematics(base, wrist, length_1=100.0, length_2=90.0):
+    bx, by = base
+    wx, wy = wrist
+    dx, dy = wx - bx, wy - by
+    distance = max(1e-6, math.hypot(dx, dy))
+    distance = min(distance, length_1 + length_2 - 0.01)
+    ux, uy = dx / distance, dy / distance
+    along = (length_1 * length_1 - length_2 * length_2 + distance * distance) / (2.0 * distance)
+    height = math.sqrt(max(0.0, length_1 * length_1 - along * along))
+    mx, my = bx + along * ux, by + along * uy
+    candidates = [
+        (mx - height * uy, my + height * ux),
+        (mx + height * uy, my - height * ux),
+    ]
+    elbow = min(candidates, key=lambda point: point[1])
+    return elbow
