+def solve_two_link(base, target, l1, l2, elbow_up=True):
+    base = np.asarray(base, dtype=float)
+    target = np.asarray(target, dtype=float)
+    delta = target - base
+    dist = float(np.linalg.norm(delta))
+    dist = float(np.clip(dist, abs(l1 - l2) + 0.15, l1 + l2 - 0.15))
+    unit = delta / max(float(np.linalg.norm(delta)), 1e-6)
+    a = (l1 * l1 - l2 * l2 + dist * dist) / (2.0 * dist)
+    h = np.sqrt(max(0.0, l1 * l1 - a * a))
+    normal = np.array([-unit[1], unit[0]])
+    sign = 1.0 if elbow_up else -1.0
+    joint = base + a * unit + sign * h * normal
+    end = base + dist * unit
+    return joint, end
