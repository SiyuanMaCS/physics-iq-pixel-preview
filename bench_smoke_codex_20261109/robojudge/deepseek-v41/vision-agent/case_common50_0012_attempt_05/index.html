+from pathlib import Path
+
+import cv2
+import numpy as np
+from PIL import Image, ImageDraw, ImageFilter
+
+import artifact_io
+
+
+CASE_DIR = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 144
+FRAME_COUNT = 20
+FPS = 4
+
+
+def smoothstep(value):
+    value = np.clip(value, 0.0, 1.0)
+    return value * value * (3.0 - 2.0 * value)
+
+
+def polygon_mask(shape, points):
+    mask = Image.new("L", (shape[1], shape[0]), 0)
+    ImageDraw.Draw(mask).polygon(points, fill=255)
+    return np.asarray(mask, dtype=np.uint8)
+
+
+def softened(mask, radius=0.8):
+    image = Image.fromarray(mask)
+    image = image.filter(ImageFilter.GaussianBlur(radius))
+    return np.asarray(image, dtype=np.float32) / 255.0
+
+
+def build_end_effector(frame):
+    """Matte the existing wrist and gripper without changing its pixels."""
+    rough = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    rough |= polygon_mask(
+        frame.shape,
+        [(124, 0), (169, 0), (170, 22), (164, 28), (157, 25),
+         (154, 32), (149, 28), (146, 36), (140, 28), (134, 25),
+         (127, 17)],
+    )
+    rough |= polygon_mask(
+        frame.shape,
+        [(136, 21), (159, 21), (158, 34), (154, 53), (157, 68),
+         (153, 74), (149, 66), (146, 73), (141, 67), (137, 72),
+         (135, 58), (140, 44), (138, 31)],
+    )
+
+    # Retain only the visibly different robotic parts inside the two rough
+    # regions. This avoids carrying a rectangular piece of wall with the arm.
+    bg_left = frame[:, 112:120].mean(axis=1)
+    bg_right = frame[:, 177:185].mean(axis=1)
+    expected = np.empty_like(frame, dtype=np.float32)
+    for y in range(HEIGHT):
+        for x in range(WIDTH):
+            mix = np.clip((x - 116.0) / 65.0, 0.0, 1.0)
+            expected[y, x] = (1.0 - mix) * bg_left[y] + mix * bg_right[y]
+
+    color_distance = np.linalg.norm(frame.astype(np.float32) - expected, axis=2)
+    color_alpha = smoothstep((color_distance - 7.0) / 18.0)
+    alpha = color_alpha * (rough.astype(np.float32) / 255.0)
+
+    # The dark wrist and bright fingers are low-contrast against portions of
+    # the window. Keep their compact silhouettes fully opaque.
+    core = polygon_mask(
+        frame.shape,
+        [(138, 22), (156, 22), (156, 38), (153, 54), (156, 69),
+         (152, 73), (149, 67), (146, 72), (141, 66), (138, 70),
+         (140, 54), (137, 37)],
+    )
+    alpha = np.maximum(alpha, core.astype(np.float32) / 255.0)
+
+    # The orange collar is a useful, reliable foreground cue.
+    rgb = frame.astype(np.int16)
+    orange = (
+        (rgb[:, :, 0] > 70)
+        & (rgb[:, :, 0] > rgb[:, :, 1] * 1.45)
+        & (rgb[:, :, 0] > rgb[:, :, 2] * 1.35)
+        & (rough > 0)
+    )
+    alpha = np.maximum(alpha, orange.astype(np.float32))
+    return softened((alpha * 255).astype(np.uint8), 0.65)
+
+
+def build_clean_plate(frame, end_effector_alpha):
+    removal = end_effector_alpha > 0.08
+    removal = cv2.dilate(removal.astype(np.uint8), np.ones((3, 3), np.uint8), iterations=1) > 0
+
+    # The end effector starts at the top border, so inpaint from a synthetic
+    # continuation of the window above it before filling its former footprint.
+    source = frame.copy()
+    source[0:12, 124:171] = frame[12:24, 124:171]
+    mask = removal.astype(np.uint8) * 255
+    clean = cv2.inpaint(
+        cv2.cvtColor(source, cv2.COLOR_RGB2BGR),
+        mask,
+        5,
+        cv2.INPAINT_TELEA,
+    )
+    clean = cv2.cvtColor(clean, cv2.COLOR_BGR2RGB)
+
+    # Replace the part of the top link hidden by the original motor with a
+    # continuation of the nearby upper-arm cable and dark mount.
+    patch = clean[0:22, 120:134].astype(np.float32)
+    clean[0:22, 121:135] = np.clip(patch * 0.72 + 25.0, 0, 255).astype(np.uint8)
+    return clean
+
+
+def build_object(frame):
+    """Extract the stopper and remove it from the table."""
+    crop_box = (132, 128, 164, 144)
+    x0, y0, x1, y1 = crop_box
+    crop = frame[y0:y1, x0:x1]
+
+    # The stopper is the dark compact object on the pale table. A small
+    # dilation preserves its antialiased outline and the center metal handle.
+    crop_i = crop.astype(np.int16)
+    dark = crop_i.max(axis=2) < 92
+    neutral_shadow = (
+        (crop_i.max(axis=2) < 128)
+        & (crop_i.max(axis=2) - crop_i.min(axis=2) < 35)
+    )
+    mask = cv2.dilate((dark | neutral_shadow).astype(np.uint8), np.ones((2, 2), np.uint8), iterations=1)
+
+    # Keep only the connected stopper silhouette near the image center.
+    count, labels, stats, _ = cv2.connectedComponentsWithStats(mask, 8)
+    keep = np.zeros_like(mask)
+    for label in range(1, count):
+        x, y, w, h, area = stats[label]
+        if area >= 18 and x + w > 8 and x < 26:
+            keep[labels == label] = 1
+    mask = keep
+
+    alpha = softened((mask * 255).astype(np.uint8), 0.55)
+    rgb = crop.astype(np.float32)
+
+    # The source touches the bottom border. Extend the silhouette by a few
+    # rows so the object remains complete when it is lifted into view.
+    extended = np.zeros((crop.shape[0] + 6, crop.shape[1], 3), dtype=np.uint8)
+    extended_alpha = np.zeros((crop.shape[0] + 6, crop.shape[1]), dtype=np.float32)
+    extended[:crop.shape[0]] = crop
+    extended_alpha[:crop.shape[0]] = alpha
+    last_rows = crop[-3:].mean(axis=0).astype(np.uint8)
+    extended[crop.shape[0]:] = last_rows
+    for row in range(crop.shape[0], extended.shape[0]):
+        taper = np.clip(1.0 - (row - crop.shape[0] + 1) / 7.0, 0.0, 1.0)
+        extended_alpha[row] = alpha[-1] * taper
+
+    # Restore the metallic handle where the clipped source ends.
+    center = extended.shape[1] // 2
+    for row in range(crop.shape[0] - 2, extended.shape[0] - 1):
+        half_width = 2 if row < extended.shape[0] - 3 else 1
+        for x in range(center - half_width, center + half_width + 1):
+            metallic = np.array([104, 108, 105], dtype=np.float32)
+            extended[row, x] = np.clip(
+                0.65 * extended[row, x].astype(np.float32) + 0.35 * metallic,
+                0,
+                255,
+            ).astype(np.uint8)
+            extended_alpha[row, x] = max(extended_alpha[row, x], 0.92)
+
+    # Remove the stopper and its contact shadow from the fixed table.
+    clean = frame.copy()
+    object_removal = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    object_removal[y0:y1, x0:x1] = mask * 255
+    object_removal = cv2.dilate(object_removal, np.ones((3, 3), np.uint8), iterations=1)
+    clean_bgr = cv2.inpaint(
+        cv2.cvtColor(clean, cv2.COLOR_RGB2BGR),
+        object_removal,
+        4,
+        cv2.INPAINT_TELEA,
+    )
+    clean = cv2.cvtColor(clean_bgr, cv2.COLOR_BGR2RGB)
+
+    return extended, extended_alpha, clean, (x0, y0)
+
+
+def alpha_composite(dst, src, alpha, left, top):
+    h, w = src.shape[:2]
+    x0, y0 = max(0, left), max(0, top)
+    x1, y1 = min(WIDTH, left + w), min(HEIGHT, top + h)
+    if x0 >= x1 or y0 >= y1:
+        return
+    sx0, sy0 = x0 - left, y0 - top
+    sx1, sy1 = sx0 + (x1 - x0), sy0 + (y1 - y0)
+    a = alpha[sy0:sy1, sx0:sx1, None]
+    region = dst[y0:y1, x0:x1].astype(np.float32)
+    patch = src[sy0:sy1, sx0:sx1].astype(np.float32)
+    dst[y0:y1, x0:x1] = np.clip(region * (1.0 - a) + patch * a, 0, 255).astype(np.uint8)
+
+
+def draw_extension(frame, dy):
+    if dy < 1.0:
+        return
+    rgb = frame.astype(np.float32)
+    overlay = Image.new("RGBA", (WIDTH, HEIGHT), (0, 0, 0, 0))
+    draw = ImageDraw.Draw(overlay)
+    bottom = int(round(8 + dy))
+    # A slim telescoping shaft bridges the stationary upper arm and the moving
+    # motor. It stays behind the end-effector patch.
+    draw.line((130, 0, 146, bottom), fill=(20, 24, 27, 255), width=8)
+    draw.line((132, 0, 147, bottom), fill=(53, 60, 64, 255), width=3)
+    draw.line((133, 1, 146, bottom - 1), fill=(91, 101, 105, 190), width=1)
+    overlay_np = np.asarray(overlay, dtype=np.float32)
+    a = overlay_np[:, :, 3:4] / 255.0
+    frame[:] = np.clip(rgb * (1.0 - a) + overlay_np[:, :, :3] * a, 0, 255).astype(np.uint8)
+
+
+def motion_curves(index):
+    # Hold, descend, close, lift, and settle. All interpolation is monotonic
+    # and eased so the arm does not jitter between four-fps frames.
+    if index <= 1:
+        descend = 0.0
+    elif index <= 10:
+        descend = smoothstep((index - 1) / 9.0)
+    else:
+        descend = 1.0
+
+    reach = 59.0 * descend
+    if index <= 12:
+        lift = 0.0
+    else:
+        lift = 23.0 * smoothstep((index - 12) / 7.0)
+    grip = smoothstep((index - 9) / 3.0)
+    return reach - lift, lift, grip
+
+
+def make_frames():
+    first = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    assert first.shape == (HEIGHT, WIDTH, 3)
+
+    end_alpha = build_end_effector(first)
+    clean = build_clean_plate(first, end_alpha)
+    object_sprite, object_alpha, clean, object_origin = build_object(clean)
+
+    # The end-effector sprite is the original unmatted RGB. At time zero the
+    # untouched first frame is emitted, guaranteeing pixel identity.
+    ee_box = (124, 0, 170, 75)
+    ex0, ey0, ex1, ey1 = ee_box
+    ee_rgb = first[ey0:ey1, ex0:ex1]
+    ee_alpha = end_alpha[ey0:ey1, ex0:ex1]
+
+    frames = []
+    ox, oy = object_origin
+    for index in range(FRAME_COUNT):
+        if index == 0:
+            frames.append(first.copy())
+            continue
+
+        dy, lift, grip = motion_curves(index)
+        frame = clean.copy()
+        draw_extension(frame, dy)
+
+        # Small, non-deforming jaw closure is represented by adding two short
+        # metallic contact pads on the existing fingers as they meet the handle.
+        pad_alpha = smoothstep((grip - 0.25) / 0.55)
+        if pad_alpha > 0:
+            overlay = Image.new("RGBA", (WIDTH, HEIGHT), (0, 0, 0, 0))
+            draw = ImageDraw.Draw(overlay)
+            top = int(round(dy + 62))
+            inset = int(round(2.0 - 1.5 * pad_alpha))
+            draw.rectangle((139 + inset, top, 142 + inset, top + 7), fill=(150, 158, 159, int(210 * pad_alpha)))
+            draw.rectangle((151 - inset, top, 154 - inset, top + 7), fill=(174, 181, 181, int(220 * pad_alpha)))
+            overlay_np = np.asarray(overlay, dtype=np.float32)
+            a = overlay_np[:, :, 3:4] / 255.0
+            frame = np.clip(frame.astype(np.float32) * (1 - a) + overlay_np[:, :, :3] * a, 0, 255).astype(np.uint8)
+
+        alpha_composite(frame, ee_rgb, ee_alpha, ex0, int(round(ey0 + dy)))
+
+        # The stopper is stationary until the jaws close, then rises with the
+        # wrist. A very slight forward offset keeps it visually in the jaws.
+        object_dx = 0
+        object_dy = int(round(lift))
+        alpha_composite(
+            frame,
+            object_sprite,
+            object_alpha,
+            ox + object_dx,
+            oy - object_dy,
+        )
+
+        # Preserve the source grain and avoid over-clean synthetic regions.
+        rng = np.random.default_rng(1000 + index)
+        grain = rng.normal(0.0, 0.8, frame.shape).astype(np.float32)
+        frame = np.clip(frame.astype(np.float32) + grain, 0, 255).astype(np.uint8)
+        frames.append(frame)
+
+    return np.stack(frames)
+
+
+def main():
+    frames = make_frames()
+    assert frames.shape == (FRAME_COUNT, HEIGHT, WIDTH, 3)
+    assert np.array_equal(frames[0], np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB")))
+    artifact_io.save(frames, CASE_DIR)
+
+    # Also provide a directly playable video alongside the benchmark payload.
+    import imageio.v2 as imageio
+
+    imageio.mimsave(CASE_DIR / "output.mp4", frames, fps=FPS, macro_block_size=None)
+
+
+if __name__ == "__main__":
+    main()
