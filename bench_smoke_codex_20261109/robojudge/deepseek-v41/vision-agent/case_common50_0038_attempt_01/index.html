+def smoothstep(value):
+    value = float(np.clip(value, 0.0, 1.0))
+    return value * value * (3.0 - 2.0 * value)

+def interpolate_keyframes(frame_index, keyframes):
+    if frame_index <= keyframes[0][0]:
+        return keyframes[0][1]
+    if frame_index >= keyframes[-1][0]:
+        return keyframes[-1][1]
+    for (t0, v0), (t1, v1) in zip(keyframes[:-1], keyframes[1:]):
+        if t0 <= frame_index <= t1:
+            u = smoothstep((frame_index - t0) / max(1, t1 - t0))
+            return np.asarray(v0, dtype=np.float32) * (1.0 - u) + np.asarray(v1, dtype=np.float32) * u
+    return keyframes[-1][1]

+def polygon_mask(points, shape):
