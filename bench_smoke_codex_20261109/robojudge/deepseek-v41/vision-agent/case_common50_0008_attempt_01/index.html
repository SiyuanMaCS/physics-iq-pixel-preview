+def _polygon_mask(points: list[tuple[int, int]], blur: float = 0.7) -> np.ndarray:
+    mask = np.zeros((H, W), dtype=np.uint8)
+    cv2.fillPoly(mask, [np.asarray(points, dtype=np.int32)], 255, lineType=cv2.LINE_AA)

+def _combine_masks(polygons: list[list[tuple[int, int]]], lines=()) -> np.ndarray:

+def _seam_x(y: np.ndarray | float) -> np.ndarray:
+    """Perspective seam: x=161 at the top, x=132 at the bottom."""
+    return 161.0 - 0.151 * np.asarray(y, dtype=np.float32)
+

+def _smoothstep(x: np.ndarray | float) -> np.ndarray:
+    x = np.clip(np.asarray(x, dtype=np.float32), 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)

+def _door_masks() -> tuple[np.ndarray, np.ndarray]:
+    yy = np.arange(H, dtype=np.float32)[:, None]
+    xx = np.arange(W, dtype=np.float32)[None, :]
+    seam = _seam_x(yy)
+    left = ((xx >= 108.0) & (xx <= seam - 1.0)).astype(np.float32)
+    right = ((xx >= seam + 1.0) & (xx <= 198.0)).astype(np.float32)
+    # A one-pixel softened edge avoids a hard aliased cut when the panels move.
+    left = cv2.GaussianBlur(left, (0, 0), 0.55)
+    right = cv2.GaussianBlur(right, (0, 0), 0.55)
+    return left, right

+def _translate_rgba(rgb: np.ndarray, alpha: np.ndarray, dx: float, dy: float):
+    matrix = np.float32([[1.0, 0.0, dx], [0.0, 1.0, dy]])
+    warped_rgb = cv2.warpAffine(
+        rgb, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=(0, 0, 0),
+    )
+    warped_alpha = cv2.warpAffine(
+        alpha, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0.0,
+    )
+    return warped_rgb, warped_alpha

+def _overlay(base: np.ndarray, rgb: np.ndarray, alpha: np.ndarray) -> np.ndarray:
+    a = alpha[:, :, None]
+    return np.clip(rgb.astype(np.float32) * a + base.astype(np.float32) * (1.0 - a), 0, 255).astype(np.uint8)
+
+
+def _interior(base: np.ndarray, gap: float) -> np.ndarray:
+    """Add a dark recess between the opening panels, with cool edge glints."""
+    out = base.astype(np.float32)
+    yy = np.arange(H, dtype=np.float32)[:, None]
+    xx = np.arange(W, dtype=np.float32)[None, :]
+    seam = _seam_x(yy)
+    left_edge = seam - gap
+    right_edge = seam + gap
+    inside = ((xx >= left_edge) & (xx <= right_edge)).astype(np.float32)
+    inside = cv2.GaussianBlur(inside, (0, 0), 0.7)
+
+    # The interior is darker near the center and catches a faint blue-grey bounce
+    # from the metallic door faces.
+    depth = np.clip(np.abs(xx - seam) / np.maximum(gap, 1.0), 0.0, 1.0)
+    vertical = np.clip(yy / float(H), 0.0, 1.0)
+    interior_rgb = np.empty((H, W, 3), dtype=np.float32)
+    interior_rgb[:, :, 0] = 16.0 + 8.0 * depth + 3.0 * vertical
+    interior_rgb[:, :, 1] = 21.0 + 11.0 * depth + 4.0 * vertical
+    interior_rgb[:, :, 2] = 24.0 + 13.0 * depth + 5.0 * vertical
+    out = interior_rgb * inside[:, :, None] + out * (1.0 - inside[:, :, None])
+
+    # Thin, bright inner lips sell the thickness of the silver panels.
+    if gap > 1.5:
+        lip_alpha = np.clip((gap - 1.0) / 5.0, 0.0, 0.85)
+        for side, edge in ((-1.0, left_edge), (1.0, right_edge)):
+            d = np.abs(xx - edge)
+            line = np.clip(1.0 - d / 1.35, 0.0, 1.0) * lip_alpha
+            line = cv2.GaussianBlur(line.astype(np.float32), (0, 0), 0.35)
+            tone = np.array([151.0, 166.0, 169.0], dtype=np.float32) if side < 0 else np.array([112.0, 126.0, 131.0], dtype=np.float32)
+            out = out * (1.0 - line[:, :, None]) + tone[None, None, :] * line[:, :, None]
+    return np.clip(out, 0, 255).astype(np.uint8)
+
+
+def _add_panel_shading(frame: np.ndarray, gap: float) -> np.ndarray:
+    """Subtle moving sheen and occlusion shadows on the opening door edges."""
+    out = frame.astype(np.float32)
+    yy = np.arange(H, dtype=np.float32)[:, None]
+    xx = np.arange(W, dtype=np.float32)[None, :]
+    seam = _seam_x(yy)
+
+    # Occlusion shadow just inside each panel lip.
+    for direction in (-1.0, 1.0):
+        edge = seam + direction * (gap + 2.0)
+        delta = (xx - edge) * direction
+        shadow = np.exp(-np.maximum(delta, 0.0) / 5.0) * (delta >= 0.0) * (delta < 18.0)
+        out *= 1.0 - 0.12 * shadow[:, :, None]
+
+    # A soft vertical reflection travels slowly across the left silver face.
+    phase = np.clip(gap / 15.0, 0.0, 1.0)
+    band_center = 123.0 + 4.0 * phase
+    band = np.exp(-((xx - band_center) / 8.0) ** 2) * np.ones((H, 1), dtype=np.float32)
+    band *= ((yy > 12.0) & (yy < 126.0)).astype(np.float32)
+    out += np.array([7.0, 7.0, 6.0], dtype=np.float32)[None, None, :] * band[:, :, None]
+    return np.clip(out, 0, 255).astype(np.uint8)
+

+def render() -> np.ndarray:
+    original = np.asarray(Image.open(ROOT / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    assert original.shape == (H, W, 3)
+
+    left_alpha, right_alpha = _robot_masks()
+    robot_alpha = np.clip(left_alpha + right_alpha, 0.0, 1.0)
+    robot_mask = (robot_alpha * 255.0).astype(np.uint8)
+    clean_bg = cv2.cvtColor(
+        cv2.inpaint(
+            cv2.cvtColor(original, cv2.COLOR_RGB2BGR),
+            robot_mask,
+            4.5,
+            cv2.INPAINT_TELEA,
+        ),
+        cv2.COLOR_BGR2RGB,
+    )
+    # Remove any residual robot-colored pixels that the inpaint could leave at
+    # the broad mask boundary.
+    clean_bg = cv2.GaussianBlur(clean_bg, (0, 0), 0.35)
+
+    left_panel_alpha, right_panel_alpha = _door_masks()
+    left_arm = original.copy()
+    right_arm = original.copy()
+
+    frames: list[np.ndarray] = []
+    for index in range(N_FRAMES):
+        if index == 0:
+            frames.append(original.copy())
+            continue
+
+        t = index / float(N_FRAMES - 1)
+        # Close around the seam briefly, then pull the two faces apart.
+        grip = 1.65 * np.sin(np.pi * np.clip(index / 4.5, 0.0, 1.0)) if index < 6 else 0.0
+        open_t = _smoothstep(np.clip((index - 3.0) / 16.0, 0.0, 1.0))
+        gap = 15.0 * open_t
+        sway = 0.25 * np.sin(index * 1.65) * open_t
+
+        frame = clean_bg.copy()
+        frame = _interior(frame, gap)
+
+        # Move the two door faces outward around the perspective seam.
+        left_rgb, left_a = _translate_rgba(clean_bg, left_panel_alpha, -gap, 0.0)
+        right_rgb, right_a = _translate_rgba(clean_bg, right_panel_alpha, gap, 0.0)
+        frame = _overlay(frame, left_rgb, left_a)
+        frame = _overlay(frame, right_rgb, right_a)
+        frame = _add_panel_shading(frame, gap)
+
+        # The grippers are rigid foreground assemblies. Their only motion is a
+        # translation that follows the corresponding door face.
+        ldx = -gap + grip + sway
+        rdx = gap - grip - sway
+        ldy = -0.8 * open_t
+        rdy = -0.8 * open_t
+        l_rgb, l_a = _translate_rgba(left_arm, left_alpha, ldx, ldy)
+        r_rgb, r_a = _translate_rgba(right_arm, right_alpha, rdx, rdy)
+        frame = _overlay(frame, l_rgb, l_a)
+        frame = _overlay(frame, r_rgb, r_a)
+
+        # Keep the render photometric and temporally calm at this small size.
+        if index >= 4:
+            frame = cv2.GaussianBlur(frame, (0, 0), 0.18)
+        frames.append(np.clip(frame, 0, 255).astype(np.uint8))
+
+    result = np.stack(frames, axis=0)
+    assert result.shape == (N_FRAMES, H, W, 3), result.shape
+    assert result.dtype == np.uint8
+    assert np.array_equal(result[0], original)
+    return result
+
+
+def main() -> None:
+    frames = render()
+    artifact_io.save(frames, ROOT)
+
+
+if __name__ == "__main__":
+    main()
