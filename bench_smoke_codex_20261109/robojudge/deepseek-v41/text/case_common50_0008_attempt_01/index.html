+def two_bone_ik(
+    base: tuple[float, float],
+    target: tuple[float, float],
+    l1: float,
+    l2: float,
+    bend: int,
+) -> tuple[tuple[float, float], tuple[float, float]]:
+    vx, vy = target[0] - base[0], target[1] - base[1]
+    distance = math.hypot(vx, vy)
+    reachable = min(l1 + l2 - 0.8, max(abs(l1 - l2) + 0.8, distance))
+    if distance < 1e-6:
+        ux, uy = 0.0, -1.0
+    else:
+        ux, uy = vx / distance, vy / distance
+    end = (base[0] + ux * reachable, base[1] + uy * reachable)
+    along = (l1 * l1 - l2 * l2 + reachable * reachable) / (2.0 * reachable)
+    height = math.sqrt(max(0.0, l1 * l1 - along * along))
+    px, py = -uy, ux
+    elbow = (
+        base[0] + ux * along + bend * px * height,
+        base[1] + uy * along + bend * py * height,
+    )
+    return elbow, end

+def composite_arm(frame: np.ndarray, layer: Image.Image, shadow: Image.Image) -> np.ndarray:
+    shadow_small = shadow.resize((W, H), Image.Resampling.LANCZOS)
+    layer_small = layer.resize((W, H), Image.Resampling.LANCZOS)
+    shadow_alpha = np.asarray(shadow_small.getchannel("A"), dtype=np.float32) / 255.0
+    shadow_rgb = np.asarray(shadow_small.convert("RGB"), dtype=np.float32)
+    out = frame.astype(np.float32)
+    out = out * (1.0 - 0.34 * shadow_alpha[..., None]) + shadow_rgb * (0.34 * shadow_alpha[..., None])
+
+    alpha = np.asarray(layer_small.getchannel("A"), dtype=np.float32) / 255.0
+    rgb = np.asarray(layer_small.convert("RGB"), dtype=np.float32)
+    out = out * (1.0 - alpha[..., None]) + rgb * alpha[..., None]
+    return np.clip(out + 0.5, 0, 255).astype(np.uint8)

+def render_door(source: np.ndarray, opening: float) -> np.ndarray:

+    frame = source.copy()

+    top, bottom = 0, 97

+    gap_width = max(1, right_edge - left_edge)
+    yy = np.linspace(0.0, 1.0, bottom - top, dtype=np.float32)[:, None]
+    xx = np.linspace(0.0, 1.0, gap_width, dtype=np.float32)[None, :]

+    for channel, (lo, hi) in enumerate(((20, 7), (27, 12), (29, 13))):
+        interior[..., channel] = lo + (hi - lo) * yy
+    interior[..., 2] += 2.0 * (1.0 - yy)

+    interior[..., 0] *= 0.80 + 0.20 * xx

+    if gap_width >= 3:

+def generate() -> np.ndarray:
+    source = np.asarray(Image.open(CASE_DIR / "first_frame.png").convert("RGB"), dtype=np.uint8)
+    if source.shape != (H, W, 3):
+        raise ValueError(f"unexpected first frame shape: {source.shape}")
+
+    frames = np.empty((N_FRAMES, H, W, 3), dtype=np.uint8)
+    for index in range(N_FRAMES):
+        if index == 0:
+            frames[index] = source
+            continue
+
+        # The manipulators enter first, clamp the seam, then pry it apart.
+        approach = smoothstep((index - 0.5) / 6.2)
+        pry = smoothstep((index - 6.0) / 11.0)
+        settle = smoothstep((index - 15.5) / 3.0)
+        opening = pry * (1.0 - 0.10 * settle)
+
+        frame = render_door(source, opening)
+        frame = render_arms(frame, approach, opening)
+
+        # A restrained exposure lift keeps the metallic edges legible without changing the camera.
+        if index > 1:
+            frame = np.clip(frame.astype(np.float32) + (0.8 * smoothstep((index - 1) / 5.0)), 0, 255).astype(np.uint8)
+        frames[index] = frame
+
+    if not np.array_equal(frames[0], source):
+        raise AssertionError("frame zero must be pixel-identical to first_frame.png")
+    return frames
+
