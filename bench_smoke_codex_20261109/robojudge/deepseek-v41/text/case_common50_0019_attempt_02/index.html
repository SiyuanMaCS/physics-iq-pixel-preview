+def smoothstep(value):

+def interpolate_keyframes(frame, keys):
+    """Interpolate (frame, value, value, ...) rows with eased motion."""
+    if frame <= keys[0][0]:
+        return np.asarray(keys[0][1:], dtype=np.float64)
+    if frame >= keys[-1][0]:
+        return np.asarray(keys[-1][1:], dtype=np.float64)
+    for left, right in zip(keys[:-1], keys[1:]):
+        if left[0] <= frame <= right[0]:
+            t = smoothstep((frame - left[0]) / (right[0] - left[0]))
+            a = np.asarray(left[1:], dtype=np.float64)
+            b = np.asarray(right[1:], dtype=np.float64)
+            return a + (b - a) * t
+    return np.asarray(keys[-1][1:], dtype=np.float64)

+def capsule_distance(xx, yy, p0, p1):
+    p0 = np.asarray(p0, dtype=np.float32)
+    p1 = np.asarray(p1, dtype=np.float32)
+    segment = p1 - p0
+    length2 = float(np.dot(segment, segment))
+    if length2 < 1e-6:
+        return np.sqrt((xx - p0[0]) ** 2 + (yy - p0[1]) ** 2)
+    projection = ((xx - p0[0]) * segment[0] + (yy - p0[1]) * segment[1]) / length2
+    projection = np.clip(projection, 0.0, 1.0)
+    qx = p0[0] + projection * segment[0]
+    qy = p0[1] + projection * segment[1]
+    return np.sqrt((xx - qx) ** 2 + (yy - qy) ** 2)
