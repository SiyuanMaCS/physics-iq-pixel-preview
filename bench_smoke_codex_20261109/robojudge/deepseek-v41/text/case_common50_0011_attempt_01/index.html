+def similarity_transform(src, dst):
+    src = np.asarray(src, dtype=np.float32)
+    dst = np.asarray(dst, dtype=np.float32)
+    src_mean = src.mean(axis=0)
+    dst_mean = dst.mean(axis=0)
+    a = src - src_mean

+    u, singular, vt = np.linalg.svd(a.T @ b)
