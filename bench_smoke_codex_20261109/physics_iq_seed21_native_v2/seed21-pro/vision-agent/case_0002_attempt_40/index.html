import numpy as np, cv2
from PIL import Image
im = np.array(Image.open('first_frame.png').convert('RGB'))
f = im.astype(float)

# local wall colors sampled in clear strips near each object

def region_mask(x0,y0,x1,y1, wall, thr):
    reg = f[y0:y1, x0:x1]
    d = np.linalg.norm(reg-wall, axis=2)
    return d>thr
