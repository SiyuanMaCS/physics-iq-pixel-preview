+"""Procedural t2v: a cat starts on the right of a rock and runs to its left.
+
+Renders 20 RGB frames (256x144, 4 fps) entirely with PIL/numpy geometry:
+sky gradient, hills, ground, a fixed rock, and an articulated running cat.
+"""
+from pathlib import Path
+import math
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+W, H, N, FPS = 256, 144, 20, 4.0
+GROUND_Y = 112          # top of the ground band
+ROCK_X = 128            # rock center, stays fixed
+CAT_START_X = 206       # clearly right of the rock
+CAT_END_X = 48          # clearly left of the rock
+
+
+def smoothstep(u):
+    u = max(0.0, min(1.0, u))
+    return u * u * (3.0 - 2.0 * u)
+
+
+def lerp(a, b, t):
+    return a + (b - a) * t
+
+
+def draw_background(draw, frame):
+    # vertical sky gradient
+    top = np.array([122, 178, 226], dtype=float)
+    bottom = np.array([206, 230, 240], dtype=float)
+    for y in range(GROUND_Y):
+        t = y / GROUND_Y
+        color = tuple((top * (1 - t) + bottom * t).astype(int))
+        draw.line([(0, y), (W, y)], fill=color)
+
+    # slow drifting clouds
+    for i, (cx, cy, s) in enumerate([(46, 24, 1.0), (150, 16, 0.8), (218, 34, 0.9)]):
+        drift = (frame * (0.35 + i * 0.12)) % (W + 60) - 30
+        x = (cx + drift) % (W + 60) - 30
+        col = (245, 248, 252)
+        for dx, dy, r in [(-14, 2, 9), (0, -2, 11), (14, 2, 8)]:
+            draw.ellipse([x + (dx - r) * s, cy + (dy - r) * s,
+                          x + (dx + r) * s, cy + (dy + r) * s], fill=col)
+
+    # distant hills
+    draw.pieslice([-40, 62, 130, 150], 180, 360, fill=(138, 184, 132))
+    draw.pieslice([96, 70, 280, 160], 180, 360, fill=(118, 168, 118))
+
+    # ground
+    draw.rectangle([0, GROUND_Y, W, H], fill=(116, 178, 84))
+    draw.rectangle([0, GROUND_Y, W, GROUND_Y + 4], fill=(138, 196, 102))
+    # grass texture (deterministic)
+    rng = np.random.default_rng(7)
+    for _ in range(90):
+        gx = int(rng.integers(0, W))
+        gy = int(rng.integers(GROUND_Y + 6, H - 2))
+        h = int(rng.integers(2, 5))
+        draw.line([(gx, gy), (gx + 1, gy - h)], fill=(96, 158, 70))
+
+
+def draw_shadow(d, cx, cy, w, alpha=110):
+    d.ellipse([cx - w, cy - 4, cx + w, cy + 4], fill=(40, 60, 35, alpha))
+
+
+def draw_rock(draw):
+    # shadow
+    draw.ellipse([ROCK_X - 34, GROUND_Y - 4, ROCK_X + 34, GROUND_Y + 6], fill=(80, 100, 70))
+    # main boulder polygon
+    pts = [(-32, 0), (-26, -20), (-12, -30), (8, -32), (24, -24),
+           (32, -8), (28, 0)]
+    pts = [(ROCK_X + dx, GROUND_Y + dy) for dx, dy in pts]
+    draw.polygon(pts, fill=(128, 130, 136), outline=(92, 94, 102))
+    # lit facets
+    draw.polygon([(ROCK_X - 26, GROUND_Y - 20), (ROCK_X - 12, GROUND_Y - 30),
+                  (ROCK_X + 8, GROUND_Y - 32), (ROCK_X - 4, GROUND_Y - 12),
+                  (ROCK_X - 18, GROUND_Y - 8)], fill=(158, 160, 166))
+    # shaded facets
+    draw.polygon([(ROCK_X + 8, GROUND_Y - 32), (ROCK_X + 24, GROUND_Y - 24),
+                  (ROCK_X + 32, GROUND_Y - 8), (ROCK_X + 10, GROUND_Y - 6)],
+                 fill=(104, 106, 114))
+    # cracks
+    draw.line([(ROCK_X - 14, GROUND_Y - 14), (ROCK_X - 6, GROUND_Y - 6),
+               (ROCK_X - 9, GROUND_Y - 2)], fill=(92, 94, 102), width=1)
+    for dx, dy in [(-20, -4), (2, -18), (16, -10), (-2, -4)]:
+        draw.point((ROCK_X + dx, GROUND_Y + dy), fill=(80, 82, 90))
+
+
+def draw_cat(base, cx, phase, moving):
+    """Draw a left-facing cat centered at (cx, GROUND_Y). phase drives gait."""
+    layer = Image.new("RGBA", (W, H), (0, 0, 0, 0))
+    d = ImageDraw.Draw(layer)
+
+    bob = math.sin(phase * 2) * 1.2 if moving else 0
+    by = GROUND_Y - 14 + bob          # body center y
+    fur = (214, 138, 58)
+    fur_dark = (170, 96, 34)
+    cream = (240, 214, 172)
+
+    # ground shadow
+    draw_shadow(d, cx, GROUND_Y + 2, 22, alpha=95)
+
+    # legs: two pairs swinging in counter-phase
+    swing = math.sin(phase) * (6 if moving else 0)
+    hips = [cx - 10, cx + 8]
+    for i, hx in enumerate(hips):
+        s = swing if i == 0 else -swing
+        d.line([(hx - 2, by + 5), (hx - 2 + s * 0.6, GROUND_Y - 1)],
+               fill=(150, 82, 28), width=4)
+        d.line([(hx + 2, by + 5), (hx + 2 + s, GROUND_Y - 1)],
+               fill=fur_dark, width=4)
+
+    # tail curves up from the rear (right side), swishing
+    wag = math.sin(phase * 1.0 + 1.2) * (5 if moving else 2)
+    tx, ty = cx + 15, by - 2
+    d.line([(tx, ty), (tx + 10, ty - 8 + wag), (tx + 8, ty - 18 + wag)],
+           fill=fur, width=5, joint="curve")
+    d.line([(tx + 8, ty - 18 + wag), (tx + 10, ty - 20 + wag)],
+           fill=fur_dark, width=3)
+
+    # body
+    d.ellipse([cx - 16, by - 9, cx + 16, by + 9], fill=fur, outline=fur_dark)
+    d.ellipse([cx - 10, by + 1, cx + 12, by + 9], fill=cream)
+    # tabby stripes
+    for sx in (-4, 2, 8):
+        d.line([(cx + sx, by - 8), (cx + sx + 2, by - 2)], fill=fur_dark, width=2)
+
+    # head at the front (left)
+    hx, hy = cx - 17, by - 6
+    d.ellipse([hx - 9, hy - 8, hx + 9, hy + 8], fill=fur, outline=fur_dark)
+    # ears
+    d.polygon([(hx - 8, hy - 5), (hx - 9, hy - 13), (hx - 2, hy - 8)],
+              fill=fur, outline=fur_dark)
+    d.polygon([(hx + 1, hy - 8), (hx + 5, hy - 13), (hx + 8, hy - 5)],
+              fill=fur, outline=fur_dark)
+    d.polygon([(hx - 7, hy - 7), (hx - 7, hy - 11), (hx - 4, hy - 8)], fill=(232, 170, 160))
+    # muzzle
+    d.ellipse([hx - 10, hy + 1, hx - 1, hy + 8], fill=cream)
+    # eye (facing left)
+    d.ellipse([hx - 8, hy - 4, hx - 4, hy + 1], fill=(30, 40, 30))
+    d.point((hx - 7, hy - 3), fill=(255, 255, 255))
+    # nose
+    d.polygon([(hx - 11, hy + 2), (hx - 8, hy + 2), (hx - 9.5, hy + 4)], fill=(120, 60, 60))
+    # whiskers
+    for wy in (hy + 3, hy + 5):
+        d.line([(hx - 6, wy), (hx - 16, wy - 1)], fill=(120, 100, 80), width=1)
+        d.line([(hx - 6, wy), (hx - 15, wy + 2)], fill=(120, 100, 80), width=1)
+
+    base.alpha_composite(layer)
+
+
+def render_frame(f):
+    img = Image.new("RGB", (W, H))
+    draw = ImageDraw.Draw(img)
+    draw_background(draw, f)
+
+    # hold at both ends, smooth run across the middle frames
+    u = smoothstep((f - 1.0) / 17.0)
+    cx = lerp(CAT_START_X, CAT_END_X, u)
+    moving = 0 < f < 19
+    phase = f * 1.9
+
+    rgba = img.convert("RGBA")
+    draw_rock(draw)
+    draw_cat(rgba, cx, phase, moving)
+    return np.asarray(rgba.convert("RGB"), dtype=np.uint8)
+
+
+def main():
+    frames = np.stack([render_frame(f) for f in range(N)], axis=0)
+    artifact_io.save(frames, Path(__file__).parent)
+    print(f"saved {frames.shape}")
+
+
+if __name__ == "__main__":
+    main()
