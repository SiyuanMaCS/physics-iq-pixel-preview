+"""Procedural text-to-video generation.
+
+Prompt: A bird is in front of a tree, then the bird flies to above the tree.
+
+Pure numpy/PIL rendering: sky, ground, a layered tree, and a flapping bird
+that eases from a position in front of the trunk to a point above the canopy.
+"""
+from pathlib import Path
+import math
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+W, H, N, FPS = 256, 144, 20, 4.0
+CASE_DIR = Path(__file__).parent
+
+
+def make_background(rng):
+    """Sky gradient, sun, soft clouds, and grassy ground."""
+    top = np.array([122, 178, 232], dtype=np.float32)
+    bottom = np.array([206, 232, 248], dtype=np.float32)
+    t = np.linspace(0.0, 1.0, H, dtype=np.float32)[:, None, None]
+    grad = top[None, None, :] * (1 - t) + bottom[None, None, :] * t
+    bg = np.repeat(grad, W, axis=1)
+
+    img = Image.fromarray(bg.astype(np.uint8), "RGB")
+    d = ImageDraw.Draw(img, "RGBA")
+
+    # Sun, upper-left.
+    d.ellipse([18, 12, 46, 40], fill=(255, 244, 200, 255))
+    d.ellipse([14, 8, 50, 44], outline=(255, 244, 200, 90), width=4)
+
+    # Static soft clouds built from overlapping ellipses.
+    for cx, cy, s in [(70, 26, 1.0), (205, 18, 0.8), (175, 44, 0.65)]:
+        for ox, oy, r in [(-14, 2, 11), (0, -3, 14), (15, 1, 10), (0, 5, 16)]:
+            r *= s
+            d.ellipse([cx + ox - r, cy + oy - r, cx + ox + r, cy + oy + r],
+                      fill=(255, 255, 255, 150))
+
+    # Grassy ground with a gentle vertical shade.
+    ground_top = 126
+    d.rectangle([0, ground_top, W, H], fill=(96, 168, 84, 255))
+    d.rectangle([0, ground_top, W, ground_top + 4], fill=(132, 196, 110, 255))
+    for x in range(0, W, 5):
+        h = int(rng.integers(2, 6))
+        d.line([(x, ground_top + 2), (x + int(rng.integers(-1, 2)), ground_top + 2 - h)],
+               fill=(72, 140, 64, 255))
+    return img
+
+
+def draw_tree(base):
+    """Trunk plus layered foliage canopy, centered near x=128."""
+    img = base.copy()
+    d = ImageDraw.Draw(img, "RGBA")
+    cx, ground = 128, 128
+
+    # Trunk with a darker shaded side and a few bark strokes.
+    d.polygon([(cx - 8, ground), (cx + 7, ground), (cx + 5, 66), (cx - 6, 66)],
+              fill=(112, 74, 45, 255))
+    d.polygon([(cx, ground), (cx + 7, ground), (cx + 5, 66), (cx, 66)],
+              fill=(86, 55, 33, 255))
+    for yy in (78, 94, 110):
+        d.line([(cx - 4, yy), (cx + 3, yy + 4)], fill=(72, 46, 28, 200), width=1)
+
+    # Short branches reaching into the canopy.
+    d.line([(cx - 4, 78), (cx - 20, 64)], fill=(96, 62, 38, 255), width=4)
+    d.line([(cx + 3, 80), (cx + 18, 66)], fill=(96, 62, 38, 255), width=4)
+
+    # Layered foliage: dark back layer, mid layer, light highlight layer.
+    canopy = [
+        (-24, 52, 22), (0, 44, 26), (24, 52, 22),
+        (-14, 30, 22), (12, 28, 23), (0, 18, 20),
+    ]
+    for ox, oy, r in canopy:
+        x, y = cx + ox, oy
+        d.ellipse([x - r, y - r, x + r, y + r], fill=(52, 122, 58, 255))
+    for ox, oy, r in [(-18, 48, 16), (4, 40, 18), (16, 46, 14), (-8, 28, 15), (8, 24, 13)]:
+        x, y = cx + ox, oy
+        d.ellipse([x - r, y - r, x + r, y + r], fill=(74, 152, 76, 255))
+    for ox, oy, r in [(-12, 42, 7), (6, 34, 8), (14, 42, 6), (-4, 22, 6)]:
+        x, y = cx + ox, oy
+        d.ellipse([x - r, y - r, x + r, y + r], fill=(112, 184, 104, 255))
+    return img
+
+
+def draw_bird(base, x, y, scale, flap, facing=1.0):
+    """Cartoon bluebird. flap in [-1, 1] drives the wing angle."""
+    img = base.copy()
+    d = ImageDraw.Draw(img, "RGBA")
+    s = scale
+    fx = facing
+
+    def poly(pts, color):
+        d.polygon([(x + px * s, y + py * s) for px, py in pts], fill=color)
+
+    def ell(dx, dy, rx, ry, color):
+        d.ellipse([x + (dx - rx) * s, y + (dy - ry) * s,
+                   x + (dx + rx) * s, y + (dy + ry) * s], fill=color)
+
+    # Tail feathers, behind the body.
+    poly([(-9 * fx, 0), (-20 * fx, -5), (-18 * fx, 1), (-20 * fx, 6)],
+         (40, 96, 168, 255))
+    poly([(-9 * fx, 2), (-19 * fx, 8), (-15 * fx, 3)], (30, 78, 140, 255))
+
+    # Body and lighter belly.
+    ell(0, 0, 11, 9, (66, 132, 214, 255))
+    ell(2 * fx, 3, 7, 5, (178, 214, 245, 255))
+
+    # Flapping wing: angle lifts with flap.
+    lift = flap * 8
+    poly([(-2 * fx, -2), (2 * fx, -3 - lift), (9 * fx, -7 - lift),
+          (7 * fx, -1 - lift * 0.4), (1 * fx, 2)],
+         (36, 88, 160, 255))
+    poly([(-1 * fx, -1), (3 * fx, -2 - lift), (8 * fx, -5 - lift)],
+         (58, 116, 190, 255))
+
+    # Head, cheek, beak (points in travel direction), and eye.
+    ell(8 * fx, -5, 7, 7, (74, 142, 222, 255))
+    ell(10 * fx, -3, 3, 2.5, (200, 226, 250, 255))
+    poly([(14 * fx, -6), (21 * fx, -4), (14 * fx, -2)], (240, 158, 54, 255))
+    d.ellipse([x + (10 * fx - 1.6) * s, y + (-7 - 1.6) * s,
+               x + (10 * fx + 1.6) * s, y + (-7 + 1.6) * s], fill=(20, 28, 40, 255))
+
+    # Small orange feet.
+    d.line([(x - 2 * s, y + 8 * s), (x - 4 * s, y + 11 * s)],
+           fill=(226, 140, 48, 255), width=max(1, int(s)))
+    d.line([(x + 3 * s, y + 8 * s), (x + 2 * s, y + 11 * s)],
+           fill=(226, 140, 48, 255), width=max(1, int(s)))
+    return img
+
+
+def ease_in_out(u):
+    return 0.5 - 0.5 * math.cos(math.pi * u)
+
+
+def main():
+    rng = np.random.default_rng(21)
+    background = make_background(rng)
+    scene = draw_tree(background)
+
+    # Flight path: start in front of the trunk, end above the canopy top.
+    start = np.array([128.0, 96.0])    # overlaps trunk -> clearly "in front"
+    end = np.array([170.0, 14.0])      # above the tree crown
+    frames = []
+    for i in range(N):
+        u = i / (N - 1)
+        e = ease_in_out(u)
+        pos = start + (end - start) * e
+        pos[1] += math.sin(u * math.pi * 2) * 2.5
+        # Hold the perch for the first two frames before lift-off.
+        if i <= 1:
+            pos = np.array(start)
+        flap = math.sin(u * math.pi * 6)
+        frame = draw_bird(scene, pos[0], pos[1], 1.0, flap, facing=1.0)
+        frames.append(np.asarray(frame, dtype=np.uint8))
+
+    frames = np.stack(frames, axis=0)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"Saved {frames.shape} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
