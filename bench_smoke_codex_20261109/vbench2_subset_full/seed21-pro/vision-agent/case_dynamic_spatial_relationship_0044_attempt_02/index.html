+"""Text-to-video: a bird above a tree flies to the left of the tree.
+
+Pure PIL/numpy procedural rendering. 20 frames, 256x144, 4 fps.
+"""
+import math
+from pathlib import Path
+
+import numpy as np
+from PIL import Image, ImageDraw
+
+import artifact_io
+
+W, H, N, FPS = 256, 144, 20, 4.0
+CASE_DIR = Path(__file__).parent
+
+# Fixed layout --------------------------------------------------------------
+GROUND_Y = 118
+TREE_X = 166
+CANOPY = (TREE_X, 60)
+
+BIRD_START = (165.0, 30.0)    # above the tree
+BIRD_END = (76.0, 74.0)       # clearly left of the canopy
+ARC_CTRL = (112.0, 20.0)      # flight arc bulges upward
+
+HOVER_FRAMES = 6
+FLY_FRAMES = 10               # frames 6..15
+
+
+def make_background(rng):
+    """Sky gradient, sun, clouds, ground and grass tufts (static scene)."""
+    img = Image.new("RGB", (W, H))
+    px = img.load()
+    top = np.array([122, 178, 232])
+    bottom = np.array([201, 229, 246])
+    for y in range(H):
+        t = y / (H - 1)
+        color = tuple((top * (1 - t) + bottom * t).astype(np.uint8))
+        for x in range(W):
+            px[x, y] = color
+    d = ImageDraw.Draw(img)
+
+    # Sun with soft halo.
+    for r, c in ((16, (255, 243, 205)), (12, (255, 236, 168)), (8, (255, 226, 130))):
+        d.ellipse([226 - r, 18 - r, 226 + r, 18 + r], fill=c)
+
+    # Two simple clouds.
+    for cx, cy, s in ((54, 26, 1.0), (118, 16, 0.75)):
+        for dx, dy, rr in ((-12, 2, 9), (0, -2, 11), (12, 2, 9), (0, 4, 10)):
+            r = rr * s
+            d.ellipse([cx + dx * s - r, cy + dy * s - r,
+                       cx + dx * s + r, cy + dy * s + r],
+                      fill=(245, 250, 255))
+
+    # Ground.
+    d.rectangle([0, GROUND_Y, W, H], fill=(96, 158, 78))
+    d.rectangle([0, GROUND_Y, W, GROUND_Y + 2], fill=(118, 178, 92))
+    for x, h in zip(rng.integers(0, W, 60), rng.integers(2, 6, 60)):
+        d.line([x, GROUND_Y + 3, x, GROUND_Y + 3 - h], fill=(74, 132, 60))
+    return img
+
+
+def draw_tree(base):
+    img = base.copy()
+    d = ImageDraw.Draw(img)
+    cx, cy = CANOPY
+
+    # Trunk and branches.
+    d.rounded_rectangle([cx - 6, 82, cx + 6, GROUND_Y + 2], 3,
+                        fill=(111, 74, 43))
+    d.line([cx - 3, 92, cx - 16, 78], fill=(111, 74, 43), width=4)
+    d.line([cx + 3, 88, cx + 15, 74], fill=(111, 74, 43), width=4)
+    d.line([cx - 3, 90, cx - 3, 86], fill=(139, 96, 58), width=2)
+
+    # Canopy: layered foliage blobs, dark back to light front.
+    blobs = [
+        (-18, 6, 20, (52, 118, 54)),
+        (18, 6, 20, (52, 118, 54)),
+        (0, -14, 22, (58, 132, 60)),
+        (-12, -6, 19, (70, 148, 68)),
+        (12, -4, 19, (70, 148, 68)),
+        (0, 4, 22, (82, 162, 76)),
+        (-6, -2, 12, (104, 182, 92)),
+        (8, 6, 10, (104, 182, 92)),
+    ]
+    for dx, dy, r, color in blobs:
+        d.ellipse([cx + dx - r, cy + dy - r, cx + dx + r, cy + dy + r],
+                  fill=color)
+    # A few leaf highlights.
+    for dx, dy in ((-14, -8), (4, -16), (14, 0), (-4, 10)):
+        d.ellipse([cx + dx - 2, cy + dy - 2, cx + dx + 2, cy + dy + 2],
+                  fill=(140, 204, 120))
+    return img
+
+
+def draw_bird(base, x, y, flap, facing=1):
+    """Local forward is +x. flap: -1 wing down, +1 wing up. facing: +/-1."""
+    img = base.copy()
+    d = ImageDraw.Draw(img)
+    s = 1.25
+    x, y = float(x), float(y)
+
+    def poly(points, color):
+        d.polygon([(x + facing * px * s, y + py * s) for px, py in points],
+                  fill=color)
+
+    def ellipse(px, py, rx, ry, color):
+        d.ellipse([x + facing * px * s - rx * s, y + py * s - ry * s,
+                   x + facing * px * s + rx * s, y + py * s + ry * s],
+                  fill=color)
+
+    body = (52, 101, 178)
+    dark = (36, 74, 138)
+    belly = (176, 205, 240)
+
+    # Tail (behind body).
+    poly([(-6, -2), (-13, -4), (-12, 1), (-6, 3)], dark)
+    # Body and belly.
+    ellipse(0, 0, 7, 5, body)
+    ellipse(-1, 2.2, 4.5, 2.2, belly)
+    # Head + beak + eye.
+    ellipse(6, -2.5, 4.2, 4.2, body)
+    poly([(9.5, -3), (14, -1.6), (9.5, -0.4)], (232, 158, 61))
+    ellipse(7.4, -3.6, 0.9, 0.9, (20, 26, 38))
+
+    # Flapping wing: shoulder fixed, tip sweeps with flap.
+    tip_y = -3 - 7 * flap
+    poly([(-2, -2), (3, -2), (4, tip_y), (-1, tip_y + 2)], dark)
+    poly([(-1, -1.6), (2.4, -1.6), (3, tip_y + 0.8), (-0.6, tip_y + 2.2)],
+         (86, 138, 208))
+    return img
+
+
+def bezier(u):
+    p0, p1, p2 = BIRD_START, ARC_CTRL, BIRD_END
+    a = (1 - u) ** 2
+    b = 2 * (1 - u) * u
+    c = u ** 2
+    return (a * p0[0] + b * p1[0] + c * p2[0],
+            a * p0[1] + b * p1[1] + c * p2[1])
+
+
+def ease(u):
+    return 0.5 - 0.5 * math.cos(math.pi * u)
+
+
+def main():
+    rng = np.random.default_rng(7)
+    scene = draw_tree(make_background(rng))
+    frames = []
+    for i in range(N):
+        if i < HOVER_FRAMES:
+            # Hover above the tree with a gentle bob and slow wing beat.
+            x, y = BIRD_START
+            y += math.sin(i * math.pi / 2) * 1.6
+            flap = math.sin(i * math.pi / 2)
+        elif i < HOVER_FRAMES + FLY_FRAMES:
+            u = ease((i - HOVER_FRAMES) / (FLY_FRAMES - 1))
+            x, y = bezier(u)
+            # Quick wing beats during flight.
+            flap = math.sin((i - HOVER_FRAMES) * 1.7 * math.pi)
+        else:
+            x, y = BIRD_END
+            y += math.sin((i - HOVER_FRAMES - FLY_FRAMES) * math.pi / 2) * 1.2
+            flap = math.sin((i - HOVER_FRAMES - FLY_FRAMES) * math.pi / 2)
+        frame = draw_bird(scene, x, y, flap, facing=-1)
+        frames.append(np.asarray(frame, dtype=np.uint8))
+
+    frames = np.stack(frames)
+    artifact_io.save(frames, CASE_DIR)
+    print(f"saved {frames.shape} frames to {CASE_DIR}")
+
+
+if __name__ == "__main__":
+    main()
