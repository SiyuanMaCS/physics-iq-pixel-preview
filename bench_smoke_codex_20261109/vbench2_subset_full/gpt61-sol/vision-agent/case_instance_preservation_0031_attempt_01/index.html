def bezier(points,steps=20):
    result=[points[0]]
    p=points[0]
    for i in range(1,len(points),3):
        a,b,c=points[i:i+3]
        for t in np.linspace(0,1,steps)[1:]:
            q=(1-t)**3*np.array(p)+3*(1-t)**2*t*np.array(a)+3*(1-t)*t*t*np.array(b)+t**3*np.array(c)
            result.append(tuple(q))
        p=c
    return result

class Painter:
    def __init__(self,im): self.im=im; self.d=ImageDraw.Draw(im)
    def poly(self,ps,c): self.d.polygon([(round(x*S),round(y*S)) for x,y in ps],fill=c)
    def path(self,ps,c): self.poly(bezier(ps),c)
    def ellipse(self,bb,c): self.d.ellipse(tuple(round(x*S) for x in bb),fill=c)
    def line(self,ps,c,w=1): self.d.line([(round(x*S),round(y*S)) for x,y in ps],fill=c,width=max(1,round(w*S)),joint='curve')
    def capsule(self,a,b,ra,rb,c):
        a=np.array(a); b=np.array(b); delta=b-a; n=np.array([-delta[1],delta[0]])/np.linalg.norm(delta)
        self.poly([a+n*ra,b+n*rb,b-n*rb,a-n*ra],c)
        self.ellipse([a[0]-ra,a[1]-ra,a[0]+ra,a[1]+ra],c)
        self.ellipse([b[0]-rb,b[1]-rb,b[0]+rb,b[1]+rb],c)
