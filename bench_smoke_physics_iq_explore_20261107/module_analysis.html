def components(frame, hue, saturation, minimum_value, roi=None):
    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
    x0, y0, x1, y1 = roi or (0, 0, 256, 144)
    mask = cv2.inRange(
        hsv[y0:y1, x0:x1],
        np.array([hue[0], saturation, minimum_value]),
        np.array([hue[1], 255, 255]),
    )
    count, labels, stats, centres = cv2.connectedComponentsWithStats(mask)
    result = []
    for j in range(1, count):
        if stats[j, cv2.CC_STAT_AREA] < 5:
            continue
        contour, _ = cv2.findContours(
            np.uint8(labels == j) * 255,
            cv2.RETR_EXTERNAL,
            cv2.CHAIN_APPROX_NONE,
        )
        contour = max(contour, key=cv2.contourArea)
        ellipse_centre = None
        if len(contour) >= 5:
            ellipse_centre = (
                np.array(cv2.fitEllipse(contour)[0]) + [x0, y0]
            ).tolist()
        result.append({
            "area_pixels": int(stats[j, cv2.CC_STAT_AREA]),
            "centre_xy": (centres[j] + [x0, y0]).tolist(),
            "bounding_box_xywh": [
                int(stats[j, 0] + x0), int(stats[j, 1] + y0),
                int(stats[j, 2]), int(stats[j, 3]),
            ],
            "ellipse_centre_xy": ellipse_centre,
        })
    return sorted(result, key=lambda item: item["area_pixels"], reverse=True)

"""Measure only the supplied conditioning history and write state.json.

Run with local OpenCV and NumPy: python estimate_state.py
Coordinates are original-frame pixel coordinates, x right and y down.
"""
import csv
import json
from pathlib import Path

import cv2
import numpy as np


ROOT = Path(__file__).resolve().parent
INFO = json.loads((ROOT / "conditioning_info.json").read_text())
FPS = float(INFO["history_fps"])
LAST = int(INFO["history_frames"]) - 1
LAST_TIME = float(INFO["last_frame_time_seconds"])
THRESHOLDS = {
    "blue": ((90, 80, 35), (130, 255, 255)),
    "yellow": ((25, 70, 75), (50, 255, 255)),
}


def measure(frame_index, name, saturation=None, value=None):
    im = cv2.imread(str(ROOT / "conditioning_frames" / f"{frame_index:02d}.png"))
    assert im is not None and im.shape[:2] == (144, 256)
    hsv = cv2.cvtColor(im, cv2.COLOR_BGR2HSV)
    lower, upper = THRESHOLDS[name]
    lower = list(lower)
    if saturation is not None:
        lower[1] = saturation
    if value is not None:
        lower[2] = value
    mask = cv2.inRange(hsv, np.array(lower), np.array(upper))
    # Visually identified static background: upper yellow strip and right tube.
    # All frames used for fitting contain fully visible balls outside these ROIs.
    mask[:15, :] = 0
    mask[:, 240:] = 0
    count, labels, stats, centres = cv2.connectedComponentsWithStats(mask)
    if count < 2:
        return None
    component = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
    if stats[component, cv2.CC_STAT_AREA] < 10:
        return None
    contour = cv2.findContours(
        (labels == component).astype(np.uint8),
        cv2.RETR_EXTERNAL,
        cv2.CHAIN_APPROX_NONE,
    )[0][0]
    ellipse_centre = cv2.fitEllipse(contour)[0] if len(contour) >= 5 else centres[component]
    return {
        "frame_index": frame_index,
        "time_seconds": float(INFO["history_start_seconds"]) + frame_index / FPS,
        "centre_xy": centres[component].tolist(),
        "area_pixels": int(stats[component, cv2.CC_STAT_AREA]),
        "ellipse_centre_xy": list(ellipse_centre),
    }


def fit(rows):
    t = np.array([r["time_seconds"] - LAST_TIME for r in rows])
    xy = np.array([r["centre_xy"] for r in rows])
    linear_design = np.column_stack([np.ones(len(t)), t])
    quadratic_design = np.column_stack([np.ones(len(t)), t, t * t / 2])
    linear = np.linalg.lstsq(linear_design, xy, rcond=None)[0]
    quadratic = np.linalg.lstsq(quadratic_design, xy, rcond=None)[0]
    quadratic_error = xy - quadratic_design @ quadratic
    standard_errors = (
        np.sqrt(np.diag(np.linalg.inv(quadratic_design.T @ quadratic_design)))[:, None]
        * np.sqrt(np.sum(quadratic_error ** 2, axis=0) / (len(t) - 3))
    )
    return {
        "first_frame_index": rows[0]["frame_index"],
        "last_frame_index": rows[-1]["frame_index"],
        "time_window_seconds": [rows[0]["time_seconds"], rows[-1]["time_seconds"]],
        "number_of_observations": len(rows),
        "linear_velocity_xy_pixels_per_second": linear[1].tolist(),
        "linear_residual_rms_xy_pixels": np.sqrt(
            np.mean((xy - linear_design @ linear) ** 2, axis=0)
        ).tolist(),
        "quadratic_final_velocity_xy_pixels_per_second": quadratic[1].tolist(),
        "quadratic_acceleration_xy_pixels_per_second_squared": quadratic[2].tolist(),
        "quadratic_acceleration_standard_error_xy": standard_errors[2].tolist(),
        "quadratic_residual_rms_xy_pixels": np.sqrt(
            np.mean(quadratic_error ** 2, axis=0)
        ).tolist(),
    }


states = []
csv_rows = []
for name in THRESHOLDS:
    observations = [measure(i, name) for i in range(79, LAST + 1)]
    assert all(row is not None for row in observations)
    recent = [row for row in observations if row["frame_index"] >= 83]
    primary = fit(recent)
    comparison_starts = [81, 82, 84, 85]
    comparisons = [fit([r for r in observations if r["frame_index"] >= start])
                   for start in comparison_starts]
    final = observations[-1]
    velocity = primary["linear_velocity_xy_pixels_per_second"].copy()
    if name == "yellow":
        # Horizontal slowing is stable under different windows; vertical
        # acceleration changes sign and is therefore not used in the estimate.
        velocity[0] = primary["quadratic_final_velocity_xy_pixels_per_second"][0]
    sensitivity = []
    values = [25, 35, 50, 70] if name == "blue" else [60, 75, 100, 120]
    for saturation in [60, 80, 100, 120]:
        for value in values:
            sensitivity.append(measure(LAST, name, saturation, value)["centre_xy"])
    sensitivity = np.array(sensitivity)
    if name == "blue":
        assessment = (
            "No robust acceleration estimate. The primary quadratic fit hints at "
            "a small positive horizontal acceleration, but short-window estimates "
            "are sensitive to window length and centroid noise. Use the recent "
            "linear velocity; vertical acceleration is unresolved."
        )
        accel_estimate = None
        centre_error = [0.8, 0.8]
        velocity_error = [6.0, 5.0]
        confidence = {"overall": 0.94, "centre": 0.98, "velocity": 0.92}
    else:
        assessment = (
            "The negative horizontal velocity becomes less negative during the "
            "observed history. Quadratic fits starting at frames 81 through 85 "
            "give horizontal acceleration about +155 to +209 pixels/s^2 and "
            "final horizontal velocity about -240 to -244 pixels/s. The primary "
            "estimate is +194 pixels/s^2, with an approximate practical uncertainty "
            "of +/-80 pixels/s^2. Vertical acceleration is unresolved: fitted "
            "values change sign with the window and vertical residuals are larger. "
            "Acceleration describes image-plane motion only; it does not identify "
            "physical force or distinguish perspective from physical deceleration."
        )
        accel_estimate = [round(primary["quadratic_acceleration_xy_pixels_per_second_squared"][0], 1), None]
        centre_error = [1.0, 1.0]
        velocity_error = [15.0, 12.0]
        confidence = {"overall": 0.88, "centre": 0.97, "velocity": 0.85}
    states.append({
        "name": name,
        "last_time_seconds": LAST_TIME,
        "centre_xy": [round(x, 2) for x in final["centre_xy"]],
        "velocity_xy_pixels_per_second": [round(x, 2) for x in velocity],
        "confidence": {
            **confidence,
            "interpretation": "Subjective reliability scores from 0 to 1; not calibrated probabilities.",
        },
        "evidence": {
            "sources": ["conditioning_info.json", "conditioning_frames/79.png through 89.png", "conditioning.mp4 metadata only"],
            "image_size_wh": [256, 144],
            "coordinate_convention": "Origin at upper-left pixel centre; x increases right, y increases down. Measurements use original frames, not contact-sheet coordinates.",
            "timing_method": "Frame time = history_start_seconds + frame_index/history_fps; 30 fps, 90 frames. Frame 89 is the last observation at 89/30 seconds, not 3.0 seconds.",
            "centre_method": "OpenCV HSV color threshold, exclude static top strip (y<15) and right tube (x>=240), choose largest remaining connected color component, then use its unweighted pixel centroid. Shadows are not part of the color mask. Visual inspection confirms both final balls are fully visible.",
            "hsv_threshold_lower_upper": [list(x) for x in THRESHOLDS[name]],
            "last_frame": "conditioning_frames/89.png",
            "last_component_area_pixels": final["area_pixels"],
            "last_contour_ellipse_centre_xy": final["ellipse_centre_xy"],
            "velocity_method": (
                "Unweighted least-squares line independently fitted to x and y over frames 83-89 (2.7667-2.9667 s). Report these recent slopes as the velocity at the final observation; possible small acceleration is included in uncertainty."
                if name == "blue" else
                "Unweighted least-squares quadratic x(t)=x_last+vx_last*(t-t_last)+0.5*ax*(t-t_last)^2 over frames 83-89. Report its derivative at the last time for vx; report the same window's linear y slope for vy because vertical acceleration is unstable."
            ),
            "primary_fit": primary,
            "recent_observations": recent,
            "alternative_window_fits": comparisons,
            "acceleration_assessment": assessment,
        },
        "uncertainty": {
            "centre_xy_approximate_plus_minus_pixels": centre_error,
            "velocity_xy_approximate_plus_minus_pixels_per_second": velocity_error,
            "bounds_interpretation": "Practical approximate error allowances, not formal confidence intervals. Include segmentation, small ball size, temporal sampling, fit-window sensitivity and model choice; fit residuals alone understate these errors.",
            "last_centroid_threshold_sensitivity_min_xy": sensitivity.min(axis=0).tolist(),
            "last_centroid_threshold_sensitivity_max_xy": sensitivity.max(axis=0).tolist(),
            "acceleration_xy_pixels_per_second_squared": accel_estimate,
            "acceleration_assessment": assessment,
            "limitations": "Only 0.2 seconds (7 fully visible observations) enter the primary fit. Earlier entry frames can be clipped or occluded by launch tubes and are excluded from the primary fit. No depth, world-unit velocity or future trajectory is inferred. Only the supplied conditioning history is read.",
        },
    })
    for row in observations:
        csv_rows.append([name, row["frame_index"], row["time_seconds"], *row["centre_xy"], row["area_pixels"]])

(ROOT / "state.json").write_text(json.dumps(states, indent=2, allow_nan=False) + "\n")
(ROOT / ".tmp").mkdir(exist_ok=True)
with (ROOT / ".tmp" / "ball_measurements.csv").open("w", newline="") as output:
    writer = csv.writer(output)

    writer.writerows(csv_rows)
