+def smoothstep(edge0, edge1, value):
+    """Vectorized smooth transition from zero to one."""
+    x = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return x * x * (3.0 - 2.0 * x)

+def clean_original_ball(frame):
+    """Create a fixed tabletop plate by locally inpainting the frame-00 ball.
+
+    A Jacobi harmonic fill supplies the broad table color. Fine residual grain is
+    sampled from the same scanlines 34 pixels to the left, then gently tapered so
+    the fill agrees exactly with its boundary.
+    """
+    src = frame.astype(np.float32)
+    yy, xx = np.mgrid[0:H, 0:W]
+    # Covers the blurred ball and its small contact region, but no other object.
+    radius = ((xx - BALL_X0) / 14.5) ** 2 + ((yy - 49.0) / 12.8) ** 2
+    core = radius < 1.0
+
+    work = src.copy()
+    # Initialize from a left/right interpolation on each scanline.
+    for y in range(35, 64):
+        left = src[y, 88:92].mean(axis=0)
+        right = src[y, 118:122].mean(axis=0)
+        for x in range(90, 120):
+            if core[y, x]:
+                q = (x - 90.0) / 29.0
+                work[y, x] = left * (1.0 - q) + right * q
+
+    # Harmonic relaxation produces a seamless low-frequency table surface.
+    for _ in range(240):
+        avg = (
+            np.roll(work, 1, axis=0)
+            + np.roll(work, -1, axis=0)
+            + np.roll(work, 1, axis=1)
+            + np.roll(work, -1, axis=1)
+        ) * 0.25
+        work[core] = avg[core]
+
+    # Restore restrained camera/wood microtexture from nearby unobstructed pixels.
+    # Residual means local detail after a 5-pixel horizontal box estimate.
+    left_detail = np.zeros_like(src)
+    for x in range(2, W - 2):
+        left_detail[:, x] = src[:, x] - src[:, x - 2 : x + 3].mean(axis=1)
+    texture = np.zeros_like(src)
+    valid_y, valid_x = np.where(core)
+    texture[valid_y, valid_x] = left_detail[valid_y, np.maximum(valid_x - 34, 2)]
+    interior_weight = smoothstep(0.99, 0.38, radius)[..., None]
+    work += texture * interior_weight * 0.72
+
+    # Feather only the outside three pixels; the source remains untouched beyond.
+    blend = smoothstep(1.08, 0.86, radius)[..., None]
+    clean = src * (1.0 - blend) + work * blend
+    return np.clip(np.rint(clean), 0, 255).astype(np.uint8)

METHOD.md

Method

The sequence is a deterministic local pixel construction from first_frame.png; no network access, reference media, external model, or external rendering engine was used.

  1. Frame 00 is copied directly from the supplied 256 x 144 RGB image.
  2. A fixed background plate is made by masking the original ball region, solving a local harmonic (Jacobi) fill from its real tabletop boundary, and adding low-amplitude scanline texture residuals sampled from a nearby unobstructed part of the same frame.
  3. The photographed ball is extracted as a softly feathered elliptical RGB sprite. For frames 01–19 its center advances exactly 8 pixels per frame from x=112 to x=256 at y=49.
  4. A small blurred contact shadow follows the ball. A faint analytic curved seam changes angle each frame to communicate rolling while leaving the scene lighting stationary.
  5. The tube, lampshade, camera, exposure, and all remaining pixels use the same fixed background throughout.
  6. artifact_io.save writes the 20 PNGs, packed uint32 0xRRGGBB tensor, contact sheet, and standalone player. transition_check.png contains three before/after/delta triplets. The generator then reloads every PNG and the tensor and asserts exact equivalence, including exact equality of frame 00 to the supplied image.

Later generated images were assessed only by code, array statistics, and numeric checks, not with an image viewing tool.
