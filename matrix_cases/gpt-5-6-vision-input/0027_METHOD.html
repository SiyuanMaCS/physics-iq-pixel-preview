+def largest_center_component(mask: np.ndarray, center_xy: tuple[int, int]) -> np.ndarray:
+    """Keep the connected component containing the supplied pixel."""
+    count, labels, _, _ = cv2.connectedComponentsWithStats(mask.astype(np.uint8), 8)
+    x, y = center_xy
+    label = int(labels[y, x])
+    if label == 0 or label >= count:
+        raise RuntimeError("Ball center did not fall inside the chroma mask")
+    return labels == label
+
+
+def make_clean_plate(source: np.ndarray) -> np.ndarray:
+    """Remove the original ball and its compact shadow using local inpainting."""
+    yy, xx = np.mgrid[0:H, 0:W]
+    # Covers the 20 px ball plus its shallow contact shadow, while staying well
+    # inside the wooden top so table boundaries are untouched.
+    ball = ((xx - 82.0) / 12.3) ** 2 + ((yy - 58.0) / 12.0) ** 2 <= 1.0
+    contact = ((xx - 82.0) / 13.5) ** 2 + ((yy - 68.0) / 4.0) ** 2 <= 1.0
+    removal = np.where(ball | contact, 255, 0).astype(np.uint8)
+    # Navier-Stokes inpainting extends the first frame's own fine tabletop
+    # gradients inward; no outside image or synthesized reference is used.
+    return cv2.inpaint(source, removal, 5.0, cv2.INPAINT_NS)
+
+
+def extract_ball(source: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
+    """Extract real ball RGB and a soft silhouette from the first frame."""
+    x0, y0, x1, y1 = 68, 44, 97, 73
+    crop = source[y0:y1, x0:x1].copy()
+    ch_r = crop[:, :, 0].astype(np.int16)
+    ch_g = crop[:, :, 1].astype(np.int16)
+    ch_b = crop[:, :, 2].astype(np.int16)
+    yy, xx = np.mgrid[0:crop.shape[0], 0:crop.shape[1]]
+    radial = (xx - 14.0) ** 2 + (yy - 14.0) ** 2 <= 12.0**2
+    chroma = (ch_b - ch_r > 7) & (ch_b - ch_g > 1) & (ch_b > 54) & radial
+    component = largest_center_component(chroma, (14, 14)).astype(np.uint8)
+    component = cv2.morphologyEx(
+        component, cv2.MORPH_CLOSE, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3))
+    )
+    # One fractional pixel of optical softness matches the small source object.
+    alpha = cv2.GaussianBlur(component.astype(np.float32), (0, 0), 0.70)
+    alpha = np.clip(alpha / max(float(alpha.max()), 1e-6), 0.0, 1.0)
+    return crop, alpha

METHOD.md

Method

The continuation is a deterministic local pixel construction based only on first_frame.png and the task prompt.

  1. The supplied frame is loaded as RGB and retained unchanged as frame 00.
  2. A clean static plate is made by masking the original 20 px blue ball and its compact contact shadow, then extending the neighboring tabletop pixels into that mask with OpenCV Navier-Stokes inpainting. This affects only the original ball footprint; scene boundaries, the glass, legs, wall, and right-side furniture are copied from the source.
  3. The actual ball RGB texture is cropped from frame 00. Its silhouette is derived from local blue dominance, restricted to the connected component containing the measured ball center, morphologically closed, and feathered by 0.7 px.
  4. For frames 01-19, that photographed ball texture is rotated according to cumulative rolling distance (radius 9.8 px) and translated through the measured centers listed in OBSERVATION.md. A very small perspective scale reduction begins on the glass.
  5. A compact analytic elliptical support shadow follows the ball. It becomes lighter and narrower on glass, where a subtle cool contact reflection is also added. These cues are drawn before the ball so the body correctly occludes them.
  6. artifact_io.save writes the packed uint32 tensor, 20 PNGs, inspection contact sheet, and standalone HTML player. A separate transition sheet contains amplified consecutive-frame absolute differences for diagnostics.

No network resource, external image, external model, rendering engine, or later-frame visual inspection is used. Validation is performed with local numeric scripts: exact frame-00 equality, tensor/PNG equality, connected-component centroid tracking, static-region identity, changed-pixel bounds, temporal difference statistics, and file/dimension checks.

Numeric validation results

  • Frame 00 is pixel-exact to first_frame.png (maximum channel error 0).
  • All 20 decoded RGB PNGs exactly match the uint32 0xRRGGBB tensor; independent decoding of the HTML player's first-frame/delta stream also matches every packed pixel.
  • All images are RGB 256 x 144, all 20 frames are unique, and deterministic regeneration is exact.
  • A strict blue connected-component measurement tracks every planned ball center with maximum error 0.793 px and mean error 0.737 px.
  • The measured blue-ball component stays between 302 and 319 pixels (maximum/minimum ratio 1.056), so scale and visibility remain stable.
  • Measured horizontal motion is positive at every transition, from x=81.51 to x=239.05.
  • Outside conservative envelopes around the removed source footprint, moving ball, shadow, and reflection, later frames have zero changed pixels relative to the clean locked plate.
