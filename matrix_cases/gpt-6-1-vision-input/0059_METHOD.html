+def harmonic_fill(source, holes):
+    """Solve a small local discrete Laplace problem, preserving all other pixels."""
+    yy, xx = np.where(holes)
+    labels = np.full((H, W), -1, dtype=int)
+    labels[yy, xx] = np.arange(len(yy))
+    rows, columns, entries = [], [], []
+    rhs = np.zeros((len(yy), 3), dtype=float)
+    for i, (y, x) in enumerate(zip(yy, xx)):
+        neighbors = [(y-1, x), (y+1, x), (y, x-1), (y, x+1)]
+        neighbors = [(v, u) for v, u in neighbors if 0 <= v < H and 0 <= u < W]
+        rows.append(i)
+        columns.append(i)
+        entries.append(len(neighbors))
+        for v, u in neighbors:
+            if holes[v, u]:
+                rows.append(i)
+                columns.append(labels[v, u])
+                entries.append(-1.0)
