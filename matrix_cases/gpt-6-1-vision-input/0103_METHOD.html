+def rail(name, nodes, speed):
+    """Energy-based rolling on a sampled projected contact curve."""
+    global clock
+    nodes = np.asarray(nodes, dtype=float)
+    distances = np.r_[0.0, np.cumsum(np.linalg.norm(np.diff(nodes, axis=0), axis=1))]
+    u = np.linspace(0.0, distances[-1], 1201)
+    xy = np.stack([PchipInterpolator(distances, nodes[:, j])(u) for j in range(2)], axis=1)
+    s = np.r_[0.0, np.cumsum(np.linalg.norm(np.diff(xy, axis=0), axis=1))]
+    # Translational + rotational energy changes with height along the rail.
+    v = np.sqrt(np.maximum(speed**2 + 2 * G * ROLL * (xy[:, 1] - xy[0, 1]), 1.0))
+    times = cumulative_trapezoid(1.0 / v, s, initial=0.0)
+    duration = float(times[-1])
+    tangent = (xy[-1] - xy[-2]) / np.linalg.norm(xy[-1] - xy[-2])
+    velocity = tangent * v[-1]
+    stage = dict(name=name, kind='contact', start=clock, end=clock + duration,
+                 times=times, points=xy, initial_speed=float(speed), final_speed=float(v[-1]))
+    stages.append(stage)
+    clock += duration
+    return xy[-1].copy(), velocity.copy()
+
+
+def fall(name, position, velocity, receiver=None, duration=None):
+    """Unconstrained constant-gravity flight up to a receiving rail."""
+    global clock
+    position, velocity = np.asarray(position), np.asarray(velocity)
+    if receiver is not None:
+        # receiver top edge: y = intercept + slope*x. Offset by marble radius.
+        slope, intercept, xmin, xmax = receiver
+        c = position[1] - (intercept + slope * position[0] - RADIUS)
+        b = velocity[1] - slope * velocity[0]
+        roots = np.roots([0.5 * G, b, c])
+        candidates = [float(t.real) for t in roots if abs(t.imag) < 1e-9 and t.real > 1e-5]
+        duration = min(candidates)
+        x = position[0] + velocity[0] * duration
+        assert xmin - 0.1 <= x <= xmax + 0.1, (name, x, receiver)
+    stage = dict(name=name, kind='flight', start=clock, end=clock + duration,
+                 position=position, velocity=velocity)
+    stages.append(stage)
+    clock += duration
+    end_position = position + velocity * duration + np.array([0.0, G * duration**2 / 2.0])
+    end_velocity = velocity + np.array([0.0, G * duration])
+    return end_position, end_velocity

METHOD.md

Method and verification

This is a deterministic local Python pixel animation based only on the supplied photograph and prompt. The initial photograph was available visually in the user message. All generated outputs were inspected using local code and numerical checks; no generated image was opened with an image Read/View tool, and no visual review of the generated frames is claimed.

Pixel generation

generate.py keeps the original 256 × 144 RGB photograph as the background. Frame 00 is copied byte-for-byte at the decoded RGB pixel level. In later frames, a 52-pixel elliptical region covering the original marble and its immediate halo is reconstructed using a locally fitted whiteboard color plane. A shallow curved ramp-edge model retains the green edge immediately below that region. All other background pixels remain the original photograph except where the moving marble and its small soft shadow overlap them.

The moving marble has a fixed radius of 2.05 pixels, calibrated to the supplied amber-yellow bead. It is drawn by an analytic sphere shader with amber-tinted background transmission, directional shading, a small reflection, a rotating internal dark variation, and a soft photographic boundary. Sixfold spatial supersampling is reduced to the native frame resolution. Three samples over a short exposure introduce a small amount of natural motion blur. The sphere and shadow are composited directly into RGB pixels; there are no generated replacement scenes or external assets.

Motion model

The ramp guide paths are measured manually in image coordinates. Contact travel follows a sampled continuous curve with speed determined by gravitational energy and a 5/7 rolling-sphere acceleration factor. Free transfers use the constant-acceleration equations with g = 600 image pixels/s², and flight ends at an analytically calculated intersection with a receiving guide. Each receiving ramp prescribes a reduced capture speed and the guide's travel direction, approximating dissipation and redirection in the channels. The magenta bend uses a projected curved guide through its visible lower portion.

This is a constrained 2D projected animation, not a recovered 3D geometry or a general rigid-body collision simulation. The physical size, true depth, and friction of the photographed setup cannot be measured from this one input. The effective gravity and capture speeds are local animation parameters. No external AI model, reference image, physics engine, network connection, or subagent was used.

The marble crosses upper green, purple, magenta, blue, lower green, and yellow guides in that order. It leaves the lowest guide at approximately 3.890 s and is fully outside the image by the 4.25 s sample. Samples 17–19 show the empty static setup. The 20 samples are at 0.00, 0.25, …, 4.75 s and each is held for 0.25 s in the 4 fps, 5-second player.
