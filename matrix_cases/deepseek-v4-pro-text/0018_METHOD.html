+# Sample background left of ball and right of ball for gradient
+def estimate_bg_gradient(src, y0, y1, x0, x1):
+    """Estimate per-pixel background by interpolating left/right edge samples."""
+    bg = np.zeros_like(src[y0:y1, x0:x1])
+    Hr = y1 - y0
+    Wr = x1 - x0
+    for dy in range(Hr):
+        gy = y0 + dy
+        # Left edge: take 5-pixel strip well left of ball
+        left_strip = src[gy, max(0, x0-15):x0]
+        # Right edge: take 5-pixel strip well right of ball
+        right_strip = src[gy, x1:x1+15]
+        if len(left_strip) > 0 and len(right_strip) > 0:
+            left_color = left_strip.mean(axis=0)
+            right_color = right_strip.mean(axis=0)
+        elif len(left_strip) > 0:
+            left_color = left_strip.mean(axis=0)
+            right_color = left_color
+        elif len(right_strip) > 0:
+            right_color = right_strip.mean(axis=0)
+            left_color = right_color
+        else:
+            left_color = src[gy, 50:70].mean(axis=0)
+            right_color = src[gy, 150:170].mean(axis=0)
+        # Linear gradient
+        for dx in range(Wr):
+            t = dx / max(Wr - 1, 1)
+            bg[dy, dx] = left_color * (1 - t) + right_color * t
+    return bg
+
+bg_est = estimate_bg_gradient(src, ball_y0, ball_y1, ball_x0, ball_x1)
+ball_region = src[ball_y0:ball_y1, ball_x0:ball_x1].copy()

+# ── 2. Create ball mask via difference from estimated background ───────────
+diff = ball_region - bg_est
+# Ball is bluer than background: B channel elevated
+blue_excess = diff[:, :, 2]  # B channel difference
+# Also the ball is darker in R and G
