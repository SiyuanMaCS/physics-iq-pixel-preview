+#!/usr/bin/env python3
+"""generate.py — 20-frame physics animation: grabber releases yellow mug over concrete brick.
+
+Method: Pixel-level sprite extraction and compositing. The yellow ceramic mug is identified
+by its warm yellowness signature (high R, low B). The sprite is cut from frame 0, the
+background hole is inpainted via horizontal bilinear interpolation from the smooth projection-
+screen gradient, and the sprite is re-composited at downward-shifted positions following a
+simple kinematic free-fall trajectory with a damped bounce on impact.
+
+Frame 00 is an exact byte-for-byte copy of first_frame.png. All pixel operations are
+local numpy/PIL; no external models or networks are used.
+"""
+
+import numpy as np
+from PIL import Image
+from pathlib import Path
+import sys
+
+# ── paths ──────────────────────────────────────────────────────────────────
+CASE_DIR = Path(__file__).resolve().parent
+FIRST_FRAME = CASE_DIR / "first_frame.png"
+TMP = CASE_DIR / ".tmp"
+TMP.mkdir(exist_ok=True)
+
+# ── load first frame ───────────────────────────────────────────────────────
+src = Image.open(FIRST_FRAME)
+arr0 = np.array(src).astype(np.uint8)
+H, W = arr0.shape[:2]  # 144, 256
+assert arr0.shape == (144, 256, 3), f"Unexpected shape: {arr0.shape}"
+
+# ── mug extraction ─────────────────────────────────────────────────────────
+# The yellow ceramic mug has a strong warm signature: (R+G)/2 - B >> 0.
+yellowness = (arr0[:,:,0].astype(np.float64) + arr0[:,:,1].astype(np.float64)) / 2.0 - arr0[:,:,2].astype(np.float64)
+
+# Mug bounding box (determined via connected-components on yellowness > 15)
+MUG_Y0, MUG_Y1 = 47, 66   # global rows (inclusive)
+MUG_X0, MUG_X1 = 115, 133 # global cols (inclusive)
+MUG_H = MUG_Y1 - MUG_Y0 + 1  # 20
+MUG_W = MUG_X1 - MUG_X0 + 1  # 19
+MUG_CENTER_Y = (MUG_Y0 + MUG_Y1) / 2.0  # 56.5
+MUG_CENTER_X = (MUG_X0 + MUG_X1) / 2.0  # 124.0
+
+# Alpha mask for the mug sprite: yellowness > 8 within the bounding box.
+# This captures the mug body plus some darker edges while excluding pure background.
+mug_yellowness_roi = yellowness[MUG_Y0:MUG_Y1+1, MUG_X0:MUG_X1+1]
+mug_alpha = mug_yellowness_roi > 8.0
+
+# Extract mug sprite pixels
+mug_sprite = arr0[MUG_Y0:MUG_Y1+1, MUG_X0:MUG_X1+1].copy()
+
+print(f"Mug sprite: {MUG_H}x{MUG_W}, alpha pixels={mug_alpha.sum()}", flush=True)
+
+# ── background inpainting for the mug hole ─────────────────────────────────
+# The background behind the mug is the projection screen: a smooth blue-gray
+# vertical gradient. We estimate it by interpolating horizontally between
+# sample columns just left and right of the mug, per row.
+#
+# Also compute a full "clean background" canvas that equals frame 0 everywhere
+# except in the mug hole, where it's filled with interpolated background.
+
+bg_canvas = arr0.copy().astype(np.float64)
+
+LEFT_SAMPLE_COL = MUG_X0 - 8   # col 107
+RIGHT_SAMPLE_COL = MUG_X1 + 8  # col 141
+
+for dy in range(MUG_H):
+    gy = MUG_Y0 + dy
+    # Sample background from left and right of the mug
+    left_pix = arr0[gy, LEFT_SAMPLE_COL].astype(np.float64)
+    right_pix = arr0[gy, RIGHT_SAMPLE_COL].astype(np.float64)
+    
+    for dx in range(MUG_W):
+        gx = MUG_X0 + dx
+        if mug_alpha[dy, dx]:
+            # Linear interpolation weighted by horizontal position
+            t = (gx - LEFT_SAMPLE_COL) / (RIGHT_SAMPLE_COL - LEFT_SAMPLE_COL)
+            bg_color = left_pix * (1 - t) + right_pix * t
+            bg_canvas[gy, gx] = bg_color
+
+# Round to uint8
+bg_canvas_uint8 = np.clip(np.round(bg_canvas), 0, 255).astype(np.uint8)
+

+# Simple kinematic model: constant gravitational acceleration with damped bounce.
+# Pixel units; gravity chosen so mug falls ~44 pixels (bottom 66→110) in ~6 frames.
+

METHOD.md

METHOD.md — Generation Approach

Method summary

All frames are generated via local pixel-level sprite extraction and compositing. No external models, networks, or engines are used.

Step-by-step

1. Mug sprite extraction

The yellow ceramic mug is identified by its warm color signature: yellowness = (R+G)/2 − B. In frame 0, the mug occupies rows 47–66, cols 115–133 (20×19 px). An alpha mask is computed from yellowness > 8 within this bounding box, yielding 273 opaque pixels. The sprite includes the mug body, its ceramic texture, and partial grabber-jaw overlap at the top edges.

2. Background inpainting

The hole left by the mug is filled by horizontal linear interpolation. For each row in the mug region, we sample the projection-screen background from columns just left (col 107) and right (col 141) of the mug, then interpolate across the hole. The background is a smooth blue-gray vertical gradient, so this produces a seamless fill.

3. Physics trajectory

A kinematic free-fall model with damped elastic bounce:

  • Gravity: 3.0 px/frame²
  • Mug center falls from row 56.5 to row 100.0 (≈43.5 px)
  • First impact at frame 6 (1.5 s)
  • Damped bounce with 35% velocity retention
  • Squish/stretch deformation on impact frames 6–8

4. Frame compositing

For each frame t ∈ [1,19]:

  1. Start from the inpainted background canvas.
  2. Composite the mug sprite at the physics-computed Y position (same X column).
  3. For impact frames (6–8), apply vertical squash/stretch.
  4. Add a subtle drop shadow as the mug nears the surface (frames 5+).

5. Verification

  • Frame 00 is a direct byte copy of first_frame.png (verified by zero-diff assert).
  • All 20 frames are 256×144 RGB uint8.
  • Outputs: pixels.npz, frames/00.png–19.png, inspection.png, transition_check.png, index.html.
