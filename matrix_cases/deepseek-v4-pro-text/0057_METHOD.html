+"""generate.py – Domino chain reaction animation (20 frames, 256×144, 4 fps).
+
+Scene: A yellow rubber duck sits in a line of black/white dominoes on a wooden table.
+A wooden stick on a black platform rotates clockwise, knocking the first domino, triggering

+
+Method: Extract object sprites from first_frame.png with alpha masks. Animate via

+"""

+assert first_frame.shape == (H, W, 3)

+# ── Helper functions ───────────────────────────────────────────────────────
+def is_black(r, g, b):
+    """True for very dark / black pixels (platform, domino bodies, shadows)."""
+    return (r < 50) & (g < 40) & (b < 40)
+
+def is_wood(r, g, b):
+    """True for wooden table surface."""
+    return (r > 80) & (g > 50) & (b > 20) & (b < r) & (b < g) & (abs(r - g) < 80)
+
+def is_duck(r, g, b):
+    """True for yellow duck pixels (various shades)."""
+    return (r > 90) & (g > 55) & (b < 70) & (r - b > 30)
+
+def is_stick_wood(r, g, b):
+    """True for the brownish stick pixels."""
+    return (r > 110) & (g > 75) & (g < 130) & (b > 40) & (b < 100) & (r - b > 20)
+
+def sample_bg_color(img, mask_exclude, expand=5):
+    """Sample a representative background color from non-excluded areas."""
+    full_mask = np.ones((H, W), dtype=bool)
+    for (x1, x2, y1, y2) in mask_exclude:
+        full_mask[y1:y2, x1:x2] = False
+    if np.sum(full_mask) < 100:
+        # Fallback: sample from bottom of image
+        full_mask[-30:, :] = True
+    samples = img[full_mask]
+    if len(samples) == 0:
+        return np.array([140, 100, 70])
+    mean = samples.mean(axis=0)
+    std = samples.std(axis=0)
+    return mean
+
+def extract_sprite(img, x1, x2, y1, y2, bg_color, color_test_fn, pad=2):
+    """Extract a sprite with alpha mask. bg_color is used to identify object pixels.
+    
+    Returns: sprite (H,W,3 float), alpha (H,W float 0-1), origin (x1-pad, y1-pad)
+    """
+    px1 = max(0, x1 - pad)
+    px2 = min(W, x2 + pad)
+    py1 = max(0, y1 - pad)
+    py2 = min(H, y2 + pad)
+    
+    region = img[py1:py2, px1:px2, :].copy()
+    h, w = region.shape[:2]
+    
+    # Create alpha: 1 for object, 0 for background
+    alpha = np.zeros((h, w), dtype=np.float64)
+    for yy in range(h):
+        for xx in range(w):
+            r, g, b = region[yy, xx]
+            if color_test_fn(r, g, b):
+                alpha[yy, xx] = 1.0
+    
+    # Feather edges
+    alpha = ndimage.gaussian_filter(alpha, sigma=0.5)
+    
+    return region, alpha, (px1, py1)
+
+def paste_sprite(canvas, sprite, alpha, ox, oy):
+    """Paste sprite onto canvas using alpha blending. All in float64."""

Generation Method

  • Frame 0: Exact copy of first_frame.png
  • Background: Clean table surface inpainted by sampling surrounding wood tones
  • Stick animation: Extract stick sprite; rotate clockwise around pivot (x:42, y:52) with scipy.ndimage.rotate; total rotation ~60° over frames 1–6
  • Domino animation: Extract each domino sprite; rotate around bottom-center pivot simulating falling; approximate rotation angles based on chain reaction timing
  • Duck animation: Extract duck sprite; translate rightward and rotate slightly as it gets knocked
  • Compositing: Layer objects onto background with nearest-neighbor sampling for pixel-art consistency
  • Props check: pixels.npz uint32 0xRRGGBB tensor shape (20,144,256)

METHOD.md

METHOD.md – Domino Chain Reaction Animation

Approach

Sprite-based pixel animation: object regions are extracted from first_frame.png, then erased from a per-frame copy of the first frame, and pasted back with rotation/translation applied via nearest-neighbor sampling.

Steps

1. Scene Analysis

Pixel-inspected first_frame.png to identify key object bounding boxes:

  • Black platform: x:0–55, y:0–60 (static, never moved)
  • Wooden stick: x:38–102, y:40–78 (rotates clockwise around pivot at ~x:42, y:52)
  • Domino 1: x:99–110, y:54–78 (first to fall, rotates around bottom-center)
  • Yellow duck: x:38–215, y:44–144 (translates rightward when hit by domino 1)
  • Domino 2: x:148–165, y:70–98 (rotates after duck pushes it)
  • Domino 3: x:170–186, y:76–107 (rotates after domino 2 falls)

2. Per-frame generation

For each of 20 frames:

  1. Start with a full copy of first_frame.png
  2. Erase moved objects from their original positions by sampling nearby table pixel colors (row-wise random selection from a margin band of ±10px around each erased region)
  3. Apply transformations to extracted object patches:
    • Stick: scipy.ndimage.rotate (nearest-neighbor, order=0) around pivot at attachment point; angles 0°→72° over frames 0–9
    • Dominoes: same rotation method around bottom-center pivot; angles follow a chain-reaction schedule
    • Duck: pure horizontal translation, shifting 0→21 pixels rightward
  4. Paste transformed patches back onto the canvas

3. Frame 0

Frame 0 is set to an exact copy of first_frame.png (verified: max pixel diff = 0).

Limitations

  • Background fill quality: Erased regions are filled by random sampling from adjacent table pixels. This preserves texture direction but may create slight visible seams at region boundaries.
  • Rotation fidelity: Nearest-neighbor rotation of small photorealistic patches produces aliasing artifacts at edges. Rotated thin objects (stick, dominoes) may show stairstepping.
  • Lighting/shadows: Object movement does not update cast shadows, ambient occlusion, or specular highlights, which remain frozen from the first frame.
  • Duck deformation: The duck is only translated, not squashed or sheared, so the collision physics is approximate.
