+#!/usr/bin/env python3
+"""generate.py — Produce 20 identical frames for a static rubber duck scene.
+
+The prompt describes a stationary yellow rubber duck on a wooden table
+with a plain white background. The scene is static: all 20 frames are
+pixel-identical to first_frame.png.
+
+Method: Direct pixel copy. Load first_frame.png as a uint8 RGB array,
+replicate it 20 times to form a (20, 144, 256, 3) tensor, then save
+all required artifacts via artifact_io.save().
+"""
+
+import sys
+from pathlib import Path
+import numpy as np
+from PIL import Image
+
+# Ensure we can import the helper from the workspace root
+WORKSPACE = Path(__file__).resolve().parent
+sys.path.insert(0, str(WORKSPACE))
+
+import artifact_io  # type: ignore
+
+
+def main() -> None:
+    """Load first frame, replicate 20 times, save all artifacts."""
+    first_frame_path = WORKSPACE / "first_frame.png"
+    first = np.array(Image.open(first_frame_path))
+    assert first.shape == (144, 256, 3), f"Unexpected shape: {first.shape}"
+    assert first.dtype == np.uint8, f"Unexpected dtype: {first.dtype}"
+
+    # Replicate 20 times: shape (20, 144, 256, 3)
+    frames = np.tile(first[np.newaxis, ...], (20, 1, 1, 1)).astype(np.uint8)
+
+    # Frame 00 must exactly equal first_frame.png
+    assert np.array_equal(frames[0], first), "Frame 00 does not match first_frame.png!"
+
+    # Save via the project helper
+    artifact_io.save(frames, str(WORKSPACE))
+
+    # Create transition_check.png: a diff-like diagnostic sheet
+    # Since all frames are identical, the transition check shows no diffs
+    _make_transition_check(frames)
+
+    print("Done. All artifacts generated successfully.", flush=True)
+
+
+def _make_transition_check(frames: np.ndarray) -> None:
+    """Create transition_check.png — a visual diff between consecutive frames.
+
+    For this static scene, all frames are identical, so the diff images
+    are uniformly black. The sheet shows 19 diff panels (frame 01-01,
+    02-01, ..., 19-18) arranged in a grid.
+    """
+    h, w = 144, 256
+    cols = 5
+    rows = 4  # ceil(19 / 5) = 4
+    sheet = np.zeros((rows * h, cols * w, 3), dtype=np.uint8)
+
+    from PIL import ImageDraw, ImageFont
+
+    pil_sheet = Image.fromarray(sheet)
+    draw = ImageDraw.Draw(pil_sheet)
+
+    for i in range(19):
+        row = i // cols
+        col = i % cols
+        x0 = col * w
+        y0 = row * h
+
+        # Diff: amplify differences ×10 (there are none for static scene)
+        diff = frames[i + 1].astype(np.int16) - frames[i].astype(np.int16)
+        # Since all identical, diff is zero → black
+        draw.rectangle([x0, y0, x0 + w - 1, y0 + h - 1], fill=(0, 0, 0))
+        draw.text((x0 + 4, y0 + 2), f"{i:02d}→{i+1:02d}", fill=(60, 60, 60))
+        # If there were any diffs, draw red dots at changed pixels
+        if np.any(diff):
+            changed = np.any(diff != 0, axis=2)
+            ys, xs = np.where(changed)
+            for y, x in zip(ys, xs):
+                draw.point((x0 + x, y0 + y), fill=(255, 0, 0))
+
+    pil_sheet.save(WORKSPACE / "transition_check.png")
+    print("Saved transition_check.png", flush=True)
+

METHOD.md — Generation Approach

Method: Direct Pixel Replication

Summary

The prompt describes a static scene: a yellow rubber duck sitting stationary on a light brown wooden table with a plain white background, filmed as a locked-off single shot with a fixed frame. There is no motion, no animation, no lighting change — all 20 frames are pixel-identical to first_frame.png.

Technique

Aspect Detail
Core operation np.tile(first_frame[np.newaxis, ...], (20, 1, 1, 1))
Pixel manipulation None — raw copy, no drawing/warp/inpaint/shading
Physics simulation None needed — object is stationary (Newton's First Law)
External models None

Rationale

The prompt explicitly states:

  • "Static locked-off single-shot with fixed frame throughout"
  • "The yellow rubber duck sits stationary"

Under these conditions, every frame is identical to the first. This is a faithful rendering of the described scene: a scientific demonstration of static equilibrium where a rubber duck rests motionless on a wooden table.

Verification

  • frames/00.png is byte-identical to first_frame.png
  • All 20 frames are pixel-identical (verified via np.array_equal)
  • pixels.npz uses uint32 0xRRGGBB packing, confirmed by round-trip decode
  • transition_check.png shows all-black diff panels (no inter-frame changes)
  • inspection.png shows the same static scene at all 8 sampled time points
