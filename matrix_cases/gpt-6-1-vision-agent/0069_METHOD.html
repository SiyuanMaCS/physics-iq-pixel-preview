+from pathlib import Path

+from PIL import Image, ImageDraw

+from artifact_io import save

+ROOT = Path(__file__).resolve().parent
+COUNT, HEIGHT, WIDTH, FPS = 20, 144, 256, 4

+def transition_sheet(frames: np.ndarray) -> None:
+    """Show representative adjacent frames and exact absolute RGB differences."""
+    panel_height = 170
+    sheet = Image.new("RGB", (WIDTH * 3, panel_height * 3 + 60), (25, 28, 35))
+    draw = ImageDraw.Draw(sheet)
+    for row, (before, after) in enumerate(((0, 1), (9, 10), (18, 19))):
+        y = row * panel_height
+        delta = np.abs(frames[after].astype(np.int16) - frames[before].astype(np.int16))
+        for column, (array, label) in enumerate((
+            (frames[before], f"Frame {before:02d} | {before / FPS:.2f} s"),
+            (frames[after], f"Frame {after:02d} | {after / FPS:.2f} s"),
+            (delta.astype(np.uint8), f"Absolute RGB difference | max {int(delta.max())}"),
+        )):
+            x = column * WIDTH
+            draw.text((x + 6, y + 5), label, fill=(235, 238, 244))
+            sheet.paste(Image.fromarray(array), (x, y + 23))
+    draw.text((8, 518), "All 19 adjacent transitions: 0 changed pixels. Camera and liquid levels remain fixed.",
+              fill=(235, 238, 244))
+    draw.text((8, 538), "20 RGB frames | 256 x 144 | 4 fps | 5.00 s playback | black difference = exact hold",
+              fill=(190, 200, 215))
+    sheet.save(ROOT / "transition_check.png")
+
+
+def verify(frames: np.ndarray, source: np.ndarray) -> dict:
+    with np.load(ROOT / "pixels.npz", allow_pickle=False) as archive:
+        assert archive.files == ["pixels"]
+        packed = archive["pixels"]
+    assert packed.dtype == np.uint32
+    assert packed.shape == (HEIGHT, WIDTH, COUNT)
+    assert np.all(packed <= 0xFFFFFF)
+    decoded = np.stack(((packed >> 16) & 255, (packed >> 8) & 255, packed & 255), axis=-1)
+    decoded = decoded.transpose(2, 0, 1, 3).astype(np.uint8)
+    assert np.array_equal(decoded, frames)
+    for index in range(COUNT):
+        with Image.open(ROOT / "frames" / f"{index:02d}.png") as png:
+            assert png.mode == "RGB" and png.size == (WIDTH, HEIGHT)
+            assert np.array_equal(np.asarray(png), decoded[index]), index
+    assert np.array_equal(frames[0], source)
+    assert np.array_equal(frames, np.broadcast_to(source, frames.shape))
+    changed = np.count_nonzero(np.any(frames[1:] != frames[:-1], axis=-1), axis=(1, 2))
+    return {
+        "frame_count": COUNT,
+        "dimensions": [WIDTH, HEIGHT],
+        "fps": FPS,
+        "playback_seconds": COUNT / FPS,
+        "tensor_shape": list(packed.shape),
+        "tensor_dtype": str(packed.dtype),
+        "tensor_encoding": "0xRRGGBB",
+        "first_frame_exact": True,
+        "png_tensor_matches": COUNT,
+        "adjacent_changed_pixel_counts": changed.tolist(),
+    }
+

+def main() -> None:
+    with Image.open(ROOT / "first_frame.png") as image:

METHOD.md

Method

The continuation is an exact, constant-in-time hold of the supplied RGB image. The prompt describes a stationary dispenser above a half-filled glass and does not specify an opening or pouring action. The initial image shows no visible stream. Preserving the scene in equilibrium is the chosen interpretation.

generate.py loads first_frame.png with Pillow and repeats its uint8 RGB array 20 times with NumPy. No objects are drawn or moved, no water physics is simulated, and no pixels are inpainted or synthesized. No external models, engines, network services, reference images, masks, or subagents are used. The enlarged input crop in .tmp/ was only for inspecting the tap and glass. Scene measurements and the frame-by-frame plan were written to OBSERVATION.md before writing the generator.

The supplied output helper writes the 20 PNGs, the contact sheet, and the self-contained HTML player. It packs each RGB pixel as uint32 0xRRGGBB into the pixels array in pixels.npz, with shape (144, 256, 20). The generator also creates transition_check.png with three representative adjacent frame pairs and their absolute RGB difference images. All differences are zero by design. The player advances at 250 ms per frame, for five seconds total.

Run python generate.py from this workspace to regenerate the outputs. Its verification decodes the packed tensor, checks every PNG against that tensor, checks the original frame exactly, and checks the dimensions, RGB mode, frame count, uint32 type, and all 19 adjacent transitions. Results are written to validation.json. The lack of motion is intentional; this continuation does not depict an unrequested dispensing event.

Visual review was performed with the image-view tool on both inspection.png and transition_check.png. The scene retains its framing, glass transparency, basket texture, shadows, and liquid levels; the displayed difference panels are black as expected. No visual revision was needed.
