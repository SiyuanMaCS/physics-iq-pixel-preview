+def smoothstep(edge0, edge1, x):
+    q = np.clip((x - edge0) / (edge1 - edge0), 0.0, 1.0)

+def make_clean_plate(source):
+    """Remove only the source ball and its tiny contact shadow by local inpaint."""
+    yy, xx = np.mgrid[0:H, 0:W]
+    cx, cy = SOURCE_CENTER
+    ball = ((xx - cx) / 10.2) ** 2 + ((yy - cy) / 10.4) ** 2 <= 1.0
+    shadow = ((xx - (cx + 0.8)) / 10.0) ** 2 + ((yy - (cy + 7.0)) / 3.7) ** 2 <= 1.0
+    mask = np.where(ball | shadow, 255, 0).astype(np.uint8)
+    # TELEA uses neighboring real table pixels and touches no other scene region.
+    clean = cv2.inpaint(source, mask, 4.0, cv2.INPAINT_TELEA)
+    return clean

+def extract_ball_residual(source, clean):
+    """Capture the real ball as a feathered additive residual over the clean plate."""
+    yy, xx = np.mgrid[0:H, 0:W]
+    cx, cy = SOURCE_CENTER
+    radial = np.sqrt(((xx - cx) / 9.4) ** 2 + ((yy - cy) / 9.2) ** 2)
+    alpha = 1.0 - smoothstep(0.78, 1.02, radial)

+    residual = (source.astype(np.float32) - clean.astype(np.float32))
+    residual *= alpha[..., None]
+    return residual[y0:y1, x0:x1], (x0, y0)

+    return cv2.warpAffine(

METHOD.md

Method

The animation is a local, deterministic pixel construction from first_frame.png. No external images, models, masks, references, engines, or network resources were used.

Construction

  1. Load the supplied 256 x 144 RGB frame.
  2. Keep frame 00 as a direct array copy of the input.
  3. Build a clean continuation plate by applying OpenCV TELEA inpainting only to a 20 px-wide ellipse covering the orange ball and its small contact-shadow footprint. The inpaint source is the immediately surrounding real tabletop.
  4. Extract the source ball as the feathered RGB difference between the real frame and clean plate. This additive residual retains the ball's actual highlight, color variation, antialiasing, and edge shading.
  5. For frames 01–19, shift that residual at constant speed from x=47.5 through x=227.0 while keeping y=45.5. Fractional x positions use local bilinear interpolation. A subtle multiplicative elliptical contact shadow moves with the ball while preserving the table texture underneath.
  6. Leave the red ball, pipe, table, lighting, and camera pixels static.

The ball's translation across nearly the entire tabletop is the main rolling cue at this very small (about 16 px diameter) scale. Its lighting pattern remains fixed in screen space, consistent with a stationary light and avoiding an artificially rotating highlight.

Output and verification

artifact_io.save writes the PNG sequence, packed uint32 0xRRGGBB tensor, inspection contact sheet, and standalone player. generate.py then reloads every PNG and verifies exact equality against both the source frame (for frame 00) and the corresponding pixels.npz plane. It also confirms every transition changes only a compact part of the locked-off frame.

transition_check.png contains previous/current, amplified absolute-difference, and changed-pixel-mask panels for four representative transitions.

Scene analysis and validation were numeric only. I did not use an image viewing tool and therefore do not claim visual review of the generated sheets.
