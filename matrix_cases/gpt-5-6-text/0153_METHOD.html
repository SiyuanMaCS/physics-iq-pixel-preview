+BALL0 = np.array([103.0, 51.0], dtype=np.float32)

+def ellipse_field(cx, cy, rx, ry, shape=(H, W)):
+    """A feathered analytic ellipse: 1 inside, fading over about 1 pixel."""
+    yy, xx = np.indices(shape, dtype=np.float32)
+    radial = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
+    # A normalized feather of ~1.2 pixels at the horizontal edge.
+    feather = 1.2 / max(rx, 1.0)
+    return np.clip((1.0 + feather - radial) / feather, 0.0, 1.0)
+
+
+def make_clean_plate(source):
+    """Remove the initial ball and its compact shadow from the source table."""
+    mask = np.zeros((H, W), np.uint8)
+    cv2.ellipse(mask, (104, 52), (24, 16), 0, 0, 360, 255, -1, cv2.LINE_AA)
+    # Telea uses only surrounding pixels and retains the source's local grain.
+    return cv2.inpaint(source, mask, 5.0, cv2.INPAINT_TELEA)
+
+
+def deform_box(plate, amount):
+    """Give the fabric collision contour a small, localized inward displacement."""
+    if amount <= 0:
+        return plate.copy()
+    yy, xx = np.indices((H, W), dtype=np.float32)
+    local_y = np.exp(-0.5 * ((yy - 51.0) / 10.0) ** 2)
+    local_x = np.exp(-0.5 * ((xx - 208.0) / 13.0) ** 2)
+    gate = np.clip((xx - 192.0) / 10.0, 0.0, 1.0) * np.clip((237.0 - xx) / 12.0, 0.0, 1.0)
+    shift = amount * local_y * local_x * gate
+    # Sampling from the left moves the visible soft edge to the right.
+    map_x = xx - shift
+    map_y = yy
+    return cv2.remap(plate, map_x, map_y, cv2.INTER_LINEAR,
+                     borderMode=cv2.BORDER_REFLECT_101)
+
+
+def affine_for_ball(target, angle_deg, scale_x, scale_y):
+    """Forward affine mapping from source ball coordinates to target coordinates."""
+    angle = np.deg2rad(angle_deg)
+    c, s = float(np.cos(angle)), float(np.sin(angle))
+    # Rotation after the impact-dependent squash/stretch.
+    linear = np.array([[c * scale_x, -s * scale_y],
+                       [s * scale_x,  c * scale_y]], dtype=np.float32)
+    offset = np.asarray(target, np.float32) - linear @ BALL0
+    return np.column_stack((linear, offset)).astype(np.float32)
+

+def render_ball(plate, source, source_alpha, center, angle, sx=1.0, sy=1.0):
+    """Composite translated/rotated source ball material plus a soft cast shadow."""
+    result = plate.astype(np.float32)

OBSERVATION.md

Numeric observations and event plan

Source measurements

  • Input is RGB, 256 x 144. All measurements below come from local numeric RGB arrays, scanlines, color clustering, and gradient coordinates; no image-viewing tool was used.
  • The camera is locked. The source already contains the complete setup, so every frame should preserve the source outside the small ball/collision region.
  • The light wood/table field around the moving object is a softly textured, low-saturation gray-brown. At the ball height, unobstructed pixels are generally around RGB (120-139, 114-137, 109-138).
  • The brown ball is centered near (103, 51). Its high-gradient circular extent is approximately x=91..115, y=40..62, giving an effective sphere radius of about 11-13 px. Brown pixels are concentrated on the lit/left portion and dark pixels on the right/lower portion. A short dark cast-shadow region continues right/down beyond the sphere.
  • The olive-green fabric box occupies the right side. Around the ball height, its collision-facing dark contour is near x=203..207, while the more plainly green front face begins near x=224..226. The left contour drifts from about x=203 at y=40 to x=209 at y=70 because of perspective/soft construction.
  • The black tube/setup is already represented in the fixed source and is not redrawn. Preserving the source pixels keeps its irregular lighting and material detail intact.
  • The ball starts already at/just beyond the tube exit in frame 00. For later frames, its original footprint and shadow will be reconstructed from nearby table pixels, then a source-derived ball cutout will be translated along one level horizontal line.

METHOD.md

Method

Actual generation method

I used only local code and numeric RGB inspection. I did not use an image Read/View tool and therefore did not perform or claim visual review.

  1. Loaded first_frame.png as a 256 x 144 RGB uint8 array and measured the scene with quantized color counts, scanline runs, connected-component statistics, Sobel-gradient coordinates, and sampled RGB grids.
  2. Kept frame 00 as an exact copy of the decoded source pixels.
  3. Built a static clean plate for frames 01-19 by masking the measured original ball/shadow footprint around (104, 52) and applying OpenCV Telea inpainting from the immediately surrounding table pixels. Numeric checks show that the filled patch has essentially the same mean luminance as its surrounding ring and lower gradient energy, without zeros or hard seams.
  4. Defined a softly feathered elliptical alpha matte for the source ball itself around (103, 51). The real source RGB material inside this matte is affine-transformed to each new position. This preserves the source's brown color variation and small-scale texture rather than inventing a flat replacement.
  5. Advanced the ball along a level horizontal path. Its source texture rotates according to accumulated travel divided by the measured radius. A small fixed-world highlight is reapplied after rotation so the lighting direction does not rotate with the material.
  6. Rebuilt the compact table contact shadow analytically as a soft, offset ellipse at each position. It is composited before the ball and follows the ball without rotating.
  7. Modeled contact at frames 11-13. The soft box contour receives a localized 0.8/2.2/0.7-pixel remap into the box, while the ball briefly changes scale from (0.92, 1.03) to (0.76, 1.10) and then (0.96, 1.02). The ball then rebounds left with damped displacement and settles for the final two frames.
  8. Saved the canonical uint32 0xRRGGBB tensor using the provided artifact_io.save, which also writes the RGB PNG sequence, inspection sheet, and self-contained HTML player. transition_check.png is a six-tile sheet of amplified absolute RGB deltas for selected consecutive-frame pairs.

Verification performed

  • 20 frames, each RGB uint8 and exactly 256 x 144.
  • frames/00.png decodes pixel-for-pixel equal to first_frame.png.
  • Every PNG decodes pixel-for-pixel equal to the corresponding frame unpacked from pixels.npz.
  • The tensor is uint32 with shape (144, 256, 20) and contains packed 0xRRGGBB values.
  • Strong-difference centroids advance steadily from about x=111 in frame 01 to x=192 at contact, then retreat to about x=175 and hold.
  • Changes are localized to the measured ball path and small box-impact region; the rest of the locked shot remains source-identical.
  • The final identical frame pair is an intentional settled hold, not a dropped frame.
