+"""Generate a slow, clockwise in-plane rotation of the marked tennis ball."""
+
+from pathlib import Path
+

+from scipy.ndimage import gaussian_filter, map_coordinates
+
+import artifact_io

+
+ROOT = Path(__file__).resolve().parent
+SOURCE = ROOT / "first_frame.png"
+FRAME_COUNT = 20
+HEIGHT, WIDTH = 144, 256
+
+# Numeric fit to the projected ball. Coordinates are in image pixel space.
+CX, CY = 139.0, 75.3
+RX, RY = 19.0, 17.7
+DEGREES_PER_FRAME = 2.0
+
+
+def bilinear_sample(rgb: np.ndarray, sx: np.ndarray, sy: np.ndarray) -> np.ndarray:
+    """Sample RGB at floating point coordinates with deterministic bilinear interpolation."""
+    channels = [
+        map_coordinates(
+            rgb[:, :, c], [sy, sx], order=1, mode="nearest", prefilter=False
+        )
+        for c in range(3)
+    ]
+    return np.stack(channels, axis=-1)
+
+
+def make_frames(first: np.ndarray) -> np.ndarray:
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH].astype(np.float32)
+    nx = (xx - CX) / RX
+    ny = (yy - CY) / RY
+    radius = np.sqrt(nx * nx + ny * ny)
+
+    # The outer 2–3 source pixels retain their original antialiased silhouette.
+    # smoothstep gives a fixed scene-space transition rather than a moving cutout edge.
+    alpha = np.clip((0.985 - radius) / (0.985 - 0.84), 0.0, 1.0)
+    alpha = alpha * alpha * (3.0 - 2.0 * alpha)
+
+    source_f = first.astype(np.float32)
+    # A wide blur estimates illumination. Rotating the source and then compensating
+    # its broad field toward this fixed field keeps the lamp direction stationary.
+    illumination = np.stack(
+        [gaussian_filter(source_f[:, :, c], sigma=4.2, mode="nearest") for c in range(3)],
+        axis=-1,
+    )
+
+    frames = np.empty((FRAME_COUNT, HEIGHT, WIDTH, 3), dtype=np.uint8)
+    frames[0] = first
+
+    for t in range(1, FRAME_COUNT):
+        angle = np.deg2rad(DEGREES_PER_FRAME * t)
+        c, s = np.cos(angle), np.sin(angle)
+
+        # Inverse map a positive clockwise motion in screen coordinates. Rotation
+        # occurs in normalized ellipse coordinates so samples stay inside the ball.
+        src_nx = c * nx + s * ny
+        src_ny = -s * nx + c * ny
+        sx = CX + RX * src_nx
+        sy = CY + RY * src_ny
+
+        rotated = bilinear_sample(source_f, sx, sy)
+        rotated_light = bilinear_sample(illumination, sx, sy)
+
+        # Preserve the broad fixed lighting without suppressing the moving black
+        # facial marks. Limiting the compensation prevents halos around those marks.
+        correction = np.clip(illumination - rotated_light, -24.0, 24.0)
+        relit = rotated + 0.72 * correction
+
+        composed = source_f * (1.0 - alpha[:, :, None]) + relit * alpha[:, :, None]
+        frames[t] = np.clip(np.rint(composed), 0, 255).astype(np.uint8)
+
+    return frames
+

OBSERVATION.md

Scene observation and animation plan

Numeric observations

  • Source is an RGB uint8 image at exactly 256 x 144 pixels. Its channel extrema are 0–207 (R), 0–211 (G), and 0–193 (B).
  • The camera is locked. A warm wood-toned horizontal surface begins near y=56–60 beneath a pale neutral background. A dark apparatus is clipped by the right edge. These regions must stay pixel-identical.
  • The tennis ball occupies approximately x=120–158 and y=58–92. Its projected center is approximately (139.0, 75.5), with an apparent horizontal radius near 19 px and vertical radius near 17.5 px. Its green/yellow core is the connected region x=122–156, y=59–90.
  • The ball rests against a black platform whose main connected dark region spans approximately x=104–175, y=90–125. The platform and ball silhouette remain stationary.
  • The face and tennis-ball markings are the high-contrast dark structures within the yellow disk: paired marks in the upper half and a curved smile across the lower half. They provide the clearest rotation cue.
  • Lighting is soft and fixed in scene coordinates. The ball has a brighter upper/right yellow-green area and a darker lower edge/contact area. A physically plausible edit therefore keeps the silhouette and broad illumination anchored while rotating surface detail.
  • No camera motion, zoom, cut, object translation, platform motion, or new object is called for.

20-frame event plan (4 fps, 5 seconds)

  • Frame 00 (0.00 s): exact byte-for-byte RGB copy of first_frame.png.
  • Frames 01–04 (0.25–1.00 s): ball begins steady clockwise rotation, reaching about 8 degrees; all static scenery is unchanged.
  • Frames 05–08 (1.25–2.00 s): continued constant-speed clockwise rotation, reaching about 16 degrees.
  • Frames 09–12 (2.25–3.00 s): continued constant-speed clockwise rotation, reaching about 24 degrees.
  • Frames 13–16 (3.25–4.00 s): continued constant-speed clockwise rotation, reaching about 32 degrees.
  • Frames 17–19 (4.25–4.75 s): continued constant-speed clockwise rotation, ending near 38 degrees.

For each generated frame, only the elliptical ball interior will be resampled. A feathered, stationary silhouette preserves edge antialiasing; low-frequency illumination remains mostly fixed, while facial/seam detail follows the clockwise material rotation. Motion advances by 2 degrees per frame with no easing, matching a constant-framerate, real-time scientific demonstration.

METHOD.md

Method

The animation is generated entirely with local Python pixel operations from first_frame.png; no external image, model, mask, engine, or reference is used.

I measured the projected tennis ball numerically and fitted it with an ellipse centered at approximately (139.0, 75.3), with radii 19.0 x 17.7 pixels. Frame 00 is assigned directly from the decoded RGB source and is never synthesized.

For frames 01–19, inverse bilinear resampling rotates pixels inside normalized ball coordinates clockwise by a constant 2 degrees per frame. The stationary support mask has a smooth feather from normalized radius 0.84 to 0.985, leaving the antialiased silhouette, contact edge, platform, table, background, and apparatus unchanged. A Gaussian low-frequency estimate of the original ball illumination is held in scene coordinates; a bounded correction partially removes the rotated low-frequency lighting from the resampled result. Thus the face, seam, and fine ball texture rotate, while broad shading continues to agree with the fixed lamp and camera.

The final motion covers 38 degrees over the 19 transitions. There is no easing, translation, camera movement, or platform motion.

artifact_io.save writes the 20 PNGs, the packed uint32 tensor, the inspection contact sheet, and the standalone HTML player. generate.py additionally writes transition_check.png, then verifies source-frame identity, exact static pixels outside the fitted ball, per-transition motion, all PNG decodes, and bit-exact equality between those RGB frames and the 0xRRGGBB tensor in pixels.npz.
