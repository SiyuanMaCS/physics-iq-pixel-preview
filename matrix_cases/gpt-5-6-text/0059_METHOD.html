+def horizontal_fill(image: np.ndarray, rect: tuple[int, int, int, int]) -> None:
+    """Replace a small rectangle by row-wise interpolation from nearby wall."""
+    x0, y0, x1, y1 = rect
+    height, width = image.shape[:2]
+    left0, left1 = max(0, x0 - 5), x0
+    right0, right1 = x1, min(width, x1 + 5)
+    for y in range(y0, min(y1, height)):
+        left = np.median(image[y, left0:left1].astype(np.float32), axis=0)
+        right = np.median(image[y, right0:right1].astype(np.float32), axis=0)
+        count = x1 - x0
+        for i, x in enumerate(range(x0, x1)):
+            u = (i + 1.0) / (count + 1.0)
+            image[y, x] = np.clip(np.rint(left * (1.0 - u) + right * u), 0, 255)
+
+
+def alpha_from_residual(source_patch: np.ndarray, background_patch: np.ndarray) -> np.ndarray:
+    """Estimate antialias coverage from separation against a neutral wall."""
+    delta = source_patch.astype(np.float32) - background_patch.astype(np.float32)
+    residual = np.sqrt(np.sum(delta * delta, axis=2))
+    alpha = np.clip(residual / 120.0, 0.0, 1.0)
+    alpha[residual < 4.0] = 0.0
+    return alpha
+
+
+def composite_sprite(
+    frame: np.ndarray,
+    source: np.ndarray,
+    clean_background: np.ndarray,
+    rect: tuple[int, int, int, int],
+    dx: int,
+    dy: int,
+    clip_below: int | None = None,
+) -> None:
+    """Move a source-composited patch while compensating for its old backdrop."""
+    x0, y0, x1, y1 = rect
+    sprite = source[y0:y1, x0:x1].astype(np.float32)
+    old_bg = clean_background[y0:y1, x0:x1].astype(np.float32)
+    alpha = alpha_from_residual(sprite, old_bg)
+
+    h, w = alpha.shape
+    for sy in range(h):
+        ty = y0 + sy + dy
+        if ty < 0 or ty >= frame.shape[0]:
+            continue
+        if clip_below is not None and ty >= clip_below:
+            continue
+        for sx in range(w):
+            a = float(alpha[sy, sx])
+            if a <= 0.0:
+                continue
+            tx = x0 + sx + dx
+            if tx < 0 or tx >= frame.shape[1]:
+                continue
+            destination = frame[ty, tx].astype(np.float32)
+            # If source = a*foreground + (1-a)*old_bg, this changes only the
+            # background contribution when placing the antialiased pixel.
+            value = sprite[sy, sx] + (1.0 - a) * (destination - old_bg[sy, sx])
+            frame[ty, tx] = np.clip(np.rint(value), 0, 255).astype(np.uint8)
+

OBSERVATION.md

Source observation and event plan

Numeric scene measurements

  • Source: first_frame.png, RGB uint8, 256×144. All measurements below come from local array/color inspection; no image viewer was used.
  • Camera/background: locked frame. The background is a softly varying warm gray wall across roughly y=0–80. The lower table region begins around y=116 and has warm brown/orange pixels, strongest from x≈47–216.
  • Grabber: two narrow orange/black vertical prongs enter from the top. Their principal runs are x=120–122 and x=132–134 from y=0–12. The gap between them is about 10 px.
  • Duck: compact yellow/orange component centered near (127,24), with antialiased extent approximately x=119–135, y=14–33. It is initially immediately below the grabber.
  • Box: centered dark green fabric box, approximately x=100–154 and y=81–139. The top opening/lip is darkest around y=82–84. The broad front fabric panel is dark olive (central mean about RGB 25,24,10) down to y≈134. The box is wide enough to receive the duck with at least 15 px clearance on either side.
  • Motion-safe fall line: x≈127 through the wall area into the middle of the opening. The visible free-fall distance from duck center y≈24 to opening y≈82 is about 58 px.
  • Occlusion rule: the duck is visible above the top lip, then is progressively clipped below the lip so it goes into the box rather than being painted over its front face.

METHOD.md

Method

I generated the continuation with local Python/NumPy/Pillow pixel operations only. I did not use an image viewer, external model, external asset, network source, mask, or rendering engine.

The source frame was inspected numerically as an RGB array. I measured the two grabber prongs, the compact duck component, the fall line, and the top edge of the dark box. generate.py reconstructs the wall behind the two narrow prongs and the duck using row-wise color interpolation from neighboring wall pixels. It derives soft sprite coverage from each source patch's RGB residual against that reconstructed wall, preserving the original antialiased, shaded pixels while moving them.

Frames 00 and 01 are an exact source hold. From frame 02 the two prongs shift outward by one then two pixels. The duck follows increasing vertical offsets sampled at 4 fps (0, 2, 6, 16, 29, 44, 55, 64 after release). Pixels at or below y=83 are clipped, using the box's top lip as a foreground occluder. Thus the duck progressively disappears into the box instead of appearing on the front fabric panel. A small, source-derived darkening at the opening is applied only during approach. Once contained, the remaining frames hold the unchanged apparatus.

artifact_io.save writes the 20 PNGs, packed uint32 tensor, inspection sheet, and standalone HTML player. The generator also writes a 19-transition amplified difference sheet and verifies every saved PNG against the generated uint8 RGB data, verifies the (144,256,20) uint32 0xRRGGBB tensor, and verifies frame 00 exactly equals first_frame.png.
