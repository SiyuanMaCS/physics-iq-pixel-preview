+"""Generate the 20-frame PixelWorld continuation from the supplied first frame.
+
+The photographed frame is retained wholesale.  Only a small fitted circular ball
+surface is changed: its high-frequency/color residual is rotated clockwise over
+a stationary low-order illumination field.  All inspection is numeric.
+"""

+
+ROOT = Path(__file__).resolve().parent
+WIDTH, HEIGHT = 256, 144
+CX, CY = 130.0, 58.5
+BALL_R = 20.8
+DEGREES_PER_FRAME = 2.0
+
+
+def polynomial_features(dx: np.ndarray, dy: np.ndarray) -> np.ndarray:
+    """Low-order spatial basis used for the fixed illumination/color field."""
+    return np.stack(
+        [
+            np.ones_like(dx),
+            dx,
+            dy,
+            dx * dx,
+            dy * dy,
+            dx * dy,
+            dx * dx * dx,
+            dy * dy * dy,
+        ],
+        axis=-1,
+    )
+
+
+def fit_ball_illumination(source: np.ndarray) -> np.ndarray:
+    """Robustly fit smooth RGB shading from unmarked yellow-green ball pixels."""
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    dx = (xx - CX) / BALL_R
+    dy = (yy - CY) / BALL_R
+    radius = np.hypot(dx, dy)
+    rgb = source.astype(np.float32)
+    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
+
+    # Exclude the black smiley, pale seam, mixed silhouette pixels, and platform.
+    sample = (
+        (radius < 0.91)
+        & (g > 105)
+        & (r > 95)
+        & (g - b > 65)
+        & (r - b > 55)
+        & (b < 105)
+    )
+    design = polynomial_features(dx, dy)
+    a = design[sample]
+    values = rgb[sample]
+
+    # Three robust reweighted least-squares passes keep small surface markings
+    # from influencing the broad stationary lighting field.
+    weights = np.ones(len(values), dtype=np.float64)
+    coeff = np.zeros((design.shape[-1], 3), dtype=np.float64)
+    for _ in range(3):
+        aw = a * np.sqrt(weights[:, None])
+        bw = values * np.sqrt(weights[:, None])
+        coeff = np.linalg.lstsq(aw, bw, rcond=None)[0]
+        error = np.linalg.norm(values - a @ coeff, axis=1)
+        scale = max(float(np.median(np.abs(error - np.median(error)))) * 1.4826, 3.0)
+        weights = 1.0 / (1.0 + (error / (2.5 * scale)) ** 2)
+
+    smooth = design @ coeff
+    return np.clip(smooth, 0, 255).astype(np.float32)

OBSERVATION.md

Pixel observations and event plan

All measurements below come from local numeric RGB-array inspection of first_frame.png; no image viewer or image-reading UI was used.

Scene measurements

  • Source is RGB, 256 x 144 pixels. The camera is locked off.
  • A softly varying neutral gray backdrop occupies approximately rows 0--35. Its corner samples are RGB (165,164,169), (172,170,181), (106,79,51), and (122,89,56), with the lower two samples belonging to the tabletop.
  • A warm brown wooden tabletop starts at about row 36 and fills the remainder of the frame. Its lighting and grain are already present in the source and must stay unchanged.
  • The tennis ball is centered at approximately (130,59), with a visible radius near 21 pixels and nominal bounds x=109--151, y=38--80. Green/yellow chroma pixels have measured bounds x=110--149, y=38--84; the lower tail includes edge/contact colors mixed with the platform.

METHOD.md

Method

I inspected first_frame.png only through local Python numeric RGB arrays, including sampled colors, threshold masks, row/column runs, color clustering, and measured bounds. I did not use an image viewer or image Read/View tool.

The source photograph is retained unchanged outside the tennis ball. For the ball, generate.py robustly fits a low-order two-dimensional RGB illumination field from unmarked yellow-green interior pixels. Subtracting that fit from the source isolates the smiley, tennis seam, surface texture, and small photographic variations. This residual surface is remapped around the measured center by 2 degrees clockwise per frame, then added back to the stationary illumination field. A measured circular mask with a soft edge and a small lower contact occlusion composites the result over the untouched source. This lets the markings and texture turn while the broad highlight, ball silhouette, black platform, table, backdrop, and camera remain fixed.

Frame 00 is explicitly copied from first_frame.png. The sequence contains 20 RGB uint8 frames at 256 x 144 and 4 fps. artifact_io.save writes the PNG sequence, packed uint32 0xRRGGBB tensor, inspection sheet, and standalone player. transition_check.png contains selected frames plus 4x amplified absolute differences for numeric/local inspection support.
