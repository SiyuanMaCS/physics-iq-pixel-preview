+def elliptical_alpha(shape, center, radii, feather):
+    """Return a soft elliptical mask in source-image coordinates."""
+    yy, xx = np.mgrid[: shape[0], : shape[1]]
+    d = np.sqrt(((xx - center[0]) / radii[0]) ** 2 +
+                ((yy - center[1]) / radii[1]) ** 2)
+    return np.clip((1.0 + feather - d) / (2.0 * feather), 0.0, 1.0).astype(np.float32)

+def repaired_table(source, masks):
+    """Remove the two initial balls while retaining local tabletop texture."""
+    combined = np.zeros((H, W), np.uint8)
+    for mask in masks:
+        combined = np.maximum(combined, (mask > 0.045).astype(np.uint8) * 255)
+    # A modest Telea radius reconstructs from the table pixels immediately around
+    # each ball instead of flattening the broader wood grain.
+    return cv2.inpaint(source, combined, 4, cv2.INPAINT_TELEA)

+def add_contact_shadow(frame, cx, cy, rx, ry, opacity=0.23):
+    """Darken a small blurred ellipse directly under a ball."""
+    yy, xx = np.mgrid[:H, :W]
+    shadow_y = cy + ry * 0.78
+    field = ((xx - (cx + rx * 0.10)) / (rx * 0.82)) ** 2
+    field += ((yy - shadow_y) / max(1.0, ry * 0.24)) ** 2
+    matte = np.clip(1.0 - field, 0.0, 1.0).astype(np.float32)
+    matte = cv2.GaussianBlur(matte, (0, 0), 1.35)
+    matte = np.clip(matte * opacity, 0.0, 0.35)[..., None]
+    return frame * (1.0 - matte) + frame * 0.54 * matte
+

+def transformed_sample(source_f, alpha, center, dest, angle_deg, scale):
+    """Warp a premultiplied source-ball sample directly onto the canvas."""
+    matrix = cv2.getRotationMatrix2D(center, angle_deg, scale)
+    matrix[0, 2] += dest[0] - center[0]
+    matrix[1, 2] += dest[1] - center[1]
+    premult = source_f * alpha[..., None]
+    warped_p = cv2.warpAffine(
+        premult, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=(0, 0, 0)
+    )
+    warped_a = cv2.warpAffine(
+        alpha, matrix, (W, H), flags=cv2.INTER_LINEAR,
+        borderMode=cv2.BORDER_CONSTANT, borderValue=0
+    )
+    return warped_p, warped_a

+                          angle_deg, scale):
+    """Composite five short-exposure samples into one restrained motion blur."""
+    offsets = (-0.13, -0.065, 0.0, 0.065, 0.13)
+    weights = (0.11, 0.22, 0.34, 0.22, 0.11)
+    sum_p = np.zeros((H, W, 3), np.float32)

+    color = sum_p / np.maximum(sum_a[..., None], 1e-6)
+    return frame * (1.0 - sum_a[..., None]) + color * sum_a[..., None]

OBSERVATION.md

Scene observation and event plan

Numeric observations

  • Source: first_frame.png, read locally as an RGB array only (no image viewer).
  • Canvas: 256 x 144 pixels, RGB, with a fixed camera.
  • The light wooden tabletop occupies most of the image below the dark upper/background band. Its projected surface has strong horizontal transitions near y=23, y=39-40, y=55-56, y=76-77, and y=140-142, plus fine natural color variation.
  • Left black pipe: connected very-dark region approximately x=0..20, y=10..68. Its opening points diagonally into the table toward the lower right.
  • Right black pipe: connected very-dark region approximately x=238..255, y=26..55, with additional rim/background structure up to y=11. Its opening points diagonally into the table toward the upper left.
  • Gray tennis ball: center approximately (70, 69), visual radius about 10-11 px. It lies on the left-pipe axis and is already clear of the pipe. Its projected continuation is down-right, approximately y = 69 + 0.33 * (x - 70).
  • Brown tennis ball: center approximately (139, 36), visual radius about 7-8 px. It lies on the right-pipe axis and is already clear of the pipe. Its projected continuation is up-left, approximately y = 36 + 0.08 * (x - 139).
  • The tracks are offset in depth and do not intersect in the visible frame. The brown ball is farther back and consequently appears smaller.

Animation method planned

Frame 00 will be byte-for-byte pixel identical to the source. For later frames, the two source ball appearances will be isolated with feathered circular alpha masks. Their original locations will be repaired using local, texture-preserving OpenCV inpainting and nearby tabletop texture. Each cutout will be rotated according to distance traveled, rescaled slightly along the receding/projected tracks, and alpha-composited at a subpixel position. A short multi-sample shutter blend will provide restrained directional motion blur appropriate to high speed at 4 fps. Soft contact shadows will be drawn under the moving balls. Pipes and camera remain completely stationary.

METHOD.md

Method

This sequence was generated entirely with local deterministic Python pixel operations. No image viewer, external model, external engine, network resource, mask, or other case was used.

first_frame.png was loaded as a numeric RGB array. Dark connected regions, luminance/color statistics, edge projections, and circle candidates were inspected numerically to measure the two pipe mouths and the two balls. The measured source appearances of the gray and brown balls were isolated with soft elliptical alpha masks. For frames after frame 00, the two small original ball regions were reconstructed from immediately surrounding tabletop pixels with OpenCV's Telea inpainting.

The gray ball follows the projected left-pipe axis down and right at 10.5 px/frame horizontally and 3.5 px/frame vertically. The brown ball follows the offset right-pipe axis up and left at 11.8 px/frame horizontally and 0.944 px/frame vertically. This keeps both paths straight in the projected tabletop plane and prevents an invented collision. Apparent ball size changes gently with depth. Source ball texture is rotated according to traveled distance, composited with premultiplied alpha, and sampled five times across a short shutter interval for restrained high-speed motion blur. Small blurred contact shadows anchor the balls to the tabletop. All pixels outside the swept ball tracks remain locked.

Frame 00 is the exact supplied PNG (including a direct file copy after artifact generation). The final verification decodes every output PNG, repacks all RGB values as uint32 0xRRGGBB, and confirms equality with the (144, 256, 20) tensor stored in pixels.npz. transition_check.png contains selected before/after transitions and four-times-amplified absolute pixel differences; it is a generated diagnostic, not a claim of visual review.
