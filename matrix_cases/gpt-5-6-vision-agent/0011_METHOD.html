+def reconstruct_table(source: np.ndarray) -> np.ndarray:
+    """Remove the initial orange ball and its shadow with local inpainting."""
+    mask = np.zeros((HEIGHT, WIDTH), dtype=np.uint8)
+    # The first ellipse covers the ball and contact edge.  The second follows
+    # the long, soft shadow cast down and to the right.
+    cv2.ellipse(mask, (55, 51), (11, 10), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (66, 55), (21, 8), -5, 0, 360, 255, -1)
+    mask = cv2.GaussianBlur(mask, (3, 3), 0)
+    mask = np.where(mask > 10, 255, 0).astype(np.uint8)
+    bgr = cv2.cvtColor(source, cv2.COLOR_RGB2BGR)
+    repaired = cv2.inpaint(bgr, mask, 4, cv2.INPAINT_NS)
+    return cv2.cvtColor(repaired, cv2.COLOR_BGR2RGB)
+

+def shadow_alpha(cx: float) -> np.ndarray:
+    """Return a soft, grounded cast-shadow alpha field."""
+    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
+    broad = 0.25 * np.exp(
+        -0.5 * (((xx - (cx + 9.0)) / 12.0) ** 2 + ((yy - (CY + 5.0)) / 3.7) ** 2)
+    )
+    contact = 0.17 * np.exp(
+        -0.5 * (((xx - (cx + 2.0)) / 6.0) ** 2 + ((yy - (CY + 6.0)) / 1.8) ** 2)
+    )
+    # Limit the very faint Gaussian tail so distant pixels stay perfectly fixed.
+    alpha = np.clip(broad + contact, 0.0, 0.36)
+    alpha[alpha < 0.006] = 0.0
+    return alpha

Scene observation and event plan

Measured scene

  • The pale wood tabletop occupies approximately y=34..125; its front dark edge is at y=128..134.

METHOD.md

Method

The animation is produced entirely with local Python pixel operations using Pillow, NumPy, and OpenCV.

  1. first_frame.png is loaded as the exact RGB source. Frame 00 is copied byte-for-byte from that array.
  2. For frames 01–19, the original orange ball and its cast shadow are removed once with OpenCV's Navier–Stokes inpainting over two small elliptical masks. This creates the fixed tabletop plate exposed behind the moving ball.
  3. The photographed orange ball is sampled from the source frame and translated 11 pixels to the right per frame. Its soft elliptical silhouette is anti-aliased, while the original upper-left highlight remains fixed relative to the room lighting.
  4. A compact two-Gaussian, neutral multiplicative shadow moves with the ball. It is offset down/right and includes a tighter contact component so the ball remains grounded on the tabletop.
  5. The ball keeps a constant vertical center at approximately y=51, producing a straight roll. It reaches the right border in frame 18 and is fully outside the shot in frame 19. Every other part of the locked-off scene is held static.
  6. artifact_io.save writes the 20 PNGs, packed uint32 pixels.npz, inspection sheet, and standalone HTML player. A separate ordered transition sheet is written, then every PNG is programmatically compared with its corresponding 0xRRGGBB NPZ plane.

No external images, models, engines, masks, or network resources are used.
