+def smoothstep(edge0, edge1, value):
+    """Hermite interpolation that also works with reversed edge order."""
+    t = np.clip((value - edge0) / (edge1 - edge0), 0.0, 1.0)
+    return t * t * (3.0 - 2.0 * t)

+def make_clean_plate(source):
+    """Remove the source ball and its cast shadow with local RGB inpainting."""
+    mask = np.zeros((H, W), np.uint8)
+    # The offset ellipse covers the 22 px ball plus the soft right-hand shadow.
+    cv2.ellipse(mask, (111, 60), (18, 15), 0, 0, 360, 255, -1)
+    repaired = cv2.inpaint(cv2.cvtColor(source, cv2.COLOR_RGB2BGR), mask, 6,
+                           cv2.INPAINT_TELEA)
+    repaired = cv2.cvtColor(repaired, cv2.COLOR_BGR2RGB)
+
+    # Feather the repaired area into untouched source pixels, avoiding a hard
+    # inpaint boundary while leaving all remote scene pixels exactly static.
+    feather = cv2.GaussianBlur(mask.astype(np.float32) / 255.0, (0, 0), 2.0)
+    feather = np.clip(feather[..., None], 0.0, 1.0)
+    return np.clip(source * (1.0 - feather) + repaired * feather, 0, 255).astype(np.uint8)
+

+def extract_ball_tile(source):
+    """Make a soft RGBA tile from the actual ball pixels in frame 00."""
+    yy, xx = np.mgrid[0:TILE_SIZE, 0:TILE_SIZE]
+    gx = np.rint(BALL_X0 + xx - TILE_C).astype(np.int32)
+    gy = np.rint(BALL_Y0 + yy - TILE_C).astype(np.int32)
+    rgb = source[gy, gx].astype(np.float32)
+
+    # The measured body is close to circular.  A one-pixel antialiased rim
+    # preserves the soft photographic edge and the pale curved tennis seam.
+    radius = np.sqrt(((xx - TILE_C) / BALL_RX) ** 2 +
+                     ((yy - TILE_C) / BALL_RY) ** 2)
+    alpha = smoothstep(1.055, 0.925, radius).astype(np.float32)
+
+    # Pull edge color slightly inward so translucent pixels do not carry wood
+    # color when the tile is moved over a different part of the grain.
+    inner_x = np.rint(TILE_C + (xx - TILE_C) * 0.90).astype(np.int32)
+    inner_y = np.rint(TILE_C + (yy - TILE_C) * 0.90).astype(np.int32)
+    inner_x = np.clip(inner_x, 0, TILE_SIZE - 1)
+    inner_y = np.clip(inner_y, 0, TILE_SIZE - 1)
+    rim = (alpha > 0.0) & (alpha < 0.92)
+    rgb[rim] = rgb[inner_y[rim], inner_x[rim]]
+    return rgb, alpha

OBSERVATION.md

Scene observation and event plan

Measured scene

  • Source frame: 256 x 144 RGB, fixed landscape camera.
  • The light wood tabletop begins at about y=34 and fills the rest of the frame. Its horizontal grain and the rear table seam must remain fixed.
  • A black cylindrical tube enters from the left. Its visible bounds are approximately x=0..21, y=40..71; the round opening/nozzle is near x=14 and points to the right.
  • The small white, finely ribbed lampshade occupies approximately x=112..167, y=3..43. It is stationary on the rear part of the table.
  • The brown ball in frame 00 is centered near (109,59), with a visible body about 18 x 18 pixels and a soft contact/cast shadow extending slightly left and right. It is already well clear of the tube.
  • The ball's travel line is nearly horizontal, left-to-right, across the front/middle part of the tabletop. A very small downward drift is acceptable from the table perspective, but the camera, tube, lampshade, lighting, and wood texture must not move.
  • Lighting is soft and diffuse. The ball has a dark brown lower-left region, a warmer brown highlight toward its upper-right, a narrow curved tennis-ball seam, and a blurred dark shadow directly beneath it.

20-frame event plan (4 fps, 5 seconds)

The source image is treated as the first observed instant of an already ongoing roll. Frame 00 is copied byte-for-byte from first_frame.png. For later frames, only the old-ball area is repaired into a clean tabletop and a translated/rotated ball plus contact shadow is composited.

METHOD.md

Method

Actual generation method

This sequence is generated locally from first_frame.png with NumPy, Pillow, and OpenCV pixel operations. No external images, models, engines, masks, or references are used.

  1. The source PNG is decoded as a 256 x 144 RGB array. Frame 00 is copied directly from that array so it is pixel-identical to the input.
  2. A static clean plate is made for frames 01–19. The measured ball-and-shadow area is covered by an ellipse centered near (111,60). Because the table grain is predominantly horizontal, every masked scanline is reconstructed by RGB interpolation between clean pixels immediately to its left and right. This preserves the local illumination ramp and avoids disturbing the tube, lampshade, or the rest of the table.
  3. A 31 x 31 tile centered on the original ball is sampled from frame 00. A measured soft elliptical alpha mask isolates the ball while retaining its real brown texture and pale curved seam. Edge colors are pulled slightly inward to prevent wood-colored halos.
  4. For each later frame, the ball center advances exactly 9 pixels to the right. Its vertical center drifts only 3 pixels over the full visible path, consistent with the tabletop perspective. The texture rotates clockwise by distance / radius, coupling angular rotation to linear distance for rolling without slipping.
  5. A two-component analytic shadow is composited before the ball: a compact contact shadow and a wider, softer tail offset to the lower right. Both move with the ball. The rotated ball tile is then composited with premultiplied-alpha resampling and clipped naturally at the right frame edge.
  6. Frames 18 and 19 show the repaired static tabletop after the ball has cleared the right edge. All other scene pixels remain locked to the clean plate.
  7. artifact_io.save writes the packed uint32 0xRRGGBB tensor, 20 PNG frames, inspection contact sheet, and standalone player. transition_check.png shows frames 00–03 together with 4x-amplified consecutive-frame differences.

Revision and visual inspection

I visually inspected the source crop, the first generated inspection.png, the early-frame transition_check.png, candidate clean plates, the late empty frame, and the revised inspection.png using the image viewing tool.

The first pass used diffusion inpainting for the source ball area. It left a faint cloudy oval visible after the ball moved away. I compared Telea, Navier–Stokes, donor-patch, and scanline reconstruction candidates, then replaced the diffusion repair with horizontal scanline interpolation. The revised contact sheet shows a clean tabletop behind the moving ball, stable tube and lampshade geometry, continuous rightward motion, visible clockwise seam rotation, consistent contact shadow, and natural right-edge exit.
