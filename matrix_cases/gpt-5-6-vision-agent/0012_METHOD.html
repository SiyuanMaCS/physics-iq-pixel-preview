+def clean_table_plate(source: np.ndarray) -> np.ndarray:
+    """Remove the source orange ball and its broad contact shadow."""
+    mask = np.zeros((H, W), np.uint8)
+    # The asymmetric second ellipse includes the faint down/right cast shadow.
+    cv2.ellipse(mask, (37, 45), (11, 10), 0, 0, 360, 255, -1)
+    cv2.ellipse(mask, (42, 50), (17, 8), 0, 0, 360, 255, -1)
+    # Telea propagation follows the very low-frequency wood surface well here.
+    return cv2.inpaint(source, mask, 5.0, cv2.INPAINT_TELEA)

+def blend_shadow(frame: np.ndarray, cx: float, cy: float, progress: float) -> None:
+    """Blend a compact, soft tabletop contact/cast shadow in-place."""
+    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
+    sx = cx + 1.7
+    sy = cy + 6.2
+    # A sharp contact core nested in a wide diffuse shadow.
+    wide = np.exp(-0.5 * (((xx - sx) / 8.1) ** 2 + ((yy - sy) / 2.7) ** 2))
+    core = np.exp(-0.5 * (((xx - (cx + 0.7)) / 4.1) ** 2 + ((yy - (cy + 6.0)) / 1.25) ** 2))
+    alpha = np.clip((0.115 + 0.018 * progress) * wide + 0.095 * core, 0, 0.24)
+    # Restrict to the region below the sphere; avoids darkening its anti-aliased rim.
+    alpha *= 1.0 / (1.0 + np.exp(-(yy - (cy + 3.8)) * 1.7))
+    f = frame.astype(np.float32)
+    neutral_shadow = np.array([48.0, 45.0, 42.0], np.float32)
+    f[:] = f * (1.0 - alpha[..., None]) + neutral_shadow * alpha[..., None]
+    frame[:] = np.clip(f, 0, 255).astype(np.uint8)
+

+def render_ball(frame: np.ndarray, cx: float, cy: float, theta: float) -> None:
+    """Render an anti-aliased shaded orange sphere with a material-space seam."""
+    radius = 7.15
+    x0 = max(0, int(np.floor(cx - radius - 2)))
+    x1 = min(W, int(np.ceil(cx + radius + 2)))
+    y0 = max(0, int(np.floor(cy - radius - 2)))
+    y1 = min(H, int(np.ceil(cy + radius + 2)))

+    # Four-by-four deterministic supersampling keeps the sphere photographic at 14 px.
+    samples = 4
+    offsets = (np.arange(samples, dtype=np.float32) + 0.5) / samples - 0.5
+    py, px, oy, ox = np.meshgrid(
+        np.arange(y0, y1, dtype=np.float32),
+        np.arange(x0, x1, dtype=np.float32),
+        offsets,
+        offsets,
+        indexing="ij",

+    u = (px + ox - cx) / radius
+    v = (py + oy - cy) / radius
+    rr = u * u + v * v
+    inside = rr <= 1.0

OBSERVATION.md

Scene observation and event plan

Measured scene

  • Source: first_frame.png, RGB, 256 x 144 pixels.
  • Camera: static, low oblique view. The tabletop occupies most of the frame and is bounded approximately by the polygon (0,29) -> (220,37) -> (235,143) -> (21,143) -> (0,119).
  • Background: olive/mustard sofa across the upper part of the frame; dark wooden floor at far right; black pipe/nozzle enters from the left at about x=0..12, y=25..48.
  • Moving orange ball in frame 00: center approximately (37,45), visible radius about 7 px, with a soft contact shadow extending mostly down/right. This is already just outside the pipe.
  • Stationary small red ball: center approximately (97,35), radius 4 px.
  • Stationary brown ball: center approximately (105,31), radius 6 px. It overlaps the red ball slightly and must remain unchanged.
  • The table's far edge rises only slightly from left to right (y≈29 to y≈37). A straight path at nearly constant physical depth therefore projects as a gently descending screen path, from y=45 to about y=52.
  • Illumination comes mainly from upper left. The orange ball has a bright upper-left face, a darker lower-right rim, and a soft shadow below/right.

Animation and physics

  • Fixed camera and constant 4 fps throughout; only the orange ball and its contact shadow change.
  • The photographed ball and its original shadow are removed from a clean background plate for frames 01-19 using a locally reconstructed continuation of the pale wood grain.
  • The orange ball travels at nearly constant rolling speed from (37,45) to (217,52), approximately 9.5 px per frame. The slight downward drift follows table perspective.
  • A small subpixel vertical oscillation represents minute tabletop unevenness, without lateral wobble.
  • The ball appearance is reconstructed as a shaded orange sphere with a rotating curved tennis-ball seam and subtle surface texture. Seam phase advances consistently with distance (angle = distance / radius) so the visible rotation agrees with no-slip rolling.
  • Its soft elliptical shadow remains attached to the contact point and darkens slightly toward the right, where the tabletop is farther from the dominant light.
  • The ball stays on the tabletop through the final frame, reaching the right side without introducing a fall or bounce that the prompt does not request.

METHOD.md

Method

Actual generation method

This sequence was made entirely with deterministic local Python pixel operations. No external image, model, engine, mask, or reference was used.

  1. first_frame.png is loaded as an RGB uint8 array. Frame 00 is assigned directly from that array and is never redrawn or recompressed before the output PNG is saved.
  2. A clean locked-camera plate is created for frames 01-19. Two overlapping local ellipse masks cover the photographed orange ball and its faint down/right shadow. OpenCV's local Telea inpainting propagates the surrounding pale wood surface into only that masked area. The sofa, pipe, table boundaries, brown ball, and red ball are untouched.
  3. The orange ball follows a nearly constant-speed path from (37,45) to (217,52). The small downward slope is the image-space projection of a straight course at roughly constant tabletop depth. A subpixel vertical term of at most 0.16 px represents minor surface unevenness.
  4. The replacement ball is an analytic shaded sphere rendered with 4 x 4 supersampling per output pixel. Its color and lighting are matched to the photographed orange ball: warm orange diffuse material, upper-left illumination, dark lower-right limb, subtle front highlight, and fine deterministic surface variation.
  5. A faint curved tennis-ball seam is evaluated in the ball's material coordinate system. The material coordinates rotate about the screen-vertical axis by distance / radius, producing no-slip rolling rather than a sliding texture.
  6. A soft neutral contact shadow is composited before the sphere. It combines a compact contact core with a wider Gaussian cast shadow and remains anchored below/right of the ball throughout.
  7. artifact_io.save writes the 20 PNGs, the packed uint32 0xRRGGBB tensor, inspection sheet, and standalone player. generate.py separately creates transition_check.png from every consecutive pair.

Visual review and revision result

I visually reviewed inspection.png, .tmp/clean_plate.png, and transition_check.png with the image-view tool. The clean plate does not retain an orange silhouette; the only temporal changes after frame 01 are the moving sphere and its attached shadow; the course is straight and stays on the tabletop; and the stationary balls and all scene boundaries remain stable. The initial 00-to-01 difference correctly contains both the removal at the photographed start position and the ball's new position.

Verification

The generator performs these checks after saving:

  • frame 00 RGB pixels are exactly equal to first_frame.png;
  • all 20 PNGs are RGB, 256 x 144, and exactly equal to the in-memory frames;
  • pixels.npz contains a uint32 tensor with shape (144, 256, 20);
  • every packed tensor pixel exactly matches the corresponding PNG's 0xRRGGBB value.
